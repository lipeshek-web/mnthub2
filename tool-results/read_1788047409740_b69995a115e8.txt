    49→type PaymentMethodValue = 'PIX' | 'CREDIT_CARD'

    51→const METHOD_LABEL: Record<string, string> = {
    52→  PIX: 'PIX',

    60→/** Resumo normalizado do item em checkout (curso ou trilha) */
    61→interface ItemSummary {
    62→  kind: 'course' | 'track'
    63→  id: string
    64→  title: string
    65→  coverUrl: string | null
    66→  category: string
    67→  level: string
    68→  price: number
    69→  mentor: {
    70→    id: string
    71→    name: string
    72→    rating: number
    73→    reviewCount: number
    74→    avatarUrl?: string | null
    75→  }
    76→  isEnrolled: boolean
    77→  /** Presente apenas em trilhas: composição da jornada */
    78→  trackStats?: { courseCount: number; mentorshipSessions: number }

    80→
    81→export function CheckoutView({ courseId, trackId }: { courseId?: string; trackId?: string }) {
    82→  const navigate = useAppStore((s) => s.navigate)
    83→  const setExploreTab = useAppStore((s) => s.setExploreTab)
    84→  const user = useAppStore((s) => s.user)
    85→
    86→  const [course, setCourse] = useState<CourseDetailDTO | null>(null)
    87→  const [track, setTrack] = useState<TrackDetailDTO | null>(null)
    88→  const [loading, setLoading] = useState(true)
    89→  const [error, setError] = useState<string | null>(null)
    90→
    91→  const [paymentMethod, setPaymentMethod] = useState<PaymentMethodValue>('PIX')
    92→  const [paying, setPaying] = useState(false)
    93→  const [freeEnrolling, setFreeEnrolling] = useState(false)
    94→  const [order, setOrder] = useState<CheckoutResultDTO | null>(null)
    95→
    96→  // Cupom de desconto (apenas itens pagos)
    97→  const [couponInput, setCouponInput] = useState('')
    98→  const [couponApplied, setCouponApplied] = useState<CouponValidationDTO | null>(null)
    99→  const [couponLoading, setCouponLoading] = useState(false)
   100→  const [couponError, setCouponError] = useState<string | null>(null)
   101→
   102→  // Guarda: begin_checkout apenas na primeira montagem bem-sucedida
   103→  // (evita duplicidade em re-render/StrictMode)
   104→  const trackedRef = useRef(false)
   105→
   106→  const fetchSummary = useCallback(async () => {
   107→    setLoading(true)
   108→    setError(null)
   109→    try {
   110→      if (trackId) {
   111→        const d = await api.getTrack(trackId, user?.id)
   112→        setTrack(d)
   113→        setCourse(null)
   114→
   115→        // Pixels da plataforma recebem as conversões deste checkout
   116→        if (!trackedRef.current) {
   117→          trackedRef.current = true
   118→          loadTrackingScripts()
   119→          trackEvent('begin_checkout', {
   120→            mentorId: d.mentor.id,
   121→            value: d.price,
   122→            contentName: d.title,
   123→          })
   124→        }
   125→      } else if (courseId) {
   126→        const d = await api.getCourse(courseId, user?.id)
   127→        setCourse(d)
   128→        setTrack(null)
   129→
   130→        // Pixels do mentor recebem as conversões deste checkout
   131→        if (!trackedRef.current) {
   132→          trackedRef.current = true
   133→          loadTrackingScripts({
   134→            mentorGaId: d.mentor.tracking?.gaMeasurementId,
   135→            mentorPixelId: d.mentor.tracking?.metaPixelId,
   136→          })
   137→          trackEvent('begin_checkout', {
   138→            mentorId: d.mentor.id,
   139→            courseId: d.id,
   140→            value: d.price,
   141→            contentName: d.title,
   142→          })
   143→        }
   144→      } else {
   145→        setError('Item não encontrado.')
   146→      }
   147→    } catch (err) {
   148→      setError(err instanceof Error ? err.message : 'Não foi possível carregar o checkout.')
   149→    } finally {
   150→      setLoading(false)
   151→    }
   152→  }, [courseId, trackId, user?.id])
   153→
   154→  useEffect(() => {
   155→    void fetchSummary()
   156→  }, [fetchSummary])
   157→
   158→  // Trocou o item do checkout: volta o cupom ao estado inicial
   159→  useEffect(() => {
   160→    setCouponInput('')
   161→    setCouponApplied(null)
   162→    setCouponError(null)
   163→    setCouponLoading(false)
   164→  }, [courseId, trackId])
   165→
   166→  /* ---------- Ações ---------- */
   167→
   168→  const applyCoupon = async () => {
   169→    if (!item || couponLoading) return
   170→    const code = couponInput.trim().toUpperCase()
   171→    if (!code) {
   172→      setCouponApplied(null)
   173→      setCouponError('Digite um cupom para aplicar.')
   174→      return
   175→    }
   176→    setCouponLoading(true)
   177→    setCouponError(null)
   178→    try {
   179→      const result = await api.validateCoupon({
   180→        code,
   181→        courseId: item.kind === 'course' ? item.id : undefined,
   182→        trackId: item.kind === 'track' ? item.id : undefined,
   183→      })
   184→      setCouponApplied(result)
   185→      setCouponInput('')
   186→    } catch (err) {
   187→      setCouponApplied(null)
   188→      setCouponError(err instanceof Error ? err.message : 'Não foi possível validar o cupom.')
   189→    } finally {
   190→      setCouponLoading(false)
   191→    }
   192→  }
   193→
   194→  const removeCoupon = () => {
   195→    setCouponApplied(null)
   196→    setCouponInput('')
   197→    setCouponError(null)
   198→  }
   199→
   200→  const doEnrollFree = async () => {
   201→    if (!user || !item) return
   202→    setFreeEnrolling(true)
   203→    try {
   204→      if (item.kind === 'track') {
   205→        await api.enrollTrack(item.id, user.id)
   206→        toast.success('Inscrição realizada! Boa jornada 🎉')
   207→        navigate({ name: 'track', trackId: item.id })
   208→      } else {
   209→        const res = await api.enrollCourse(item.id, user.id)
   210→        toast.success(
   211→          res.alreadyEnrolled
   212→            ? 'Você já estava inscrito neste curso.'
   213→            : 'Inscrição realizada! Boa jornada 🎉'
   214→        )
   215→        navigate({ name: 'course', courseId: item.id })
   216→      }
   217→    } catch (err) {
   218→      toast.error(err instanceof Error ? err.message : 'Erro ao realizar inscrição.')
   219→    } finally {
   220→      setFreeEnrolling(false)
   221→    }
   222→  }
   223→
   224→  const doPay = async () => {
   225→    if (!user || !item || paying) return
   226→    setPaying(true)
   227→    try {
   228→      // A atribuição (utm/gclid/fbclid/canal/landing) viaja no mesmo corpo —
   229→      // o servidor lê body.attribution para o pedido e o evento purchase.
   230→      const payload = {
   231→        userId: user.id,
   232→        courseId: item.kind === 'course' ? item.id : undefined,
   233→        trackId: item.kind === 'track' ? item.id : undefined,
   234→        paymentMethod,
   235→        couponCode: couponApplied?.code || undefined,
   236→        attribution: getAttribution(),
   237→      }
   238→      const result = await api.checkout(payload)
   239→
   240→      trackEvent('purchase', {
   241→        mentorId: item.mentor.id,
   242→        courseId: item.kind === 'course' ? item.id : null,
   243→        value: item.price,
   244→        transactionId: result.order.id,
   245→        contentName: item.title,
   246→      })
   247→      toast.success('Pagamento aprovado! 🎉')
   248→      setOrder(result)
   249→    } catch (err) {
   250→      const message = err instanceof Error ? err.message : 'Não foi possível concluir o pagamento.'
   251→      toast.error(message)
   252→      // Cupom pode ter esgotado/expirado entre a validação e o pagamento:
   253→      // mostra o erro no fluxo normal e volta o cupom ao estado inicial.
   254→      if (/cupom/i.test(message)) {
   255→        setCouponApplied(null)
   256→        setCouponInput('')
   257→        setCouponError(message)
   258→      }
   259→      // Matrícula já existia (409): recarrega para exibir o estado "já inscrito"
   260→      if (message.includes('já tem acesso')) void fetchSummary()
   261→    } finally {
   262→      setPaying(false)
   263→    }
   264→  }
   265→
   266→  /* ---------- Skeleton ---------- */
   267→  if (loading) {
   268→    return (
   269→      <div className="mx-auto w-full max-w-3xl px-4 py-6 sm:py-8" aria-busy="true">
   270→        <Skeleton className="h-9 w-28 rounded-full" />
   271→        <Skeleton className="mt-4 h-64 w-full rounded-2xl" />
   272→        <Skeleton className="mt-6 h-24 w-full rounded-2xl" />
   273→        <Skeleton className="mt-6 h-12 w-full rounded-full" />
   274→        <p className="sr-only">Carregando checkout…</p>
   275→      </div>
   276→    )
   277→  }
   278→
   279→  /* ---------- Erro ---------- */
   280→  if (error || (!course && !track)) {
   281→    return (
   282→      <div className="mx-auto w-full max-w-3xl px-4 py-16">
   283→        <Card className="mx-auto max-w-md border-dashed">
   284→          <CardContent className="flex flex-col items-center gap-3 p-10 text-center">
   285→            <span className="flex h-14 w-14 items-center justify-center rounded-full bg-rose-50 dark:bg-rose-950/50">
   286→              <AlertCircle aria-hidden className="h-7 w-7 text-rose-500" />
   287→            </span>
   288→            <p className="font-bold text-stone-900 dark:text-stone-50">Não foi possível carregar o checkout</p>
   289→            <p className="max-w-sm text-sm leading-relaxed text-stone-500 dark:text-stone-400">
   290→              {error ?? 'Item não encontrado.'}
   291→            </p>
   292→            <Button variant="outline" className="rounded-full" onClick={() => void fetchSummary()}>
   293→              Tentar novamente
   294→            </Button>
   295→          </CardContent>
   296→        </Card>
   297→      </div>
   298→    )
   299→  }
   300→
   301→  /* ---------- Resumo normalizado (curso ou trilha) ---------- */
   302→  const item: ItemSummary | null = track
   303→    ? {
   304→        kind: 'track',
   305→        id: track.id,
   306→        title: track.title,
   307→        coverUrl: track.coverUrl ?? null,
   308→        category: track.category,
   309→        level: track.level,
   310→        price: track.price,
   311→        mentor: {
   312→          id: track.mentor.id,
   313→          name: track.mentor.name,
   314→          rating: track.mentor.rating,
   315→          reviewCount: track.mentor.reviewCount,
   316→          avatarUrl: track.mentor.avatarUrl,
   317→        },
   318→        isEnrolled: track.myEnrollment !== null,
   319→        trackStats: {
   320→          courseCount: track.courseCount,
   321→          mentorshipSessions: track.mentorshipSessions,
   322→        },
   323→      }
   324→    : course
   325→      ? {
   326→          kind: 'course',
   327→          id: course.id,
   328→          title: course.title,
   329→          coverUrl: course.coverUrl ?? null,
   330→          category: course.category,
   331→          level: course.level,
   332→          price: course.price,
   333→          mentor: {
   334→            id: course.mentor.id,
   335→            name: course.mentor.name,
   336→            rating: course.mentor.rating,
   337→            reviewCount: course.mentor.reviewCount,
   338→            avatarUrl: course.mentor.avatarUrl,
   339→          },
   340→          isEnrolled: course.enrollment !== null,
   341→        }
   342→      : null
   343→
   344→  if (!item) return null
   345→
   346→  const isTrack = item.kind === 'track'
   347→
   348→  /* ---------- Tela de SUCESSO (substitui o formulário) ---------- */
   349→  if (order) {
   350→    return (
   351→      <div className="mx-auto w-full max-w-3xl px-4 py-10 sm:py-14">
   352→        <Card className="mx-auto max-w-md rounded-2xl">
   353→          <CardContent className="flex flex-col items-center gap-1.5 p-8 text-center">
   354→            <span className="flex h-16 w-16 items-center justify-center rounded-full bg-emerald-50 dark:bg-emerald-950/50">
   355→              <CheckCircle2 aria-hidden className="h-9 w-9 text-emerald-600 dark:text-emerald-400" />
   356→            </span>
   357→            <h1 className="mt-2 text-xl font-extrabold tracking-tight text-stone-900 dark:text-stone-50 sm:text-2xl">
   358→              Pagamento confirmado!
   359→            </h1>
   360→            <p className="text-sm leading-relaxed text-stone-500 dark:text-stone-400">
   361→              Você já tem acesso a{' '}
   362→              <span className="font-semibold text-stone-700 dark:text-stone-200">“{order.order.itemTitle}”</span>
   363→            </p>
   364→
   365→            <dl className="mt-5 w-full space-y-2 rounded-xl border border-stone-200 dark:border-stone-800 bg-stone-50/60 dark:bg-stone-950/50 p-4 text-left text-sm">
   366→              <div className="flex items-center justify-between gap-3">
   367→                <dt className="text-stone-500 dark:text-stone-400">Pedido</dt>
   368→                <dd className="font-bold text-stone-800 dark:text-stone-200">#{order.order.id.slice(0, 8).toUpperCase()}</dd>
   369→              </div>
   370→              <div className="flex items-center justify-between gap-3">
   371→                <dt className="text-stone-500 dark:text-stone-400">Método</dt>
   372→                <dd className="font-bold text-stone-800 dark:text-stone-200">
   373→                  {METHOD_LABEL[order.order.paymentMethod] ?? order.order.paymentMethod}
   374→                </dd>
   375→              </div>
   376→              <div className="flex items-center justify-between gap-3">
   377→                <dt className="text-stone-500 dark:text-stone-400">Total</dt>
   378→                <dd className="font-extrabold text-emerald-700 dark:text-emerald-300">{formatBRL(order.order.amount)}</dd>
   379→              </div>
   380→            </dl>
   381→
   382→            {isTrack ? (
   383→              <>
   384→                <Button
   385→                  onClick={() => navigate({ name: 'track', trackId: item.id })}
   386→                  className="mt-6 h-12 w-full rounded-full font-bold"
   387→                >
   388→                  Acessar a trilha
   389→                </Button>
   390→                <Button
   391→                  variant="outline"
   392→                  onClick={() => {
   393→                    setExploreTab('tracks')
   394→                    navigate({ name: 'marketplace' })
   395→                  }}
   396→                  className="w-full rounded-full"
   397→                >
   398→                  Explorar mais trilhas
   399→                </Button>
   400→              </>
   401→            ) : (
   402→              <>
   403→                <Button
   404→                  onClick={() => navigate({ name: 'course', courseId: item.id })}
   405→                  className="mt-6 h-12 w-full rounded-full font-bold"
   406→                >
   407→                  Acessar o curso
   408→                </Button>
   409→                <Button
   410→                  variant="outline"
   411→                  onClick={() => navigate({ name: 'marketplace' })}
   412→                  className="w-full rounded-full"
   413→                >
   414→                  Explorar mais cursos
   415→                </Button>
   416→              </>
   417→            )}
   418→          </CardContent>
   419→        </Card>
   420→      </div>
   421→    )
   422→  }
   423→
   424→  /* ---------- Checkout ---------- */
   425→
   426→  return (
   427→    <div className="mx-auto w-full max-w-3xl px-4 py-6 sm:py-8">
   428→      <Button
   429→        variant="ghost"
   430→        onClick={() =>
   431→          isTrack
   432→            ? navigate({ name: 'track', trackId: item.id })
   433→            : navigate({ name: 'course', courseId: item.id })
   434→        }
   435→        className="-ml-2 h-10 gap-1.5 rounded-full px-3 font-semibold text-stone-600 dark:text-stone-300 hover:text-stone-900 dark:hover:text-stone-50"
   436→      >
   437→        <ArrowLeft aria-hidden className="h-4 w-4" />
   438→        {isTrack ? 'Voltar à trilha' : 'Voltar ao curso'}
   439→      </Button>
   440→      <h1 className="mt-1 text-xl font-extrabold tracking-tight text-stone-900 dark:text-stone-50 sm:text-2xl">
   441→        Checkout
   442→      </h1>
   443→
   444→      {/* Sem login: pedir para entrar */}
   445→      {!user && (
   446→        <Card className="mt-5 border-dashed">
   447→          <CardContent className="flex flex-col items-center gap-3 p-10 text-center">
   448→            <span className="flex h-14 w-14 items-center justify-center rounded-full bg-emerald-50 dark:bg-emerald-950/50">
   449→              <LogIn aria-hidden className="h-7 w-7 text-emerald-700 dark:text-emerald-300" />
   450→            </span>
   451→            <p className="font-bold text-stone-900 dark:text-stone-50">Entre para concluir a compra</p>
   452→            <p className="max-w-sm text-sm leading-relaxed text-stone-500 dark:text-stone-400">
   453→              Entre com uma conta para concluir a compra de “{item.title}”.
   454→            </p>
   455→            <Button
   456→              className="mt-1 h-11 rounded-full px-8 font-bold"
   457→              onClick={() => navigate({ name: 'auth', mode: 'login' })}
   458→            >
   459→              Entrar ou criar conta
   460→            </Button>
   461→          </CardContent>
   462→        </Card>
   463→      )}
   464→
   465→      {/* Já inscrito: acesso liberado */}
   466→      {user && item.isEnrolled && (
   467→        <Card className="mt-5 rounded-2xl border-emerald-200 dark:border-emerald-900 bg-emerald-50/40 dark:bg-emerald-950/50">
   468→          <CardContent className="flex flex-col items-center gap-3 p-10 text-center">
   469→            <span className="flex h-14 w-14 items-center justify-center rounded-full bg-emerald-100 dark:bg-emerald-950/50">
   470→              <CheckCircle2 aria-hidden className="h-7 w-7 text-emerald-700 dark:text-emerald-300" />
   471→            </span>
   472→            <p className="font-bold text-stone-900 dark:text-stone-50">
   473→              {isTrack ? 'Você já tem acesso a esta trilha' : 'Você já tem acesso a este curso'}
   474→            </p>
   475→            <p className="max-w-sm text-sm leading-relaxed text-stone-500 dark:text-stone-400">
   476→              Sua matrícula em “{item.title}” já está ativa — bons estudos!
   477→            </p>
   478→            <Button
   479→              onClick={() =>
   480→                isTrack
   481→                  ? navigate({ name: 'track', trackId: item.id })
   482→                  : navigate({ name: 'course', courseId: item.id })
   483→              }
   484→              className="mt-1 h-11 rounded-full px-6 font-bold"
   485→            >
   486→              {isTrack ? 'Acessar a trilha' : 'Ir para o curso'}
   487→            </Button>
   488→          </CardContent>
   489→        </Card>
   490→      )}
   491→
   492→      {/* Item gratuito: matrícula direta (defensivo — trilhas gratuitas normalmente
   493→          são inscritas direto na tela da trilha, sem passar pelo checkout) */}
   494→      {user && !item.isEnrolled && item.price === 0 && (
   495→        <div className="mt-5 flex flex-col items-center gap-2.5 rounded-2xl border border-emerald-200 dark:border-emerald-900 bg-emerald-50/60 dark:bg-emerald-950/50 p-8 text-center">
   496→          <p className="font-bold text-stone-900 dark:text-stone-50">
   497→            {isTrack ? 'Esta trilha é gratuita' : 'Este curso é gratuito'}
   498→          </p>
   499→          <p className="max-w-sm text-sm leading-relaxed text-stone-500 dark:text-stone-400">
   500→            Não é preciso pagar nada: matricule-se e comece a aprender agora mesmo.
   501→          </p>
   502→          <Button
   503→            onClick={() => void doEnrollFree()}
   504→            disabled={freeEnrolling}
   505→            className="mt-1 h-11 w-full rounded-full px-8 font-bold sm:w-auto"
   506→          >
   507→            {freeEnrolling && <Loader2 aria-hidden className="h-4 w-4 animate-spin" />}
   508→            Inscrever-se
   509→          </Button>
   510→        </div>
   511→      )}
   512→
   513→      {/* Fluxo de pagamento (item pago + logado + não inscrito) */}
   514→      {user && !item.isEnrolled && item.price > 0 && (
   515→        <>
   516→          {/* ---- RESUMO DO PEDIDO ---- */}
   517→          <Card className="mt-5 gap-0 overflow-hidden rounded-2xl py-0">
   518→            {item.coverUrl ? (
