   191→function BookingCard({
   192→  booking: b,
   193→  userId,
   194→  busy,
   195→  onConfirm,
   196→  onReject,
   197→  onComplete,
   198→  onCancel,
   199→  onJoin,
   200→  onReview,
   201→}: {
   202→  booking: BookingDTO
   203→  userId: string
   204→  busy: boolean
   205→  onConfirm: (booking: BookingDTO) => void
   206→  onReject: (booking: BookingDTO) => void
   207→  onComplete: (booking: BookingDTO) => void
   208→  onCancel: (booking: BookingDTO) => void
   209→  onJoin: (booking: BookingDTO) => void
   210→  onReview: (booking: BookingDTO, rating: number, comment: string) => Promise<void>
   211→}) {
   212→  const isMentorSide = b.mentor.userId === userId
   213→  const otherName = isMentorSide ? b.mentee.name : b.mentor.name
   214→  const relative = relativeDayLabel(b.startsAt)
   215→  // #8: "Google Calendar" só para sessões futuras aguardando confirmação/confirmadas
   216→  const isUpcomingEvent =
   217→    (b.status === 'PENDING' || b.status === 'CONFIRMED') && b.startsAt >= nowNaive()
   218→  const gcalUrl = (() => {
   219→    const start = new Date(b.startsAt) // naive → Date local
   220→    const end = new Date(start.getTime() + b.durationMin * 60_000)
   221→    const fmt = (d: Date) => d.toISOString().replace(/[-:]/g, '').replace(/\.\d{3}/, '')
   222→    const params = new URLSearchParams({
   223→      action: 'TEMPLATE',
   224→      text: `Mentoria: ${b.topic}`,
   225→      dates: `${fmt(start)}/${fmt(end)}`,
   226→      details: `Sessão de mentoria no MentorHub com ${otherName}.`,
   227→      location: b.meetingRoom,
   228→    })
   229→    return `https://calendar.google.com/calendar/render?${params.toString()}`
   230→  })()
   231→  const statusMeta = STATUS_META[b.status] ?? {
   232→    label: b.status,
   233→    className: 'bg-stone-100 dark:bg-stone-800 text-stone-600 dark:text-stone-300 border-stone-200 dark:border-stone-800',
   234→  }
   235→
   236→  const [dialogAction, setDialogAction] = useState<ConfirmKind | null>(null)
   237→  const [reviewOpen, setReviewOpen] = useState(false)
   238→  const [rating, setRating] = useState(0)
   239→  const [hoverRating, setHoverRating] = useState(0)
   240→  const [comment, setComment] = useState('')
   241→  const [submitting, setSubmitting] = useState(false)
   242→
   243→  const canJoin = b.status === 'CONFIRMED'
   244→  const canReview = !isMentorSide && b.status === 'COMPLETED' && !b.reviewed
   245→
   246→  const dialogContent: Record<ConfirmKind, { title: string; description: string; actionLabel: string }> = {
   247→    reject: {
   248→      title: 'Recusar solicitação?',
   249→      description: `A solicitação de ${b.mentee.name} sobre "${b.topic}" será cancelada e o horário será liberado na sua agenda.`,
   250→      actionLabel: 'Recusar',
   251→    },
   252→    complete: {
   253→      title: 'Marcar como concluída?',
   254→      description: `Confirme que a sessão com ${b.mentee.name} aconteceu. Ela irá para o histórico e o aluno poderá avaliá-la.`,
   255→      actionLabel: 'Concluir sessão',
   256→    },
   257→    cancel: {
   258→      title: 'Cancelar sessão?',
   259→      description: `Sua sessão com ${b.mentor.name} será cancelada e o horário será liberado para outras pessoas.`,
   260→      actionLabel: 'Cancelar sessão',
   261→    },
   262→  }
   263→  const currentDialog = dialogAction ? dialogContent[dialogAction] : null
   264→
   265→  const submitReview = async () => {
   266→    if (rating < 1) {
   267→      toast.error('Escolha uma nota de 1 a 5 estrelas antes de enviar.')
   268→      return
   269→    }
   270→    setSubmitting(true)
   271→    try {
   272→      await onReview(b, rating, comment.trim())
   273→      setReviewOpen(false)
   274→      setRating(0)
   275→      setHoverRating(0)
   276→      setComment('')
   277→    } catch (err) {
   278→      toast.error(err instanceof Error ? err.message : 'Não foi possível enviar sua avaliação.')
   279→    } finally {
   280→      setSubmitting(false)
   281→    }
   282→  }
   283→
   284→  return (
   285→    <Card className="gap-4 py-6">
   286→      <CardContent className="flex flex-col gap-4 px-6">
   287→        <div className="flex flex-wrap items-start justify-between gap-3">
   288→          <div className="flex min-w-0 items-start gap-3">
   289→            <Avatar name={otherName} size="lg" />
   290→            <div className="min-w-0">
   291→              <div className="flex flex-wrap items-center gap-2">
   292→                <p className="font-semibold leading-tight">{otherName}</p>
   293→                <Badge
   294→                  className={
   295→                    isMentorSide
   296→                      ? 'border-emerald-200 dark:border-emerald-900 bg-emerald-50 dark:bg-emerald-950/50 text-emerald-700 dark:text-emerald-300'
   297→                      : 'border-stone-200 dark:border-stone-800 bg-stone-100 dark:bg-stone-800 text-stone-600 dark:text-stone-300'
   298→                  }
   299→                >
   300→                  {isMentorSide ? 'Como mentor' : 'Como aluno'}
   301→                </Badge>
   302→              </div>
   303→              {!isMentorSide && b.mentor.headline ? (
   304→                <p className="mt-0.5 text-sm text-muted-foreground">{b.mentor.headline}</p>
   305→              ) : null}
   306→              <p className="mt-1.5 font-medium">{b.topic}</p>
   307→              {b.notes ? <p className="mt-1 text-sm text-muted-foreground">{b.notes}</p> : null}
   308→            </div>
   309→          </div>
   310→          <Badge variant="outline" className={statusMeta.className}>
   311→            {statusMeta.label}
   312→          </Badge>
   313→        </div>
   314→
   315→        <div className="flex flex-wrap items-center gap-x-4 gap-y-1.5 text-sm text-muted-foreground">
   316→          <span className="inline-flex items-center gap-1.5">
   317→            <CalendarDays className="size-4 text-emerald-600 dark:text-emerald-400" aria-hidden />
   318→            {relative ? `${relative} · ` : ''}
   319→            {formatDayLabelLong(b.startsAt)} · {formatTimeLabel(b.startsAt)} →{' '}
   320→            {addMinutesToTime(b.startsAt, b.durationMin)}
   321→          </span>
   322→          <span className="inline-flex items-center gap-1.5">
   323→            <Clock className="size-4" aria-hidden /> {b.durationMin} min
   324→          </span>
   325→          <span className="inline-flex items-center gap-1.5">
   326→            <Wallet className="size-4" aria-hidden /> {currencyBRL(b.price)}
   327→          </span>
   328→          {isUpcomingEvent ? (
   329→            <a
   330→              href={gcalUrl}
   331→              target="_blank"
   332→              rel="noopener noreferrer"
   333→              aria-label={`Adicionar "${b.topic}" ao Google Calendar`}
   334→              className="inline-flex items-center gap-1.5 rounded px-1 py-1 -my-1 text-sm font-medium text-emerald-700 underline-offset-4 hover:underline focus-visible:outline-2 focus-visible:outline-emerald-600 dark:text-emerald-400"
   335→            >
   336→              <ExternalLink className="size-3.5" aria-hidden /> Google Calendar
   337→            </a>
   338→          ) : null}
   339→        </div>
   340→
   341→        <div className="flex flex-wrap gap-2">
   342→          {isMentorSide && b.status === 'PENDING' ? (
   343→            <>
   344→              <Button size="sm" disabled={busy} onClick={() => onConfirm(b)}>
   345→                <Check className="size-4" aria-hidden /> Confirmar
   346→              </Button>
   347→              <Button size="sm" variant="outline" disabled={busy} onClick={() => setDialogAction('reject')}>
   348→                <X className="size-4" aria-hidden /> Recusar
   349→              </Button>
   350→            </>
   351→          ) : null}
   352→
   353→          {isMentorSide && b.status === 'CONFIRMED' ? (
   354→            <Button size="sm" variant="outline" disabled={busy} onClick={() => setDialogAction('complete')}>
   355→              <CheckCheck className="size-4" aria-hidden /> Marcar como concluída
   356→            </Button>
   357→          ) : null}
   358→
   359→          {canJoin ? (
   360→            <Button size="sm" onClick={() => onJoin(b)}>
   361→              <Video className="size-4" aria-hidden /> Entrar na sala
   362→            </Button>
   363→          ) : null}
   364→
   365→          {!isMentorSide && (b.status === 'PENDING' || b.status === 'CONFIRMED') ? (
   366→            <Button
   367→              size="sm"
   368→              variant="ghost"
   369→              className="text-rose-600 dark:text-rose-400 hover:bg-rose-50 dark:hover:bg-rose-950/50 hover:text-rose-700 dark:hover:text-rose-400"
   370→              disabled={busy}
   371→              onClick={() => setDialogAction('cancel')}
   372→            >
   373→              Cancelar sessão
   374→            </Button>
   375→          ) : null}
   376→
   377→          {canReview ? (
   378→            <Button size="sm" variant="outline" onClick={() => setReviewOpen(true)}>
   379→              <Star className="size-4 fill-amber-400 text-amber-400" aria-hidden /> Avaliar sessão
   380→            </Button>
   381→          ) : null}
   382→        </div>
   383→      </CardContent>
   384→
   385→      {/* Confirmações: recusar / concluir / cancelar */}
   386→      <AlertDialog
   387→        open={dialogAction !== null}
   388→        onOpenChange={(open) => {
   389→          if (!open) setDialogAction(null)
   390→        }}
   391→      >
   392→        <AlertDialogContent>
   393→          <AlertDialogHeader>
   394→            <AlertDialogTitle>{currentDialog?.title}</AlertDialogTitle>
   395→            <AlertDialogDescription>{currentDialog?.description}</AlertDialogDescription>
   396→          </AlertDialogHeader>
   397→          <AlertDialogFooter>
   398→            <AlertDialogCancel>Voltar</AlertDialogCancel>
   399→            <AlertDialogAction
   400→              className="bg-rose-600 text-white hover:bg-rose-700"
   401→              disabled={busy}
   402→              onClick={() => {
   403→                const action = dialogAction
   404→                setDialogAction(null)
   405→                if (action === 'reject') onReject(b)
   406→                else if (action === 'complete') onComplete(b)
   407→                else if (action === 'cancel') onCancel(b)
   408→              }}
   409→            >
   410→              {currentDialog?.actionLabel}
   411→            </AlertDialogAction>
   412→          </AlertDialogFooter>
   413→        </AlertDialogContent>
   414→      </AlertDialog>
   415→
   416→      {/* Avaliar sessão */}
   417→      <Dialog open={reviewOpen} onOpenChange={setReviewOpen}>
   418→        <DialogContent className="sm:max-w-md">
   419→          <DialogHeader>
   420→            <DialogTitle>Avaliar sessão</DialogTitle>
   421→            <DialogDescription>
   422→              Como foi sua mentoria com {b.mentor.name} sobre &quot;{b.topic}&quot;? Seu feedback ajuda outros
   423→              alunos.
   424→            </DialogDescription>
   425→          </DialogHeader>
   426→          <div className="flex flex-col gap-4">
   427→            <div className="flex items-center justify-center gap-1" role="radiogroup" aria-label="Nota da avaliação">
   428→              {[1, 2, 3, 4, 5].map((value) => {
   429→                const active = (hoverRating || rating) >= value
   430→                return (
   431→                  <button
   432→                    key={value}
   433→                    type="button"
   434→                    role="radio"
   435→                    aria-checked={rating === value}
   436→                    aria-label={`${value} ${value === 1 ? 'estrela' : 'estrelas'}`}
   437→                    onClick={() => setRating(value)}
   438→                    onMouseEnter={() => setHoverRating(value)}
   439→                    onMouseLeave={() => setHoverRating(0)}
   440→                    className="rounded-full p-1 transition-transform hover:scale-110 focus-visible:outline-2 focus-visible:outline-amber-500"
   441→                  >
   442→                    <Star
   443→                      size={28}
   444→                      className={cn(
   445→                        'transition-colors',
   446→                        active ? 'fill-amber-400 text-amber-400' : 'fill-stone-200 dark:fill-stone-800 text-stone-300 dark:text-stone-600'
   447→                      )}
   448→                      aria-hidden
   449→                    />
   450→                  </button>
   451→                )
   452→              })}
   453→            </div>
   454→            <div className="flex flex-col gap-2">
   455→              <Label htmlFor={`review-comment-${b.id}`}>Como foi sua experiência?</Label>
   456→              <Textarea
   457→                id={`review-comment-${b.id}`}
   458→                rows={4}
   459→                value={comment}
   460→                onChange={(event) => setComment(event.target.value)}
   461→                placeholder="Conte o que funcionou bem e o que pode melhorar..."
   462→              />
   463→            </div>
   464→          </div>
   465→          <DialogFooter>
   466→            <Button variant="ghost" onClick={() => setReviewOpen(false)} disabled={submitting}>
   467→              Cancelar
   468→            </Button>
   469→            <Button onClick={() => void submitReview()} disabled={submitting}>
   470→              {submitting ? 'Enviando...' : 'Enviar avaliação'}
   471→            </Button>
   472→          </DialogFooter>
   473→        </DialogContent>
   474→      </Dialog>
   475→    </Card>
   476→  )
   477→}

   964→function EnrolledCourseCard({
   965→  enrollment,
   966→  onOpen,
   967→}: {
   968→  enrollment: EnrolledCourseDTO
   969→  onOpen: () => void
   970→}) {
   971→  const { course, completedLessonIds } = enrollment
   972→  const total = Math.max(course.lessonCount, 1)
   973→  const completed = completedLessonIds.length
   974→  const pct = Math.round((completed / total) * 100)
   975→  const isDone = completed >= course.lessonCount && course.lessonCount > 0
   976→
   977→  return (
   978→    <Card className="overflow-hidden p-0">
   979→      <div className="flex flex-col gap-4 p-5 sm:flex-row sm:items-center">
   980→        {course.coverUrl ? (
   981→          <img
   982→            src={course.coverUrl}
   983→            alt=""
   984→            aria-hidden
   985→            className="h-14 w-14 shrink-0 rounded-2xl object-cover"
   986→          />
   987→        ) : (
   988→          <span
   989→            aria-hidden
   990→            style={avatarGradient(course.title)}
   991→            className="flex h-14 w-14 shrink-0 items-center justify-center rounded-2xl text-white"
   992→          >
   993→            <Library className="size-6" />
   994→          </span>
   995→        )}
   996→
   997→        <div className="min-w-0 flex-1">
   998→          <div className="flex flex-wrap items-center gap-2">
   999→            <p className="truncate font-bold text-stone-900 dark:text-stone-50">{course.title}</p>
  1000→            {isDone && (
  1001→              <Badge className="border-emerald-200 dark:border-emerald-900 bg-emerald-100 dark:bg-emerald-950/50 text-emerald-800 dark:text-emerald-300">
  1002→                <CheckCircle2 className="size-3" aria-hidden /> Concluído
  1003→              </Badge>
  1004→            )}
  1005→          </div>
  1006→          <p className="mt-0.5 text-xs text-muted-foreground">
  1007→            por {course.mentor.name} · {course.lessonCount} aulas ·{' '}
  1008→            {formatTotalDuration(course.totalDurationMin)}
  1009→          </p>
  1010→          <div className="mt-2.5 flex items-center gap-3">
  1011→            <Progress value={pct} aria-label={`${pct}% do curso concluído`} className="h-2 max-w-64 flex-1" />
  1012→            <span className="shrink-0 text-xs font-semibold text-stone-500 dark:text-stone-400">
  1013→              {completed}/{course.lessonCount} aulas
  1014→            </span>
  1015→          </div>
  1016→        </div>
  1017→
  1018→        <Button onClick={onOpen} className="shrink-0 self-start rounded-full sm:self-auto">
  1019→          <PlayCircle className="size-4" aria-hidden /> {isDone ? 'Revisar curso' : 'Continuar'}
  1020→        </Button>
  1021→      </div>
  1022→    </Card>
  1023→  )
  1024→}
