   853→/* ==================== NAVEGAÇÃO DE CONTEÚDOS (sidebar + dialog mobile) ==================== */
   854→
   855→function ContentsNav({
   856→  groups,
   857→  themeList,
   858→  collapsedThemes,
   859→  onToggleTheme,
   860→  currentLessonId,
   861→  completedIds,
   862→  onSelectLesson,
   863→  className,
   864→}: {
   865→  groups: LessonGroup[]
   866→  themeList: CourseThemeDTO[]
   867→  collapsedThemes: Record<string, boolean>
   868→  onToggleTheme: (key: string) => void
   869→  currentLessonId: string | null
   870→  completedIds: string[]
   871→  onSelectLesson: (lessonId: string) => void
   872→  className?: string
   873→}) {
   874→  const totalLessons = groups.reduce((n, g) => n + g.lessons.length, 0)
   875→
   876→  return (
   877→    <nav
   878→      aria-label="Conteúdos do curso"
   879→      className={cn(
   880→        'min-h-0 flex-1 overflow-y-auto px-3 pb-3 [scrollbar-width:thin] [&::-webkit-scrollbar-thumb]:rounded-full [&::-webkit-scrollbar-thumb]:bg-stone-300 [&::-webkit-scrollbar-track]:bg-stone-100 [&::-webkit-scrollbar]:w-1.5',
   881→        className
   882→      )}
   883→    >
   884→      {groups.map((group) => {
   885→        const collapsible = themeList.length > 0
   886→        const isCollapsed = collapsible && Boolean(collapsedThemes[group.key])
   887→        const groupTotal = group.lessons.length
   888→        const groupDone = group.lessons.filter((l) => completedIds.includes(l.id)).length
   889→        return (
   890→          <div key={group.key} className="space-y-1 py-1">
   891→            {collapsible ? (
   892→              <button
   893→                type="button"
