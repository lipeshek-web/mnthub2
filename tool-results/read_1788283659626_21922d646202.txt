   878→/* ==================== NAVEGAÇÃO DE CONTEÚDOS (sidebar + dialog mobile) ==================== */
   879→
   880→function ContentsNav({
   881→  groups,
   882→  themeList,
   883→  collapsedThemes,
   884→  onToggleTheme,
   885→  currentLessonId,
   886→  completedIds,
   887→  onSelectLesson,
   888→  className,
   889→}: {
   890→  groups: LessonGroup[]
   891→  themeList: CourseThemeDTO[]
   892→  collapsedThemes: Record<string, boolean>
   893→  onToggleTheme: (key: string) => void
   894→  currentLessonId: string | null
   895→  completedIds: string[]
   896→  onSelectLesson: (lessonId: string) => void
   897→  className?: string
   898→}) {
   899→  const totalLessons = groups.reduce((n, g) => n + g.lessons.length, 0)
   900→
   901→  return (
   902→    <nav
   903→      aria-label="Conteúdos do curso"
   904→      className={cn(
   905→        'min-h-0 flex-1 overflow-y-auto px-3 pb-3 [scrollbar-width:thin] [&::-webkit-scrollbar-thumb]:rounded-full [&::-webkit-scrollbar-thumb]:bg-stone-300 dark:[&::-webkit-scrollbar-thumb]:bg-stone-700 [&::-webkit-scrollbar-track]:bg-stone-100 dark:[&::-webkit-scrollbar-track]:bg-stone-900 [&::-webkit-scrollbar]:w-1.5',
   906→        className
   907→      )}
   908→    >
