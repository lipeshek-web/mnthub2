   372→
   373→function BookingWidget({ mentor }: { mentor: MentorDetailDTO }) {
   374→  const navigate = useAppStore((s) => s.navigate)
   375→  const user = useAppStore((s) => s.user)
   376→
   377→  const days = Array.from({ length: 14 }, (_, i) => addDays(new Date(), i))
   378→  const [selectedDate, setSelectedDate] = useState(() => dateKey(new Date()))
   379→  const [slots, setSlots] = useState<string[]>([])
   380→  const [slotsLoading, setSlotsLoading] = useState(true)
   381→  const [selectedSlot, setSelectedSlot] = useState<string | null>(null)
   382→  const [topic, setTopic] = useState('')
   383→  const [notes, setNotes] = useState('')
   384→  const [confirmOpen, setConfirmOpen] = useState(false)
   385→  const [submitting, setSubmitting] = useState(false)
   386→
   387→  const loadSlots = useCallback(async (date: string) => {
   388→    setSlotsLoading(true)
   389→    setSelectedSlot(null)
   390→    try {
   391→      const { slots } = await api.getSlots(mentor.id, date)
   392→      setSlots(slots)
   393→    } catch {
   394→      setSlots([])
   395→    } finally {
   396→      setSlotsLoading(false)
   397→    }
   398→  }, [mentor.id])
   399→
   400→  useEffect(() => {
   401→    loadSlots(selectedDate)
   402→  }, [loadSlots, selectedDate])
   403→
   404→  const isOwner = user?.id === mentor.userId
   405→
   406→  const submit = async () => {
   407→    if (!user) {
   408→      toast.error('Entre com um usuário no topo da página para agendar.')
   409→      return
   410→    }
   411→    if (!selectedSlot) {
   412→      toast.error('Escolha um horário disponível.')
   413→      return
   414→    }
   415→    if (topic.trim().length < 5) {
   416→      toast.error('Descreva o tema da sessão (mín. 5 caracteres).')
   417→      return
   418→    }
   419→    setSubmitting(true)
   420→    try {
   421→      await api.createBooking({
   422→        menteeId: user.id,
   423→        mentorId: mentor.id,
   424→        startsAt: selectedSlot,
   425→        durationMin: 60,
   426→        topic: topic.trim(),
   427→        notes: notes.trim() || undefined,
   428→      })
   429→      toast.success('Solicitação enviada! O mentor confirmará em breve. 🎉')
   430→      navigate({ name: 'dashboard' })
   431→    } catch (e) {
   432→      toast.error(e instanceof Error ? e.message : 'Erro ao agendar')
   433→      loadSlots(selectedDate)
   434→    } finally {
   435→      setSubmitting(false)
   436→      setConfirmOpen(false)
   437→    }
   438→  }
   439→
   440→  if (isOwner) {
   441→    return (
   442→      <Card className="border-emerald-200">
   443→        <CardContent className="flex flex-col items-center gap-3 p-6 text-center">
   444→          <Avatar name={mentor.name} size="lg" />
   445→          <p className="font-bold">Este é o seu perfil público de mentor</p>
   446→          <p className="text-sm text-muted-foreground">
   447→            Gerencie suas informações, agenda e mural de conteúdos.
   448→          </p>
   449→          <Button className="w-full" onClick={() => navigate({ name: 'onboarding' })}>
   450→            Gerenciar meu perfil
   451→          </Button>
   452→        </CardContent>
   453→      </Card>
   454→    )
   455→  }
   456→
   457→  return (
   458→    <Card className="border-stone-200 shadow-md">
   459→      <CardContent className="space-y-5 p-6">
   460→        <div className="flex items-end justify-between">
   461→          <p className="text-3xl font-black tracking-tight text-stone-900">
   462→            {currencyBRL(mentor.hourlyRate)}
   463→            <span className="text-sm font-medium text-muted-foreground"> / sessão</span>
   464→          </p>
   465→          <Badge variant="secondary" className="bg-stone-100 text-stone-600">60 min · vídeo</Badge>
   466→        </div>
   467→
   468→        {/* Dias */}
   469→        <div>
   470→          <Label className="text-xs uppercase tracking-wide text-muted-foreground">Escolha o dia</Label>
   471→          <div className="mt-2 flex gap-2 overflow-x-auto pb-1.5 [scrollbar-width:thin]">
   472→            {days.map((d) => {
   473→              const key = dateKey(d)
   474→              const active = key === selectedDate
   475→              return (
   476→                <button
   477→                  key={key}
   478→                  onClick={() => setSelectedDate(key)}
   479→                  aria-pressed={active}
   480→                  aria-label={`${WEEKDAYS_PT[d.getDay()]}, ${d.getDate()} de ${MONTHS_PT[d.getMonth()]}`}
   481→                  className={cn(
   482→                    'flex h-[62px] w-[52px] shrink-0 flex-col items-center justify-center rounded-xl border text-center transition-all',
   483→                    active
   484→                      ? 'border-emerald-700 bg-emerald-700 text-white shadow-md'
   485→                      : 'border-stone-200 bg-white text-stone-600 hover:border-emerald-300 hover:bg-emerald-50'
   486→                  )}
   487→                >
   488→                  <span className={cn('text-[10px] font-semibold uppercase', active ? 'text-emerald-200' : 'text-stone-400')}>
   489→                    {WEEKDAYS_PT[d.getDay()]}
   490→                  </span>
   491→                  <span className="text-lg font-extrabold leading-tight">{d.getDate()}</span>
   492→                  <span className={cn('text-[10px]', active ? 'text-emerald-200' : 'text-stone-400')}>
   493→                    {MONTHS_PT[d.getMonth()]}
   494→                  </span>
   495→                </button>
   496→              )
   497→            })}
   498→          </div>
   499→        </div>
   500→
   501→        {/* Horários */}
   502→        <div>
   503→          <Label className="text-xs uppercase tracking-wide text-muted-foreground">
   504→            Horários livres · {formatDayLabel(`${selectedDate}T12:00`)}
   505→          </Label>
   506→          {slotsLoading ? (
   507→            <div className="mt-2 grid grid-cols-4 gap-2">
   508→              {Array.from({ length: 4 }).map((_, i) => (
   509→                <Skeleton key={i} className="h-10 rounded-lg" />
   510→              ))}
   511→            </div>
   512→          ) : slots.length === 0 ? (
   513→            <div className="mt-2 flex items-center gap-2 rounded-xl border border-dashed border-stone-200 bg-stone-50 p-4 text-sm text-stone-500">
   514→              <CalendarOff className="h-4 w-4 shrink-0" />
   515→              Sem horários livres neste dia. Tente outra data.
   516→            </div>
   517→          ) : (
   518→            <div className="mt-2 grid max-h-44 grid-cols-4 gap-2 overflow-y-auto pr-1">
   519→              {slots.map((s) => {
   520→                const active = s === selectedSlot
   521→                return (
   522→                  <button
   523→                    key={s}
   524→                    onClick={() => setSelectedSlot(s)}
   525→                    aria-pressed={active}
   526→                    className={cn(
   527→                      'h-10 rounded-lg border text-sm font-semibold transition-all',
   528→                      active
   529→                        ? 'border-emerald-700 bg-emerald-700 text-white shadow'
   530→                        : 'border-stone-200 bg-white text-stone-700 hover:border-emerald-400 hover:text-emerald-700'
   531→                    )}
   532→                  >
   533→                    {formatTimeLabel(s)}
   534→                  </button>
   535→                )
   536→              })}
   537→            </div>
   538→          )}
   539→        </div>
   540→
   541→        {/* Resumo */}
   542→        {selectedSlot && (
   543→          <div className="rounded-xl border border-emerald-200 bg-emerald-50 p-3.5 text-sm">
   544→            <p className="font-bold text-emerald-900">{formatDayLabelLong(selectedSlot)}</p>
   545→            <p className="mt-0.5 text-emerald-800">
   546→              {formatTimeLabel(selectedSlot)} → {addMinutesToTime(selectedSlot, 60)} · {currencyBRL(mentor.hourlyRate)}
   547→            </p>
   548→          </div>
   549→        )}
   550→
   551→        {/* Tema */}
   552→        <div className="space-y-3">
   553→          <div className="space-y-1.5">
   554→            <Label htmlFor="booking-topic">Tema da sessão *</Label>
   555→            <Input
   556→              id="booking-topic"
   557→              value={topic}
   558→              onChange={(e) => setTopic(e.target.value)}
   559→              placeholder="Ex.: Revisão de portfólio e próximos passos"
   560→            />
   561→          </div>
   562→          <div className="space-y-1.5">
   563→            <Label htmlFor="booking-notes">Observações (opcional)</Label>
   564→            <Textarea
   565→              id="booking-notes"
   566→              value={notes}
   567→              onChange={(e) => setNotes(e.target.value)}
   568→              placeholder="Contexto, links ou materiais para o mentor se preparar..."
   569→              rows={2}
   570→            />
   571→          </div>
   572→        </div>
   573→
   574→        <Button
   575→          className="h-11 w-full text-base font-bold"
   576→          disabled={submitting}
   577→          onClick={() => {
   578→            if (!user) {
   579→              toast.error('Entre com um usuário no topo da página para agendar.')
   580→              return
   581→            }
   582→            setConfirmOpen(true)
   583→          }}
   584→        >
   585→          {submitting ? 'Enviando...' : 'Solicitar agendamento'}
   586→        </Button>
   587→        <p className="-mt-2 text-center text-[11px] leading-relaxed text-muted-foreground">
   588→          A sessão nasce como solicitação e é confirmada pelo mentor. Acompanhe tudo em{' '}
   589→          <strong>Minhas sessões</strong>.
   590→        </p>
   591→      </CardContent>
   592→
   593→      {/* Confirmação */}
   594→      <Dialog open={confirmOpen} onOpenChange={setConfirmOpen}>
   595→        <DialogContent className="sm:max-w-md">
   596→          <DialogHeader>
   597→            <DialogTitle>Confirmar solicitação</DialogTitle>
   598→            <DialogDescription>
   599→              Revise os dados antes de enviar para {mentor.name.split(' ')[0]}.
   600→            </DialogDescription>
   601→          </DialogHeader>
   602→          <div className="space-y-2.5 rounded-xl border border-stone-200 bg-stone-50 p-4 text-sm">
   603→            <div className="flex justify-between gap-4">
   604→              <span className="text-muted-foreground">Mentor</span>
   605→              <span className="font-semibold">{mentor.name}</span>
   606→            </div>
   607→            <div className="flex justify-between gap-4">
   608→              <span className="text-muted-foreground">Quando</span>
   609→              <span className="text-right font-semibold">
   610→                {selectedSlot && formatDayLabelLong(selectedSlot)}
   611→                <br />
   612→                {selectedSlot &&
   613→                  `${formatTimeLabel(selectedSlot)} → ${addMinutesToTime(selectedSlot, 60)}`}
   614→              </span>
   615→            </div>
   616→            <div className="flex justify-between gap-4">
   617→              <span className="text-muted-foreground">Tema</span>
   618→              <span className="max-w-56 text-right font-semibold">{topic || '—'}</span>
   619→            </div>
   620→            <div className="flex justify-between gap-4 border-t border-stone-200 pt-2.5">
   621→              <span className="text-muted-foreground">Valor</span>
   622→              <span className="font-extrabold text-emerald-700">{currencyBRL(mentor.hourlyRate)}</span>
   623→            </div>
   624→          </div>
   625→          <DialogFooter>
   626→            <Button variant="outline" onClick={() => setConfirmOpen(false)}>
   627→              Revisar
   628→            </Button>
   629→            <Button onClick={submit} disabled={submitting}>
   630→              {submitting ? 'Enviando...' : 'Enviar solicitação'}
   631→            </Button>
   632→          </DialogFooter>
   633→        </DialogContent>
   634→      </Dialog>
   635→    </Card>
   636→  )
   637→}
