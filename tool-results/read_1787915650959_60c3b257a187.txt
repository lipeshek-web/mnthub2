   716→   716→/* ==================== Q&A DA AULA ==================== */
   717→   717→
   718→   718→function LessonQuestions({
   719→   719→  lesson,
   720→   720→  isOwner,
   721→   721→  mentorName,
   722→   722→  user,
   723→   723→}: {
   724→   724→  lesson: CourseLessonDTO
   725→   725→  isOwner: boolean
   726→   726→  mentorName: string
   727→   727→  user: { id: string; name: string } | null
   728→   728→}) {
   729→   729→  const [questions, setQuestions] = useState<LessonQuestionDTO[] | null>(null)
   730→   730→  const [body, setBody] = useState('')
   731→   731→  const [sending, setSending] = useState(false)
   732→   732→  const [answerDraft, setAnswerDraft] = useState<Record<string, string>>({})
   733→   733→  const [answeringId, setAnsweringId] = useState<string | null>(null)
   734→   734→
   735→   735→  const load = useCallback(async () => {
   736→   736→    if (!user) {
   737→   737→      setQuestions([])
   738→   738→      return
   739→   739→    }
   740→   740→    try {
   741→   741→      const list = await api.listLessonQuestions(lesson.id, user.id)
   742→   742→      setQuestions(list)
   743→   743→    } catch {
   744→   744→      setQuestions([])
   745→   745→    }
   746→   746→  }, [lesson.id, user])
   747→   747→
   748→   748→  useEffect(() => {
   749→   749→    setQuestions(null)
   750→   750→    void load()
   751→   751→  }, [load])
   752→   752→
   753→   753→  const ask = async () => {
   754→   754→    if (!user) {
   755→   755→      toast.error('Entre com uma conta para perguntar.')
   756→   756→      return
   757→   757→    }
   758→   758→    if (body.trim().length < 5) {
   759→   759→      toast.error('Escreva sua pergunta (mín. 5 caracteres).')
   760→   760→      return
   761→   761→    }
   762→   762→    setSending(true)
   763→   763→    try {
   764→   764→      const created = await api.askLessonQuestion(lesson.id, { userId: user.id, body: body.trim() })
   765→   765→      setQuestions((prev) => [created, ...(prev ?? [])])
   766→   766→      setBody('')
   767→   767→      toast.success('Pergunta enviada! O mentor será notificado.')
   768→   768→    } catch (err) {
   769→   769→      toast.error(err instanceof Error ? err.message : 'Não foi possível enviar a pergunta.')
   770→   770→    } finally {
   771→   771→      setSending(false)
   772→   772→    }
   773→   773→  }
   774→   774→
   775→   775→  const answer = async (questionId: string) => {
   776→   776→    const text = (answerDraft[questionId] ?? '').trim()
   777→   777→    if (!user || text.length < 2) {
   778→   778→      toast.error('Escreva a resposta antes de enviar.')
   779→   779→      return
   780→   780→    }
   781→   781→    setAnsweringId(questionId)
   782→   782→    try {
   783→   783→      const res = await api.answerQuestion(questionId, { userId: user.id, answer: text })
   784→   784→      setQuestions((prev) =>
   785→   785→        (prev ?? []).map((q) =>
   786→   786→          q.id === questionId ? { ...q, answer: res.answer, answeredAt: res.answeredAt } : q
   787→   787→        )
   788→   788→      )
   789→   789→      setAnswerDraft((prev) => ({ ...prev, [questionId]: '' }))
   790→   790→      toast.success('Resposta publicada!')
   791→   791→    } catch (err) {
   792→   792→      toast.error(err instanceof Error ? err.message : 'Não foi possível responder.')
   793→   793→    } finally {
   794→   794→      setAnsweringId(null)
   795→   795→    }
   796→   796→  }
   797→   797→
   798→   798→  const remove = async (questionId: string) => {
   799→   799→    if (!user) return
   800→   800→    try {
   801→   801→      await api.deleteQuestion(questionId, user.id)
   802→   802→      setQuestions((prev) => (prev ?? []).filter((q) => q.id !== questionId))
   803→   803→      toast('Pergunta removida.')
   804→   804→    } catch (err) {
   805→   805→      toast.error(err instanceof Error ? err.message : 'Não foi possível remover.')
   806→   806→    }
   807→   807→  }
   808→   808→
   809→   809→  if (!user) {
   810→   810→    return (
   811→   811→      <div className="rounded-2xl border border-stone-200 bg-white p-6 text-center text-sm text-stone-500">
   812→   812→        Entre com uma conta no topo da página para participar das perguntas.
   813→   813→      </div>
   814→   814→    )
   815→   815→  }
   816→   816→
   817→   817→  return (
   818→   818→    <div className="space-y-4">
   819→   819→      {/* Formulário de pergunta (aluno) */}
   820→   820→      {!isOwner ? (
   821→   821→        <div className="rounded-2xl border border-stone-200 bg-white p-4 sm:p-5">
   822→   822→          <label htmlFor="question-body" className="text-sm font-bold text-stone-800">
   823→   823→            Tirou uma dúvida? Pergunte ao mentor
   824→   824→          </label>
   825→   825→          <Textarea
   826→   826→            id="question-body"
   827→   827→            rows={3}
   828→   828→            value={body}
   829→   829→            onChange={(e) => setBody(e.target.value)}
   830→   830→            placeholder={`Ex.: "${mentorName.split(' ')[0]}", no minuto 4:30 você menciona X — como aplico isso na prática?`}
   831→   831→            className="mt-2 resize-none"
   832→   832→            maxLength={1200}
   833→   833→          />
   834→   834→          <div className="mt-3 flex items-center justify-between gap-3">
   835→   835→            <span className="text-xs text-stone-400">{body.length}/1200</span>
   836→   836→            <Button
   837→   837→              className="h-10 rounded-full font-semibold"
   838→   838→              onClick={() => void ask()}
   839→   839→              disabled={sending || body.trim().length < 5}
   840→   840→            >
   841→   841→              <Send aria-hidden className="h-4 w-4" />
   842→   842→              {sending ? 'Enviando…' : 'Enviar pergunta'}
   843→   843→            </Button>
   844→   844→          </div>
   845→   845→        </div>
   846→   846→      ) : null}
   847→   847→
   848→   848→      {/* Lista */}
   849→   849→      {questions === null ? (
   850→   850→        <div className="space-y-3">
   851→   851→          <Skeleton className="h-20 rounded-2xl" />

   937→   937→
   938→   938→function LessonNotes({
   939→   939→  lessonId,
   940→   940→  user,
   941→   941→  lessonTitle,
   942→   942→}: {
   943→   943→  lessonId: string
   944→   944→  user: { id: string; name: string } | null
   945→   945→  lessonTitle: string
   946→   946→}) {
   947→   947→  const [body, setBody] = useState('')
   948→   948→  const [loaded, setLoaded] = useState(false)
   949→   949→  const [saveState, setSaveState] = useState<'idle' | 'saving' | 'saved'>('idle')
   950→   950→  const [savedAt, setSavedAt] = useState<string | null>(null)
   951→   951→  const timerRef = useRef<ReturnType<typeof setTimeout> | null>(null)
   952→   952→  const latestRef = useRef('')
   953→   953→
   954→   954→  // Carrega as anotações da aula (o componente é remontado por aula via key)
   955→   955→  useEffect(() => {
   956→   956→    if (!user) return
   957→   957→    let active = true
   958→   958→    api
   959→   959→      .getLessonNote(lessonId, user.id)
   960→   960→      .then((note) => {
   961→   961→        if (!active) return
   962→   962→        setBody(note.body)
   963→   963→        latestRef.current = note.body
   964→   964→        setSavedAt(note.updatedAt)
   965→   965→      })
   966→   966→      .catch(() => {
   967→   967→        /* sem anotações ainda */
   968→   968→      })
   969→   969→      .finally(() => {
   970→   970→        if (active) setLoaded(true)
   971→   971→      })
   972→   972→    return () => {
   973→   973→      active = false
   974→   974→    }
   975→   975→  }, [lessonId, user])
   976→   976→
   977→   977→  // Autosave com debounce (o estado "salvando" aparece no envio)
   978→   978→  useEffect(() => {
   979→   979→    if (!loaded || !user) return
   980→   980→    if (body === latestRef.current) return
   981→   981→    if (timerRef.current) clearTimeout(timerRef.current)
   982→   982→    timerRef.current = setTimeout(async () => {
   983→   983→      setSaveState('saving')
   984→   984→      try {
   985→   985→        const res = await api.saveLessonNote(lessonId, { userId: user.id, body })
   986→   986→        latestRef.current = body
   987→   987→        setSavedAt(res.updatedAt)
   988→   988→        setSaveState('saved')
   989→   989→      } catch {
   990→   990→        setSaveState('idle')
   991→   991→        toast.error('Não foi possível salvar as anotações.')
   992→   992→      }
   993→   993→    }, 900)
   994→   994→    return () => {
   995→   995→      if (timerRef.current) clearTimeout(timerRef.current)
   996→   996→    }
   997→   997→  }, [body, loaded, user, lessonId])
   998→   998→
   999→   999→  const downloadNotes = () => {
  1000→  1000→    const text = body.trim()
  1001→  1001→    if (!text) {
  1002→  1002→      toast.error('Escreva algo antes de baixar suas anotações.')
  1003→  1003→      return
  1004→  1004→    }
  1005→  1005→    const blob = new Blob(
  1006→  1006→      [`Anotações — ${lessonTitle}\nMentorHub · ${new Date().toLocaleString('pt-BR')}\n\n${text}\n`],
  1007→  1007→      { type: 'text/plain;charset=utf-8' }
  1008→  1008→    )
  1009→  1009→    const url = URL.createObjectURL(blob)
  1010→  1010→    const a = document.createElement('a')
  1011→  1011→    a.href = url
  1012→  1012→    a.download = `anotacoes-${lessonTitle.toLowerCase().replace(/[^a-z0-9]+/gi, '-').slice(0, 40)}.txt`
  1013→  1013→    a.click()
  1014→  1014→    URL.revokeObjectURL(url)
  1015→  1015→    toast.success('Anotações baixadas!')
  1016→  1016→  }
  1017→  1017→
  1018→  1018→  if (!user) {
  1019→  1019→    return (
  1020→  1020→      <div className="rounded-2xl border border-stone-200 bg-white p-6 text-center text-sm text-stone-500">
  1021→  1021→        Entre com uma conta para escrever e guardar suas anotações.
  1022→  1022→      </div>
  1023→  1023→    )
  1024→  1024→  }
  1025→  1025→
  1026→  1026→  return (
  1027→  1027→    <div className="rounded-2xl border border-stone-200 bg-white p-4 sm:p-5">
  1028→  1028→      <div className="flex items-center justify-between gap-3">
  1029→  1029→        <label htmlFor="lesson-notes" className="text-sm font-bold text-stone-800">
  1030→  1030→          Minhas anotações desta aula
  1031→  1031→        </label>
  1032→  1032→        <span
  1033→  1033→          className={cn(
  1034→  1034→            'inline-flex items-center gap-1.5 text-xs font-semibold',
  1035→  1035→            saveState === 'saving' ? 'text-amber-600' : saveState === 'saved' ? 'text-emerald-600' : 'text-stone-400'
  1036→  1036→          )}
  1037→  1037→          aria-live="polite"
  1038→  1038→        >
  1039→  1039→          {saveState === 'saving' ? (
  1040→  1040→            'Salvando…'
  1041→  1041→          ) : saveState === 'saved' ? (
  1042→  1042→            <>
  1043→  1043→              <Check aria-hidden className="h-3.5 w-3.5" /> Salvo
  1044→  1044→            </>
  1045→  1045→          ) : user ? (
  1046→  1046→            'Salva automaticamente'
  1047→  1047→          ) : (
  1048→  1048→            'Entre para salvar'
  1049→  1049→          )}
  1050→  1050→        </span>
  1051→  1051→      </div>
  1052→  1052→      <Textarea
  1053→  1053→        id="lesson-notes"
  1054→  1054→        rows={8}
  1055→  1055→        value={body}
  1056→  1056→        onChange={(e) => setBody(e.target.value)}
  1057→  1057→        placeholder="Anote insights, referências e tarefas desta aula… (salvo automaticamente na sua conta)"
  1058→  1058→        className="mt-2 resize-y"
  1059→  1059→      />
  1060→  1060→      <div className="mt-3 flex items-center justify-between gap-3">
  1061→  1061→        <span className="text-xs text-stone-400">
  1062→  1062→          {savedAt ? `Última atualização: ${new Date(savedAt).toLocaleString('pt-BR')}` : 'Nenhuma edição ainda'}
  1063→  1063→        </span>
  1064→  1064→        <Button
  1065→  1065→          variant="outline"
  1066→  1066→          className="h-10 rounded-full font-semibold"
  1067→  1067→          onClick={downloadNotes}
  1068→  1068→          disabled={!body.trim()}
  1069→  1069→        >
  1070→  1070→          <FileDown aria-hidden className="h-4 w-4" /> Baixar .txt
  1071→  1071→        </Button>
  1072→  1072→      </div>
  1073→  1073→    </div>
  1074→  1074→  )
  1075→  1075→}
