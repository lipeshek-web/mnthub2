   450→function BookingWidget({ mentor }: { mentor: MentorDetailDTO }) {
   451→  const navigate = useAppStore((s) => s.navigate)
   452→  const user = useAppStore((s) => s.user)
   453→
   454→  const days = Array.from({ length: 14 }, (_, i) => addDays(new Date(), i))
   455→  const [selectedDate, setSelectedDate] = useState(() => dateKey(new Date()))
   456→  const [slots, setSlots] = useState<string[]>([])
   457→  const [slotsLoading, setSlotsLoading] = useState(true)
   458→  const [selectedSlot, setSelectedSlot] = useState<string | null>(null)
   459→  const [topic, setTopic] = useState('')
   460→  const [notes, setNotes] = useState('')
   461→  const [confirmOpen, setConfirmOpen] = useState(false)
   462→  const [submitting, setSubmitting] = useState(false)
   463→
   464→  const loadSlots = useCallback(async (date: string) => {
   465→    setSlotsLoading(true)
   466→    setSelectedSlot(null)
   467→    try {
   468→      const { slots } = await api.getSlots(mentor.id, date)
   469→      setSlots(slots)
   470→    } catch {
   471→      setSlots([])
   472→    } finally {
   473→      setSlotsLoading(false)
   474→    }
   475→  }, [mentor.id])
   476→
   477→  useEffect(() => {
   478→    loadSlots(selectedDate)
   479→  }, [loadSlots, selectedDate])
   480→
   481→  const isOwner = user?.id === mentor.userId
   482→
   483→  const submit = async () => {
   484→    if (!user) {
   485→      toast.error('Entre com um usuário no topo da página para agendar.')
   486→      return
   487→    }
   488→    if (!selectedSlot) {
   489→      toast.error('Escolha um horário disponível.')
   490→      return
   491→    }
   492→    if (topic.trim().length < 5) {
   493→      toast.error('Descreva o tema da sessão (mín. 5 caracteres).')
   494→      return
   495→    }
   496→    setSubmitting(true)
   497→    try {
   498→      await api.createBooking({
   499→        menteeId: user.id,
   500→        mentorId: mentor.id,
   501→        startsAt: selectedSlot,
   502→        durationMin: 60,
   503→        topic: topic.trim(),
   504→        notes: notes.trim() || undefined,
   505→      })
   506→      toast.success('Solicitação enviada! O mentor confirmará em breve. 🎉')
   507→      navigate({ name: 'dashboard' })
   508→    } catch (e) {
   509→      toast.error(e instanceof Error ? e.message : 'Erro ao agendar')
   510→      loadSlots(selectedDate)
   511→    } finally {
   512→      setSubmitting(false)
   513→      setConfirmOpen(false)
   514→    }
   515→  }
   516→
   517→  if (isOwner) {
   518→    return (
   519→      <Card className="border-emerald-200">
   520→        <CardContent className="flex flex-col items-center gap-3 p-6 text-center">
   521→          <Avatar name={mentor.name} size="lg" />
   522→          <p className="font-bold">Este é o seu perfil público de mentor</p>
   523→          <p className="text-sm text-muted-foreground">
   524→            Gerencie suas informações, agenda e mural de conteúdos.
   525→          </p>
   526→          <Button className="w-full" onClick={() => navigate({ name: 'onboarding' })}>
   527→            Gerenciar meu perfil
   528→          </Button>
   529→        </CardContent>
   530→      </Card>
   531→    )
   532→  }
   533→
   534→  return (
   535→    <Card className="border-stone-200 shadow-md">
   536→      <CardContent className="space-y-5 p-6">
   537→        <div className="flex items-end justify-between">
   538→          <p className="text-3xl font-black tracking-tight text-stone-900">
   539→            {currencyBRL(mentor.hourlyRate)}
   540→            <span className="text-sm font-medium text-muted-foreground"> / sessão</span>
   541→          </p>
   542→          <Badge variant="secondary" className="bg-stone-100 text-stone-600">60 min · vídeo</Badge>
   543→        </div>
   544→
   545→        {/* Dias */}
   546→        <div>
   547→          <Label className="text-xs uppercase tracking-wide text-muted-foreground">Escolha o dia</Label>
   548→          <div className="mt-2 flex gap-2 overflow-x-auto pb-1.5 [scrollbar-width:thin]">
   549→            {days.map((d) => {
   550→              const key = dateKey(d)
   551→              const active = key === selectedDate
   552→              return (
   553→                <button
   554→                  key={key}
   555→                  onClick={() => setSelectedDate(key)}
   556→                  aria-pressed={active}
   557→                  aria-label={`${WEEKDAYS_PT[d.getDay()]}, ${d.getDate()} de ${MONTHS_PT[d.getMonth()]}`}
   558→                  className={cn(
   559→                    'flex h-[62px] w-[52px] shrink-0 flex-col items-center justify-center rounded-xl border text-center transition-all',
   560→                    active
   561→                      ? 'border-emerald-700 bg-emerald-700 text-white shadow-md'
   562→                      : 'border-stone-200 bg-white text-stone-600 hover:border-emerald-300 hover:bg-emerald-50'
   563→                  )}
   564→                >
   565→                  <span className={cn('text-[10px] font-semibold uppercase', active ? 'text-emerald-200' : 'text-stone-400')}>
   566→                    {WEEKDAYS_PT[d.getDay()]}
   567→                  </span>
   568→                  <span className="text-lg font-extrabold leading-tight">{d.getDate()}</span>
   569→                  <span className={cn('text-[10px]', active ? 'text-emerald-200' : 'text-stone-400')}>
   570→                    {MONTHS_PT[d.getMonth()]}
   571→                  </span>
   572→                </button>
   573→              )
   574→            })}
   575→          </div>
   576→        </div>
   577→
   578→        {/* Horários */}
   579→        <div>
   580→          <Label className="text-xs uppercase tracking-wide text-muted-foreground">
   581→            Horários livres · {formatDayLabel(`${selectedDate}T12:00`)}
   582→          </Label>
   583→          {slotsLoading ? (
   584→            <div className="mt-2 grid grid-cols-4 gap-2">
   585→              {Array.from({ length: 4 }).map((_, i) => (
   586→                <Skeleton key={i} className="h-10 rounded-lg" />
   587→              ))}
   588→            </div>
   589→          ) : slots.length === 0 ? (
   590→            <div className="mt-2 flex items-center gap-2 rounded-xl border border-dashed border-stone-200 bg-stone-50 p-4 text-sm text-stone-500">
   591→              <CalendarOff className="h-4 w-4 shrink-0" />
   592→              Sem horários livres neste dia. Tente outra data.
   593→            </div>
   594→          ) : (
   595→            <div className="mt-2 grid max-h-44 grid-cols-4 gap-2 overflow-y-auto pr-1">
   596→              {slots.map((s) => {
   597→                const active = s === selectedSlot
   598→                return (
   599→                  <button
   600→                    key={s}
   601→                    onClick={() => setSelectedSlot(s)}
   602→                    aria-pressed={active}
   603→                    className={cn(
   604→                      'h-10 rounded-lg border text-sm font-semibold transition-all',
   605→                      active
   606→                        ? 'border-emerald-700 bg-emerald-700 text-white shadow'
   607→                        : 'border-stone-200 bg-white text-stone-700 hover:border-emerald-400 hover:text-emerald-700'
   608→                    )}
   609→                  >
   610→                    {formatTimeLabel(s)}
   611→                  </button>
   612→                )
   613→              })}
   614→            </div>
   615→          )}
   616→        </div>
   617→
   618→        {/* Resumo */}
   619→        {selectedSlot && (
   620→          <div className="rounded-xl border border-emerald-200 bg-emerald-50 p-3.5 text-sm">
   621→            <p className="font-bold text-emerald-900">{formatDayLabelLong(selectedSlot)}</p>
   622→            <p className="mt-0.5 text-emerald-800">
   623→              {formatTimeLabel(selectedSlot)} → {addMinutesToTime(selectedSlot, 60)} · {currencyBRL(mentor.hourlyRate)}
   624→            </p>
   625→          </div>
   626→        )}
   627→
   628→        {/* Tema */}
   629→        <div className="space-y-3">
   630→          <div className="space-y-1.5">
   631→            <Label htmlFor="booking-topic">Tema da sessão *</Label>
   632→            <Input
   633→              id="booking-topic"
   634→              value={topic}
   635→              onChange={(e) => setTopic(e.target.value)}
   636→              placeholder="Ex.: Revisão de portfólio e próximos passos"
   637→            />
   638→          </div>
   639→          <div className="space-y-1.5">
   640→            <Label htmlFor="booking-notes">Observações (opcional)</Label>
   641→            <Textarea
   642→              id="booking-notes"
   643→              value={notes}
   644→              onChange={(e) => setNotes(e.target.value)}
   645→              placeholder="Contexto, links ou materiais para o mentor se preparar..."
   646→              rows={2}
   647→            />
   648→          </div>
   649→        </div>
   650→
   651→        <Button
   652→          className="h-11 w-full text-base font-bold"
   653→          disabled={submitting}
   654→          onClick={() => {
   655→            if (!user) {
   656→              toast.error('Entre com um usuário no topo da página para agendar.')
   657→              return
   658→            }
   659→            setConfirmOpen(true)
   660→          }}
   661→        >
   662→          {submitting ? 'Enviando...' : 'Solicitar agendamento'}
   663→        </Button>
   664→        <p className="-mt-2 text-center text-[11px] leading-relaxed text-muted-foreground">
   665→          A sessão nasce como solicitação e é confirmada pelo mentor. Acompanhe tudo em{' '}
   666→          <strong>Minhas sessões</strong>.
   667→        </p>
   668→      </CardContent>
   669→
   670→      {/* Confirmação */}
   671→      <Dialog open={confirmOpen} onOpenChange={setConfirmOpen}>
   672→        <DialogContent className="sm:max-w-md">
   673→          <DialogHeader>
   674→            <DialogTitle>Confirmar solicitação</DialogTitle>
   675→            <DialogDescription>
   676→              Revise os dados antes de enviar para {mentor.name.split(' ')[0]}.
   677→            </DialogDescription>
   678→          </DialogHeader>
   679→          <div className="space-y-2.5 rounded-xl border border-stone-200 bg-stone-50 p-4 text-sm">
   680→            <div className="flex justify-between gap-4">
   681→              <span className="text-muted-foreground">Mentor</span>
   682→              <span className="font-semibold">{mentor.name}</span>
   683→            </div>
   684→            <div className="flex justify-between gap-4">
   685→              <span className="text-muted-foreground">Quando</span>
   686→              <span className="text-right font-semibold">
   687→                {selectedSlot && formatDayLabelLong(selectedSlot)}
   688→                <br />
   689→                {selectedSlot &&
   690→                  `${formatTimeLabel(selectedSlot)} → ${addMinutesToTime(selectedSlot, 60)}`}
   691→              </span>
   692→            </div>
   693→            <div className="flex justify-between gap-4">
   694→              <span className="text-muted-foreground">Tema</span>
   695→              <span className="max-w-56 text-right font-semibold">{topic || '—'}</span>
   696→            </div>
   697→            <div className="flex justify-between gap-4 border-t border-stone-200 pt-2.5">
   698→              <span className="text-muted-foreground">Valor</span>
   699→              <span className="font-extrabold text-emerald-700">{currencyBRL(mentor.hourlyRate)}</span>
   700→            </div>
   701→          </div>
   702→          <DialogFooter>
   703→            <Button variant="outline" onClick={() => setConfirmOpen(false)}>
   704→              Revisar
   705→            </Button>
   706→            <Button onClick={submit} disabled={submitting}>
   707→              {submitting ? 'Enviando...' : 'Enviar solicitação'}
   708→            </Button>
   709→          </DialogFooter>
   710→        </DialogContent>
   711→      </Dialog>
   712→    </Card>
   713→  )
   714→}
