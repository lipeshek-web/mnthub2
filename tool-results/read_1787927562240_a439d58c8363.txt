    88→    88→export function ClassroomView({ courseId }: { courseId: string }) {

    90→    90→  const user = useAppStore((s) => s.user)
    91→    91→
    92→    92→  const [data, setData] = useState<CourseDetailDTO | null>(null)
    93→    93→  const [loading, setLoading] = useState(true)
    94→    94→  const [error, setError] = useState<string | null>(null)
    95→    95→
    96→    96→  const [currentLessonId, setCurrentLessonId] = useState<string | null>(null)
    97→    97→  const [completedIds, setCompletedIds] = useState<string[]>([])
    98→    98→  const [toggling, setToggling] = useState(false)
    99→    99→
   100→   100→  const fetchCourse = useCallback(async () => {
   101→   101→    setLoading(true)
   102→   102→    setError(null)
   103→   103→    try {
   104→   104→      const d = await api.getCourse(courseId, user?.id)
   105→   105→      setData(d)
   106→   106→    } catch (err) {
   107→   107→      setError(err instanceof Error ? err.message : 'Não foi possível carregar o curso.')
   108→   108→    } finally {
   109→   109→      setLoading(false)
   110→   110→    }
   111→   111→  }, [courseId, user?.id])
   112→   112→
   113→   113→  useEffect(() => {
   114→   114→    void fetchCourse()
   115→   115→  }, [fetchCourse])
   116→   116→
   117→   117→  const course = data
   118→   118→  const isOwner = Boolean(user && course && course.mentor.userId === user.id)
   119→   119→  const isEnrolled = course?.enrollment != null
   120→   120→  const hasAccess = isEnrolled || isOwner
   121→   121→
   122→   122→  const lessons = useMemo<CourseLessonDTO[]>(
   123→   123→    () => [...(course?.lessons ?? [])].sort((a, b) => a.order - b.order),
   124→   124→    [course]
   125→   125→  )
   126→   126→
   127→   127→  // Temas (módulos) em ordem; aulas sem tema (ou com tema removido) vão para o

   130→   130→    () => [...(course?.themes ?? [])].sort((a, b) => a.order - b.order),
   131→   131→    [course]
   132→   132→  )
   133→   133→
   134→   134→  const lessonGroups = useMemo<LessonGroup[]>(() => {
   135→   135→    const byTheme = new Map<string, CourseLessonDTO[]>()
   136→   136→    const loose: CourseLessonDTO[] = []
   137→   137→    for (const lesson of lessons) {
   138→   138→      if (lesson.themeId && themeList.some((t) => t.id === lesson.themeId)) {
   139→   139→        const bucket = byTheme.get(lesson.themeId) ?? []
   140→   140→        bucket.push(lesson)
   141→   141→        byTheme.set(lesson.themeId, bucket)
   142→   142→      } else {
   143→   143→        loose.push(lesson)
   144→   144→      }
   145→   145→    }
   146→   146→    const groups: LessonGroup[] = themeList.map((theme) => ({
   147→   147→      key: theme.id,
   148→   148→      title: theme.title,
   149→   149→      theme,
   150→   150→      lessons: byTheme.get(theme.id) ?? [],
   151→   151→    }))
   152→   152→    if (loose.length > 0) {
   153→   153→      groups.push({ key: 'none', title: 'Outros conteúdos', theme: null, lessons: loose })
   154→   154→    }
   155→   155→    return groups
   156→   156→  }, [lessons, themeList])
   157→   157→
   158→   158→  // Lista achatada (temas em ordem → aulas em ordem): base do índice global,
   159→   159→  // do avançar/voltar e do progresso geral.
   160→   160→  const orderedLessons = useMemo(() => lessonGroups.flatMap((g) => g.lessons), [lessonGroups])
   161→   161→
   162→   162→  // Seções recolhidas da sidebar (default: todas expandidas)
   163→   163→  const [collapsedThemes, setCollapsedThemes] = useState<Record<string, boolean>>({})
   164→   164→  const [certOpen, setCertOpen] = useState(false)
   165→   165→
   166→   166→  // Aula inicial: primeira ainda não concluída (só define uma vez por curso)
   167→   167→  const initializedRef = useRef<string | null>(null)
   168→   168→  useEffect(() => {
   169→   169→    if (!course || !hasAccess) return
   170→   170→    if (initializedRef.current === course.id) return
   171→   171→    initializedRef.current = course.id
   172→   172→    const enrollment = course.enrollment
   173→   173→    const completed = enrollment?.completedLessonIds ?? []
   174→   174→    setCompletedIds(completed)
   175→   175→    const firstIncomplete = orderedLessons.find((l) => !completed.includes(l.id))

   177→   177→  }, [course, hasAccess, orderedLessons])
   178→   178→
   179→   179→  const currentLesson = useMemo(
   180→   180→    () => orderedLessons.find((l) => l.id === currentLessonId) ?? orderedLessons[0] ?? null,

   184→   184→  const percent =
   185→   185→    orderedLessons.length > 0 ? Math.round((completedIds.length / orderedLessons.length) * 100) : 0
   186→   186→  const courseCompleted =
   187→   187→    orderedLessons.length > 0 && completedIds.length >= orderedLessons.length
   188→   188→
   189→   189→  const selectLesson = (lessonId: string) => {
   190→   190→    setCurrentLessonId(lessonId)
   191→   191→    // Ao selecionar uma aula, garante o tema dela expandido
   192→   192→    const lessonThemeId = orderedLessons.find((l) => l.id === lessonId)?.themeId
   193→   193→    if (lessonThemeId) setCollapsedThemes((prev) => ({ ...prev, [lessonThemeId]: false }))
   194→   194→  }
   195→   195→
   196→   196→  const toggleComplete = async (autoAdvance = false) => {
   197→   197→    if (!user || !course || !currentLesson) return
   198→   198→    setToggling(true)
   199→   199→    try {
   200→   200→      const res = await api.toggleLessonComplete(course.id, {
   201→   201→        userId: user.id,
   202→   202→        lessonId: currentLesson.id,
   203→   203→      })
   204→   204→      const nowCompleted = res.completedLessonIds.includes(currentLesson.id)
   205→   205→      setCompletedIds(res.completedLessonIds)
   206→   206→      if (nowCompleted && res.completedLessonIds.length === orderedLessons.length) {
   207→   207→        toast.success('Curso concluído! Parabéns 🎉')
   208→   208→      } else if (nowCompleted) {
   209→   209→        toast.success('Aula marcada como concluída.')
   210→   210→      } else {
   211→   211→        toast('Marcação removida desta aula.')
   212→   212→      }
   213→   213→      if (autoAdvance && nowCompleted && currentIndex < orderedLessons.length - 1) {
   214→   214→        setCurrentLessonId(orderedLessons[currentIndex + 1].id)
   215→   215→      }
   216→   216→    } catch (err) {
   217→   217→      toast.error(err instanceof Error ? err.message : 'Não foi possível atualizar o progresso.')
   218→   218→    } finally {
   219→   219→      setToggling(false)
   220→   220→    }
   221→   221→  }
   222→   222→
   223→   223→  const goPrev = () => {
   224→   224→    if (currentLesson && currentIndex > 0) setCurrentLessonId(orderedLessons[currentIndex - 1].id)
   225→   225→  }
   226→   226→  const goNext = () => {
   227→   227→    if (currentLesson && currentIndex < orderedLessons.length - 1) {
   228→   228→      setCurrentLessonId(orderedLessons[currentIndex + 1].id)
   229→   229→    }
   230→   230→  }
   231→   231→
   232→   232→  /* ---------- Estados de tela ---------- */
   233→   233→
   234→   234→  if (loading) {
   235→   235→    return (
   236→   236→      <div className="flex h-full min-h-0 flex-col" aria-busy="true">
   237→   237→        <div className="flex items-center gap-3 border-b border-stone-200 bg-white px-4 py-3 sm:px-6">
   238→   238→          <Skeleton className="h-9 w-9 rounded-full" />
   239→   239→          <Skeleton className="h-5 w-48" />
   240→   240→          <Skeleton className="ml-auto h-6 w-28 rounded-full" />
   241→   241→        </div>
   242→   242→        <div className="grid flex-1 grid-cols-1 lg:grid-cols-[1fr_380px]">
   243→   243→          <div className="space-y-4 p-4 sm:p-6">
   244→   244→            <Skeleton className="aspect-video w-full rounded-2xl" />
   245→   245→            <Skeleton className="h-7 w-2/3" />
   246→   246→            <Skeleton className="h-24 w-full rounded-2xl" />
   247→   247→          </div>
   248→   248→          <div className="hidden space-y-2 border-l border-stone-200 bg-white p-4 lg:block">
   249→   249→            {Array.from({ length: 6 }).map((_, i) => (
   250→   250→              <Skeleton key={i} className="h-14 rounded-xl" />
   251→   251→            ))}
   252→   252→          </div>
   253→   253→        </div>
   254→   254→        <p className="sr-only">Abrindo sala de aula…</p>
   255→   255→      </div>
   256→   256→    )
   257→   257→  }
   258→   258→
   259→   259→  if (!course) {
   260→   260→    return (
   261→   261→      <div className="flex h-full items-center justify-center p-6">
   262→   262→        <div className="flex w-full max-w-md flex-col items-center gap-3 rounded-2xl border border-dashed border-stone-300 px-6 py-14 text-center">
   263→   263→          <span className="flex h-14 w-14 items-center justify-center rounded-full bg-rose-50">
   264→   264→            <AlertCircle aria-hidden className="h-7 w-7 text-rose-500" />
   265→   265→          </span>
   266→   266→          <p className="font-bold text-stone-900">Não foi possível abrir a sala de aula</p>
   267→   267→          <p className="max-w-sm text-sm leading-relaxed text-stone-500">
   268→   268→            {error ?? 'Curso não encontrado.'}
   269→   269→          </p>
   270→   270→          <Button variant="outline" className="rounded-full" onClick={() => void fetchCourse()}>
   271→   271→            Tentar novamente
   272→   272→          </Button>
   273→   273→        </div>
   274→   274→      </div>
   275→   275→    )
   276→   276→  }
   277→   277→
   278→   278→  if (!hasAccess) {
   279→   279→    return (
   280→   280→      <div className="flex h-full items-center justify-center p-6">
   281→   281→        <Card className="w-full max-w-md rounded-2xl border-stone-200 p-8 text-center shadow-none">
   282→   282→          <span className="mx-auto flex h-14 w-14 items-center justify-center rounded-full bg-amber-50">
   283→   283→            <Lock aria-hidden className="h-6 w-6 text-amber-600" />
   284→   284→          </span>
   285→   285→          <h1 className="mt-4 text-lg font-extrabold tracking-tight text-stone-900">
   286→   286→            {course.title}
   287→   287→          </h1>
   288→   288→          <p className="mt-2 text-sm leading-relaxed text-stone-500">
   289→   289→            Você ainda não tem acesso a este curso. Inscreva-se para desbloquear todas as aulas,
   290→   290→            perguntas e anotações.
   291→   291→          </p>
   292→   292→          <Button
   293→   293→            className="mt-5 h-11 w-full rounded-full font-bold"
   294→   294→            onClick={() => navigate({ name: 'course', courseId })}
   295→   295→          >
   296→   296→            Ver o curso e se inscrever
   297→   297→          </Button>
   298→   298→        </Card>
   299→   299→      </div>
   300→   300→    )
   301→   301→  }
   302→   302→
   303→   303→  const firstName = course.mentor.name.split(' ')[0]
   304→   304→
   305→   305→  return (
   306→   306→    <div className="flex h-full min-h-0 flex-col bg-stone-50" aria-label={`Sala de aula: ${course.title}`}>
   307→   307→      {/* CSS de impressão: imprime apenas o certificado quando aberto */}
   308→   308→      <style>{`@media print { body * { visibility: hidden; } .certificate-print, .certificate-print * { visibility: visible; } .certificate-print { position: absolute; inset: 0; } }`}</style>
   309→   309→      {/* ---------- BARRA SUPERIOR COMPACTA ---------- */}
   310→   310→      <div className="flex shrink-0 items-center gap-2 border-b border-stone-200 bg-white px-3 py-2.5 sm:gap-3 sm:px-6">
   311→   311→        <Button
   312→   312→          variant="ghost"
   313→   313→          size="icon"
   314→   314→          className="h-9 w-9 shrink-0 rounded-full"
   315→   315→          aria-label="Voltar para a visão geral do curso"
   316→   316→          onClick={() => navigate({ name: 'course', courseId })}
   317→   317→        >
   318→   318→          <ArrowLeft aria-hidden className="h-4 w-4" />
   319→   319→        </Button>
   320→   320→        <div className="min-w-0 flex-1">
   321→   321→          <p className="truncate text-sm font-extrabold tracking-tight text-stone-900 sm:text-base">
   322→   322→            {course.title}
   323→   323→          </p>
   324→   324→          <p className="hidden text-xs text-stone-400 sm:block">
   325→   325→            {orderedLessons.length} {orderedLessons.length === 1 ? 'aula' : 'aulas'} ·{' '}
   326→   326→            {formatTotalDuration(course.totalDurationMin)}
   327→   327→          </p>
   328→   328→        </div>
   329→   329→        <div className="hidden items-center gap-2 sm:flex" aria-hidden>
   330→   330→          <Avatar
   331→   331→            name={course.mentor.name}
   332→   332→            src={course.mentor.avatarUrl}
   333→   333→            size="sm"
   334→   334→            className="h-6 w-6 text-[10px]"
   335→   335→          />
   336→   336→          <span className="max-w-36 truncate text-xs font-semibold text-stone-600">
   337→   337→            {course.mentor.name}
   338→   338→          </span>
   339→   339→        </div>
   340→   340→        <div className="flex w-32 shrink-0 items-center gap-2 sm:w-44">
   341→   341→          <Progress
   342→   342→            value={percent}
   343→   343→            aria-label="Progresso do curso"
   344→   344→            className="h-2 [&_[data-slot=progress-indicator]]:bg-emerald-600"
   345→   345→          />
   346→   346→          <span className="w-9 shrink-0 text-right text-xs font-bold tabular-nums text-stone-600">
   347→   347→            {percent}%
   348→   348→          </span>
   349→   349→        </div>
   350→   350→      </div>
   351→   351→
   352→   352→      {/* ---------- CORPO: PLAYER + CONTEÚDOS ---------- */}
   353→   353→      <div className="grid min-h-0 flex-1 grid-cols-1 lg:grid-cols-[1fr_380px]">
   354→   354→        {/* Coluna esquerda: aula */}
   355→   355→        <div className="min-h-0 overflow-y-auto p-4 sm:p-6 lg:p-8">
   356→   356→          <div className="mx-auto max-w-4xl">
   357→   357→            {currentLesson ? (
   358→   358→              <>
   359→   359→                {currentLesson.kind === 'LIVE' ? (
   360→   360→                  <LivePanel lesson={currentLesson} isOwner={isOwner} />
   361→   361→                ) : currentLesson.kind === 'READING' ? (
   362→   362→                  <ReadingMaterial lesson={currentLesson} />
   363→   363→                ) : (
   364→   364→                  <LessonPlayer lesson={currentLesson} />
   365→   365→                )}
   366→   366→
   367→   367→                {/* Celebração + certificado: 100% das aulas concluídas */}
   368→   368→                {courseCompleted ? (
   369→   369→                  <div className="mt-5 flex flex-col items-center gap-3 rounded-2xl border border-emerald-200 bg-emerald-50/80 p-6 text-center">
   370→   370→                    <CheckCircle2 aria-hidden className="h-12 w-12 fill-emerald-600 text-white" />
   371→   371→                    <p className="text-lg font-extrabold tracking-tight text-emerald-900">
   372→   372→                      Parabéns! Você concluiu este curso 🎉
   373→   373→                    </p>
   374→   374→                    <Button
   375→   375→                      onClick={() => setCertOpen(true)}
   376→   376→                      className="h-11 rounded-full bg-emerald-700 px-6 font-bold text-white hover:bg-emerald-800"
   377→   377→                    >
   378→   378→                      <GraduationCap aria-hidden className="h-4 w-4" /> Emitir certificado
   379→   379→                    </Button>
   380→   380→                  </div>
   381→   381→                ) : null}
   382→   382→
   383→   383→                <div className="mt-5 flex flex-wrap items-center gap-2">
   384→   384→                  <Badge variant="outline" className="border-stone-200 bg-white text-stone-600">
   385→   385→                    Aula {currentIndex + 1} de {orderedLessons.length}
   386→   386→                  </Badge>
   387→   387→                  {currentLesson.kind === 'LIVE' ? (
   388→   388→                    <Badge className="border-transparent bg-rose-100 text-rose-700">
   389→   389→                      <Radio aria-hidden className="h-3 w-3" /> Ao vivo
   390→   390→                    </Badge>
   391→   391→                  ) : currentLesson.kind === 'READING' ? (
   392→   392→                    <Badge className="border-transparent bg-amber-100 text-amber-800">
   393→   393→                      <BookOpen aria-hidden className="h-3 w-3" /> {readingKindLabel(currentLesson.reading?.kind)}
   394→   394→                    </Badge>
   395→   395→                  ) : currentLesson.videoUrl ? (
   396→   396→                    <Badge variant="outline" className="border-stone-200 bg-white text-stone-600">
   397→   397→                      Vídeo
   398→   398→                    </Badge>
   399→   399→                  ) : (
   400→   400→                    <Badge variant="outline" className="border-stone-200 bg-white text-stone-600">
   401→   401→                      Leitura
   402→   402→                    </Badge>
   403→   403→                  )}
   404→   404→                  {completedIds.includes(currentLesson.id) && (
   405→   405→                    <Badge className="border-transparent bg-emerald-100 text-emerald-800">
   406→   406→                      <Check aria-hidden className="h-3 w-3" /> Concluída
   407→   407→                    </Badge>
   408→   408→                  )}
   409→   409→                </div>
   410→   410→
   411→   411→                <h1 className="mt-2.5 text-xl font-extrabold tracking-tight text-stone-900 sm:text-2xl">
   412→   412→                  {currentLesson.title}
   413→   413→                </h1>
   414→   414→                {currentLesson.description ? (
   415→   415→                  <p className="mt-1.5 text-sm leading-relaxed text-stone-500">
   416→   416→                    {currentLesson.description}
   417→   417→                  </p>

   419→   419→
   420→   420→                {/* Anexos para download */}
   421→   421→                {currentLesson.attachments.length > 0 ? (
   422→   422→                  <div className="mt-4 flex flex-wrap gap-2">

  1246→  1246→
  1247→  1247→function LessonNotes({
  1248→  1248→  lessonId,
  1249→  1249→  user,
  1250→  1250→  lessonTitle,
  1251→  1251→}: {
  1252→  1252→  lessonId: string
  1253→  1253→  user: { id: string; name: string } | null
  1254→  1254→  lessonTitle: string
  1255→  1255→}) {
  1256→  1256→  const [body, setBody] = useState('')
  1257→  1257→  const [loaded, setLoaded] = useState(false)
  1258→  1258→  const [saveState, setSaveState] = useState<'idle' | 'saving' | 'saved'>('idle')
  1259→  1259→  const [savedAt, setSavedAt] = useState<string | null>(null)
  1260→  1260→  const timerRef = useRef<ReturnType<typeof setTimeout> | null>(null)
  1261→  1261→  const latestRef = useRef('')
  1262→  1262→
  1263→  1263→  // Carrega as anotações da aula (o componente é remontado por aula via key)
  1264→  1264→  useEffect(() => {
  1265→  1265→    if (!user) return
  1266→  1266→    let active = true
  1267→  1267→    api
  1268→  1268→      .getLessonNote(lessonId, user.id)
  1269→  1269→      .then((note) => {
  1270→  1270→        if (!active) return
  1271→  1271→        setBody(note.body)
  1272→  1272→        latestRef.current = note.body
  1273→  1273→        setSavedAt(note.updatedAt)
  1274→  1274→      })
  1275→  1275→      .catch(() => {
  1276→  1276→        /* sem anotações ainda */
  1277→  1277→      })
  1278→  1278→      .finally(() => {
  1279→  1279→        if (active) setLoaded(true)
  1280→  1280→      })
  1281→  1281→    return () => {
  1282→  1282→      active = false
  1283→  1283→    }
  1284→  1284→  }, [lessonId, user])
  1285→  1285→
  1286→  1286→  // Autosave com debounce (o estado "salvando" aparece no envio)
  1287→  1287→  useEffect(() => {
  1288→  1288→    if (!loaded || !user) return
  1289→  1289→    if (body === latestRef.current) return
  1290→  1290→    if (timerRef.current) clearTimeout(timerRef.current)
  1291→  1291→    timerRef.current = setTimeout(async () => {
  1292→  1292→      setSaveState('saving')
  1293→  1293→      try {
  1294→  1294→        const res = await api.saveLessonNote(lessonId, { userId: user.id, body })
  1295→  1295→        latestRef.current = body
  1296→  1296→        setSavedAt(res.updatedAt)
  1297→  1297→        setSaveState('saved')
  1298→  1298→      } catch {
  1299→  1299→        setSaveState('idle')
  1300→  1300→        toast.error('Não foi possível salvar as anotações.')
  1301→  1301→      }
  1302→  1302→    }, 900)
  1303→  1303→    return () => {
  1304→  1304→      if (timerRef.current) clearTimeout(timerRef.current)
  1305→  1305→    }
  1306→  1306→  }, [body, loaded, user, lessonId])
  1307→  1307→
  1308→  1308→  const downloadNotes = () => {
  1309→  1309→    const text = body.trim()
  1310→  1310→    if (!text) {
  1311→  1311→      toast.error('Escreva algo antes de baixar suas anotações.')
  1312→  1312→      return
  1313→  1313→    }
  1314→  1314→    const blob = new Blob(
  1315→  1315→      [`Anotações — ${lessonTitle}\nMentorHub · ${new Date().toLocaleString('pt-BR')}\n\n${text}\n`],
  1316→  1316→      { type: 'text/plain;charset=utf-8' }
  1317→  1317→    )
  1318→  1318→    const url = URL.createObjectURL(blob)
  1319→  1319→    const a = document.createElement('a')
  1320→  1320→    a.href = url
  1321→  1321→    a.download = `anotacoes-${lessonTitle.toLowerCase().replace(/[^a-z0-9]+/gi, '-').slice(0, 40)}.txt`
  1322→  1322→    a.click()
  1323→  1323→    URL.revokeObjectURL(url)
  1324→  1324→    toast.success('Anotações baixadas!')
  1325→  1325→  }
  1326→  1326→
  1327→  1327→  if (!user) {
  1328→  1328→    return (
  1329→  1329→      <div className="rounded-2xl border border-stone-200 bg-white p-6 text-center text-sm text-stone-500">
  1330→  1330→        Entre com uma conta para escrever e guardar suas anotações.
  1331→  1331→      </div>
  1332→  1332→    )
  1333→  1333→  }
  1334→  1334→
  1335→  1335→  return (
  1336→  1336→    <div className="rounded-2xl border border-stone-200 bg-white p-4 sm:p-5">
  1337→  1337→      <div className="flex items-center justify-between gap-3">
  1338→  1338→        <label htmlFor="lesson-notes" className="text-sm font-bold text-stone-800">
  1339→  1339→          Minhas anotações desta aula
  1340→  1340→        </label>
  1341→  1341→        <span
  1342→  1342→          className={cn(
  1343→  1343→            'inline-flex items-center gap-1.5 text-xs font-semibold',
  1344→  1344→            saveState === 'saving' ? 'text-amber-600' : saveState === 'saved' ? 'text-emerald-600' : 'text-stone-400'
  1345→  1345→          )}
  1346→  1346→          aria-live="polite"
  1347→  1347→        >
  1348→  1348→          {saveState === 'saving' ? (
  1349→  1349→            'Salvando…'
  1350→  1350→          ) : saveState === 'saved' ? (
  1351→  1351→            <>
  1352→  1352→              <Check aria-hidden className="h-3.5 w-3.5" /> Salvo
  1353→  1353→            </>
  1354→  1354→          ) : user ? (
  1355→  1355→            'Salva automaticamente'
  1356→  1356→          ) : (
  1357→  1357→            'Entre para salvar'
  1358→  1358→          )}
  1359→  1359→        </span>
  1360→  1360→      </div>
  1361→  1361→      <Textarea
  1362→  1362→        id="lesson-notes"
  1363→  1363→        rows={8}
  1364→  1364→        value={body}
  1365→  1365→        onChange={(e) => setBody(e.target.value)}
  1366→  1366→        placeholder="Anote insights, referências e tarefas desta aula… (salvo automaticamente na sua conta)"
  1367→  1367→        className="mt-2 resize-y"
  1368→  1368→      />
  1369→  1369→      <div className="mt-3 flex items-center justify-between gap-3">
  1370→  1370→        <span className="text-xs text-stone-400">
  1371→  1371→          {savedAt ? `Última atualização: ${new Date(savedAt).toLocaleString('pt-BR')}` : 'Nenhuma edição ainda'}
  1372→  1372→        </span>
  1373→  1373→        <Button
  1374→  1374→          variant="outline"
  1375→  1375→          className="h-10 rounded-full font-semibold"
  1376→  1376→          onClick={downloadNotes}
  1377→  1377→          disabled={!body.trim()}
  1378→  1378→        >
  1379→  1379→          <FileDown aria-hidden className="h-4 w-4" /> Baixar .txt
  1380→  1380→        </Button>
  1381→  1381→      </div>
  1382→  1382→    </div>
  1383→  1383→  )
  1384→  1384→}
