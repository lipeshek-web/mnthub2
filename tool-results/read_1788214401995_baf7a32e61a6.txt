     4→import {
     5→  Eye,
     6→  EyeOff,
     7→  GraduationCap,
     8→  Library,
     9→  Loader2,
    10→  ShieldCheck,
    11→  Users,

    13→} from 'lucide-react'
    14→import { toast } from 'sonner'
    15→import { Button } from '@/components/ui/button'
    16→import { Input } from '@/components/ui/input'
    17→import { Label } from '@/components/ui/label'
    18→import { Skeleton } from '@/components/ui/skeleton'
    19→import { Tabs, TabsContent, TabsList, TabsTrigger } from '@/components/ui/tabs'
    20→import {
    21→  InputOTP,

    24→} from '@/components/ui/input-otp'
    25→import { Avatar, Stars } from '@/components/platform/avatar'
    26→import { api } from '@/lib/api'
    27→import { firstName } from '@/lib/helpers'
    28→import { clearStoredRefCode, getStoredRefCode } from '@/lib/referral'
    29→import { useAppStore } from '@/lib/store'
    30→import type { UserDTO } from '@/lib/types'
    31→
    32→const EMAIL_RE = /^[^\s@]+@[^\s@]+\.[^\s@]+$/
    33→const DEMO_PASSWORD = 'demo123'
    34→
    35→const PANEL_BULLETS = [
    36→  { icon: Video, text: 'Cursos com aulas ao vivo, temas e materiais' },
    37→  { icon: Users, text: 'Mentorias 1:1 com especialistas de verdade' },
    38→  { icon: Library, text: 'Biblioteca com artigos e livros para ler na plataforma' },
    39→]
    40→
    41→interface FieldErrors {
    42→  name?: string
    43→  email?: string
    44→  password?: string
    45→}
    46→
    47→export function AuthView({ initialMode }: { initialMode?: 'login' | 'register' }) {
    48→  const setUser = useAppStore((s) => s.setUser)
    49→  const navigate = useAppStore((s) => s.navigate)
    50→
    51→  const [tab, setTab] = useState<'login' | 'register'>(initialMode ?? 'login')
    52→
    53→  // ----- Entrar -----
    54→  const [loginEmail, setLoginEmail] = useState('')
    55→  const [loginPassword, setLoginPassword] = useState('')
    56→  const [showLoginPassword, setShowLoginPassword] = useState(false)
    57→  const [loginErrors, setLoginErrors] = useState<FieldErrors>({})
    58→  const [loginFormError, setLoginFormError] = useState<string | null>(null)
    59→  const [loginLoading, setLoginLoading] = useState(false)
    60→
    61→  // ----- Criar conta -----
    62→  const [regName, setRegName] = useState('')
    63→  const [regEmail, setRegEmail] = useState('')
    64→  const [regPassword, setRegPassword] = useState('')
    65→  const [showRegPassword, setShowRegPassword] = useState(false)
    66→  const [regErrors, setRegErrors] = useState<FieldErrors>({})
    67→  const [regFormError, setRegFormError] = useState<string | null>(null)
    68→  const [regLoading, setRegLoading] = useState(false)
    69→
    70→  // ----- Contas de demonstração -----
    71→  const [demoUsers, setDemoUsers] = useState<
    72→    { id: string; name: string; email: string; avatarUrl: string | null }[]
    73→  >([])
    74→  const [demoLoading, setDemoLoading] = useState(true)
    75→  const [demoBusyId, setDemoBusyId] = useState<string | null>(null)
    76→
    77→  // ----- MFA (segundo fator): ativo após senha correta de conta com MFA -----
    78→  const [mfaTicket, setMfaTicket] = useState<string | null>(null)
    79→  const [mfaCode, setMfaCode] = useState('')
    80→  const [mfaLoading, setMfaLoading] = useState(false)
    81→  const [mfaError, setMfaError] = useState<string | null>(null)
    82→  const [mfaMode, setMfaMode] = useState<'totp' | 'recovery'>('totp')
    83→  const [mfaRecoveryCode, setMfaRecoveryCode] = useState('')
    84→
    85→  // Sincroniza quando a view chega com outro modo (ex.: clique em "Criar conta" no navbar)
    86→  useEffect(() => {
    87→    if (initialMode) setTab(initialMode)
    88→  }, [initialMode])
    89→
    90→  useEffect(() => {
    91→    let active = true
    92→    api
    93→      .demoAccounts()
    94→      .then((users) => {
    95→        if (active) setDemoUsers(users)
    96→      })
    97→      .catch(() => {
    98→        if (active) setDemoUsers([])
    99→      })
   100→      .finally(() => {
   101→        if (active) setDemoLoading(false)
   102→      })
   103→    return () => {
   104→      active = false
   105→    }
   106→  }, [])
   107→
   108→  /** Ao trocar de tab, limpa os erros inline */
   109→  const switchTab = (next: string) => {
   110→    setTab(next as 'login' | 'register')
   111→    setLoginErrors({})
   112→    setLoginFormError(null)
   113→    setRegErrors({})
   114→    setRegFormError(null)
   115→  }
   116→
   117→  // ---------- Entrar ----------
   118→  const handleLogin = async (e: FormEvent<HTMLFormElement>) => {
   119→    e.preventDefault()
   120→    if (loginLoading) return // bloqueia duplo submit
   121→
   122→    const errors: FieldErrors = {}
   123→    if (!EMAIL_RE.test(loginEmail.trim())) errors.email = 'Informe um e-mail válido.'
   124→    if (loginPassword.length < 6) errors.password = 'A senha deve ter pelo menos 6 caracteres.'
   125→    setLoginErrors(errors)
   126→    if (Object.keys(errors).length > 0) return
   127→
   128→    setLoginLoading(true)
   129→    setLoginFormError(null)
   130→    try {
   131→      const result = await api.login({ email: loginEmail.trim(), password: loginPassword })
   132→
   133→      // Conta com MFA: mostra a etapa do código de 6 dígitos
   134→      if ('mfaRequired' in result && result.mfaRequired) {
   135→        setMfaTicket(result.mfaTicket)
   136→        setMfaCode('')
   137→        setMfaError(null)
   138→        setMfaMode('totp')
   139→        setMfaRecoveryCode('')
   140→        toast.info('Confirme com o código do seu app autenticador 🔐')
   141→        return
   142→      }
   143→
   144→      const user = result as UserDTO
   145→      setUser(user)
   146→      toast.success(`Bem-vindo(a) de volta, ${firstName(user.name)}! 👋`)
   147→      navigate({ name: 'home' })
   148→    } catch (err) {
   149→      const msg = err instanceof Error ? err.message : 'Não foi possível entrar. Tente novamente.'
   150→      toast.error(msg)
   151→      setLoginFormError(msg)
   152→    } finally {
   153→      setLoginLoading(false)
   154→    }
   155→  }
   156→
   157→  // ---------- MFA: verifica o código TOTP ou de recuperação ----------
   158→  const handleMfaVerify = async (code?: string) => {
   159→    if (!mfaTicket || mfaLoading) return
   160→    const value =
   161→      mfaMode === 'totp'
   162→        ? (code ?? mfaCode).replace(/\D/g, '')
   163→        : (code ?? mfaRecoveryCode).trim().toUpperCase()
   164→    if (mfaMode === 'totp' && value.length !== 6) return
   165→    if (mfaMode === 'recovery' && value.length < 6) return
   166→    setMfaLoading(true)
   167→    setMfaError(null)
   168→    try {
   169→      const user = await api.verifyMfa({ ticket: mfaTicket, code: value })
   170→      setUser(user)
   171→      if (user.usedRecoveryCode) {
   172→        toast.warning(
   173→          `Você entrou com um código de recuperação. Restam ${user.recoveryCodesRemaining ?? 0} — gere novos no painel admin. ⚠️`
   174→        )
   175→      } else {
   176→        toast.success(`Bem-vindo(a) de volta, ${firstName(user.name)}! 👋`)
   177→      }
   178→      setMfaTicket(null)
   179→      setMfaCode('')
   180→      setMfaRecoveryCode('')
   181→      setMfaMode('totp')
   182→      navigate({ name: 'home' })
   183→    } catch (err) {
   184→      const msg = err instanceof Error ? err.message : 'Código inválido.'
   185→      setMfaError(msg)
   186→      setMfaCode('')
   187→      setMfaRecoveryCode('')
   188→    } finally {
   189→      setMfaLoading(false)
   190→    }
   191→  }
   192→
   193→  // ---------- Criar conta ----------
   194→  const handleRegister = async (e: FormEvent<HTMLFormElement>) => {
   195→    e.preventDefault()
   196→    if (regLoading) return // bloqueia duplo submit
   197→
   198→    const errors: FieldErrors = {}
   199→    if (regName.trim().length < 2) errors.name = 'Informe seu nome completo.'
   200→    if (!EMAIL_RE.test(regEmail.trim())) errors.email = 'Informe um e-mail válido.'
   201→    if (regPassword.length < 6) errors.password = 'A senha deve ter pelo menos 6 caracteres.'
   202→    setRegErrors(errors)
   203→    if (Object.keys(errors).length > 0) return
   204→
   205→    setRegLoading(true)
   206→    setRegFormError(null)
   207→    try {
   208→      // Convite capturado em ?ref= (localStorage, janela de 7 dias)
   209→      const refCode = getStoredRefCode() ?? undefined
   210→      const user = await api.register({
   211→        name: regName.trim(),
   212→        email: regEmail.trim(),
   213→        password: regPassword,
   214→        refCode,
   215→      })
   216→      if (refCode) clearStoredRefCode()
   217→      setUser(user)
   218→      toast.success(
   219→        user.referralApplied
   220→          ? `Conta criada! R$ 10 de crédito de convite adicionados, ${firstName(user.name)}! 🎁`
   221→          : `Conta criada! Bem-vindo(a), ${firstName(user.name)}! 🎉`
   222→      )
   223→      navigate({ name: 'home' })
   224→    } catch (err) {
   225→      const msg =
   226→        err instanceof Error ? err.message : 'Não foi possível criar sua conta. Tente novamente.'
   227→      toast.error(msg)
   228→      if (msg.includes('Já existe') || msg.includes('e-mail')) {
   229→        setRegErrors((prev) => ({ ...prev, email: 'Já existe conta com este e-mail. Faça login.' }))
   230→      } else {
   231→        setRegFormError(msg)
   232→      }
   233→    } finally {
   234→      setRegLoading(false)
   235→    }
   236→  }
   237→
   238→  // ---------- Contas demo ----------
   239→  const handleDemoLogin = async (u: UserDTO) => {
   240→    if (demoBusyId) return
   241→    setDemoBusyId(u.id)
   242→    try {
   243→      const result = await api.login({ email: u.email, password: DEMO_PASSWORD })
   244→      // Contas demo não têm MFA — mas o tipo cobre o caso
   245→      if ('mfaRequired' in result && result.mfaRequired) {
   246→        setMfaTicket(result.mfaTicket)
   247→        toast.info('Confirme com o código do seu app autenticador 🔐')
   248→        return
   249→      }
   250→      const user = result as UserDTO
   251→      setUser(user)
   252→      toast.success(`Entrou como ${user.name}`)
   253→      navigate({ name: 'home' })
   254→    } catch (err) {
   255→      // Mostra o erro real (ex.: 429 de rate limit) em vez de mascarar
   256→      const msg = err instanceof Error ? err.message : ''
   257→      toast.info(msg || 'Esta conta demo ainda está sendo preparada. Crie a sua!')
   258→    } finally {
   259→      setDemoBusyId(null)
   260→    }
   261→  }
   262→
   263→  const handleForgotPassword = () => {
   264→    toast.info('Em breve! Por enquanto, use uma conta demo ou crie a sua.')
   265→  }
   266→
   267→  return (
   268→    <div className="grid min-h-full bg-stone-50 dark:bg-stone-950 lg:grid-cols-2">
   269→      {/* ---------- Painel esquerdo (header compacto no mobile, completo no lg+) ---------- */}
   270→      <section
   271→        aria-label="Sobre o MentorHub"
   272→        className="relative flex flex-col gap-10 overflow-hidden bg-gradient-to-br from-emerald-800 via-emerald-700 to-teal-700 p-6 text-white lg:justify-between lg:gap-12 lg:p-10 xl:p-14"
   273→      >
   274→        {/* Blobs decorativos */}
   275→        <div aria-hidden className="pointer-events-none absolute inset-0">
   276→          <div className="absolute -left-28 -top-28 h-80 w-80 rounded-full bg-white/10 blur-3xl" />
   277→          <div className="absolute -right-24 top-1/3 h-96 w-96 rounded-full bg-white/10 blur-3xl" />
   278→          <div className="absolute -bottom-32 left-1/4 h-72 w-72 rounded-full bg-white/10 blur-3xl" />
   279→        </div>
   280→
   281→        {/* Logo */}
   282→        <div className="relative flex items-center gap-3">
   283→          <span className="flex h-10 w-10 items-center justify-center rounded-xl bg-white/15">
   284→            <GraduationCap className="h-5 w-5" aria-hidden />
   285→          </span>
   286→          <span className="text-lg font-extrabold tracking-tight">MentorHub</span>
   287→        </div>
   288→
   289→        {/* Headline + bullets */}
   290→        <div className="relative">
   291→          <h1 className="max-w-lg text-2xl font-extrabold leading-tight tracking-tight lg:text-4xl lg:leading-[1.1]">
   292→            Aprenda com quem vive o que ensina.
   293→          </h1>
   294→          <p className="mt-3 max-w-md text-sm text-emerald-100/80 lg:mt-4 lg:text-base">
   295→            Cursos ao vivo, mentorias 1:1 e biblioteca — tudo o que você precisa para evoluir de
   296→            verdade, em um só lugar.
   297→          </p>
   298→          <ul className="mt-8 hidden space-y-4 lg:block">
   299→            {PANEL_BULLETS.map(({ icon: Icon, text }) => (
   300→              <li key={text} className="flex items-center gap-3.5">
   301→                <span className="flex h-10 w-10 shrink-0 items-center justify-center rounded-full bg-white/10">
   302→                  <Icon className="h-5 w-5" aria-hidden />
   303→                </span>
   304→                <span className="text-sm text-emerald-50/90">{text}</span>
   305→              </li>
   306→            ))}
   307→          </ul>
   308→        </div>
   309→
   310→        {/* Prova social (rodapé do painel) */}
   311→        <div className="relative hidden lg:block">
   312→          <div className="max-w-md rounded-2xl border border-white/15 bg-white/10 p-4">
   313→            <Stars rating={5} size={16} />
   314→            <p className="mt-2 text-sm text-emerald-50/90">
   315→              4,9 de média · +2.400 alunos aprendendo todos os dias
   316→            </p>
   317→          </div>
   318→        </div>
   319→      </section>
   320→
   321→      {/* ---------- Painel direito: formulário ---------- */}
   322→      <section className="flex items-center justify-center p-4 sm:p-6 lg:p-8 xl:p-12">
   323→        <div className="w-full max-w-md rounded-2xl border border-stone-200 bg-white dark:border-stone-800 dark:bg-stone-900 p-6 shadow-sm sm:p-8">
   324→          <Tabs value={tab} onValueChange={switchTab}>
   325→            <TabsList className="grid h-11 w-full grid-cols-2 rounded-full bg-stone-100 dark:bg-stone-800 p-1">
   326→              <TabsTrigger
   327→                value="login"
   328→                className="rounded-full text-sm font-semibold text-stone-500 data-[state=active]:bg-white data-[state=active]:text-stone-900 data-[state=active]:shadow-sm dark:text-stone-400 dark:data-[state=active]:bg-stone-900 dark:data-[state=active]:text-stone-50"
   329→              >
   330→                Entrar
   331→              </TabsTrigger>
   332→              <TabsTrigger
   333→                value="register"
   334→                className="rounded-full text-sm font-semibold text-stone-500 data-[state=active]:bg-white data-[state=active]:text-stone-900 data-[state=active]:shadow-sm dark:text-stone-400 dark:data-[state=active]:bg-stone-900 dark:data-[state=active]:text-stone-50"
   335→              >
   336→                Criar conta
   337→              </TabsTrigger>
   338→            </TabsList>
   339→
   340→            {/* ----- Tab Entrar ----- */}
   341→            <TabsContent value="login" className="mt-6">
   342→              {mfaTicket ? (
   343→                <div className="flex flex-col items-center text-center">
   344→                  <span className="flex h-14 w-14 items-center justify-center rounded-2xl bg-emerald-50 text-emerald-700 dark:bg-emerald-950/60 dark:text-emerald-300">
   345→                    <ShieldCheck className="h-7 w-7" aria-hidden />
   346→                  </span>
   347→                  <h2 className="mt-4 text-xl font-extrabold tracking-tight text-stone-900 dark:text-stone-50">
   348→                    Verificação em duas etapas
   349→                  </h2>
   350→                  <p className="mt-1.5 max-w-xs text-sm leading-relaxed text-stone-500 dark:text-stone-400">
   351→                    {mfaMode === 'totp' ? (
   352→                      <>Digite o código de 6 dígitos do seu app autenticador (Google Authenticator, Authy…).</>
   353→                    ) : (
   354→                      <>Digite um dos seus códigos de recuperação (formato XXXX-XXXX). Cada código funciona uma única vez.</>
   355→                    )}
   356→                  </p>
   357→
   358→                  <div className="mt-6">
   359→                    {mfaMode === 'totp' ? (
   360→                      <InputOTP
   361→                        maxLength={6}
   362→                        value={mfaCode}
   363→                        onChange={(v) => {
   364→                          setMfaCode(v)
   365→                          setMfaError(null)
   366→                          if (v.length === 6) void handleMfaVerify(v)
   367→                        }}
   368→                        disabled={mfaLoading}
   369→                      >
   370→                        <InputOTPGroup>
   371→                          {[0, 1, 2, 3, 4, 5].map((i) => (
   372→                            <InputOTPSlot key={i} index={i} className="h-12 w-11 rounded-lg border-stone-300 text-lg font-bold dark:border-stone-700" />
   373→                          ))}
   374→                        </InputOTPGroup>
   375→                      </InputOTP>
   376→                    ) : (
   377→                      <Input
   378→                        value={mfaRecoveryCode}
   379→                        onChange={(e) => {
   380→                          setMfaRecoveryCode(e.target.value.toUpperCase())
   381→                          setMfaError(null)
   382→                        }}
   383→                        onKeyDown={(e) => {
   384→                          if (e.key === 'Enter') void handleMfaVerify()
   385→                        }}
   386→                        placeholder="XXXX-XXXX"
   387→                        autoComplete="off"
   388→                        disabled={mfaLoading}
   389→                        className="h-12 w-56 rounded-xl text-center font-mono text-lg font-bold tracking-widest uppercase"
   390→                        aria-label="Código de recuperação"
   391→                      />
   392→                    )}
   393→                  </div>
   394→
   395→                  {mfaError && (
   396→                    <p role="alert" className="mt-4 text-xs font-medium text-rose-600 dark:text-rose-400">
   397→                      {mfaError}
   398→                    </p>
   399→                  )}
   400→
   401→                  <Button
   402→                    onClick={() => void handleMfaVerify()}
   403→                    disabled={
   404→                      mfaLoading ||
   405→                      (mfaMode === 'totp'
   406→                        ? mfaCode.replace(/\D/g, '').length !== 6
   407→                        : mfaRecoveryCode.trim().length < 6)
   408→                    }
   409→                    className="mt-6 h-11 w-full rounded-full text-sm font-bold"
   410→                  >
   411→                    {mfaLoading ? (
   412→                      <>
   413→                        <Loader2 className="h-4 w-4 animate-spin" aria-hidden />
   414→                        Verificando…
   415→                      </>
   416→                    ) : (
   417→                      'Confirmar código'
   418→                    )}
   419→                  </Button>
   420→                  <button
   421→                    type="button"
   422→                    onClick={() => {
   423→                      if (mfaMode === 'recovery') {
   424→                        setMfaMode('totp')
   425→                        setMfaRecoveryCode('')
   426→                      } else {
   427→                        setMfaMode('recovery')
   428→                        setMfaCode('')
   429→                      }
   430→                      setMfaError(null)
   431→                    }}
   432→                    className="mt-4 text-xs font-medium text-stone-400 underline-offset-2 transition-colors hover:text-stone-600 hover:underline dark:text-stone-500 dark:hover:text-stone-300"
   433→                  >
   434→                    {mfaMode === 'recovery'
   435→                      ? 'Usar o app autenticador'
   436→                      : 'Não tem o app? Usar código de recuperação'}
   437→                  </button>
   438→                  <button
   439→                    type="button"
   440→                    onClick={() => {
   441→                      setMfaTicket(null)
   442→                      setMfaCode('')
   443→                      setMfaRecoveryCode('')
   444→                      setMfaError(null)
   445→                      setMfaMode('totp')
   446→                    }}
   447→                    className="mt-2 text-xs font-medium text-stone-400 underline-offset-2 transition-colors hover:text-stone-600 hover:underline dark:text-stone-500 dark:hover:text-stone-300"
   448→                  >
   449→                    Voltar para o login
   450→                  </button>
   451→                </div>
   452→              ) : (
   453→              <form onSubmit={handleLogin} noValidate className="space-y-4">
   454→                <div>
   455→                  <h2 className="text-xl font-extrabold tracking-tight text-stone-900 dark:text-stone-50">
   456→                    Bem-vindo de volta
   457→                  </h2>
   458→                  <p className="mt-1 text-sm text-stone-500 dark:text-stone-400">
   459→                    Entre para continuar sua jornada de aprendizado.
   460→                  </p>
   461→                </div>
   462→
   463→                <div className="space-y-2">
   464→                  <Label htmlFor="login-email" className="text-sm font-medium text-stone-700 dark:text-stone-200">
   465→                    E-mail
   466→                  </Label>
   467→                  <Input
   468→                    id="login-email"
   469→                    type="email"
   470→                    value={loginEmail}
   471→                    onChange={(e) => {
   472→                      setLoginEmail(e.target.value)
   473→                      if (loginErrors.email)
   474→                        setLoginErrors((prev) => ({ ...prev, email: undefined }))
   475→                    }}
   476→                    placeholder="voce@exemplo.com"
   477→                    autoComplete="email"
   478→                    className="h-11 rounded-xl"
   479→                    aria-invalid={Boolean(loginErrors.email)}
   480→                    aria-describedby={loginErrors.email ? 'login-email-error' : undefined}
   481→                  />
   482→                  {loginErrors.email && (
   483→                    <p id="login-email-error" className="text-xs text-rose-600 dark:text-rose-400">
   484→                      {loginErrors.email}
   485→                    </p>
   486→                  )}
   487→                </div>
   488→
   489→                <div className="space-y-2">
   490→                  <Label htmlFor="login-password" className="text-sm font-medium text-stone-700 dark:text-stone-200">
   491→                    Senha
   492→                  </Label>
   493→                  <div className="relative">
   494→                    <Input
   495→                      id="login-password"
   496→                      type={showLoginPassword ? 'text' : 'password'}
   497→                      value={loginPassword}
   498→                      onChange={(e) => {
   499→                        setLoginPassword(e.target.value)
   500→                        if (loginErrors.password)
   501→                          setLoginErrors((prev) => ({ ...prev, password: undefined }))
   502→                      }}
   503→                      placeholder="Sua senha"
