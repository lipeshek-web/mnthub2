    61→    61→async function request<T>(url: string, options?: RequestInit): Promise<T> {
    62→    62→  const res = await fetch(url, {
    63→    63→    headers: { 'Content-Type': 'application/json' },
    64→    64→    cache: 'no-store',
    65→    65→    ...options,
    66→    66→  })
    67→    67→  let json: unknown = {}
    68→    68→  try {
    69→    69→    json = await res.json()
    70→    70→  } catch {
    71→    71→    json = {}
    72→    72→  }
    73→    73→  if (!res.ok) {
    74→    74→    const msg = (json as { error?: string })?.error || 'Ocorreu um erro inesperado. Tente novamente.'
    75→    75→    throw new Error(msg)
    76→    76→  }
    77→    77→  return json as T
    78→    78→}

    80→    80→const qs = (params: Record<string, string | number | undefined>) => {
    81→    81→  const sp = new URLSearchParams()
    82→    82→  Object.entries(params).forEach(([k, v]) => {
    83→    83→    if (v !== undefined && v !== '') sp.set(k, String(v))
    84→    84→  })
    85→    85→  const s = sp.toString()
    86→    86→  return s ? `?${s}` : ''
    87→    87→}
