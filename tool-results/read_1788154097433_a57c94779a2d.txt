    96→export function ClassroomView({ courseId }: { courseId: string }) {
    97→  const navigate = useAppStore((s) => s.navigate)
    98→  const user = useAppStore((s) => s.user)
    99→
   100→  const [data, setData] = useState<CourseDetailDTO | null>(null)
   101→  const [loading, setLoading] = useState(true)
   102→  const [error, setError] = useState<string | null>(null)
   103→
   104→  const [currentLessonId, setCurrentLessonId] = useState<string | null>(null)
   105→  const [completedIds, setCompletedIds] = useState<string[]>([])
   106→  const [toggling, setToggling] = useState(false)
   107→
   108→  // Auto-retry para falhas transitórias de rede/servidor (ex.: dev server reiniciando)
   109→  const retriesRef = useRef(0)
   110→  const retryTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null)
   111→
   112→  const fetchCourse = useCallback(async () => {
   113→    setLoading(true)
   114→    setError(null)
   115→    try {
   116→      const d = await api.getCourse(courseId, user?.id)
   117→      retriesRef.current = 0
   118→      setData(d)
   119→    } catch (err) {
   120→      const message = err instanceof Error ? err.message : 'Não foi possível carregar o curso.'
   121→      const transient =
   122→        /inesperado|fetch|network|load failed|conex/i.test(message) && retriesRef.current < 2
   123→      if (transient) {
   124→        retriesRef.current += 1
   125→        if (retryTimerRef.current) clearTimeout(retryTimerRef.current)
   126→        retryTimerRef.current = setTimeout(() => void fetchCourse(), 1200)
   127→      } else {
   128→        setError(message)
   129→      }
   130→    } finally {
   131→      setLoading(false)
   132→    }
   133→  }, [courseId, user?.id])
   134→
   135→  useEffect(() => {
   136→    void fetchCourse()
   137→    return () => {
   138→      if (retryTimerRef.current) clearTimeout(retryTimerRef.current)
   139→    }
   140→  }, [fetchCourse])
   141→
   142→  const course = data
   143→  const isOwner = Boolean(user && course && course.mentor.userId === user.id)
   144→  const isEnrolled = course?.enrollment != null
   145→  const hasAccess = isEnrolled || isOwner
   146→
   147→  const lessons = useMemo<CourseLessonDTO[]>(
   148→    () => [...(course?.lessons ?? [])].sort((a, b) => a.order - b.order),
   149→    [course]
   150→  )
   151→
   152→  // Temas (módulos) em ordem; aulas sem tema (ou com tema removido) vão para o
   153→  // grupo virtual "Outros conteúdos", sempre no fim da lista.
   154→  const themeList = useMemo<CourseThemeDTO[]>(
   155→    () => [...(course?.themes ?? [])].sort((a, b) => a.order - b.order),
   156→    [course]
   157→  )
   158→
   159→  const lessonGroups = useMemo<LessonGroup[]>(() => {
   160→    const byTheme = new Map<string, CourseLessonDTO[]>()
   161→    const loose: CourseLessonDTO[] = []
   162→    for (const lesson of lessons) {
   163→      if (lesson.themeId && themeList.some((t) => t.id === lesson.themeId)) {
   164→        const bucket = byTheme.get(lesson.themeId) ?? []
   165→        bucket.push(lesson)
   166→        byTheme.set(lesson.themeId, bucket)
   167→      } else {
   168→        loose.push(lesson)
   169→      }
   170→    }
   171→    const groups: LessonGroup[] = themeList.map((theme) => ({
   172→      key: theme.id,
   173→      title: theme.title,
   174→      theme,
   175→      lessons: byTheme.get(theme.id) ?? [],
   176→    }))
   177→    if (loose.length > 0) {
   178→      groups.push({ key: 'none', title: 'Outros conteúdos', theme: null, lessons: loose })
   179→    }
   180→    return groups
   181→  }, [lessons, themeList])
   182→
   183→  // Lista achatada (temas em ordem → aulas em ordem): base do índice global,
   184→  // do avançar/voltar e do progresso geral.
   185→  const orderedLessons = useMemo(() => lessonGroups.flatMap((g) => g.lessons), [lessonGroups])
   186→
   187→  // Seções recolhidas da sidebar (default: todas expandidas)
   188→  const [collapsedThemes, setCollapsedThemes] = useState<Record<string, boolean>>({})
   189→  // Emissão do certificado (botão da celebração de 100%)
   190→  const [issuing, setIssuing] = useState(false)
   191→
   192→  // Modo foco (teatro): esconde sidebar e abas, centraliza o palco da aula.
   193→  // Persistido na sessão para o aluno retomar a imersão ao voltar para a sala.
   194→  const [focusMode, setFocusMode] = useState(() => {
   195→    if (typeof window === 'undefined') return false
   196→    try {
   197→      return window.sessionStorage.getItem('mentorhub-classroom-focus') === '1'
   198→    } catch {
   199→      return false
   200→    }
   201→  })
   202→  const [contentsOpen, setContentsOpen] = useState(false)
   203→
   204→  useEffect(() => {
   205→    try {
   206→      if (focusMode) window.sessionStorage.setItem('mentorhub-classroom-focus', '1')
   207→      else window.sessionStorage.removeItem('mentorhub-classroom-focus')
   208→    } catch {
   209→      /* storage indisponível */
   210→    }
   211→  }, [focusMode])
   212→
   213→  // Aula inicial: a indicada na view (retorno do leitor) ou a primeira ainda
   214→  // não concluída (só define uma vez por curso)
   215→  const initializedRef = useRef<string | null>(null)
   216→  useEffect(() => {
   217→    if (!course || !hasAccess) return
   218→    if (initializedRef.current === course.id) return
   219→    initializedRef.current = course.id
   220→    const enrollment = course.enrollment
   221→    const completed = enrollment?.completedLessonIds ?? []
   222→    setCompletedIds(completed)
   223→    const view = useAppStore.getState().view
   224→    const returnLessonId = view.name === 'classroom' ? view.lessonId : undefined
   225→    const target =
   226→      (returnLessonId && orderedLessons.some((l) => l.id === returnLessonId) ? returnLessonId : undefined) ??
   227→      orderedLessons.find((l) => !completed.includes(l.id))?.id ??
   228→      orderedLessons[0]?.id ??
   229→      null
   230→    setCurrentLessonId(target)
   231→  }, [course, hasAccess, orderedLessons])
   232→
   233→  const currentLesson = useMemo(
   234→    () => orderedLessons.find((l) => l.id === currentLessonId) ?? orderedLessons[0] ?? null,
   235→    [orderedLessons, currentLessonId]
   236→  )
   237→  const currentIndex = currentLesson ? orderedLessons.findIndex((l) => l.id === currentLesson.id) : -1
   238→  const percent =
   239→    orderedLessons.length > 0 ? Math.round((completedIds.length / orderedLessons.length) * 100) : 0
   240→  const courseCompleted =
   241→    orderedLessons.length > 0 && completedIds.length >= orderedLessons.length
   242→
   243→  const handleToggleTheme = (key: string) =>
   244→    setCollapsedThemes((prev) => ({ ...prev, [key]: !prev[key] }))
   245→
   246→  const selectLesson = (lessonId: string) => {
   247→    setCurrentLessonId(lessonId)
   248→    // Ao selecionar uma aula, garante o tema dela expandido
   249→    const lessonThemeId = orderedLessons.find((l) => l.id === lessonId)?.themeId
   250→    if (lessonThemeId) setCollapsedThemes((prev) => ({ ...prev, [lessonThemeId]: false }))
   251→  }
   252→
   253→  const toggleComplete = async (autoAdvance = false) => {
   254→    if (!user || !course || !currentLesson) return
   255→    setToggling(true)
   256→    try {
   257→      const res = await api.toggleLessonComplete(course.id, {
   258→        userId: user.id,
   259→        lessonId: currentLesson.id,
   260→      })
   261→      const nowCompleted = res.completedLessonIds.includes(currentLesson.id)
   262→      setCompletedIds(res.completedLessonIds)
   263→      if (res.xpAwarded > 0) {
   264→        toast.success(`+${res.xpAwarded} XP de estudo ⚡`)
   265→      }
   266→      if (nowCompleted && res.courseCompleted) {
   267→        toast.success('Curso concluído! Parabéns 🎉')
   268→      } else if (nowCompleted) {
   269→        toast.success('Aula marcada como concluída.')
   270→      } else {
   271→        toast('Marcação removida desta aula.')
   272→      }
   273→      if (autoAdvance && nowCompleted && currentIndex < orderedLessons.length - 1) {
   274→        setCurrentLessonId(orderedLessons[currentIndex + 1].id)
   275→      }
   276→    } catch (err) {
   277→      toast.error(err instanceof Error ? err.message : 'Não foi possível atualizar o progresso.')
   278→    } finally {
   279→      setToggling(false)
   280→    }
   281→  }
   282→
   283→  const goPrev = () => {
   284→    if (currentLesson && currentIndex > 0) setCurrentLessonId(orderedLessons[currentIndex - 1].id)
   285→  }
   286→  const goNext = () => {
   287→    if (currentLesson && currentIndex < orderedLessons.length - 1) {
   288→      setCurrentLessonId(orderedLessons[currentIndex + 1].id)
   289→    }
   290→  }
   291→
   292→  // Emite (ou retorna) o certificado e abre a página pública verificável.
   293→  // O overlay da sala fecha naturalmente: a view global muda para 'certificate'.
   294→  const handleIssueCertificate = async () => {
   295→    if (!user || !course) return
   296→    setIssuing(true)
   297→    try {
   298→      const cert = await api.issueCertificate(course.id, user.id)
   299→      toast.success('Certificado emitido! 🎓')
   300→      navigate({ name: 'certificate', code: cert.code })
   301→    } catch (err) {
   302→      toast.error(err instanceof Error ? err.message : 'Não foi possível emitir o certificado.')
   303→    } finally {
   304→      setIssuing(false)
   305→    }
   306→  }
   307→
   308→  /* ---------- Estados de tela ---------- */
   309→
   310→  if (loading) {
   311→    return (
   312→      <div className="flex h-full min-h-0 flex-col" aria-busy="true">
   313→        <div className="flex items-center gap-3 border-b border-emerald-400/15 bg-emerald-950 px-4 py-3 sm:px-6">
   314→          <Skeleton className="h-9 w-9 rounded-full bg-white/10" />
   315→          <Skeleton className="h-5 w-48 bg-white/10" />
   316→          <Skeleton className="ml-auto h-6 w-28 rounded-full bg-white/10" />
   317→        </div>
   318→        <div className="grid flex-1 grid-cols-1 lg:grid-cols-[1fr_380px]">
   319→          <div className="space-y-4 p-4 sm:p-6">
   320→            <Skeleton className="aspect-video w-full rounded-2xl" />
   321→            <Skeleton className="h-7 w-2/3" />
   322→            <Skeleton className="h-24 w-full rounded-2xl" />
   323→          </div>
   324→          <div className="hidden space-y-2 border-l border-stone-200 bg-white p-4 lg:block dark:border-stone-800 dark:bg-stone-900">
   325→            {Array.from({ length: 6 }).map((_, i) => (
   326→              <Skeleton key={i} className="h-14 rounded-xl" />
   327→            ))}
   328→          </div>
   329→        </div>
   330→        <p className="sr-only">Abrindo sala de aula…</p>
   331→      </div>
   332→    )
   333→  }
   334→
   335→  if (!course) {
   336→    return (
   337→      <div className="flex h-full items-center justify-center p-6">
   338→        <div className="flex w-full max-w-md flex-col items-center gap-3 rounded-2xl border border-dashed border-stone-300 px-6 py-14 text-center dark:border-stone-700">
   339→          <span className="flex h-14 w-14 items-center justify-center rounded-full bg-rose-50 dark:bg-rose-950/50">
   340→            <AlertCircle aria-hidden className="h-7 w-7 text-rose-500" />
   341→          </span>
   342→          <p className="font-bold text-stone-900 dark:text-stone-50">Não foi possível abrir a sala de aula</p>
   343→          <p className="max-w-sm text-sm leading-relaxed text-stone-500 dark:text-stone-400">
   344→            {error ?? 'Curso não encontrado.'}
   345→          </p>
   346→          <Button variant="outline" className="rounded-full" onClick={() => void fetchCourse()}>
   347→            Tentar novamente
   348→          </Button>
   349→        </div>
   350→      </div>
   351→    )
   352→  }
   353→
   354→  if (!hasAccess) {
   355→    return (
   356→      <div className="flex h-full items-center justify-center p-6">
   357→        <Card className="w-full max-w-md rounded-2xl border-stone-200 p-8 text-center shadow-none dark:border-stone-800">
   358→          <span className="mx-auto flex h-14 w-14 items-center justify-center rounded-full bg-amber-50 dark:bg-amber-950/50">
   359→            <Lock aria-hidden className="h-6 w-6 text-amber-600 dark:text-amber-400" />
   360→          </span>
   361→          <h1 className="mt-4 text-lg font-extrabold tracking-tight text-stone-900 dark:text-stone-50">
   362→            {course.title}
   363→          </h1>
   364→          <p className="mt-2 text-sm leading-relaxed text-stone-500 dark:text-stone-400">
   365→            Você ainda não tem acesso a este curso. Inscreva-se para desbloquear todas as aulas,
   366→            perguntas e anotações.
   367→          </p>
   368→          <Button
   369→            className="mt-5 h-11 w-full rounded-full font-bold"
   370→            onClick={() => navigate({ name: 'course', courseId })}
   371→          >
   372→            Ver o curso e se inscrever
   373→          </Button>
   374→        </Card>
   375→      </div>
   376→    )
   377→  }
   378→
   379→  const firstName = course.mentor.name.split(' ')[0]
   380→  // Código do certificado do usuário neste curso (null = ainda não emitido)
   381→  const certificateCode = course.certificateCode
   382→
   383→  return (
   384→    <div className="flex h-full min-h-0 flex-col bg-stone-50 dark:bg-stone-950" aria-label={`Sala de aula: ${course.title}`}>
   385→      {/* ---------- BARRA SUPERIOR IMERSIVA ---------- */}
   386→      <div className="flex shrink-0 items-center gap-1 border-b border-emerald-400/15 bg-emerald-950 px-3 py-2.5 text-white sm:gap-2.5 sm:px-6">
   387→        <Button
   388→          variant="ghost"
   389→          size="icon"
   390→          className="h-9 w-9 shrink-0 rounded-full text-white hover:bg-white/10 hover:text-white"
   391→          aria-label="Sair da sala de aula"
   392→          onClick={() => navigate({ name: 'course', courseId })}
   393→        >
   394→          <ArrowLeft aria-hidden className="h-4 w-4" />
   395→        </Button>
   396→        <div className="min-w-0 flex-1">
   397→          <p className="truncate text-sm font-extrabold tracking-tight text-white sm:text-base">
   398→            {course.title}
   399→          </p>
   400→          <p className="hidden text-xs text-emerald-100/70 sm:block">
   401→            {orderedLessons.length} {orderedLessons.length === 1 ? 'aula' : 'aulas'} ·{' '}
   402→            {formatTotalDuration(course.totalDurationMin)}
   403→          </p>
   404→        </div>
   405→        <Button
   406→          variant="ghost"
   407→          size="icon"
   408→          className="h-9 w-9 shrink-0 rounded-full text-emerald-100 hover:bg-white/10 hover:text-white lg:hidden"
   409→          aria-label="Ver conteúdos do curso"
   410→          onClick={() => setContentsOpen(true)}
   411→        >
   412→          <PanelRight aria-hidden className="h-4 w-4" />
   413→        </Button>
   414→        <Button
   415→          variant="ghost"
   416→          size="icon"
   417→          className="h-9 w-9 shrink-0 rounded-full text-emerald-100 hover:bg-white/10 hover:text-white"
   418→          aria-label={focusMode ? 'Sair do modo foco' : 'Ativar modo foco'}
   419→          title={focusMode ? 'Sair do modo foco' : 'Ativar modo foco'}
   420→          onClick={() => setFocusMode((v) => !v)}
   421→        >
   422→          {focusMode ? (
   423→            <Minimize2 aria-hidden className="h-4 w-4" />
   424→          ) : (
   425→            <Maximize2 aria-hidden className="h-4 w-4" />
   426→          )}
   427→        </Button>
   428→        <div className="hidden items-center gap-2 sm:flex" aria-hidden>
   429→          <Avatar
   430→            name={course.mentor.name}
   431→            src={course.mentor.avatarUrl}
   432→            size="sm"
   433→            className="h-6 w-6 text-[10px]"
   434→          />
   435→          <span className="max-w-36 truncate text-xs font-semibold text-emerald-100/70">
   436→            {course.mentor.name}
   437→          </span>
   438→        </div>
   439→        <div className="flex w-32 shrink-0 items-center gap-2 sm:w-44">
   440→          <Progress
   441→            value={percent}
   442→            aria-label="Progresso do curso"
   443→            className="h-2 bg-white/20 [&_[data-slot=progress-indicator]]:bg-emerald-400"
   444→          />
   445→          <span className="w-9 shrink-0 text-right text-xs font-bold tabular-nums text-emerald-100">
   446→            {percent}%
   447→          </span>
   448→        </div>
   449→      </div>
   450→
   451→      {/* ---------- CORPO: PLAYER + CONTEÚDOS (sidebar oculta no modo foco) ---------- */}
   452→      <div
   453→        className={cn(
   454→          'grid min-h-0 flex-1 grid-cols-1',
   455→          focusMode ? 'lg:grid-cols-1' : 'lg:grid-cols-[1fr_380px]'
   456→        )}
   457→      >
   458→        {/* Coluna esquerda: aula — conteúdo interno largo p/ vídeos grandes */}
   459→        <div className="min-h-0 overflow-y-auto p-4 sm:p-6 lg:p-8">
   460→          <div className={cn('mx-auto w-full', focusMode ? 'max-w-4xl' : 'max-w-5xl')}>
   461→            <AnimatePresence mode="wait">
   462→              {currentLesson ? (
   463→                <motion.div
   464→                  key={currentLesson.id}
   465→                  initial={{ opacity: 0, y: 12 }}
   466→                  animate={{ opacity: 1, y: 0 }}
   467→                  exit={{ opacity: 0, y: -8 }}
   468→                  transition={{ duration: 0.22 }}
   469→                >
   470→                {currentLesson.kind === 'LIVE' ? (
   471→                  <LivePanel lesson={currentLesson} isOwner={isOwner} />
   472→                ) : currentLesson.kind === 'READING' ? (
   473→                  <ReadingMaterial lesson={currentLesson} courseId={courseId} />
   474→                ) : currentLesson.kind === 'RECORDED' && currentLesson.videoUrl ? (
   475→                  <div className="overflow-hidden rounded-2xl bg-black shadow-2xl ring-1 ring-stone-900/10">
   476→                    <LessonPlayer lesson={currentLesson} />
   477→                  </div>
   478→                ) : (
   479→                  <LessonPlayer lesson={currentLesson} />
   480→                )}
   481→
   482→                {/* Celebração + certificado: 100% das aulas concluídas */}
   483→                {courseCompleted ? (
   484→                  <div className="mt-5 flex flex-col items-center gap-3 rounded-2xl border border-emerald-200 bg-emerald-50/80 p-6 text-center dark:border-emerald-900 dark:bg-emerald-950/50">
   485→                    <CheckCircle2 aria-hidden className="h-12 w-12 fill-emerald-600 text-white" />
   486→                    <p className="text-lg font-extrabold tracking-tight text-emerald-900 dark:text-emerald-200">
   487→                      Parabéns! Você concluiu este curso 🎉
   488→                    </p>
   489→                    {certificateCode ? (
   490→                      <Button
