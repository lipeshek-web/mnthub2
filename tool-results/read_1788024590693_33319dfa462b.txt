   501→                        {mentor.rating.toFixed(1)}
   502→                      </p>
   503→                      <div>
   504→                        <Stars rating={mentor.rating} size={17} />
   505→                        <p className="mt-1 text-xs text-stone-500">
   506→                          Baseado em {mentor.reviewCount} avaliações de sessões reais
   507→                        </p>
   508→                      </div>

   510→                  </Card>
   511→                  {mentor.reviews.map((r) => (

   513→                      <CardContent className="p-5">
   514→                        <div className="flex items-center gap-3">
   515→                          <Avatar name={r.authorName} size="md" />
   516→                          <div className="flex-1">
   517→                            <p className="text-sm font-bold text-stone-900">{r.authorName}</p>
   518→                            <p className="text-xs text-muted-foreground">
   519→                              {formatDayLabel(r.createdAt.replace('T00:00:00.000Z', 'T12:00'))}
   520→                            </p>
   521→                          </div>
   522→                          <Stars rating={r.rating} size={14} />
   523→                        </div>
   524→                        <p className="mt-3 text-sm leading-relaxed text-stone-600" style={bodyStyle}>
   525→                          {r.comment}
   526→                        </p>
   527→                      </CardContent>
   528→                    </Card>
   529→                  ))}
   530→                </>
   531→              )}
   532→            </TabsContent>
   533→          </Tabs>
   534→        </div>
   535→
   536→        <aside className="lg:sticky lg:top-24 lg:self-start">
   537→          <BookingWidget mentor={mentor} />

   539→      </div>
   540→    </div>
   541→  )

   546→function BookingWidget({ mentor }: { mentor: MentorDetailDTO }) {
   547→  const navigate = useAppStore((s) => s.navigate)
   548→  const user = useAppStore((s) => s.user)
   549→  const bookingTopic = useAppStore((s) => s.bookingTopic)
   550→  const setBookingTopic = useAppStore((s) => s.setBookingTopic)
   551→
   552→  const days = Array.from({ length: 14 }, (_, i) => addDays(new Date(), i))
   553→  const [selectedDate, setSelectedDate] = useState(() => dateKey(new Date()))
   554→  const [slots, setSlots] = useState<string[]>([])
   555→  const [slotsLoading, setSlotsLoading] = useState(true)
   556→  const [selectedSlot, setSelectedSlot] = useState<string | null>(null)
   557→  const [topic, setTopic] = useState(() => bookingTopic)
   558→  const [notes, setNotes] = useState('')
   559→  const [confirmOpen, setConfirmOpen] = useState(false)
   560→  const [submitting, setSubmitting] = useState(false)
   561→
   562→  // Consome o pré-preenchimento (mentoria inclusa em curso/trilha) uma única vez
   563→  useEffect(() => {
   564→    if (bookingTopic) {
   565→      setTopic((prev) => prev || bookingTopic)
   566→      setBookingTopic('')
   567→    }
   568→  }, [bookingTopic, setBookingTopic])
   569→
   570→  const loadSlots = useCallback(async (date: string) => {
   571→    setSlotsLoading(true)
   572→    setSelectedSlot(null)
   573→    try {
   574→      const { slots } = await api.getSlots(mentor.id, date)
   575→      setSlots(slots)
   576→    } catch {
   577→      setSlots([])
   578→    } finally {
   579→      setSlotsLoading(false)
   580→    }
   581→  }, [mentor.id])
   582→
   583→  useEffect(() => {
   584→    loadSlots(selectedDate)
   585→  }, [loadSlots, selectedDate])
   586→
   587→  const isOwner = user?.id === mentor.userId
   588→
   589→  const submit = async () => {
   590→    if (!user) {
   591→      toast.error('Entre com um usuário no topo da página para agendar.')
   592→      return
   593→    }
   594→    if (!selectedSlot) {
   595→      toast.error('Escolha um horário disponível.')
   596→      return
   597→    }
   598→    if (topic.trim().length < 5) {
   599→      toast.error('Descreva o tema da sessão (mín. 5 caracteres).')
   600→      return
   601→    }
   602→    setSubmitting(true)
   603→    try {
   604→      await api.createBooking({
   605→        menteeId: user.id,
   606→        mentorId: mentor.id,
   607→        startsAt: selectedSlot,
   608→        durationMin: 60,
   609→        topic: topic.trim(),
   610→        notes: notes.trim() || undefined,
   611→      })
   612→      toast.success('Solicitação enviada! O mentor confirmará em breve. 🎉')
   613→      navigate({ name: 'dashboard' })
   614→    } catch (e) {
   615→      toast.error(e instanceof Error ? e.message : 'Erro ao agendar')
   616→      loadSlots(selectedDate)
   617→    } finally {
   618→      setSubmitting(false)
   619→      setConfirmOpen(false)
   620→    }
   621→  }
   622→
   623→  if (isOwner) {
   624→    return (
   625→      <Card className="border-emerald-200">
   626→        <CardContent className="flex flex-col items-center gap-3 p-6 text-center">
   627→          <Avatar name={mentor.name} size="lg" />
   628→          <p className="font-bold">Este é o seu perfil público de mentor</p>
   629→          <p className="text-sm text-muted-foreground">
   630→            Gerencie suas informações, agenda e mural de conteúdos.
   631→          </p>
   632→          <Button className="w-full" onClick={() => navigate({ name: 'onboarding' })}>
   633→            Gerenciar meu perfil
   634→          </Button>
   635→        </CardContent>
   636→      </Card>
   637→    )
   638→  }
   639→
   640→  return (
   641→    <Card className="border-stone-200 shadow-md">
   642→      <CardContent className="space-y-5 p-6">
   643→        <div className="flex items-end justify-between">
   644→          <p className="text-3xl font-black tracking-tight text-stone-900">
   645→            {currencyBRL(mentor.hourlyRate)}
   646→            <span className="text-sm font-medium text-muted-foreground"> / sessão</span>
   647→          </p>
   648→          <Badge variant="secondary" className="bg-stone-100 text-stone-600">60 min · vídeo</Badge>
   649→        </div>
   650→
   651→        {/* Dias */}
   652→        <div>
   653→          <Label className="text-xs uppercase tracking-wide text-muted-foreground">Escolha o dia</Label>
   654→          <div className="mt-2 flex gap-2 overflow-x-auto pb-1.5 [scrollbar-width:thin]">
   655→            {days.map((d) => {
   656→              const key = dateKey(d)
   657→              const active = key === selectedDate
   658→              return (
   659→                <button
   660→                  key={key}
   661→                  onClick={() => setSelectedDate(key)}
   662→                  aria-pressed={active}
   663→                  aria-label={`${WEEKDAYS_PT[d.getDay()]}, ${d.getDate()} de ${MONTHS_PT[d.getMonth()]}`}
   664→                  className={cn(
   665→                    'flex h-[62px] w-[52px] shrink-0 flex-col items-center justify-center rounded-xl border text-center transition-all',
   666→                    active
   667→                      ? 'border-emerald-700 bg-emerald-700 text-white shadow-md'
   668→                      : 'border-stone-200 bg-white text-stone-600 hover:border-emerald-300 hover:bg-emerald-50'
   669→                  )}
   670→                >
   671→                  <span className={cn('text-[10px] font-semibold uppercase', active ? 'text-emerald-200' : 'text-stone-400')}>
   672→                    {WEEKDAYS_PT[d.getDay()]}
   673→                  </span>
   674→                  <span className="text-lg font-extrabold leading-tight">{d.getDate()}</span>
   675→                  <span className={cn('text-[10px]', active ? 'text-emerald-200' : 'text-stone-400')}>
   676→                    {MONTHS_PT[d.getMonth()]}
   677→                  </span>
   678→                </button>
   679→              )
   680→            })}
   681→          </div>
   682→        </div>
   683→
   684→        {/* Horários */}
   685→        <div>
   686→          <Label className="text-xs uppercase tracking-wide text-muted-foreground">
   687→            Horários livres · {formatDayLabel(`${selectedDate}T12:00`)}
   688→          </Label>
   689→          {slotsLoading ? (
   690→            <div className="mt-2 grid grid-cols-4 gap-2">
   691→              {Array.from({ length: 4 }).map((_, i) => (
   692→                <Skeleton key={i} className="h-10 rounded-lg" />
   693→              ))}
   694→            </div>
   695→          ) : slots.length === 0 ? (
   696→            <div className="mt-2 flex items-center gap-2 rounded-xl border border-dashed border-stone-200 bg-stone-50 p-4 text-sm text-stone-500">
   697→              <CalendarOff className="h-4 w-4 shrink-0" />
   698→              Sem horários livres neste dia. Tente outra data.
   699→            </div>
   700→          ) : (
   701→            <div className="mt-2 grid max-h-44 grid-cols-4 gap-2 overflow-y-auto pr-1">
   702→              {slots.map((s) => {
   703→                const active = s === selectedSlot
   704→                return (

   706→                    key={s}
   707→                    onClick={() => setSelectedSlot(s)}
   708→                    aria-pressed={active}
   709→                    className={cn(
   710→                      'h-10 rounded-lg border text-sm font-semibold transition-all',
   711→                      active
   712→                        ? 'border-emerald-700 bg-emerald-700 text-white shadow'
   713→                        : 'border-stone-200 bg-white text-stone-700 hover:border-emerald-400 hover:text-emerald-700'
   714→                    )}
   715→                  >
   716→                    {formatTimeLabel(s)}
   717→                  </button>

   719→              })}

   768→          {submitting ? 'Enviando...' : 'Solicitar agendamento'}

   796→                  `${formatTimeLabel(selectedSlot)} → ${addMinutesToTime(selectedSlot, 60)}`}
