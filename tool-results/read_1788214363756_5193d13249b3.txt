    62→async function request<T>(url: string, options?: RequestInit): Promise<T> {
    63→  // Timeout padrão de 15s: rede travada não deixa skeleton eterno nem empilha polls.
    64→  // Callers podem passar o próprio signal (AbortController) para cancelar.
    65→  const signal = options?.signal ?? AbortSignal.timeout(15_000)
    66→  let res: Response
    67→  try {
    68→    res = await fetch(url, {
    69→      ...options,
    70→      cache: 'no-store',
    71→      signal,
    72→      headers: {
    73→        ...(hasBody(options) ? { 'Content-Type': 'application/json' } : {}),
    74→        ...authHeaders(),
    75→        ...options?.headers,
    76→      },
    77→    })
    78→  } catch (err) {
    79→    if (err instanceof DOMException && err.name === 'TimeoutError') {
    80→      throw new Error('A conexão demorou demais. Verifique sua internet e tente de novo.')
    81→    }
    82→    throw err
    83→  }
    84→  let json: unknown = {}
    85→  try {
    86→    json = await res.json()
    87→  } catch {
    88→    json = {}
    89→  }
    90→  if (!res.ok) {
    91→    // Sessão inválida/expirada: limpa a sessão local e pede novo login
    92→    if (res.status === 401) {
    93→      clearSession()
    94→      const msg401 = (json as { error?: string })?.error || 'Sessão expirada. Entre novamente.'
    95→      throw new Error(msg401)
    96→    }
    97→    const msg = (json as { error?: string })?.error || 'Ocorreu um erro inesperado. Tente novamente.'
    98→    throw new Error(msg)
    99→  }
   100→  return json as T
   101→}

   103→/** true quando a requisição tem corpo (POST/PUT/PATCH) — só então envia Content-Type JSON */
   104→function hasBody(options?: RequestInit): boolean {
   105→  const method = (options?.method || 'GET').toUpperCase()
   106→  return options?.body != null || method === 'POST' || method === 'PUT' || method === 'PATCH'
   107→}

   109→/** Limpa a sessão local (token + usuário) quando o servidor rejeita a sessão */
   110→function clearSession() {
   111→  if (typeof window === 'undefined') return
   112→  try {
   113→    window.localStorage.removeItem('mentorhub-session')
   114→    window.dispatchEvent(new CustomEvent('mentorhub:session-expired'))
   115→  } catch {
   116→    /* ignora */
   117→  }
   118→}

   120→/**
   121→ * Header Authorization com o token de sessão assinado (emitido no login/
   122→ * registro/verificação MFA e persistido junto ao usuário no zustand).
   123→ * Anexado em TODAS as chamadas — o servidor decide quais rotas exigem.
   124→ */
   125→function authHeaders(): Record<string, string> {
   126→  try {
   127→    const token = useAppStore.getState().user?.sessionToken
   128→    return token ? { Authorization: `Bearer ${token}` } : {}
   129→  } catch {
   130→    return {}
   131→  }
   132→}

   134→const qs = (params: Record<string, string | number | undefined>) => {
   135→  const sp = new URLSearchParams()
   136→  Object.entries(params).forEach(([k, v]) => {
   137→    if (v !== undefined && v !== '') sp.set(k, String(v))
   138→  })
   139→  const s = sp.toString()
   140→  return s ? `?${s}` : ''
   141→}
