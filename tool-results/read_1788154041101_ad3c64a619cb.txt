   302→  }, [baseCourses])

   313→
   314→  const trackStats = useMemo(

   355→  const allView = useMemo(() => {

   359→      q === '' || parts.some((p) => normalizeText(p ?? '').includes(q))

   447→  const searchTotal =
   448→    tab === 'all'

   450→      : tab === 'mentors'
   451→        ? mentors.length
   452→        : tab === 'courses'
   453→          ? courses.length
   454→          : tab === 'tracks'
   455→            ? tracks.length
   456→            : tab === 'library'
   457→              ? libItems.length
   458→              : null
   459→  const clearAllFilters = clearAllSearch
   460→
   461→  // Títulos por aba (modo navegar)
   462→  const tabTitle =
   463→    tab === 'all'
   464→      ? 'Explorar tudo'
   465→      : tab === 'mentors'
   466→        ? 'Explorar mentores'
   467→        : tab === 'courses'
   468→          ? 'Explorar cursos'
   469→          : tab === 'tracks'
   470→            ? 'Explorar trilhas'
   471→            : tab === 'bundles'
   472→              ? 'Pacotes de cursos'
   473→              : 'Explorar a Biblioteca'
   474→  const tabSubtitle =
   475→    tab === 'all'
   476→      ? totalContents === 0
   477→        ? 'Carregando conteúdos...'
   478→        : `${totalContents} conteúdos — ${baseMentors.length} mentores, ${baseCourses.length} cursos, ${baseTracks.length} trilhas e ${baseLibItems.length} leituras`
   479→      : tab === 'mentors'
   480→        ? loading
   481→          ? 'Carregando especialistas...'
   482→          : `${mentors.length} ${mentors.length === 1 ? 'especialista' : 'especialistas'} pronto${
   483→              mentors.length === 1 ? '' : 's'
   484→            } para mentoria 1:1`
   485→        : tab === 'courses'
   486→          ? coursesLoading
   487→            ? 'Carregando cursos...'
   488→            : `${courses.length} ${courses.length === 1 ? 'curso publicado' : 'cursos publicados'}`
   489→          : tab === 'tracks'
   490→            ? tracksLoading
   491→              ? 'Carregando trilhas...'
   492→              : `${tracks.length} ${tracks.length === 1 ? 'trilha publicada' : 'trilhas publicadas'}`
   493→            : tab === 'bundles'
   494→              ? 'Vários cursos do mesmo mentor por um preço especial — economize comprando o conjunto'
   495→              : libLoading
   496→                ? 'Carregando conteúdos...'
   497→                : `${libItems.length} ${libItems.length === 1 ? 'conteúdo publicado' : 'conteúdos publicados'}`
   498→  const searchSubtitle =

   502→
   503→  return (
   504→    <div>
   505→      {/* ---------- BARRA SUPERIOR: título, ordenação, busca e categorias ---------- */}

   520→                tab === 'all'
   521→                  ? 'bg-white dark:bg-stone-900 text-stone-900 dark:text-stone-50 shadow-sm'
   522→                  : 'text-stone-500 dark:text-stone-400 hover:text-stone-700 dark:hover:text-stone-200'
