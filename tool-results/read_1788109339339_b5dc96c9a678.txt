     2→
     3→import { useEffect, useState, type FormEvent } from 'react'
     4→import {
     5→  Eye,
     6→  EyeOff,
     7→  GraduationCap,
     8→  Library,
     9→  Loader2,
    10→  ShieldCheck,
    11→  Users,
    12→  Video,
    13→} from 'lucide-react'
    14→import { toast } from 'sonner'
    15→import { Button } from '@/components/ui/button'
    16→import { Input } from '@/components/ui/input'
    17→import { Label } from '@/components/ui/label'
    18→import { Skeleton } from '@/components/ui/skeleton'
    19→import { Tabs, TabsContent, TabsList, TabsTrigger } from '@/components/ui/tabs'
    20→import {
    21→  InputOTP,
    22→  InputOTPGroup,
    23→  InputOTPSlot,
    24→} from '@/components/ui/input-otp'
    25→import { Avatar, Stars } from '@/components/platform/avatar'
    26→import { api } from '@/lib/api'
    27→import { firstName } from '@/lib/helpers'
    28→import { clearStoredRefCode, getStoredRefCode } from '@/lib/referral'
    29→import { useAppStore } from '@/lib/store'
    30→import type { UserDTO } from '@/lib/types'
    31→
    32→const EMAIL_RE = /^[^\s@]+@[^\s@]+\.[^\s@]+$/
    33→const DEMO_PASSWORD = 'demo123'
    34→
    35→const PANEL_BULLETS = [
    36→  { icon: Video, text: 'Cursos com aulas ao vivo, temas e materiais' },
    37→  { icon: Users, text: 'Mentorias 1:1 com especialistas de verdade' },
    38→  { icon: Library, text: 'Biblioteca com artigos e livros para ler na plataforma' },
    39→]
    40→
    41→interface FieldErrors {
    42→  name?: string
    43→  email?: string
    44→  password?: string
    45→}
    46→
    47→export function AuthView({ initialMode }: { initialMode?: 'login' | 'register' }) {
    48→  const setUser = useAppStore((s) => s.setUser)
    49→  const navigate = useAppStore((s) => s.navigate)
    50→
    51→  const [tab, setTab] = useState<'login' | 'register'>(initialMode ?? 'login')
    52→
    53→  // ----- Entrar -----
    54→  const [loginEmail, setLoginEmail] = useState('')
    55→  const [loginPassword, setLoginPassword] = useState('')
    56→  const [showLoginPassword, setShowLoginPassword] = useState(false)
    57→  const [loginErrors, setLoginErrors] = useState<FieldErrors>({})
    58→  const [loginFormError, setLoginFormError] = useState<string | null>(null)
    59→  const [loginLoading, setLoginLoading] = useState(false)
    60→
    61→  // ----- Criar conta -----
    62→  const [regName, setRegName] = useState('')
    63→  const [regEmail, setRegEmail] = useState('')
    64→  const [regPassword, setRegPassword] = useState('')
    65→  const [showRegPassword, setShowRegPassword] = useState(false)
    66→  const [regErrors, setRegErrors] = useState<FieldErrors>({})
    67→  const [regFormError, setRegFormError] = useState<string | null>(null)
    68→  const [regLoading, setRegLoading] = useState(false)
    69→
    70→  // ----- Contas de demonstração -----
    71→  const [demoUsers, setDemoUsers] = useState<UserDTO[]>([])
    72→  const [demoLoading, setDemoLoading] = useState(true)
    73→  const [demoBusyId, setDemoBusyId] = useState<string | null>(null)
    74→
    75→  // ----- MFA (segundo fator): ativo após senha correta de conta com MFA -----
    76→  const [mfaTicket, setMfaTicket] = useState<string | null>(null)
    77→  const [mfaCode, setMfaCode] = useState('')
    78→  const [mfaLoading, setMfaLoading] = useState(false)
    79→  const [mfaError, setMfaError] = useState<string | null>(null)
    80→  const [mfaMode, setMfaMode] = useState<'totp' | 'recovery'>('totp')
    81→  const [mfaRecoveryCode, setMfaRecoveryCode] = useState('')
    82→
    83→  // Sincroniza quando a view chega com outro modo (ex.: clique em "Criar conta" no navbar)
    84→  useEffect(() => {
    85→    if (initialMode) setTab(initialMode)
    86→  }, [initialMode])
    87→
    88→  useEffect(() => {
    89→    let active = true
    90→    api
    91→      .listUsers()
    92→      .then((users) => {
    93→        if (active) setDemoUsers(users)
    94→      })
    95→      .catch(() => {
    96→        if (active) setDemoUsers([])
    97→      })
    98→      .finally(() => {
    99→        if (active) setDemoLoading(false)
   100→      })
   101→    return () => {
   102→      active = false
   103→    }
   104→  }, [])
   105→
   106→  /** Ao trocar de tab, limpa os erros inline */
   107→  const switchTab = (next: string) => {
   108→    setTab(next as 'login' | 'register')
   109→    setLoginErrors({})
   110→    setLoginFormError(null)
   111→    setRegErrors({})
   112→    setRegFormError(null)
   113→  }
   114→
   115→  // ---------- Entrar ----------
   116→  const handleLogin = async (e: FormEvent<HTMLFormElement>) => {
   117→    e.preventDefault()
   118→    if (loginLoading) return // bloqueia duplo submit
   119→
   120→    const errors: FieldErrors = {}
   121→    if (!EMAIL_RE.test(loginEmail.trim())) errors.email = 'Informe um e-mail válido.'
   122→    if (loginPassword.length < 6) errors.password = 'A senha deve ter pelo menos 6 caracteres.'
   123→    setLoginErrors(errors)
   124→    if (Object.keys(errors).length > 0) return
   125→
   126→    setLoginLoading(true)
   127→    setLoginFormError(null)
   128→    try {
   129→      const result = await api.login({ email: loginEmail.trim(), password: loginPassword })
   130→
   131→      // Conta com MFA: mostra a etapa do código de 6 dígitos
   132→      if ('mfaRequired' in result && result.mfaRequired) {
   133→        setMfaTicket(result.mfaTicket)
   134→        setMfaCode('')
   135→        setMfaError(null)
   136→        setMfaMode('totp')
   137→        setMfaRecoveryCode('')
   138→        toast.info('Confirme com o código do seu app autenticador 🔐')
   139→        return
   140→      }
   141→
   142→      const user = result as UserDTO
   143→      setUser(user)
   144→      toast.success(`Bem-vindo(a) de volta, ${firstName(user.name)}! 👋`)
   145→      navigate({ name: 'home' })
   146→    } catch (err) {
   147→      const msg = err instanceof Error ? err.message : 'Não foi possível entrar. Tente novamente.'
   148→      toast.error(msg)
   149→      setLoginFormError(msg)
   150→    } finally {
   151→      setLoginLoading(false)
   152→    }
   153→  }
   154→
   155→  // ---------- MFA: verifica o código TOTP ou de recuperação ----------
   156→  const handleMfaVerify = async (code?: string) => {
   157→    if (!mfaTicket || mfaLoading) return
   158→    const value =
   159→      mfaMode === 'totp'
   160→        ? (code ?? mfaCode).replace(/\D/g, '')
   161→        : (code ?? mfaRecoveryCode).trim().toUpperCase()
   162→    if (mfaMode === 'totp' && value.length !== 6) return
   163→    if (mfaMode === 'recovery' && value.length < 6) return
   164→    setMfaLoading(true)
   165→    setMfaError(null)
   166→    try {
   167→      const user = await api.verifyMfa({ ticket: mfaTicket, code: value })
   168→      setUser(user)
   169→      if (user.usedRecoveryCode) {
   170→        toast.warning(
   171→          `Você entrou com um código de recuperação. Restam ${user.recoveryCodesRemaining ?? 0} — gere novos no painel admin. ⚠️`
   172→        )
   173→      } else {
   174→        toast.success(`Bem-vindo(a) de volta, ${firstName(user.name)}! 👋`)
   175→      }
   176→      setMfaTicket(null)
   177→      setMfaCode('')
   178→      setMfaRecoveryCode('')
   179→      setMfaMode('totp')
   180→      navigate({ name: 'home' })
   181→    } catch (err) {
   182→      const msg = err instanceof Error ? err.message : 'Código inválido.'
   183→      setMfaError(msg)
   184→      setMfaCode('')
   185→      setMfaRecoveryCode('')
   186→    } finally {
   187→      setMfaLoading(false)
   188→    }
   189→  }
   190→
   191→  // ---------- Criar conta ----------
   192→  const handleRegister = async (e: FormEvent<HTMLFormElement>) => {
   193→    e.preventDefault()
   194→    if (regLoading) return // bloqueia duplo submit
   195→
   196→    const errors: FieldErrors = {}
   197→    if (regName.trim().length < 2) errors.name = 'Informe seu nome completo.'
   198→    if (!EMAIL_RE.test(regEmail.trim())) errors.email = 'Informe um e-mail válido.'
   199→    if (regPassword.length < 6) errors.password = 'A senha deve ter pelo menos 6 caracteres.'
   200→    setRegErrors(errors)
   201→    if (Object.keys(errors).length > 0) return
   202→
   203→    setRegLoading(true)
   204→    setRegFormError(null)
   205→    try {
   206→      // Convite capturado em ?ref= (localStorage, janela de 7 dias)
   207→      const refCode = getStoredRefCode() ?? undefined
   208→      const user = await api.register({
   209→        name: regName.trim(),
   210→        email: regEmail.trim(),
   211→        password: regPassword,
   212→        refCode,
   213→      })
   214→      if (refCode) clearStoredRefCode()
   215→      setUser(user)
   216→      toast.success(
   217→        user.referralApplied
   218→          ? `Conta criada! R$ 10 de crédito de convite adicionados, ${firstName(user.name)}! 🎁`
   219→          : `Conta criada! Bem-vindo(a), ${firstName(user.name)}! 🎉`
   220→      )
   221→      navigate({ name: 'home' })
   222→    } catch (err) {
   223→      const msg =
   224→        err instanceof Error ? err.message : 'Não foi possível criar sua conta. Tente novamente.'
   225→      toast.error(msg)
   226→      if (msg.includes('Já existe') || msg.includes('e-mail')) {
   227→        setRegErrors((prev) => ({ ...prev, email: 'Já existe conta com este e-mail. Faça login.' }))
   228→      } else {
   229→        setRegFormError(msg)
   230→      }
   231→    } finally {
   232→      setRegLoading(false)
   233→    }
   234→  }
   235→
   236→  // ---------- Contas demo ----------
   237→  const handleDemoLogin = async (u: UserDTO) => {
   238→    if (demoBusyId) return
   239→    setDemoBusyId(u.id)
   240→    try {
   241→      const result = await api.login({ email: u.email, password: DEMO_PASSWORD })
   242→      // Contas demo não têm MFA — mas o tipo cobre o caso
   243→      if ('mfaRequired' in result && result.mfaRequired) {
   244→        setMfaTicket(result.mfaTicket)
   245→        toast.info('Confirme com o código do seu app autenticador 🔐')
   246→        return
   247→      }
   248→      const user = result as UserDTO
   249→      setUser(user)
   250→      toast.success(`Entrou como ${user.name}`)
   251→      navigate({ name: 'home' })
   252→    } catch {
   253→      toast.info('Esta conta demo ainda está sendo preparada. Crie a sua!')
   254→    } finally {
   255→      setDemoBusyId(null)
   256→    }
   257→  }
   258→
   259→  const handleForgotPassword = () => {
   260→    toast.info('Em breve! Por enquanto, use uma conta demo ou crie a sua.')
   261→  }
   262→
   263→  return (
   264→    <div className="grid min-h-full bg-stone-50 dark:bg-stone-950 lg:grid-cols-2">
   265→      {/* ---------- Painel esquerdo (header compacto no mobile, completo no lg+) ---------- */}
   266→      <section
   267→        aria-label="Sobre o MentorHub"
   268→        className="relative flex flex-col gap-10 overflow-hidden bg-gradient-to-br from-emerald-800 via-emerald-700 to-teal-700 p-6 text-white lg:justify-between lg:gap-12 lg:p-10 xl:p-14"
   269→      >
   270→        {/* Blobs decorativos */}
   271→        <div aria-hidden className="pointer-events-none absolute inset-0">
   272→          <div className="absolute -left-28 -top-28 h-80 w-80 rounded-full bg-white/10 blur-3xl" />
   273→          <div className="absolute -right-24 top-1/3 h-96 w-96 rounded-full bg-white/10 blur-3xl" />
   274→          <div className="absolute -bottom-32 left-1/4 h-72 w-72 rounded-full bg-white/10 blur-3xl" />
   275→        </div>
   276→
   277→        {/* Logo */}
   278→        <div className="relative flex items-center gap-3">
   279→          <span className="flex h-10 w-10 items-center justify-center rounded-xl bg-white/15">
   280→            <GraduationCap className="h-5 w-5" aria-hidden />
   281→          </span>
   282→          <span className="text-lg font-extrabold tracking-tight">MentorHub</span>
   283→        </div>
   284→
   285→        {/* Headline + bullets */}
   286→        <div className="relative">
   287→          <h1 className="max-w-lg text-2xl font-extrabold leading-tight tracking-tight lg:text-4xl lg:leading-[1.1]">
   288→            Aprenda com quem vive o que ensina.
   289→          </h1>
   290→          <p className="mt-3 max-w-md text-sm text-emerald-100/80 lg:mt-4 lg:text-base">
   291→            Cursos ao vivo, mentorias 1:1 e biblioteca — tudo o que você precisa para evoluir de
   292→            verdade, em um só lugar.
   293→          </p>
   294→          <ul className="mt-8 hidden space-y-4 lg:block">
   295→            {PANEL_BULLETS.map(({ icon: Icon, text }) => (
   296→              <li key={text} className="flex items-center gap-3.5">
   297→                <span className="flex h-10 w-10 shrink-0 items-center justify-center rounded-full bg-white/10">
   298→                  <Icon className="h-5 w-5" aria-hidden />
   299→                </span>
   300→                <span className="text-sm text-emerald-50/90">{text}</span>
   301→              </li>
   302→            ))}
   303→          </ul>
   304→        </div>
   305→
   306→        {/* Prova social (rodapé do painel) */}
   307→        <div className="relative hidden lg:block">
   308→          <div className="max-w-md rounded-2xl border border-white/15 bg-white/10 p-4">
   309→            <Stars rating={5} size={16} />
   310→            <p className="mt-2 text-sm text-emerald-50/90">
   311→              4,9 de média · +2.400 alunos aprendendo todos os dias
   312→            </p>
   313→          </div>
   314→        </div>
   315→      </section>
   316→
   317→      {/* ---------- Painel direito: formulário ---------- */}
   318→      <section className="flex items-center justify-center p-4 sm:p-6 lg:p-8 xl:p-12">
   319→        <div className="w-full max-w-md rounded-2xl border border-stone-200 bg-white dark:border-stone-800 dark:bg-stone-900 p-6 shadow-sm sm:p-8">
   320→          <Tabs value={tab} onValueChange={switchTab}>
   321→            <TabsList className="grid h-11 w-full grid-cols-2 rounded-full bg-stone-100 dark:bg-stone-800 p-1">
   322→              <TabsTrigger
   323→                value="login"
   324→                className="rounded-full text-sm font-semibold text-stone-500 data-[state=active]:bg-white data-[state=active]:text-stone-900 data-[state=active]:shadow-sm dark:text-stone-400 dark:data-[state=active]:bg-stone-900 dark:data-[state=active]:text-stone-50"
   325→              >
   326→                Entrar
   327→              </TabsTrigger>
   328→              <TabsTrigger
   329→                value="register"
   330→                className="rounded-full text-sm font-semibold text-stone-500 data-[state=active]:bg-white data-[state=active]:text-stone-900 data-[state=active]:shadow-sm dark:text-stone-400 dark:data-[state=active]:bg-stone-900 dark:data-[state=active]:text-stone-50"
   331→              >
   332→                Criar conta
   333→              </TabsTrigger>
   334→            </TabsList>
   335→
   336→            {/* ----- Tab Entrar ----- */}
   337→            <TabsContent value="login" className="mt-6">
   338→              {mfaTicket ? (
   339→                <div className="flex flex-col items-center text-center">
   340→                  <span className="flex h-14 w-14 items-center justify-center rounded-2xl bg-emerald-50 text-emerald-700 dark:bg-emerald-950/60 dark:text-emerald-300">
   341→                    <ShieldCheck className="h-7 w-7" aria-hidden />
   342→                  </span>
   343→                  <h2 className="mt-4 text-xl font-extrabold tracking-tight text-stone-900 dark:text-stone-50">
   344→                    Verificação em duas etapas
   345→                  </h2>
   346→                  <p className="mt-1.5 max-w-xs text-sm leading-relaxed text-stone-500 dark:text-stone-400">
   347→                    {mfaMode === 'totp' ? (
   348→                      <>Digite o código de 6 dígitos do seu app autenticador (Google Authenticator, Authy…).</>
   349→                    ) : (
   350→                      <>Digite um dos seus códigos de recuperação (formato XXXX-XXXX). Cada código funciona uma única vez.</>
   351→                    )}
   352→                  </p>
   353→
   354→                  <div className="mt-6">
   355→                    {mfaMode === 'totp' ? (
   356→                      <InputOTP
   357→                        maxLength={6}
   358→                        value={mfaCode}
   359→                        onChange={(v) => {
   360→                          setMfaCode(v)
   361→                          setMfaError(null)
   362→                          if (v.length === 6) void handleMfaVerify(v)
   363→                        }}
   364→                        disabled={mfaLoading}
   365→                      >
   366→                        <InputOTPGroup>
   367→                          {[0, 1, 2, 3, 4, 5].map((i) => (
   368→                            <InputOTPSlot key={i} index={i} className="h-12 w-11 rounded-lg border-stone-300 text-lg font-bold dark:border-stone-700" />
   369→                          ))}
   370→                        </InputOTPGroup>
   371→                      </InputOTP>
   372→                    ) : (
   373→                      <Input
   374→                        value={mfaRecoveryCode}
   375→                        onChange={(e) => {
   376→                          setMfaRecoveryCode(e.target.value.toUpperCase())
   377→                          setMfaError(null)
   378→                        }}
   379→                        onKeyDown={(e) => {
   380→                          if (e.key === 'Enter') void handleMfaVerify()
   381→                        }}
   382→                        placeholder="XXXX-XXXX"
   383→                        autoComplete="off"
   384→                        disabled={mfaLoading}
   385→                        className="h-12 w-56 rounded-xl text-center font-mono text-lg font-bold tracking-widest uppercase"
   386→                        aria-label="Código de recuperação"
   387→                      />
   388→                    )}
   389→                  </div>
   390→
   391→                  {mfaError && (
   392→                    <p role="alert" className="mt-4 text-xs font-medium text-rose-600 dark:text-rose-400">
   393→                      {mfaError}
   394→                    </p>
   395→                  )}
   396→
   397→                  <Button
   398→                    onClick={() => void handleMfaVerify()}
   399→                    disabled={
   400→                      mfaLoading ||
   401→                      (mfaMode === 'totp'
   402→                        ? mfaCode.replace(/\D/g, '').length !== 6
   403→                        : mfaRecoveryCode.trim().length < 6)
   404→                    }
   405→                    className="mt-6 h-11 w-full rounded-full text-sm font-bold"
   406→                  >
   407→                    {mfaLoading ? (
   408→                      <>
   409→                        <Loader2 className="h-4 w-4 animate-spin" aria-hidden />
   410→                        Verificando…
   411→                      </>
   412→                    ) : (
   413→                      'Confirmar código'
   414→                    )}
   415→                  </Button>
   416→                  <button
   417→                    type="button"
   418→                    onClick={() => {
   419→                      if (mfaMode === 'recovery') {
   420→                        setMfaMode('totp')
   421→                        setMfaRecoveryCode('')
   422→                      } else {
   423→                        setMfaMode('recovery')
   424→                        setMfaCode('')
   425→                      }
   426→                      setMfaError(null)
   427→                    }}
   428→                    className="mt-4 text-xs font-medium text-stone-400 underline-offset-2 transition-colors hover:text-stone-600 hover:underline dark:text-stone-500 dark:hover:text-stone-300"
   429→                  >
   430→                    {mfaMode === 'recovery'
   431→                      ? 'Usar o app autenticador'
   432→                      : 'Não tem o app? Usar código de recuperação'}
   433→                  </button>
   434→                  <button
   435→                    type="button"
   436→                    onClick={() => {
   437→                      setMfaTicket(null)
   438→                      setMfaCode('')
   439→                      setMfaRecoveryCode('')
   440→                      setMfaError(null)
   441→                      setMfaMode('totp')
   442→                    }}
   443→                    className="mt-2 text-xs font-medium text-stone-400 underline-offset-2 transition-colors hover:text-stone-600 hover:underline dark:text-stone-500 dark:hover:text-stone-300"
   444→                  >
   445→                    Voltar para o login
   446→                  </button>
   447→                </div>
   448→              ) : (
   449→              <form onSubmit={handleLogin} noValidate className="space-y-4">
   450→                <div>
   451→                  <h2 className="text-xl font-extrabold tracking-tight text-stone-900 dark:text-stone-50">
   452→                    Bem-vindo de volta
   453→                  </h2>
   454→                  <p className="mt-1 text-sm text-stone-500 dark:text-stone-400">
   455→                    Entre para continuar sua jornada de aprendizado.
   456→                  </p>
   457→                </div>
   458→
   459→                <div className="space-y-2">
   460→                  <Label htmlFor="login-email" className="text-sm font-medium text-stone-700 dark:text-stone-200">
   461→                    E-mail
   462→                  </Label>
   463→                  <Input
   464→                    id="login-email"
   465→                    type="email"
   466→                    value={loginEmail}
   467→                    onChange={(e) => {
   468→                      setLoginEmail(e.target.value)
   469→                      if (loginErrors.email)
   470→                        setLoginErrors((prev) => ({ ...prev, email: undefined }))
   471→                    }}
   472→                    placeholder="voce@exemplo.com"
   473→                    autoComplete="email"
   474→                    className="h-11 rounded-xl"
   475→                    aria-invalid={Boolean(loginErrors.email)}
   476→                    aria-describedby={loginErrors.email ? 'login-email-error' : undefined}
   477→                  />
   478→                  {loginErrors.email && (
   479→                    <p id="login-email-error" className="text-xs text-rose-600 dark:text-rose-400">
   480→                      {loginErrors.email}
   481→                    </p>
   482→                  )}
   483→                </div>
   484→
   485→                <div className="space-y-2">
   486→                  <Label htmlFor="login-password" className="text-sm font-medium text-stone-700 dark:text-stone-200">
   487→                    Senha
   488→                  </Label>
   489→                  <div className="relative">
   490→                    <Input
   491→                      id="login-password"
   492→                      type={showLoginPassword ? 'text' : 'password'}
   493→                      value={loginPassword}
   494→                      onChange={(e) => {
   495→                        setLoginPassword(e.target.value)
   496→                        if (loginErrors.password)
   497→                          setLoginErrors((prev) => ({ ...prev, password: undefined }))
   498→                      }}
   499→                      placeholder="Sua senha"
   500→                      autoComplete="current-password"
   501→                      className="h-11 rounded-xl pr-11"
   502→                      aria-invalid={Boolean(loginErrors.password)}
   503→                      aria-describedby={loginErrors.password ? 'login-password-error' : undefined}
   504→                    />
