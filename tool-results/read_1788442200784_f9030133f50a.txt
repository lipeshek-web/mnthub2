     1→/**
     2→ * Detalhe do mentor + agendamento de sessão 1:1:
     3→ * - Perfil: avatar, headline, stats (avaliação/preço/experiência), categorias,
     4→ *   idiomas, redes sociais (Linking) e avaliações (Stars + comentário).
     5→ * - Agendamento: chips de dia (próximos 14 dias) → horários livres do dia
     6→ *   (listMentorSlots) → tema obrigatório → createBooking (sessão de 60 min).
     7→ *   409 mostra a mensagem da API e recarrega os horários. Sucesso mostra a
     8→ *   confirmação (status PENDING — aguardando o mentor) com atalho para as
     9→ *   Minhas sessões.
    10→ */

    55→type Stage = "profile" | "booking" | "success";
    56→
    57→export default function MentorDetailScreen() {
    58→  const params = (useRoute<any>().params ?? {}) as { id: string };
    59→  const mentorId = Array.isArray(params.id) ? params.id[0] : params.id;
    60→
    61→  const navigation = useNavigation<any>();
    62→
    63→  const [data, setData] = useState<MentorDetailResponse | null>(null);
    64→  const [loading, setLoading] = useState(true);
    65→  const [refreshing, setRefreshing] = useState(false);
    66→  const [error, setError] = useState<string | null>(null);
    67→
    68→  const [stage, setStage] = useState<Stage>("profile");
    69→  const [created, setCreated] = useState<BookingCreated | null>(null);
    70→  const [createdStartsAt, setCreatedStartsAt] = useState("");
    71→
    72→  const load = useCallback(
    73→    async (mode: "initial" | "refresh") => {
    74→      if (!mentorId) return;
    75→      if (mode === "initial") setLoading(true);
    76→      else setRefreshing(true);
    77→      setError(null);
    78→      try {
    79→        const res = await getMentor(mentorId);
    80→        setData(res);
    81→      } catch (err) {
    82→        setError(errMessage(err));
    83→      } finally {
    84→        setLoading(false);
    85→        setRefreshing(false);
    86→      }
    87→    },
    88→    [mentorId]
    89→  );
    90→
    91→  useEffect(() => {
    92→    void load("initial");
    93→  }, [load]);
    94→
    95→  function handleCreated(booking: BookingCreated, startsAt: string) {
    96→    setCreated(booking);
    97→    setCreatedStartsAt(startsAt);
    98→    setStage("success");
    99→  }
   100→
   101→  function resetToProfile() {
   102→    setCreated(null);
   103→    setCreatedStartsAt("");
   104→    setStage("profile");
   105→  }
   106→
   107→  const mentor = data?.mentor ?? null;
   108→
   109→  return (
   110→    <Screen edges={["top", "left", "right", "bottom"]}>
   111→      {loading ? (
   112→        <LoadingList label="Carregando perfil..." />
   113→      ) : error && !data ? (
   114→        <ErrorBox message={error} onRetry={() => void load("initial")} />
   115→      ) : data && mentor ? (
   116→        stage === "booking" ? (
   117→          <BookingScheduler
   118→            mentor={mentor}
   119→            onCancel={() => setStage("profile")}
   120→            onCreated={handleCreated}
   121→          />
   122→        ) : stage === "success" ? (
   123→          <BookingSuccess
   124→            mentorName={mentor.name}
   125→            created={created}
   126→            startsAt={createdStartsAt}
   127→            onBack={resetToProfile}
   128→          />
   129→        ) : (
   130→          <MentorProfile
   131→            mentor={mentor}
   132→            reviews={data.reviews}
   133→            refreshing={refreshing}
   134→            onRefresh={() => void load("refresh")}
   135→            onSchedule={() => setStage("booking")}
   136→            onMessage={() =>
   137→              mentor.userId &&
   138→              navigation.navigate("Mensagens", { peerId: mentor.userId, peerName: mentor.name })
   139→            }
   140→          />
   141→        )
   142→      ) : null}
   143→    </Screen>
   144→  );
   145→}

   346→/* --------------------------- Agendamento -------------------------------- */
   347→
   348→function BookingScheduler({
   349→  mentor,
   350→  onCancel,
   351→  onCreated,
   352→}: {
   353→  mentor: MentorDetail;
   354→  onCancel: () => void;
   355→  onCreated: (created: BookingCreated, startsAt: string) => void;
   356→}) {
   357→  const days = upcomingDays(14);
   358→  const [day, setDay] = useState<DayChip>(days[0]);
   359→  const [slots, setSlots] = useState<string[]>([]);
   360→  const [slotsLoading, setSlotsLoading] = useState(true);
   361→  const [slotsError, setSlotsError] = useState<string | null>(null);
   362→  const [slot, setSlot] = useState<string | null>(null);
   363→  const [topic, setTopic] = useState("");
   364→  const [submitting, setSubmitting] = useState(false);
   365→  const [flowError, setFlowError] = useState<string | null>(null);
   366→  const styles = makeStyles();
   367→
   368→  const loadSlots = useCallback(
   369→    async (iso: string) => {
   370→      setSlotsLoading(true);
   371→      setSlotsError(null);
   372→      try {
   373→        const res = await listMentorSlots(mentor.id, iso);
   374→        setSlots(res.slots);
   375→      } catch (err) {
   376→        setSlots([]);
   377→        setSlotsError(errMessage(err));
   378→      } finally {
   379→        setSlotsLoading(false);
   380→      }
   381→    },
   382→    [mentor.id]
   383→  );
   384→
   385→  useEffect(() => {
   386→    void loadSlots(day.iso);
   387→  }, [day.iso, loadSlots]);
   388→
   389→  function selectDay(next: DayChip) {
   390→    if (next.iso === day.iso) return;
   391→    setDay(next);
   392→    setSlot(null);
   393→    setFlowError(null);
   394→  }
   395→
   396→  async function handleConfirm() {
   397→    if (!slot || topic.trim().length === 0 || submitting) return;
   398→    setSubmitting(true);
   399→    setFlowError(null);
   400→    const startsAt = `${day.iso}T${slot}`;
   401→    try {
   402→      const booking = await createBooking({
   403→        mentorId: mentor.id,
   404→        startsAt,
   405→        durationMin: 60,
   406→        topic: topic.trim(),
   407→      });
   408→      onCreated(booking, startsAt);
   409→    } catch (err) {
   410→      // 409 → horário ficou indisponível: mostra a mensagem da API e
   411→      // recarrega os horários do dia para remover o slot ocupado.
   412→      setFlowError(errMessage(err));
   413→      setSlot(null);
   414→      void loadSlots(day.iso);
   415→    } finally {
   416→      setSubmitting(false);
   417→    }
   418→  }
   419→
   420→  return (
   421→    <View style={styles.flex}>
   422→      <ScreenHeader title={`Agendar com ${mentor.name.split(" ")[0] ?? ""}`} onBack={onCancel} />
   423→      <ScrollView
   424→        style={styles.flex}
   425→        contentContainerStyle={styles.content}
   426→        showsVerticalScrollIndicator={false}
   427→        keyboardShouldPersistTaps="handled"
   428→      >
   429→        <Text style={styles.stepTitle}>1. Escolha o dia</Text>
   430→        <ScrollView
   431→          horizontal
   432→          showsHorizontalScrollIndicator={false}
   433→          contentContainerStyle={styles.daysRow}
   434→        >
   435→          {days.map((dayChip) => (
   436→            <DayChipButton
   437→              key={dayChip.iso}
   438→              day={dayChip}
   439→              selected={dayChip.iso === day.iso}
   440→              onPress={() => selectDay(dayChip)}
   441→            />
   442→          ))}
   443→        </ScrollView>
   444→
   445→        <Text style={styles.stepTitle}>2. Escolha o horário</Text>
   446→        {slotsLoading ? (
   447→          <LoadingList compact />
   448→        ) : slotsError ? (
   449→          <ErrorBox compact message={slotsError} onRetry={() => void loadSlots(day.iso)} />
   450→        ) : slots.length === 0 ? (
   451→          <Text style={styles.emptySlots}>
   452→            Nenhum horário livre neste dia. Tente outro dia, por favor.
   453→          </Text>
   454→        ) : (
   455→          <View style={styles.slotsWrap}>
   456→            {slots.map((time) => (
   457→              <FilterChip
   458→                key={time}
   459→                label={time}
   460→                selected={slot === time}
   461→                onPress={() => {
   462→                  setSlot(time);
   463→                  setFlowError(null);
   464→                }}
   465→              />
   466→            ))}
   467→          </View>
   468→        )}
   469→
   470→        <Text style={styles.stepTitle}>3. Sobre o que você quer conversar?</Text>
   471→        <TextInput
   472→          style={styles.topicInput}
   473→          value={topic}
   474→          onChangeText={setTopic}
   475→          placeholder="Ex.: Revisar meu portfólio e planejar os próximos passos da carreira."
   476→          placeholderTextColor={theme.colors.textFaint}
   477→          multiline
   478→          maxLength={200}
   479→          textAlignVertical="top"
   480→        />
   481→
   482→        <View style={styles.summaryRow}>
   483→          <Ionicons name="time-outline" size={14} color={theme.colors.textFaint} />
   484→          <Text style={styles.summaryText}>Sessão de 60 minutos</Text>
   485→          {mentor.hourlyRate > 0 ? (
   486→            <Text style={styles.summaryPrice}>{formatPrice(mentor.hourlyRate)}</Text>
   487→          ) : null}
   488→        </View>
   489→
   490→        {flowError ? (
   491→          <View style={styles.flowError}>
   492→            <ErrorBox compact message={flowError} />
   493→          </View>
   494→        ) : null}
   495→
   496→        <TouchableOpacity
   497→          style={[
   498→            styles.cta,
   499→            (!slot || topic.trim().length === 0 || submitting) && styles.ctaDisabled,
   500→          ]}
   501→          onPress={() => void handleConfirm()}
   502→          disabled={!slot || topic.trim().length === 0 || submitting}
   503→          activeOpacity={0.85}
   504→          accessibilityRole="button"
   505→          accessibilityLabel="Confirmar agendamento"
   506→        >
   507→          {submitting ? (
   508→            <ActivityIndicator size="small" color={theme.colors.onAccent} />
   509→          ) : (
   510→            <Text style={styles.ctaText}>Confirmar agendamento</Text>
   511→          )}
   512→        </TouchableOpacity>
   513→
   514→        <View style={styles.bottomSpacer} />
   515→      </ScrollView>
   516→    </View>
   517→  );
   518→}

   520→function DayChipButton({
   521→  day,
   522→  selected,
   523→  onPress,
   524→}: {
   525→  day: DayChip;
   526→  selected: boolean;
   527→  onPress: () => void;
   528→}) {
   529→  const dayStyles = makeDayStyles();
   530→  return (
   531→    <TouchableOpacity
   532→      style={[dayStyles.pill, selected ? dayStyles.selected : dayStyles.unselected]}
   533→      onPress={onPress}
   534→      activeOpacity={0.8}
   535→      accessibilityRole="button"
   536→      accessibilityState={{ selected }}
   537→    >
   538→      <Text style={[dayStyles.weekday, selected ? dayStyles.weekdaySelected : null]}>
   539→        {day.weekday}
   540→      </Text>
   541→      <Text style={[dayStyles.label, selected ? dayStyles.labelSelected : null]}>{day.label}</Text>
   542→    </TouchableOpacity>
   543→  );
   544→}

   546→/* ----------------------------- Sucesso ---------------------------------- */
   547→
   548→function BookingSuccess({
   549→  mentorName,
   550→  created,
   551→  startsAt,
   552→  onBack,
   553→}: {
   554→  mentorName: string;
   555→  created: BookingCreated | null;
   556→  startsAt: string;
   557→  onBack: () => void;
   558→}) {
   559→  const styles = makeStyles();
   560→  const navigation = useNavigation<any>();
   561→  const { setTab } = useTabs();
   562→  return (
   563→    <View style={styles.flex}>
   564→      <ScreenHeader title="Sessão solicitada" onBack={onBack} />
   565→      <ScrollView
   566→        style={styles.flex}
   567→        contentContainerStyle={styles.successContent}
   568→        showsVerticalScrollIndicator={false}
   569→      >
   570→        <View style={styles.successCard}>
   571→          <View style={styles.successIcon}>
   572→            <Ionicons name="checkmark" size={30} color={theme.colors.accent} />
   573→          </View>
   574→          <Text style={styles.successTitle}>Agendamento enviado!</Text>
   575→          <Text style={styles.successText}>
   576→            Sua sessão com {mentorName} foi solicitada para{"\n"}
   577→            <Text style={styles.successWhen}>{formatNaiveLong(startsAt)}</Text>.
   578→          </Text>
   579→          {created ? <StatusPill status={created.status} /> : null}
   580→          <Text style={styles.successPending}>
   581→            Aguardando o mentor confirmar o pedido. Você será notificado assim que ele responder.
   582→          </Text>
   583→          {created && created.price > 0 ? (
   584→            <Text style={styles.successPrice}>Valor da sessão: {formatPrice(created.price)}</Text>
   585→          ) : null}
   586→
   587→          <TouchableOpacity
   588→            style={styles.cta}
   589→            onPress={() => {
   590→              // Aba Mentorias no pager principal (TabsContext) + desempilha o
   591→              // Mentor para revelar o pager já na aba correta.
   592→              setTab("Mentorias");
   593→              navigation.goBack();
   594→            }}
   595→            activeOpacity={0.85}
   596→            accessibilityRole="button"
   597→            accessibilityLabel="Ir para Minhas sessões"
   598→          >
   599→            <Text style={styles.ctaText}>Ver minhas sessões</Text>

   612→/** Normaliza handles/URLs das redes sociais para URLs absolutas. */
   613→function socialUrl(network: "instagram" | "linkedin" | "website", value: string): string {
   614→  const raw = value.trim();
   615→  if (!raw) return "";
   616→  if (/^https?:\/\//i.test(raw)) return raw;
   617→  if (network === "instagram") return `https://instagram.com/${raw.replace(/^@/, "")}`;
   618→  if (network === "linkedin") return `https://linkedin.com/in/${raw.replace(/^\/+/, "")}`;
   619→  return `https://${raw}`;
   620→}
