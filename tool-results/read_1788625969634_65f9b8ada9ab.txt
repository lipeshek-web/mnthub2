    71→    71→/** Tempo relativo compacto: "agora", "há 5 min", "há 2 h", "há 3 d" (data p/ antigos) */
    72→    72→function relativeTime(iso: string) {
    73→    73→  const diffMin = Math.max(0, Math.floor((Date.now() - new Date(iso).getTime()) / 60_000))
    74→    74→  if (diffMin < 1) return 'agora'
    75→    75→  if (diffMin < 60) return `há ${diffMin} min`
    76→    76→  const hours = Math.floor(diffMin / 60)
    77→    77→  if (hours < 24) return `há ${hours} h`
    78→    78→  const days = Math.floor(hours / 24)
    79→    79→  if (days < 30) return `há ${days} d`
    80→    80→  return new Date(iso).toLocaleDateString('pt-BR', { day: '2-digit', month: 'short' })
    81→    81→}

    83→    83→// ==================== Contadores do header (1 poll para os 2 sininhos) ====================
    84→    84→
    85→    85→/**
    86→    86→ * Badge único: mensagens + notificações não lidas em UM request autenticado
    87→    87→ * (antes eram 2 polls — messages/unread 45s + notifications 60s). Pausa com a
    88→    88→ * aba oculta e atualiza ao voltar o foco. `refresh()` força revalidação depois
    89→    89→ * de ações locais (ex.: marcar tudo como lido). Compartilhado com a tab bar mobile.
    90→    90→ */
    91→    91→export function useBadges() {
    92→    92→  const userId = useAppStore((s) => s.user?.id)
    93→    93→  const [badges, setBadges] = useState({ messages: 0, notifications: 0 })
    94→    94→  const aliveRef = useRef(true)
    95→    95→  const reqIdRef = useRef(0)
    96→    96→
    97→    97→  const refresh = useCallback(() => {
    98→    98→    if (!userId) return
    99→    99→    const reqId = ++reqIdRef.current
   100→   100→    api
   101→   101→      .badges()
   102→   102→      .then((res) => {
   103→   103→        if (!aliveRef.current || reqId !== reqIdRef.current) return
   104→   104→        setBadges((prev) =>
   105→   105→          prev.messages === res.messages && prev.notifications === res.notifications
   106→   106→            ? prev
   107→   107→            : { messages: res.messages, notifications: res.notifications }
   108→   108→        )
   109→   109→      })
   110→   110→      .catch(() => {
   111→   111→        /* silencioso */
   112→   112→      })
   113→   113→  }, [userId])
   114→   114→
   115→   115→  useEffect(() => {
   116→   116→    if (!userId) return
   117→   117→    aliveRef.current = true
   118→   118→    refresh()
   119→   119→    const timer = setInterval(() => {
   120→   120→      if (document.hidden) return // economia: aba em segundo plano não faz polling
   121→   121→      refresh()
   122→   122→    }, 45_000)
   123→   123→    const onVisible = () => {
   124→   124→      if (!document.hidden) refresh()
   125→   125→    }
   126→   126→    document.addEventListener('visibilitychange', onVisible)
   127→   127→    window.addEventListener('focus', onVisible)
   128→   128→    return () => {
   129→   129→      aliveRef.current = false
   130→   130→      clearInterval(timer)
   131→   131→      document.removeEventListener('visibilitychange', onVisible)
   132→   132→      window.removeEventListener('focus', onVisible)
   133→   133→    }
   134→   134→  }, [userId, refresh])
   135→   135→
   136→   136→  // Sem login os badges são sempre zero (derivado — sem setState em efeito)
   137→   137→  const shown = userId ? badges : { messages: 0, notifications: 0 }
   138→   138→  return { ...shown, refresh }
   139→   139→}

   208→   208→function NotificationsBell({
   209→   209→  badgeCount,
   210→   210→  refreshBadges,
   211→   211→}: {
   212→   212→  badgeCount: number
   213→   213→  refreshBadges: () => void
   214→   214→}) {
   215→   215→  const user = useAppStore((s) => s.user)
   216→   216→  const navigate = useAppStore((s) => s.navigate)
   217→   217→
   218→   218→  const [open, setOpen] = useState(false)
   219→   219→  const [items, setItems] = useState<NotificationDTO[]>([])
   220→   220→  // Contador exibido = sobrescrita otimista local ?? badge do header (poll 45s).
   221→   221→  // Sem setState em efeito: o badge chega como prop e a otimização vence até o
   222→   222→  // servidor confirmar.
   223→   223→  const [localOverride, setLocalOverride] = useState<number | null>(null)
   224→   224→  const unreadCount = localOverride ?? badgeCount
   225→   225→
   226→   226→  // Guard `alive` + contador de requisição: evita setState de respostas
   227→   227→  // obsoletas (unmount, troca de usuário ou fetch sobreposto)
   228→   228→  const aliveRef = useRef(true)
   229→   229→  const reqIdRef = useRef(0)
   230→   230→
   231→   231→  const loadNotifications = useCallback(async (userId: string) => {
   232→   232→    const reqId = ++reqIdRef.current
   233→   233→    try {
   234→   234→      const res = await api.listNotifications(userId)
   235→   235→      if (!aliveRef.current || reqId !== reqIdRef.current) return
   236→   236→      setItems(res.items)
   237→   237→      setLocalOverride(res.unreadCount)
   238→   238→    } catch {
   239→   239→      // Erro silencioso: badge apenas não aparece (sem toast)
   240→   240→    }
   241→   241→  }, [])
   242→   242→
   243→   243→  const userId = user?.id
   244→   244→  // Itens do dropdown: carregados ao ABRIR o painel (sem poll e sem fetch no
   245→   245→  // mount — o contador vem do badge único; a lista só importa ao abrir)
   246→   246→  useEffect(() => {
   247→   247→    aliveRef.current = true
   248→   248→    return () => {
   249→   249→      aliveRef.current = false
   250→   250→    }
   251→   251→  }, [])
   252→   252→
   253→   253→  const handleOpenChange = (next: boolean) => {
   254→   254→    setOpen(next)
   255→   255→    if (next && userId) loadNotifications(userId) // refetch ao abrir o dropdown
   256→   256→  }

   258→   258→    // Marca como lida de forma otimista (badge/zumbido atualizam na hora)
   259→   259→    if (!item.read) {
   260→   260→      setItems((prev) => prev.map((n) => (n.id === item.id ? { ...n, read: true } : n)))
   261→   261→      setLocalOverride((count) => Math.max(0, unreadCount - 1))
   262→   262→      if (userId) api.markNotificationsRead(userId, [item.id]).catch(() => {})
   263→   263→      refreshBadges()
   264→   264→    }
   265→   265→    // Navega conforme o destino sugerido pela notificação
   266→   266→    if (item.linkView === 'dashboard') navigate({ name: 'dashboard' })
   267→   267→    else if (item.linkView === 'course' && item.refId) navigate({ name: 'course', courseId: item.refId })
   268→   268→    else if (item.linkView === 'onboarding') navigate({ name: 'onboarding' })
   269→   269→    else if (item.linkView === 'messages') navigate({ name: 'messages', peerId: item.refId ?? undefined })
   270→   270→    else if (item.linkView === 'referrals') navigate({ name: 'referrals' })
   271→   271→    setOpen(false)
   272→   272→  }
   273→   273→
   274→   274→  const handleMarkAll = () => {
   275→   275→    if (!userId || unreadCount === 0) return
   276→   276→    const prevItems = items
   277→   277→    setItems((prev) => prev.map((n) => ({ ...n, read: true })))
   278→   278→    setLocalOverride(0)
   279→   279→    api.markNotificationsRead(userId).catch(() => {
   280→   280→      // Falhou: restaura o estado anterior
   281→   281→      setItems(prevItems)
   282→   282→      setLocalOverride(prevItems.filter((n) => !n.read).length)
   283→   283→    })
   284→   284→    refreshBadges()
   285→   285→  }
   286→   286→
   287→   287→  if (!user) return null
   288→   288→
   289→   289→  return (
   290→   290→    <DropdownMenu open={open} onOpenChange={handleOpenChange}>
   291→   291→      <DropdownMenuTrigger asChild>
   292→   292→        <button
   293→   293→          type="button"
   294→   294→          aria-label={
   295→   295→            unreadCount > 0
   296→   296→              ? `Notificações, ${unreadCount} não lida${unreadCount === 1 ? '' : 's'}`
   297→   297→              : 'Notificações'
   298→   298→          }
   299→   299→          title="Notificações"
   300→   300→          className="relative flex h-9 w-9 shrink-0 items-center justify-center rounded-full text-stone-500 transition-colors hover:bg-stone-100 hover:text-stone-900"
   301→   301→        >
   302→   302→          <Bell className="h-4.5 w-4.5" />
   303→   303→          {unreadCount > 0 && (
   304→   304→            <span
   305→   305→              aria-live="polite"
   306→   306→              className="absolute -right-0.5 -top-0.5 flex h-4 min-w-4 items-center justify-center rounded-full bg-rose-500 px-1 text-[10px] font-bold leading-none text-white"
   307→   307→            >
   308→   308→              {unreadCount > 99 ? '99+' : unreadCount}
   309→   309→            </span>
   310→   310→          )}
   311→   311→        </button>
   312→   312→      </DropdownMenuTrigger>
   313→   313→
   314→   314→      <DropdownMenuContent align="end" sideOffset={8} className="w-88 overflow-hidden p-0 sm:w-96">
   315→   315→        {/* Cabeçalho do painel */}

   318→   318→            Notificações

   323→   323→            disabled={unreadCount === 0}
