    71→/** Tempo relativo compacto: "agora", "há 5 min", "há 2 h", "há 3 d" (data p/ antigos) */
    72→function relativeTime(iso: string) {
    73→  const diffMin = Math.max(0, Math.floor((Date.now() - new Date(iso).getTime()) / 60_000))
    74→  if (diffMin < 1) return 'agora'
    75→  if (diffMin < 60) return `há ${diffMin} min`
    76→  const hours = Math.floor(diffMin / 60)
    77→  if (hours < 24) return `há ${hours} h`
    78→  const days = Math.floor(hours / 24)
    79→  if (days < 30) return `há ${days} d`
    80→  return new Date(iso).toLocaleDateString('pt-BR', { day: '2-digit', month: 'short' })
    81→}

    85→/**
    86→ * Badge único: mensagens + notificações não lidas em UM request autenticado
    87→ * (antes eram 2 polls — messages/unread 45s + notifications 60s). Pausa com a
    88→ * aba oculta e atualiza ao voltar o foco. `refresh()` força revalidação depois
    89→ * de ações locais (ex.: marcar tudo como lido). Compartilhado com a tab bar mobile.
    90→ */
    91→export function useBadges() {
    92→  const userId = useAppStore((s) => s.user?.id)
    93→  const [badges, setBadges] = useState({ messages: 0, notifications: 0 })
    94→  const aliveRef = useRef(true)
    95→  const reqIdRef = useRef(0)
    96→
    97→  const refresh = useCallback(() => {
    98→    if (!userId) return
    99→    const reqId = ++reqIdRef.current
   100→    api
   101→      .badges()
   102→      .then((res) => {
   103→        if (!aliveRef.current || reqId !== reqIdRef.current) return
   104→        setBadges((prev) =>
   105→          prev.messages === res.messages && prev.notifications === res.notifications
   106→            ? prev
   107→            : { messages: res.messages, notifications: res.notifications }
   108→        )
   109→      })
   110→      .catch(() => {
   111→        /* silencioso */
   112→      })
   113→  }, [userId])
   114→
   115→  useEffect(() => {
   116→    if (!userId) return
   117→    aliveRef.current = true
   118→    refresh()
   119→    const timer = setInterval(() => {
   120→      if (document.hidden) return // economia: aba em segundo plano não faz polling
   121→      refresh()
   122→    }, 45_000)
   123→    const onVisible = () => {
   124→      if (!document.hidden) refresh()
   125→    }
   126→    document.addEventListener('visibilitychange', onVisible)
   127→    window.addEventListener('focus', onVisible)
   128→    return () => {
   129→      aliveRef.current = false
   130→      clearInterval(timer)
   131→      document.removeEventListener('visibilitychange', onVisible)
   132→      window.removeEventListener('focus', onVisible)
   133→    }
   134→  }, [userId, refresh])
   135→
   136→  // Sem login os badges são sempre zero (derivado — sem setState em efeito)
   137→  const shown = userId ? badges : { messages: 0, notifications: 0 }
   138→  return { ...shown, refresh }
   139→}

   175→function ThemeToggle() {
   176→  const { resolvedTheme, setTheme } = useTheme()
   177→  // next-themes só sabe o tema no cliente — evita mismatch de hidratação
   178→  // (useSyncExternalStore em vez de setState em effect: sem render cascata)
   179→  const emptySubscribe = useCallback(() => () => {}, [])
   180→  const mounted = useSyncExternalStore(
   181→    emptySubscribe,
   182→    () => true,
   183→    () => false
   184→  )
   185→
   186→  const isDark = mounted && resolvedTheme === 'dark'
   187→  return (
   188→    <button
   189→      type="button"
   190→      onClick={() => setTheme(isDark ? 'light' : 'dark')}
   191→      aria-label={isDark ? 'Mudar para o tema claro' : 'Mudar para o tema escuro'}
   192→      title={isDark ? 'Tema claro' : 'Tema escuro'}
   193→      className="flex h-9 w-9 shrink-0 items-center justify-center rounded-full text-stone-500 transition-colors hover:bg-stone-100 hover:text-stone-900 dark:text-stone-400 dark:hover:bg-stone-800 dark:hover:text-stone-100"
   194→    >
   195→      {mounted ? (
   196→        isDark ? (
   197→          <Sun className="h-4.5 w-4.5" />
   198→        ) : (
   199→          <Moon className="h-4.5 w-4.5" />
   200→        )
   201→      ) : (
   202→        <span className="h-4.5 w-4.5" aria-hidden />
   203→      )}
   204→    </button>
   205→  )
   206→}

   208→function NotificationsBell({
   209→  badgeCount,
   210→  refreshBadges,
   211→}: {
   212→  badgeCount: number
   213→  refreshBadges: () => void
   214→}) {
   215→  const user = useAppStore((s) => s.user)
   216→  const navigate = useAppStore((s) => s.navigate)
   217→
   218→  const [open, setOpen] = useState(false)
   219→  const [items, setItems] = useState<NotificationDTO[]>([])
   220→  // Contador exibido = sobrescrita otimista local ?? badge do header (poll 45s).
   221→  // Sem setState em efeito: o badge chega como prop e a otimização vence até o
   222→  // servidor confirmar.
   223→  const [localOverride, setLocalOverride] = useState<number | null>(null)
   224→  const unreadCount = localOverride ?? badgeCount
   225→
   226→  // Guard `alive` + contador de requisição: evita setState de respostas
   227→  // obsoletas (unmount, troca de usuário ou fetch sobreposto)
   228→  const aliveRef = useRef(true)
   229→  const reqIdRef = useRef(0)
   230→
   231→  const loadNotifications = useCallback(async (userId: string) => {
   232→    const reqId = ++reqIdRef.current
   233→    try {
   234→      const res = await api.listNotifications(userId)
   235→      if (!aliveRef.current || reqId !== reqIdRef.current) return
   236→      setItems(res.items)
   237→      setLocalOverride(res.unreadCount)
   238→    } catch {
   239→      // Erro silencioso: badge apenas não aparece (sem toast)
   240→    }
   241→  }, [])
   242→
   243→  const userId = user?.id
   244→  // Itens do dropdown: carregados ao ABRIR o painel (sem poll e sem fetch no
   245→  // mount — o contador vem do badge único; a lista só importa ao abrir)
   246→  useEffect(() => {
   247→    aliveRef.current = true
   248→    return () => {
   249→      aliveRef.current = false
   250→    }
   251→  }, [])
   252→
   253→  const handleOpenChange = (next: boolean) => {
   254→    setOpen(next)
   255→    if (next && userId) loadNotifications(userId) // refetch ao abrir o dropdown
   256→  }
   257→  const handleItemClick = (item: NotificationDTO) => {
   258→    // Marca como lida de forma otimista (badge/zumbido atualizam na hora)
   259→    if (!item.read) {
   260→      setItems((prev) => prev.map((n) => (n.id === item.id ? { ...n, read: true } : n)))
   261→      setLocalOverride((count) => Math.max(0, unreadCount - 1))
   262→      if (userId) api.markNotificationsRead(userId, [item.id]).catch(() => {})
   263→      refreshBadges()
   264→    }
   265→    // Navega conforme o destino sugerido pela notificação
   266→    if (item.linkView === 'dashboard') navigate({ name: 'dashboard' })
   267→    else if (item.linkView === 'course' && item.refId) navigate({ name: 'course', courseId: item.refId })
   268→    else if (item.linkView === 'onboarding') navigate({ name: 'onboarding' })
   269→    else if (item.linkView === 'messages') navigate({ name: 'messages', peerId: item.refId ?? undefined })
   270→    else if (item.linkView === 'referrals') navigate({ name: 'referrals' })
   271→    setOpen(false)
   272→  }
   273→
   274→  const handleMarkAll = () => {
   275→    if (!userId || unreadCount === 0) return
   276→    const prevItems = items
   277→    setItems((prev) => prev.map((n) => ({ ...n, read: true })))
   278→    setLocalOverride(0)
   279→    api.markNotificationsRead(userId).catch(() => {
   280→      // Falhou: restaura o estado anterior
   281→      setItems(prevItems)
   282→      setLocalOverride(prevItems.filter((n) => !n.read).length)
   283→    })
   284→    refreshBadges()
   285→  }
   286→
   287→  if (!user) return null
   288→
   289→  return (
   290→    <DropdownMenu open={open} onOpenChange={handleOpenChange}>
   291→      <DropdownMenuTrigger asChild>
   292→        <button
   293→          type="button"
   294→          aria-label={
   295→            unreadCount > 0
   296→              ? `Notificações, ${unreadCount} não lida${unreadCount === 1 ? '' : 's'}`
   297→              : 'Notificações'
   298→          }
   299→          title="Notificações"
   300→          className="relative flex h-9 w-9 shrink-0 items-center justify-center rounded-full text-stone-500 transition-colors hover:bg-stone-100 hover:text-stone-900"
   301→        >
   302→          <Bell className="h-4.5 w-4.5" />
   303→          {unreadCount > 0 && (
   304→            <span
   305→              aria-live="polite"
   306→              className="absolute -right-0.5 -top-0.5 flex h-4 min-w-4 items-center justify-center rounded-full bg-rose-500 px-1 text-[10px] font-bold leading-none text-white"
   307→            >
   308→              {unreadCount > 99 ? '99+' : unreadCount}
   309→            </span>
   310→          )}
   311→        </button>
   312→      </DropdownMenuTrigger>
   313→
   314→      <DropdownMenuContent align="end" sideOffset={8} className="w-88 overflow-hidden p-0 sm:w-96">
   315→        {/* Cabeçalho do painel */}
   316→        <div className="flex items-center justify-between gap-2 border-b border-stone-100 px-4 py-2.5 dark:border-stone-800">
   317→          <DropdownMenuLabel className="p-0 font-semibold text-stone-900 dark:text-stone-100">
   318→            Notificações
   319→          </DropdownMenuLabel>
   320→          <button
   321→            type="button"
   322→            onClick={handleMarkAll}
   323→            disabled={unreadCount === 0}
   324→            className="rounded text-xs font-semibold text-emerald-700 transition-colors hover:text-emerald-900 disabled:pointer-events-none disabled:text-stone-300 dark:text-emerald-400 dark:hover:text-emerald-300 dark:disabled:text-stone-600"
   325→          >
   326→            Marcar todas como lidas
   327→          </button>
   328→        </div>
   329→
   330→        {items.length === 0 ? (
   331→          /* Empty state */
   332→          <div className="flex flex-col items-center gap-2 px-6 py-8 text-center">
   333→            <BellOff className="h-8 w-8 text-stone-300 dark:text-stone-600" aria-hidden />
   334→            <p className="text-sm text-stone-500 dark:text-stone-400">Você está em dia! Nenhuma notificação.</p>
   335→          </div>
   336→        ) : (
   337→          <div
   338→            role="list"
   339→            aria-label="Lista de notificações"
   340→            className="max-h-[380px] overflow-y-auto [scrollbar-width:thin] [&::-webkit-scrollbar]:h-1.5 [&::-webkit-scrollbar-thumb]:rounded-full [&::-webkit-scrollbar-thumb]:bg-stone-200 dark:[&::-webkit-scrollbar-thumb]:bg-stone-700 [&::-webkit-scrollbar-track]:bg-transparent"
   341→          >
   342→            {items.map((item) => {
   343→              const chip = NOTIFICATION_KINDS[item.kind] ?? DEFAULT_KIND
   344→              const Icon = chip.icon
   345→              return (
   346→                <div key={item.id} role="listitem">
   347→                  <button
   348→                    type="button"
   349→                    onClick={() => handleItemClick(item)}
   350→                    className={cn(
   351→                      'relative flex w-full items-start gap-3 px-4 py-3 text-left transition-colors focus-visible:outline-none',
   352→                      item.read
   353→                        ? 'hover:bg-stone-50 focus-visible:bg-stone-50 dark:hover:bg-stone-800/60 dark:focus-visible:bg-stone-800/60'
   354→                        : 'bg-emerald-50/50 hover:bg-emerald-50 focus-visible:bg-emerald-50 dark:bg-emerald-950/40 dark:hover:bg-emerald-900/30 dark:focus-visible:bg-emerald-900/30'
   355→                    )}
   356→                  >
   357→                    {!item.read && (
   358→                      <span
   359→                        aria-hidden
   360→                        className="absolute left-1.5 top-1/2 h-1.5 w-1.5 -translate-y-1/2 rounded-full bg-emerald-500"
   361→                      />
   362→                    )}
   363→                    <span
   364→                      className={cn(
   365→                        'flex h-8 w-8 shrink-0 items-center justify-center rounded-full',
   366→                        chip.chipClass
   367→                      )}
   368→                    >
   369→                      <Icon className="h-4 w-4" aria-hidden />
   370→                    </span>
   371→                    <span className="min-w-0 flex-1">
   372→                      <span className="flex items-baseline justify-between gap-2">
   373→                        <span
   374→                          className={cn(
   375→                            'truncate text-sm font-semibold',
   376→                            item.read ? 'text-stone-500' : 'text-stone-900'
   377→                          )}
   378→                        >
   379→                          {item.title}
   380→                        </span>
   381→                        <span className="shrink-0 text-[11px] text-stone-400 dark:text-stone-500">
   382→                          {relativeTime(item.createdAt)}
   383→                        </span>
   384→                      </span>
   385→                      {item.body && (
   386→                        <span className="mt-0.5 block text-xs leading-snug text-stone-500 line-clamp-2 dark:text-stone-400">
   387→                          {item.body}
   388→                        </span>
   389→                      )}
   390→                    </span>
   391→                  </button>
   392→                </div>
   393→              )
   394→            })}
   395→          </div>
   396→        )}
   397→      </DropdownMenuContent>
   398→    </DropdownMenu>
   399→  )
   400→}

   402→export function Navbar() {
   403→  // Seletores atômicos (não destruturar a store inteira: Navbar re-renderizaria
   404→  // a cada commit de qualquer campo, incluindo a cada digitação da busca)
   405→  const user = useAppStore((s) => s.user)
   406→  const view = useAppStore((s) => s.view)
   407→  const setUser = useAppStore((s) => s.setUser)
   408→  const navigate = useAppStore((s) => s.navigate)
   409→  const badges = useBadges()
   410→
   411→  // ---------- Busca global do header (a busca principal, sempre visível) ----------
   412→  const [query, setQuery] = useState('')
   413→  const [mobileSearchOpen, setMobileSearchOpen] = useState(false)
   414→  const desktopSearchRef = useRef<HTMLInputElement>(null)
   415→  const mobileSearchRef = useRef<HTMLInputElement>(null)
   416→  const searchDebounceRef = useRef<ReturnType<typeof setTimeout> | null>(null)
   417→  // O texto do campo acompanha o termo ativo em qualquer tela (ex.: busca feita
   418→  // em outra aba/session) — a busca é uma só, centralizada no header.
   419→  const externalQuery = useAppStore((s) => s.exploreQuery)
   420→
   421→  useEffect(() => {
   422→    setQuery(externalQuery)
   423→  }, [externalQuery])
   424→
   425→  useEffect(() => {
   426→    return () => {
   427→      if (searchDebounceRef.current) clearTimeout(searchDebounceRef.current)
   428→    }
   429→  }, [])
   430→
   431→  /** Aplica o termo na store e leva o usuário ao Explorar (modo resultados) */
   432→  const applySearch = (value: string) => {
   433→    const q = value.trim()
   434→    const state = useAppStore.getState()
   435→    state.setExploreQuery(q)
   436→    if (q && state.view.name !== 'marketplace') {
   437→      state.setExploreTab('all')
   438→      state.navigate({ name: 'marketplace' })
   439→    }
   440→  }
   441→
   442→  /** Digitação ao vivo: navega e filtra com debounce — o corpo vira só resultados */
   443→  const onSearchInput = (value: string) => {
   444→    setQuery(value)
   445→    if (searchDebounceRef.current) clearTimeout(searchDebounceRef.current)
   446→    const trimmed = value.trim()
   447→    if (!trimmed) {
   448→      useAppStore.getState().setExploreQuery('')
   449→      if (useAppStore.getState().view.name === 'marketplace') setMobileSearchOpen(false)
   450→      return
   451→    }
   452→    searchDebounceRef.current = setTimeout(() => {
   453→      const wasOnMarketplace = useAppStore.getState().view.name === 'marketplace'
   454→      applySearch(value)
   455→      if (!wasOnMarketplace) setMobileSearchOpen(false)
   456→    }, 250)
   457→  }
   458→
   459→  const submitSearch = (value: string) => {
   460→    const q = value.trim()
   461→    if (!q) return
   462→    if (searchDebounceRef.current) clearTimeout(searchDebounceRef.current)
   463→    useAppStore.getState().setExploreQuery(q)
   464→    useAppStore.getState().setExploreTab('all')
   465→    navigate({ name: 'marketplace' })
   466→    setMobileSearchOpen(false)
   467→  }
   468→
   469→  // Atalho "/" foca a busca principal do header (sensação de app nativo)
   470→  useEffect(() => {
   471→    const onKey = (e: KeyboardEvent) => {
   472→      const target = e.target as HTMLElement | null
   473→      const typing =
   474→        target instanceof HTMLInputElement ||
   475→        target instanceof HTMLTextAreaElement ||
   476→        target?.isContentEditable
   477→      if (e.key === '/' && !typing) {
   478→        e.preventDefault()
   479→        if (window.matchMedia('(min-width: 768px)').matches) {
   480→          desktopSearchRef.current?.focus()
   481→        } else {
   482→          setMobileSearchOpen(true)
   483→          setTimeout(() => mobileSearchRef.current?.focus(), 60)
   484→        }
   485→      }
   486→    }
   487→    window.addEventListener('keydown', onKey)
   488→    return () => window.removeEventListener('keydown', onKey)
   489→  }, [])
   490→
   491→  const searchField = (isMobile: boolean) => {
   492→    const value = query
   493→    return (
   494→      <form
   495→        role="search"
   496→        onSubmit={(e) => {
   497→          e.preventDefault()
   498→          submitSearch(value)
   499→        }}
   500→        className="relative w-full"
   501→      >
   502→        <Search
   503→          aria-hidden
   504→          className="absolute left-3.5 top-1/2 h-4 w-4 -translate-y-1/2 text-stone-400"
   505→        />
   506→        <input
   507→          ref={isMobile ? mobileSearchRef : desktopSearchRef}
   508→          type="search"
   509→          value={value}
   510→          onChange={(e) => onSearchInput(e.target.value)}
   511→          placeholder="Buscar mentores, cursos, trilhas e leituras..."
