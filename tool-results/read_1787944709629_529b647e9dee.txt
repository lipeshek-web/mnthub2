  1221→/* ==================== Q&A DA AULA ==================== */
  1222→
  1223→function LessonQuestions({
  1224→  lesson,
  1225→  isOwner,
  1226→  mentorName,
  1227→  user,
  1228→}: {
  1229→  lesson: CourseLessonDTO
  1230→  isOwner: boolean
  1231→  mentorName: string
  1232→  user: { id: string; name: string } | null
  1233→}) {
  1234→  const [questions, setQuestions] = useState<LessonQuestionDTO[] | null>(null)
  1235→  const [body, setBody] = useState('')
  1236→  const [sending, setSending] = useState(false)
  1237→  const [answerDraft, setAnswerDraft] = useState<Record<string, string>>({})
  1238→  const [answeringId, setAnsweringId] = useState<string | null>(null)
  1239→
  1240→  const load = useCallback(async () => {
  1241→    if (!user) {
  1242→      setQuestions([])
  1243→      return
  1244→    }
  1245→    try {
  1246→      const list = await api.listLessonQuestions(lesson.id, user.id)
  1247→      setQuestions(list)
  1248→    } catch {
  1249→      setQuestions([])
  1250→    }
  1251→  }, [lesson.id, user])
  1252→
  1253→  useEffect(() => {
  1254→    setQuestions(null)
  1255→    void load()
  1256→  }, [load])
  1257→
  1258→  const ask = async () => {
  1259→    if (!user) {
  1260→      toast.error('Entre com uma conta para perguntar.')
  1261→      return
  1262→    }
  1263→    if (body.trim().length < 5) {
  1264→      toast.error('Escreva sua pergunta (mín. 5 caracteres).')
  1265→      return
  1266→    }
  1267→    setSending(true)
  1268→    try {
  1269→      const created = await api.askLessonQuestion(lesson.id, { userId: user.id, body: body.trim() })
  1270→      setQuestions((prev) => [created, ...(prev ?? [])])
  1271→      setBody('')
  1272→      toast.success('Pergunta enviada! O mentor será notificado.')
  1273→    } catch (err) {
  1274→      toast.error(err instanceof Error ? err.message : 'Não foi possível enviar a pergunta.')
  1275→    } finally {
  1276→      setSending(false)
  1277→    }
  1278→  }
  1279→
  1280→  const answer = async (questionId: string) => {
  1281→    const text = (answerDraft[questionId] ?? '').trim()
  1282→    if (!user || text.length < 2) {
  1283→      toast.error('Escreva a resposta antes de enviar.')
  1284→      return
  1285→    }
  1286→    setAnsweringId(questionId)
  1287→    try {
  1288→      const res = await api.answerQuestion(questionId, { userId: user.id, answer: text })
  1289→      setQuestions((prev) =>
  1290→        (prev ?? []).map((q) =>
  1291→          q.id === questionId ? { ...q, answer: res.answer, answeredAt: res.answeredAt } : q
  1292→        )
  1293→      )
  1294→      setAnswerDraft((prev) => ({ ...prev, [questionId]: '' }))
  1295→      toast.success('Resposta publicada!')
  1296→    } catch (err) {
  1297→      toast.error(err instanceof Error ? err.message : 'Não foi possível responder.')
  1298→    } finally {
  1299→      setAnsweringId(null)
  1300→    }
  1301→  }
  1302→
  1303→  const remove = async (questionId: string) => {
  1304→    if (!user) return
  1305→    try {
  1306→      await api.deleteQuestion(questionId, user.id)
  1307→      setQuestions((prev) => (prev ?? []).filter((q) => q.id !== questionId))
  1308→      toast('Pergunta removida.')
  1309→    } catch (err) {
  1310→      toast.error(err instanceof Error ? err.message : 'Não foi possível remover.')
  1311→    }
  1312→  }
  1313→
  1314→  if (!user) {
  1315→    return (
  1316→      <div className="rounded-2xl border border-stone-200 bg-white p-6 text-center text-sm text-stone-500">
  1317→        Entre com uma conta no topo da página para participar das perguntas.
  1318→      </div>
  1319→    )
  1320→  }
  1321→
  1322→  return (
  1323→    <div className="space-y-4">
  1324→      {/* Formulário de pergunta (aluno) */}
  1325→      {!isOwner ? (
  1326→        <div className="rounded-2xl border border-stone-200 bg-white p-4 sm:p-5">
  1327→          <label htmlFor="question-body" className="text-sm font-bold text-stone-800">
  1328→            Tirou uma dúvida? Pergunte ao mentor
  1329→          </label>
  1330→          <Textarea
  1331→            id="question-body"
  1332→            rows={3}
  1333→            value={body}
  1334→            onChange={(e) => setBody(e.target.value)}
  1335→            placeholder={`Ex.: "${mentorName.split(' ')[0]}", no minuto 4:30 você menciona X — como aplico isso na prática?`}
  1336→            className="mt-2 resize-none"
  1337→            maxLength={1200}
  1338→          />
  1339→          <div className="mt-3 flex items-center justify-between gap-3">
  1340→            <span className="text-xs text-stone-400">{body.length}/1200</span>
  1341→            <Button
  1342→              className="h-10 rounded-full font-semibold"
  1343→              onClick={() => void ask()}
  1344→              disabled={sending || body.trim().length < 5}
  1345→            >
  1346→              <Send aria-hidden className="h-4 w-4" />

  1441→/* ==================== ANOTAÇÕES DA AULA ==================== */
  1442→
  1443→function LessonNotes({
  1444→  lessonId,
  1445→  user,
  1446→  lessonTitle,
  1447→}: {
  1448→  lessonId: string
  1449→  user: { id: string; name: string } | null
  1450→  lessonTitle: string
  1451→}) {
  1452→  const [body, setBody] = useState('')
  1453→  const [loaded, setLoaded] = useState(false)
  1454→  const [saveState, setSaveState] = useState<'idle' | 'saving' | 'saved'>('idle')
  1455→  const [savedAt, setSavedAt] = useState<string | null>(null)
  1456→  const timerRef = useRef<ReturnType<typeof setTimeout> | null>(null)
  1457→  const latestRef = useRef('')
  1458→
  1459→  // Carrega as anotações da aula (o componente é remontado por aula via key)
  1460→  useEffect(() => {
  1461→    if (!user) return
  1462→    let active = true
  1463→    api
  1464→      .getLessonNote(lessonId, user.id)
  1465→      .then((note) => {
  1466→        if (!active) return
  1467→        setBody(note.body)
  1468→        latestRef.current = note.body
  1469→        setSavedAt(note.updatedAt)
  1470→      })
  1471→      .catch(() => {
  1472→        /* sem anotações ainda */
  1473→      })
  1474→      .finally(() => {
  1475→        if (active) setLoaded(true)
  1476→      })
  1477→    return () => {
  1478→      active = false
  1479→    }
  1480→  }, [lessonId, user])
  1481→
  1482→  // Autosave com debounce (o estado "salvando" aparece no envio)
  1483→  useEffect(() => {
  1484→    if (!loaded || !user) return
  1485→    if (body === latestRef.current) return
  1486→    if (timerRef.current) clearTimeout(timerRef.current)
  1487→    timerRef.current = setTimeout(async () => {
  1488→      setSaveState('saving')
  1489→      try {
  1490→        const res = await api.saveLessonNote(lessonId, { userId: user.id, body })
  1491→        latestRef.current = body
  1492→        setSavedAt(res.updatedAt)
  1493→        setSaveState('saved')
  1494→      } catch {
  1495→        setSaveState('idle')
  1496→        toast.error('Não foi possível salvar as anotações.')
  1497→      }
  1498→    }, 900)
  1499→    return () => {
  1500→      if (timerRef.current) clearTimeout(timerRef.current)
  1501→    }
  1502→  }, [body, loaded, user, lessonId])
  1503→
  1504→  const downloadNotes = () => {
  1505→    const text = body.trim()
  1506→    if (!text) {
  1507→      toast.error('Escreva algo antes de baixar suas anotações.')
  1508→      return
  1509→    }
  1510→    const blob = new Blob(
  1511→      [`Anotações — ${lessonTitle}\nMentorHub · ${new Date().toLocaleString('pt-BR')}\n\n${text}\n`],
  1512→      { type: 'text/plain;charset=utf-8' }
  1513→    )
  1514→    const url = URL.createObjectURL(blob)
  1515→    const a = document.createElement('a')
  1516→    a.href = url
  1517→    a.download = `anotacoes-${lessonTitle.toLowerCase().replace(/[^a-z0-9]+/gi, '-').slice(0, 40)}.txt`
  1518→    a.click()
  1519→    URL.revokeObjectURL(url)
  1520→    toast.success('Anotações baixadas!')
  1521→  }
  1522→
  1523→  if (!user) {
  1524→    return (
  1525→      <div className="rounded-2xl border border-stone-200 bg-white p-6 text-center text-sm text-stone-500">
  1526→        Entre com uma conta para escrever e guardar suas anotações.
  1527→      </div>
  1528→    )
  1529→  }
  1530→
  1531→  return (
  1532→    <div className="rounded-2xl border border-stone-200 bg-white p-4 sm:p-5">
  1533→      <div className="flex items-center justify-between gap-3">
  1534→        <label htmlFor="lesson-notes" className="text-sm font-bold text-stone-800">
  1535→          Minhas anotações desta aula
  1536→        </label>
  1537→        <span
  1538→          className={cn(
  1539→            'inline-flex items-center gap-1.5 text-xs font-semibold',
  1540→            saveState === 'saving' ? 'text-amber-600' : saveState === 'saved' ? 'text-emerald-600' : 'text-stone-400'
  1541→          )}
  1542→          aria-live="polite"
  1543→        >
  1544→          {saveState === 'saving' ? (
  1545→            'Salvando…'
  1546→          ) : saveState === 'saved' ? (
  1547→            <>
  1548→              <Check aria-hidden className="h-3.5 w-3.5" /> Salvo
  1549→            </>
  1550→          ) : user ? (
  1551→            'Salva automaticamente'
  1552→          ) : (
  1553→            'Entre para salvar'
  1554→          )}
  1555→        </span>
  1556→      </div>
  1557→      <Textarea
  1558→        id="lesson-notes"
  1559→        rows={8}
  1560→        value={body}
  1561→        onChange={(e) => setBody(e.target.value)}
  1562→        placeholder="Anote insights, referências e tarefas desta aula… (salvo automaticamente na sua conta)"
  1563→        className="mt-2 resize-y"
  1564→      />
  1565→      <div className="mt-3 flex items-center justify-between gap-3">
  1566→        <span className="text-xs text-stone-400">
  1567→          {savedAt ? `Última atualização: ${new Date(savedAt).toLocaleString('pt-BR')}` : 'Nenhuma edição ainda'}
  1568→        </span>
  1569→        <Button
  1570→          variant="outline"
  1571→          className="h-10 rounded-full font-semibold"
  1572→          onClick={downloadNotes}
  1573→          disabled={!body.trim()}
  1574→        >
  1575→          <FileDown aria-hidden className="h-4 w-4" /> Baixar .txt
  1576→        </Button>
  1577→      </div>
  1578→    </div>
  1579→  )
  1580→}
