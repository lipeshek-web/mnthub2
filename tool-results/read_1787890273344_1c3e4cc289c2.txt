   170→function BookingCard({
   171→  booking: b,
   172→  userId,
   173→  busy,
   174→  onConfirm,
   175→  onReject,
   176→  onComplete,
   177→  onCancel,
   178→  onJoin,
   179→  onReview,
   180→}: {
   181→  booking: BookingDTO
   182→  userId: string
   183→  busy: boolean
   184→  onConfirm: (booking: BookingDTO) => void
   185→  onReject: (booking: BookingDTO) => void
   186→  onComplete: (booking: BookingDTO) => void
   187→  onCancel: (booking: BookingDTO) => void
   188→  onJoin: (booking: BookingDTO) => void
   189→  onReview: (booking: BookingDTO, rating: number, comment: string) => Promise<void>
   190→}) {
   191→  const isMentorSide = b.mentor.userId === userId
   192→  const otherName = isMentorSide ? b.mentee.name : b.mentor.name
   193→  const relative = relativeDayLabel(b.startsAt)
   194→  const statusMeta = STATUS_META[b.status] ?? {
   195→    label: b.status,
   196→    className: 'bg-stone-100 text-stone-600 border-stone-200',
   197→  }
   198→
   199→  const [dialogAction, setDialogAction] = useState<ConfirmKind | null>(null)
   200→  const [reviewOpen, setReviewOpen] = useState(false)
   201→  const [rating, setRating] = useState(0)
   202→  const [hoverRating, setHoverRating] = useState(0)
   203→  const [comment, setComment] = useState('')
   204→  const [submitting, setSubmitting] = useState(false)
   205→
   206→  const canJoin = b.status === 'CONFIRMED'
   207→  const canReview = !isMentorSide && b.status === 'COMPLETED' && !b.reviewed
   208→
   209→  const dialogContent: Record<ConfirmKind, { title: string; description: string; actionLabel: string }> = {
   210→    reject: {
   211→      title: 'Recusar solicitação?',
   212→      description: `A solicitação de ${b.mentee.name} sobre "${b.topic}" será cancelada e o horário será liberado na sua agenda.`,
   213→      actionLabel: 'Recusar',
   214→    },
   215→    complete: {
   216→      title: 'Marcar como concluída?',
   217→      description: `Confirme que a sessão com ${b.mentee.name} aconteceu. Ela irá para o histórico e o aluno poderá avaliá-la.`,
   218→      actionLabel: 'Concluir sessão',
   219→    },
   220→    cancel: {
   221→      title: 'Cancelar sessão?',
   222→      description: `Sua sessão com ${b.mentor.name} será cancelada e o horário será liberado para outras pessoas.`,
   223→      actionLabel: 'Cancelar sessão',
   224→    },
   225→  }
   226→  const currentDialog = dialogAction ? dialogContent[dialogAction] : null
   227→
   228→  const submitReview = async () => {
   229→    if (rating < 1) {
   230→      toast.error('Escolha uma nota de 1 a 5 estrelas antes de enviar.')
   231→      return
   232→    }
   233→    setSubmitting(true)
   234→    try {
   235→      await onReview(b, rating, comment.trim())
   236→      setReviewOpen(false)
   237→      setRating(0)
   238→      setHoverRating(0)
   239→      setComment('')
   240→    } catch (err) {
   241→      toast.error(err instanceof Error ? err.message : 'Não foi possível enviar sua avaliação.')
   242→    } finally {
   243→      setSubmitting(false)
   244→    }
   245→  }
   246→
   247→  return (
   248→    <Card className="gap-4 py-6">
   249→      <CardContent className="flex flex-col gap-4 px-6">
   250→        <div className="flex flex-wrap items-start justify-between gap-3">
   251→          <div className="flex min-w-0 items-start gap-3">
   252→            <Avatar name={otherName} size="lg" />
   253→            <div className="min-w-0">
   254→              <div className="flex flex-wrap items-center gap-2">
   255→                <p className="font-semibold leading-tight">{otherName}</p>
   256→                <Badge
   257→                  className={
   258→                    isMentorSide
   259→                      ? 'border-emerald-200 bg-emerald-50 text-emerald-700'
   260→                      : 'border-stone-200 bg-stone-100 text-stone-600'
   261→                  }
   262→                >
   263→                  {isMentorSide ? 'Como mentor' : 'Como aluno'}
   264→                </Badge>
   265→              </div>
   266→              {!isMentorSide && b.mentor.headline ? (
   267→                <p className="mt-0.5 text-sm text-muted-foreground">{b.mentor.headline}</p>
   268→              ) : null}
   269→              <p className="mt-1.5 font-medium">{b.topic}</p>
   270→              {b.notes ? <p className="mt-1 text-sm text-muted-foreground">{b.notes}</p> : null}
   271→            </div>
   272→          </div>
   273→          <Badge variant="outline" className={statusMeta.className}>
   274→            {statusMeta.label}
   275→          </Badge>
   276→        </div>
   277→
   278→        <div className="flex flex-wrap items-center gap-x-4 gap-y-1.5 text-sm text-muted-foreground">
   279→          <span className="inline-flex items-center gap-1.5">
   280→            <CalendarDays className="size-4 text-emerald-600" aria-hidden />
   281→            {relative ? `${relative} · ` : ''}
   282→            {formatDayLabelLong(b.startsAt)} · {formatTimeLabel(b.startsAt)} →{' '}
   283→            {addMinutesToTime(b.startsAt, b.durationMin)}
   284→          </span>
   285→          <span className="inline-flex items-center gap-1.5">
   286→            <Clock className="size-4" aria-hidden /> {b.durationMin} min
   287→          </span>
   288→          <span className="inline-flex items-center gap-1.5">
   289→            <Wallet className="size-4" aria-hidden /> {currencyBRL(b.price)}
   290→          </span>
   291→        </div>
   292→
   293→        <div className="flex flex-wrap gap-2">
   294→          {isMentorSide && b.status === 'PENDING' ? (
   295→            <>
   296→              <Button size="sm" disabled={busy} onClick={() => onConfirm(b)}>
   297→                <Check className="size-4" aria-hidden /> Confirmar
   298→              </Button>
   299→              <Button size="sm" variant="outline" disabled={busy} onClick={() => setDialogAction('reject')}>
   300→                <X className="size-4" aria-hidden /> Recusar
   301→              </Button>
   302→            </>
   303→          ) : null}
   304→
   305→          {isMentorSide && b.status === 'CONFIRMED' ? (
   306→            <Button size="sm" variant="outline" disabled={busy} onClick={() => setDialogAction('complete')}>
   307→              <CheckCheck className="size-4" aria-hidden /> Marcar como concluída
   308→            </Button>
   309→          ) : null}
   310→
   311→          {canJoin ? (
   312→            <Button size="sm" onClick={() => onJoin(b)}>
   313→              <Video className="size-4" aria-hidden /> Entrar na sala
   314→            </Button>
   315→          ) : null}
   316→
   317→          {!isMentorSide && (b.status === 'PENDING' || b.status === 'CONFIRMED') ? (
   318→            <Button
   319→              size="sm"
   320→              variant="ghost"
   321→              className="text-rose-600 hover:bg-rose-50 hover:text-rose-700"
   322→              disabled={busy}
   323→              onClick={() => setDialogAction('cancel')}
   324→            >
   325→              Cancelar sessão
   326→            </Button>
   327→          ) : null}
   328→
   329→          {canReview ? (
   330→            <Button size="sm" variant="outline" onClick={() => setReviewOpen(true)}>
   331→              <Star className="size-4 fill-amber-400 text-amber-400" aria-hidden /> Avaliar sessão
   332→            </Button>
   333→          ) : null}
   334→        </div>
   335→      </CardContent>
   336→
   337→      {/* Confirmações: recusar / concluir / cancelar */}
   338→      <AlertDialog
   339→        open={dialogAction !== null}
   340→        onOpenChange={(open) => {
   341→          if (!open) setDialogAction(null)
   342→        }}
   343→      >
   344→        <AlertDialogContent>
   345→          <AlertDialogHeader>
   346→            <AlertDialogTitle>{currentDialog?.title}</AlertDialogTitle>
   347→            <AlertDialogDescription>{currentDialog?.description}</AlertDialogDescription>
   348→          </AlertDialogHeader>
   349→          <AlertDialogFooter>
   350→            <AlertDialogCancel>Voltar</AlertDialogCancel>
   351→            <AlertDialogAction
   352→              className="bg-rose-600 text-white hover:bg-rose-700"
   353→              disabled={busy}
   354→              onClick={() => {
   355→                const action = dialogAction
   356→                setDialogAction(null)
   357→                if (action === 'reject') onReject(b)
   358→                else if (action === 'complete') onComplete(b)
   359→                else if (action === 'cancel') onCancel(b)
   360→              }}
   361→            >
   362→              {currentDialog?.actionLabel}
   363→            </AlertDialogAction>
   364→          </AlertDialogFooter>
   365→        </AlertDialogContent>
   366→      </AlertDialog>
   367→
   368→      {/* Avaliar sessão */}
   369→      <Dialog open={reviewOpen} onOpenChange={setReviewOpen}>
   370→        <DialogContent className="sm:max-w-md">
   371→          <DialogHeader>
   372→            <DialogTitle>Avaliar sessão</DialogTitle>
   373→            <DialogDescription>
   374→              Como foi sua mentoria com {b.mentor.name} sobre &quot;{b.topic}&quot;? Seu feedback ajuda outros
   375→              alunos.
   376→            </DialogDescription>
   377→          </DialogHeader>
   378→          <div className="flex flex-col gap-4">
   379→            <div className="flex items-center justify-center gap-1" role="radiogroup" aria-label="Nota da avaliação">
   380→              {[1, 2, 3, 4, 5].map((value) => {
   381→                const active = (hoverRating || rating) >= value
   382→                return (
   383→                  <button
   384→                    key={value}
   385→                    type="button"
   386→                    role="radio"
   387→                    aria-checked={rating === value}
   388→                    aria-label={`${value} ${value === 1 ? 'estrela' : 'estrelas'}`}
   389→                    onClick={() => setRating(value)}
   390→                    onMouseEnter={() => setHoverRating(value)}
   391→                    onMouseLeave={() => setHoverRating(0)}
   392→                    className="rounded-full p-1 transition-transform hover:scale-110 focus-visible:outline-2 focus-visible:outline-amber-500"
   393→                  >
   394→                    <Star
   395→                      size={28}
   396→                      className={cn(
   397→                        'transition-colors',
   398→                        active ? 'fill-amber-400 text-amber-400' : 'fill-stone-200 text-stone-300'
   399→                      )}
   400→                      aria-hidden
   401→                    />
   402→                  </button>
   403→                )
   404→              })}
   405→            </div>
   406→            <div className="flex flex-col gap-2">
   407→              <Label htmlFor={`review-comment-${b.id}`}>Como foi sua experiência?</Label>
   408→              <Textarea
   409→                id={`review-comment-${b.id}`}
   410→                rows={4}
   411→                value={comment}
   412→                onChange={(event) => setComment(event.target.value)}
   413→                placeholder="Conte o que funcionou bem e o que pode melhorar..."
   414→              />
   415→            </div>
   416→          </div>
   417→          <DialogFooter>
   418→            <Button variant="ghost" onClick={() => setReviewOpen(false)} disabled={submitting}>
   419→              Cancelar
   420→            </Button>
   421→            <Button onClick={() => void submitReview()} disabled={submitting}>
   422→              {submitting ? 'Enviando...' : 'Enviar avaliação'}
   423→            </Button>
   424→          </DialogFooter>
   425→        </DialogContent>
   426→      </Dialog>
   427→    </Card>
   428→  )
   429→}

   433→export default function DashboardView() {
   434→  const user = useAppStore((state) => state.user)
   435→  const navigate = useAppStore((state) => state.navigate)
   436→  const userId = user?.id ?? null
   437→
   438→  const [bookings, setBookings] = useState<BookingDTO[] | null>(null)
   439→  const [mentorProfile, setMentorProfile] = useState<MentorDetailDTO | null>(null)
   440→  const [loading, setLoading] = useState(true)
   441→  const [actingId, setActingId] = useState<string | null>(null)
   442→
   443→  const refetch = useCallback(async () => {
   444→    if (!userId) return
   445→    try {
   446→      const [bookingsRes, profileRes] = await Promise.all([
   447→        api.listBookings(userId),
   448→        api
   449→          .getMyMentorProfile(userId)
   450→          .then((res) => res.profile)
   451→          .catch(() => null),
   452→      ])
   453→      setBookings(bookingsRes)
   454→      setMentorProfile(profileRes)
   455→    } catch (err) {
   456→      toast.error(err instanceof Error ? err.message : 'Não foi possível carregar suas sessões.')
   457→      setBookings((prev) => prev ?? [])
   458→    } finally {
   459→      setLoading(false)
   460→    }
   461→  }, [userId])
   462→
   463→  useEffect(() => {
   464→    if (!userId) return
   465→    setLoading(true)
   466→    void refetch()
   467→  }, [userId, refetch])
   468→
   469→  const runAction = useCallback(
   470→    async (booking: BookingDTO, action: 'confirm' | 'cancel' | 'complete', successMsg: string) => {
   471→      if (!userId) return
   472→      setActingId(booking.id)
   473→      try {
   474→        await api.updateBooking(booking.id, { userId, action })
   475→        toast.success(successMsg)
   476→        await refetch()
   477→      } catch (err) {
   478→        toast.error(err instanceof Error ? err.message : 'Não foi possível atualizar a sessão.')
   479→      } finally {
   480→        setActingId(null)
   481→      }
   482→    },
   483→    [userId, refetch]
   484→  )
   485→
   486→  const handleConfirm = useCallback(
   487→    (booking: BookingDTO) => void runAction(booking, 'confirm', `Sessão com ${booking.mentee.name} confirmada!`),
   488→    [runAction]
   489→  )
   490→  const handleReject = useCallback(
   491→    (booking: BookingDTO) => void runAction(booking, 'cancel', 'Solicitação recusada.'),
   492→    [runAction]
   493→  )
   494→  const handleComplete = useCallback(
   495→    (booking: BookingDTO) =>
   496→      void runAction(booking, 'complete', 'Sessão marcada como concluída! Ela foi para o histórico.'),
   497→    [runAction]
   498→  )
   499→  const handleCancel = useCallback(
   500→    (booking: BookingDTO) => void runAction(booking, 'cancel', 'Sessão cancelada. O horário foi liberado.'),
   501→    [runAction]
   502→  )
   503→  const handleJoin = useCallback(
   504→    (booking: BookingDTO) => navigate({ name: 'meeting', bookingId: booking.id }),
   505→    [navigate]
   506→  )
   507→  const handleReview = useCallback(
   508→    async (booking: BookingDTO, rating: number, comment: string) => {
   509→      if (!userId) return
   510→      await api.createReview({ bookingId: booking.id, userId, rating, comment })
   511→      toast.success('Avaliação enviada. Obrigado pelo feedback!')
   512→      await refetch()
   513→    },
   514→    [userId, refetch]
   515→  )
   516→
   517→  // ---------- Guardas de render ----------
   518→
   519→  if (!user) {
   520→    return (
   521→      <div className="flex min-h-[70vh] items-center justify-center px-4 py-10">
   522→        <Card className="w-full max-w-md p-6 text-center">
   523→          <div className="flex flex-col items-center gap-3">
   524→            <div className="flex size-12 items-center justify-center rounded-full bg-emerald-50 text-emerald-600 ring-1 ring-emerald-200">
   525→              <LogIn className="size-6" aria-hidden />
   526→            </div>
   527→            <h1 className="text-xl font-semibold">Você precisa entrar</h1>
   528→            <p className="max-w-xs text-sm text-muted-foreground">
   529→              Selecione um usuário no seletor do topo da página para ver e gerenciar suas sessões de mentoria.
   530→            </p>
   531→          </div>
   532→        </Card>
   533→      </div>
   534→    )
   535→  }
   536→
   537→  if (loading || bookings === null) {
   538→    return (
   539→      <div className="mx-auto w-full max-w-4xl px-4 py-8 sm:px-6">
   540→        <div className="flex items-center justify-between gap-4">
   541→          <div className="space-y-2">
   542→            <Skeleton className="h-7 w-48" />
   543→            <Skeleton className="h-4 w-72 max-w-full" />
   544→          </div>
   545→          <Skeleton className="h-9 w-40 rounded-md" />
   546→        </div>
   547→        <Skeleton className="mt-6 h-10 w-full max-w-sm rounded-lg" />
   548→        <div className="mt-6 flex flex-col gap-4">
   549→          {[0, 1, 2].map((i) => (
