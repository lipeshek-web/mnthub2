    71→    71→/** Valida o tamanho (máx. 5MB), envia o arquivo e devolve a URL — falhas são exibidas via toast */
    72→    72→async function uploadImageFile(file: File | null | undefined): Promise<string | null> {
    73→    73→  if (!file) return null
    74→    74→  if (file.size > MAX_IMAGE_BYTES) {
    75→    75→    toast.error('A imagem deve ter no máximo 5MB.')
    76→    76→    return null
    77→    77→  }
    78→    78→  try {
    79→    79→    return await api.uploadImage(file)
    80→    80→  } catch (err) {
    81→    81→    toast.error(err instanceof Error ? err.message : 'Não foi possível enviar a imagem.')
    82→    82→    return null
    83→    83→  }
    84→    84→}
