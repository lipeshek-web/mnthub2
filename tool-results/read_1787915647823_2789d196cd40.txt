     4→import {
     5→  AlertCircle,
     6→  ArrowLeft,
     7→  CalendarClock,
     8→  Check,
     9→  CheckCircle2,
    10→  ChevronLeft,
    11→  ChevronRight,

    26→} from 'lucide-react'

    29→import { Button } from '@/components/ui/button'
    30→import { Card } from '@/components/ui/card'
    31→import { Progress } from '@/components/ui/progress'

    45→} from '@/lib/helpers'

    51→} from '@/lib/types'

    54→/* ==================== SALA DE AULA PROFISSIONAL ====================
    55→   Tela cheia (só o header da plataforma): player de um lado, lista de
    56→   conteúdos do outro, com Q&A, anotações e anexos por aula. */
    57→
    58→export function ClassroomView({ courseId }: { courseId: string }) {
    59→  const navigate = useAppStore((s) => s.navigate)
    60→  const user = useAppStore((s) => s.user)
    61→
    62→  const [data, setData] = useState<CourseDetailDTO | null>(null)
    63→  const [loading, setLoading] = useState(true)
    64→  const [error, setError] = useState<string | null>(null)
    65→
    66→  const [currentLessonId, setCurrentLessonId] = useState<string | null>(null)
    67→  const [completedIds, setCompletedIds] = useState<string[]>([])
    68→  const [toggling, setToggling] = useState(false)
    69→
    70→  const fetchCourse = useCallback(async () => {
    71→    setLoading(true)
    72→    setError(null)
    73→    try {
    74→      const d = await api.getCourse(courseId, user?.id)
    75→      setData(d)
    76→    } catch (err) {
    77→      setError(err instanceof Error ? err.message : 'Não foi possível carregar o curso.')
    78→    } finally {
    79→      setLoading(false)
    80→    }
    81→  }, [courseId, user?.id])
    82→
    83→  useEffect(() => {
    84→    void fetchCourse()
    85→  }, [fetchCourse])
    86→
    87→  const course = data
    88→  const isOwner = Boolean(user && course && course.mentor.userId === user.id)
    89→  const isEnrolled = course?.enrollment != null
    90→  const hasAccess = isEnrolled || isOwner
    91→
    92→  const lessons = useMemo<CourseLessonDTO[]>(
    93→    () => [...(course?.lessons ?? [])].sort((a, b) => a.order - b.order),
    94→    [course]
    95→  )
    96→
    97→  // Aula inicial: primeira ainda não concluída (só define uma vez por curso)
    98→  const initializedRef = useRef<string | null>(null)
    99→  useEffect(() => {
   100→    if (!course || !hasAccess) return
   101→    if (initializedRef.current === course.id) return
   102→    initializedRef.current = course.id
   103→    const enrollment = course.enrollment
   104→    const completed = enrollment?.completedLessonIds ?? []
   105→    setCompletedIds(completed)
   106→    const firstIncomplete = lessons.find((l) => !completed.includes(l.id))
   107→    setCurrentLessonId(firstIncomplete?.id ?? lessons[0]?.id ?? null)
   108→  }, [course, hasAccess, lessons])
   109→
   110→  const currentLesson = useMemo(
   111→    () => lessons.find((l) => l.id === currentLessonId) ?? lessons[0] ?? null,
   112→    [lessons, currentLessonId]
   113→  )
   114→  const currentIndex = currentLesson ? lessons.findIndex((l) => l.id === currentLesson.id) : -1
   115→  const percent = lessons.length > 0 ? Math.round((completedIds.length / lessons.length) * 100) : 0
   116→
   117→  const selectLesson = (lessonId: string) => setCurrentLessonId(lessonId)
   118→
   119→  const toggleComplete = async (autoAdvance = false) => {
   120→    if (!user || !course || !currentLesson) return
   121→    setToggling(true)
   122→    try {
   123→      const res = await api.toggleLessonComplete(course.id, {
   124→        userId: user.id,
   125→        lessonId: currentLesson.id,
   126→      })
   127→      const nowCompleted = res.completedLessonIds.includes(currentLesson.id)
   128→      setCompletedIds(res.completedLessonIds)
   129→      if (nowCompleted && res.completedLessonIds.length === lessons.length) {
   130→        toast.success('Curso concluído! Parabéns 🎉')
   131→      } else if (nowCompleted) {
   132→        toast.success('Aula marcada como concluída.')
   133→      } else {
   134→        toast('Marcação removida desta aula.')
   135→      }
   136→      if (autoAdvance && nowCompleted && currentIndex < lessons.length - 1) {
   137→        setCurrentLessonId(lessons[currentIndex + 1].id)
   138→      }
   139→    } catch (err) {
   140→      toast.error(err instanceof Error ? err.message : 'Não foi possível atualizar o progresso.')
   141→    } finally {
   142→      setToggling(false)
   143→    }
   144→  }
   145→
   146→  const goPrev = () => {
   147→    if (currentLesson && currentIndex > 0) setCurrentLessonId(lessons[currentIndex - 1].id)
   148→  }
   149→  const goNext = () => {
   150→    if (currentLesson && currentIndex < lessons.length - 1) {
   151→      setCurrentLessonId(lessons[currentIndex + 1].id)
   152→    }
   153→  }
   154→
   155→  /* ---------- Estados de tela ---------- */
   156→
   157→  if (loading) {
   158→    return (
   159→      <div className="flex h-full min-h-0 flex-col" aria-busy="true">
   160→        <div className="flex items-center gap-3 border-b border-stone-200 bg-white px-4 py-3 sm:px-6">
   161→          <Skeleton className="h-9 w-9 rounded-full" />
   162→          <Skeleton className="h-5 w-48" />
   163→          <Skeleton className="ml-auto h-6 w-28 rounded-full" />
   164→        </div>
   165→        <div className="grid flex-1 grid-cols-1 lg:grid-cols-[1fr_380px]">
   166→          <div className="space-y-4 p-4 sm:p-6">
   167→            <Skeleton className="aspect-video w-full rounded-2xl" />
   168→            <Skeleton className="h-7 w-2/3" />
   169→            <Skeleton className="h-24 w-full rounded-2xl" />
   170→          </div>
   171→          <div className="hidden space-y-2 border-l border-stone-200 bg-white p-4 lg:block">
   172→            {Array.from({ length: 6 }).map((_, i) => (
   173→              <Skeleton key={i} className="h-14 rounded-xl" />
   174→            ))}
   175→          </div>
   176→        </div>
   177→        <p className="sr-only">Abrindo sala de aula…</p>
   178→      </div>
   179→    )
   180→  }
   181→
   182→  if (!course) {
   183→    return (
   184→      <div className="flex h-full items-center justify-center p-6">
   185→        <div className="flex w-full max-w-md flex-col items-center gap-3 rounded-2xl border border-dashed border-stone-300 px-6 py-14 text-center">
   186→          <span className="flex h-14 w-14 items-center justify-center rounded-full bg-rose-50">
   187→            <AlertCircle aria-hidden className="h-7 w-7 text-rose-500" />
   188→          </span>
   189→          <p className="font-bold text-stone-900">Não foi possível abrir a sala de aula</p>
   190→          <p className="max-w-sm text-sm leading-relaxed text-stone-500">
   191→            {error ?? 'Curso não encontrado.'}
   192→          </p>
   193→          <Button variant="outline" className="rounded-full" onClick={() => void fetchCourse()}>
   194→            Tentar novamente
   195→          </Button>
   196→        </div>
   197→      </div>
   198→    )
   199→  }
   200→
   201→  if (!hasAccess) {
   202→    return (
   203→      <div className="flex h-full items-center justify-center p-6">
   204→        <Card className="w-full max-w-md rounded-2xl border-stone-200 p-8 text-center shadow-none">
   205→          <span className="mx-auto flex h-14 w-14 items-center justify-center rounded-full bg-amber-50">
   206→            <Lock aria-hidden className="h-6 w-6 text-amber-600" />
   207→          </span>
   208→          <h1 className="mt-4 text-lg font-extrabold tracking-tight text-stone-900">
   209→            {course.title}
   210→          </h1>
   211→          <p className="mt-2 text-sm leading-relaxed text-stone-500">
   212→            Você ainda não tem acesso a este curso. Inscreva-se para desbloquear todas as aulas,
   213→            perguntas e anotações.
   214→          </p>
   215→          <Button
   216→            className="mt-5 h-11 w-full rounded-full font-bold"
   217→            onClick={() => navigate({ name: 'course', courseId })}
   218→          >
   219→            Ver o curso e se inscrever
   220→          </Button>
   221→        </Card>
   222→      </div>
   223→    )
   224→  }
   225→
   226→  const firstName = course.mentor.name.split(' ')[0]
   227→
   228→  return (
   229→    <div className="flex h-full min-h-0 flex-col bg-stone-50" aria-label={`Sala de aula: ${course.title}`}>
   230→      {/* ---------- BARRA SUPERIOR COMPACTA ---------- */}
   231→      <div className="flex shrink-0 items-center gap-2 border-b border-stone-200 bg-white px-3 py-2.5 sm:gap-3 sm:px-6">
   232→        <Button
   233→          variant="ghost"
   234→          size="icon"
   235→          className="h-9 w-9 shrink-0 rounded-full"
   236→          aria-label="Voltar para a visão geral do curso"
   237→          onClick={() => navigate({ name: 'course', courseId })}
   238→        >
   239→          <ArrowLeft aria-hidden className="h-4 w-4" />
   240→        </Button>
   241→        <div className="min-w-0 flex-1">
   242→          <p className="truncate text-sm font-extrabold tracking-tight text-stone-900 sm:text-base">
   243→            {course.title}
   244→          </p>
   245→          <p className="hidden text-xs text-stone-400 sm:block">
   246→            {lessons.length} {lessons.length === 1 ? 'aula' : 'aulas'} ·{' '}
   247→            {formatTotalDuration(course.totalDurationMin)}
   248→          </p>
   249→        </div>
   250→        <div className="hidden items-center gap-2 sm:flex" aria-hidden>
   251→          <Avatar
   252→            name={course.mentor.name}
   253→            src={course.mentor.avatarUrl}
   254→            size="sm"
   255→            className="h-6 w-6 text-[10px]"
   256→          />
   257→          <span className="max-w-36 truncate text-xs font-semibold text-stone-600">
   258→            {course.mentor.name}
   259→          </span>
   260→        </div>
   261→        <div className="flex w-32 shrink-0 items-center gap-2 sm:w-44">
   262→          <Progress
   263→            value={percent}
   264→            aria-label="Progresso do curso"
   265→            className="h-2 [&_[data-slot=progress-indicator]]:bg-emerald-600"
   266→          />
   267→          <span className="w-9 shrink-0 text-right text-xs font-bold tabular-nums text-stone-600">
   268→            {percent}%
   269→          </span>
   270→        </div>
   271→      </div>
   272→
   273→      {/* ---------- CORPO: PLAYER + CONTEÚDOS ---------- */}
   274→      <div className="grid min-h-0 flex-1 grid-cols-1 lg:grid-cols-[1fr_380px]">
   275→        {/* Coluna esquerda: aula */}
   276→        <div className="min-h-0 overflow-y-auto p-4 sm:p-6 lg:p-8">
   277→          <div className="mx-auto max-w-4xl">
   278→            {currentLesson ? (
   279→              <>
   280→                {currentLesson.kind === 'LIVE' ? (
   281→                  <LivePanel lesson={currentLesson} isOwner={isOwner} />
   282→                ) : (
   283→                  <LessonPlayer lesson={currentLesson} />
   284→                )}
   285→
   286→                <div className="mt-5 flex flex-wrap items-center gap-2">
   287→                  <Badge variant="outline" className="border-stone-200 bg-white text-stone-600">
   288→                    Aula {currentIndex + 1} de {lessons.length}
   289→                  </Badge>
   290→                  {currentLesson.kind === 'LIVE' ? (
   291→                    <Badge className="border-transparent bg-rose-100 text-rose-700">
   292→                      <Radio aria-hidden className="h-3 w-3" /> Ao vivo
   293→                    </Badge>
   294→                  ) : currentLesson.videoUrl ? (
   295→                    <Badge variant="outline" className="border-stone-200 bg-white text-stone-600">
   296→                      Vídeo
   297→                    </Badge>
   298→                  ) : (
   299→                    <Badge variant="outline" className="border-stone-200 bg-white text-stone-600">
   300→                      Leitura
   301→                    </Badge>
   302→                  )}
   303→                  {completedIds.includes(currentLesson.id) && (
   304→                    <Badge className="border-transparent bg-emerald-100 text-emerald-800">
   305→                      <Check aria-hidden className="h-3 w-3" /> Concluída
   306→                    </Badge>
   307→                  )}
   308→                </div>
   309→
   310→                <h1 className="mt-2.5 text-xl font-extrabold tracking-tight text-stone-900 sm:text-2xl">
   311→                  {currentLesson.title}
   312→                </h1>
   313→                {currentLesson.description ? (
   314→                  <p className="mt-1.5 text-sm leading-relaxed text-stone-500">
   315→                    {currentLesson.description}
   316→                  </p>
   317→                ) : null}
   318→
   319→                {/* Anexos para download */}
   320→                {currentLesson.attachments.length > 0 ? (
   321→                  <div className="mt-4 flex flex-wrap gap-2">
   322→                    {currentLesson.attachments.map((att, i) => (
   323→                      <a
   324→                        key={`${att.url}-${i}`}
   325→                        href={att.url}
   326→                        download
   327→                        target="_blank"
   328→                        rel="noopener noreferrer"
   329→                        className="inline-flex min-h-11 items-center gap-2 rounded-full border border-stone-200 bg-white px-4 text-sm font-semibold text-stone-700 transition-colors hover:border-emerald-300 hover:bg-emerald-50 hover:text-emerald-800"
   330→                        aria-label={`Baixar anexo ${att.name}`}
   331→                      >
   332→                        <Paperclip aria-hidden className="h-3.5 w-3.5 text-stone-400" />
   333→                        <span className="max-w-52 truncate">{att.name}</span>
   334→                        <Download aria-hidden className="h-3.5 w-3.5 text-emerald-600" />
   335→                      </a>
   336→                    ))}
   337→                  </div>
   338→                ) : null}
   339→
   340→                {/* Abas: conteúdo / perguntas / anotações */}
   341→                <Tabs defaultValue="lesson" className="mt-6">
   342→                  <TabsList className="h-11 w-full justify-start overflow-x-auto rounded-full bg-stone-100 p-1 sm:w-auto">
   343→                    <TabsTrigger value="lesson" className="rounded-full">
   344→                      <FileText aria-hidden className="h-4 w-4" />
   345→                      <span className="hidden sm:inline">Material</span>
   346→                    </TabsTrigger>
   347→                    <TabsTrigger value="qa" className="rounded-full">
   348→                      <MessageCircle aria-hidden className="h-4 w-4" />
   349→                      Perguntas
   350→                      {currentLesson.questionCount > 0 ? ` (${currentLesson.questionCount})` : ''}
   351→                    </TabsTrigger>
   352→                    <TabsTrigger value="notes" className="rounded-full">
   353→                      <PencilLine aria-hidden className="h-4 w-4" />
   354→                      <span className="hidden sm:inline">Anotações</span>
   355→                    </TabsTrigger>
   356→                  </TabsList>
   357→
   358→                  <TabsContent value="lesson" className="mt-4">
   359→                    {currentLesson.content ? (
   360→                      <article className="rounded-2xl border border-stone-200 bg-white p-5 sm:p-6">
   361→                        <div className="max-w-prose space-y-4">
   362→                          {currentLesson.content.split(/\n{2,}/).map((para, i) => (
   363→                            <p
   364→                              key={i}
   365→                              className="whitespace-pre-line text-[15px] leading-relaxed text-stone-700"
   366→                            >
   367→                              {para}
   368→                            </p>
   369→                          ))}
   370→                        </div>
   371→                      </article>
   372→                    ) : currentLesson.kind === 'LIVE' ? (
   373→                      <div className="rounded-2xl border border-stone-200 bg-white p-5 text-sm leading-relaxed text-stone-600 sm:p-6">
   374→                        Esta é uma aula ao vivo — participe no horário agendado e use a aba
   375→                        <strong className="font-semibold"> Perguntas </strong>
   376→                        para enviar suas dúvidas ao mentor. Se perdeu, a gravação aparece no player
   377→                        quando disponível.
   378→                      </div>
   379→                    ) : currentLesson.videoUrl ? (
   380→                      <div className="rounded-2xl border border-stone-200 bg-white p-5 text-sm leading-relaxed text-stone-600 sm:p-6">
   381→                        Assista à aula acima. Dúvidas? Use a aba{' '}
   382→                        <strong className="font-semibold">Perguntas</strong> — {firstName} responde
   383→                        por lá. Anote ideias em <strong className="font-semibold">Anotações</strong>{' '}
   384→                        e baixe os anexos desta aula.
   385→                      </div>
   386→                    ) : (
   387→                      <div className="flex flex-col items-center gap-2 rounded-2xl border border-dashed border-stone-300 bg-white px-6 py-10 text-center">
   388→                        <FileText aria-hidden className="h-6 w-6 text-stone-300" />
   389→                        <p className="text-sm text-stone-400">
   390→                          O material desta aula será publicado em breve.
   391→                        </p>
   392→                      </div>
   393→                    )}
   394→                  </TabsContent>
   395→
   396→                  <TabsContent value="qa" className="mt-4">
   397→                    <LessonQuestions
   398→                      lesson={currentLesson}
   399→                      isOwner={isOwner}
   400→                      mentorName={course.mentor.name}
   401→                      user={user}
   402→                    />
   403→                  </TabsContent>
   404→
   405→                  <TabsContent value="notes" className="mt-4">
   406→                    <LessonNotes
   407→                      key={currentLesson.id}
   408→                      lessonId={currentLesson.id}
   409→                      user={user}
   410→                      lessonTitle={currentLesson.title}
   411→                    />
   412→                  </TabsContent>
   413→                </Tabs>
   414→
   415→                {/* Navegação */}
   416→                <div className="mt-6 flex items-center justify-between gap-3 border-t border-stone-200 pt-5">
   417→                  <Button
   418→                    variant="outline"
   419→                    className="h-11 rounded-full font-semibold"
   420→                    onClick={goPrev}
   421→                    disabled={currentIndex <= 0 || toggling}
   422→                  >
   423→                    <ChevronLeft aria-hidden className="h-4 w-4" /> Anterior
   424→                  </Button>
   425→                  <Button
   426→                    variant={completedIds.includes(currentLesson.id) ? 'outline' : 'default'}
   427→                    onClick={() => void toggleComplete(true)}
   428→                    disabled={toggling}
   429→                    className={cn(
   430→                      'h-11 rounded-full font-semibold',
   431→                      completedIds.includes(currentLesson.id) &&
   432→                        'border-emerald-200 bg-emerald-50 text-emerald-700 hover:bg-emerald-100 hover:text-emerald-800'
   433→                    )}
   434→                  >
   435→                    {toggling ? (
   436→                      'Salvando…'
   437→                    ) : completedIds.includes(currentLesson.id) ? (
   438→                      <>
   439→                        <CheckCircle2 aria-hidden className="h-4 w-4" /> Concluída
   440→                      </>
   441→                    ) : (
   442→                      <>
   443→                        <CheckCircle2 aria-hidden className="h-4 w-4" /> Concluir e avançar
   444→                      </>
   445→                    )}
   446→                  </Button>
   447→                  <Button
   448→                    variant="outline"
   449→                    className="h-11 rounded-full font-semibold"
   450→                    onClick={goNext}
   451→                    disabled={currentIndex >= lessons.length - 1 || toggling}
   452→                  >
   453→                    Próxima <ChevronRight aria-hidden className="h-4 w-4" />
   454→                  </Button>
   455→                </div>
   456→              </>
   457→            ) : (
   458→              <div className="flex flex-col items-center gap-2 rounded-2xl border border-dashed border-stone-300 bg-white px-6 py-16 text-center">
   459→                <Library aria-hidden className="h-8 w-8 text-stone-300" />
   460→                <p className="text-sm text-stone-400">
   461→                  Nenhuma aula publicada ainda — volte mais tarde.
   462→                </p>
   463→              </div>
   464→            )}
   465→          </div>
   466→        </div>
   467→
   468→        {/* Coluna direita: lista de conteúdos */}
   469→        <aside
   470→          aria-label="Conteúdos do curso"
   471→          className="flex min-h-0 flex-col border-t border-stone-200 bg-white lg:border-l lg:border-t-0"
   472→        >
   473→          <div className="flex shrink-0 items-center justify-between gap-2 px-4 pb-2 pt-4">
   474→            <p className="text-xs font-bold uppercase tracking-wider text-stone-400">
   475→              Conteúdos do curso
   476→            </p>
   477→            <span className="text-xs font-semibold text-emerald-700">
   478→              {completedIds.length}/{lessons.length}
   479→            </span>
   480→          </div>
   481→          <nav className="min-h-0 flex-1 space-y-1 overflow-y-auto px-3 pb-3 [scrollbar-width:thin] [&::-webkit-scrollbar-thumb]:rounded-full [&::-webkit-scrollbar-thumb]:bg-stone-300 [&::-webkit-scrollbar-track]:bg-stone-100 [&::-webkit-scrollbar]:w-1.5">
   482→            {lessons.map((lesson, i) => {
   483→              const isCurrent = currentLesson?.id === lesson.id
   484→              const isCompleted = completedIds.includes(lesson.id)
   485→              const live = lesson.kind === 'LIVE'
   486→              return (
   487→                <button
   488→                  key={lesson.id}
   489→                  onClick={() => selectLesson(lesson.id)}
   490→                  aria-current={isCurrent ? 'true' : undefined}
   491→                  className={cn(
   492→                    'flex min-h-14 w-full items-center gap-3 rounded-xl border p-3 text-left transition-colors',
   493→                    isCurrent
   494→                      ? 'border-emerald-300 bg-emerald-50/80 shadow-[inset_0_0_0_1px_rgba(16,185,129,0.15)]'
   495→                      : 'border-transparent hover:bg-stone-50'
   496→                  )}
   497→                >
   498→                  <span
   499→                    className={cn(

   716→/* ==================== Q&A DA AULA ==================== */
   717→
   718→function LessonQuestions({
   719→  lesson,
   720→  isOwner,
   721→  mentorName,
   722→  user,
   723→}: {
   724→  lesson: CourseLessonDTO
   725→  isOwner: boolean
   726→  mentorName: string
   727→  user: { id: string; name: string } | null
   728→}) {
   729→  const [questions, setQuestions] = useState<LessonQuestionDTO[] | null>(null)
   730→  const [body, setBody] = useState('')
   731→  const [sending, setSending] = useState(false)
   732→  const [answerDraft, setAnswerDraft] = useState<Record<string, string>>({})
   733→  const [answeringId, setAnsweringId] = useState<string | null>(null)
   734→
   735→  const load = useCallback(async () => {
   736→    if (!user) {
   737→      setQuestions([])
   738→      return
   739→    }
   740→    try {
   741→      const list = await api.listLessonQuestions(lesson.id, user.id)
   742→      setQuestions(list)
   743→    } catch {
   744→      setQuestions([])
   745→    }
   746→  }, [lesson.id, user])
   747→
   748→  useEffect(() => {
   749→    setQuestions(null)
   750→    void load()
   751→  }, [load])
   752→
   753→  const ask = async () => {
   754→    if (!user) {
   755→      toast.error('Entre com uma conta para perguntar.')
   756→      return
   757→    }
   758→    if (body.trim().length < 5) {
   759→      toast.error('Escreva sua pergunta (mín. 5 caracteres).')
   760→      return
   761→    }
   762→    setSending(true)
   763→    try {
   764→      const created = await api.askLessonQuestion(lesson.id, { userId: user.id, body: body.trim() })
   765→      setQuestions((prev) => [created, ...(prev ?? [])])
   766→      setBody('')
   767→      toast.success('Pergunta enviada! O mentor será notificado.')
   768→    } catch (err) {
   769→      toast.error(err instanceof Error ? err.message : 'Não foi possível enviar a pergunta.')
   770→    } finally {
   771→      setSending(false)
   772→    }
   773→  }
   774→
   775→  const answer = async (questionId: string) => {
   776→    const text = (answerDraft[questionId] ?? '').trim()
   777→    if (!user || text.length < 2) {
   778→      toast.error('Escreva a resposta antes de enviar.')
   779→      return
   780→    }
   781→    setAnsweringId(questionId)
   782→    try {
   783→      const res = await api.answerQuestion(questionId, { userId: user.id, answer: text })
   784→      setQuestions((prev) =>
   785→        (prev ?? []).map((q) =>
   786→          q.id === questionId ? { ...q, answer: res.answer, answeredAt: res.answeredAt } : q
   787→        )
   788→      )
   789→      setAnswerDraft((prev) => ({ ...prev, [questionId]: '' }))
   790→      toast.success('Resposta publicada!')
   791→    } catch (err) {
   792→      toast.error(err instanceof Error ? err.message : 'Não foi possível responder.')
   793→    } finally {
   794→      setAnsweringId(null)
   795→    }
   796→  }
   797→
   798→  const remove = async (questionId: string) => {
   799→    if (!user) return
   800→    try {
   801→      await api.deleteQuestion(questionId, user.id)
   802→      setQuestions((prev) => (prev ?? []).filter((q) => q.id !== questionId))
   803→      toast('Pergunta removida.')
   804→    } catch (err) {
   805→      toast.error(err instanceof Error ? err.message : 'Não foi possível remover.')
   806→    }
   807→  }
   808→
   809→  if (!user) {
   810→    return (
   811→      <div className="rounded-2xl border border-stone-200 bg-white p-6 text-center text-sm text-stone-500">
   812→        Entre com uma conta no topo da página para participar das perguntas.
   813→      </div>
   814→    )
   815→  }
   816→
   817→  return (
   818→    <div className="space-y-4">
   819→      {/* Formulário de pergunta (aluno) */}
   820→      {!isOwner ? (
   821→        <div className="rounded-2xl border border-stone-200 bg-white p-4 sm:p-5">
   822→          <label htmlFor="question-body" className="text-sm font-bold text-stone-800">
   823→            Tirou uma dúvida? Pergunte ao mentor
   824→          </label>
   825→          <Textarea
   826→            id="question-body"
   827→            rows={3}
   828→            value={body}
   829→            onChange={(e) => setBody(e.target.value)}
   830→            placeholder={`Ex.: "${mentorName.split(' ')[0]}", no minuto 4:30 você menciona X — como aplico isso na prática?`}
   831→            className="mt-2 resize-none"
   832→            maxLength={1200}
   833→          />
   834→          <div className="mt-3 flex items-center justify-between gap-3">
   835→            <span className="text-xs text-stone-400">{body.length}/1200</span>
   836→            <Button
   837→              className="h-10 rounded-full font-semibold"
   838→              onClick={() => void ask()}
   839→              disabled={sending || body.trim().length < 5}
   840→            >
   841→              <Send aria-hidden className="h-4 w-4" />
   842→              {sending ? 'Enviando…' : 'Enviar pergunta'}
   843→            </Button>
   844→          </div>
   845→        </div>
   846→      ) : null}
   847→
   848→      {/* Lista */}
   849→      {questions === null ? (
   850→        <div className="space-y-3">
   851→          <Skeleton className="h-20 rounded-2xl" />
   852→          <Skeleton className="h-20 rounded-2xl" />
   853→        </div>
   854→      ) : questions.length === 0 ? (
   855→        <div className="flex flex-col items-center gap-2 rounded-2xl border border-dashed border-stone-300 bg-white px-6 py-10 text-center">
   856→          <MessageCircle aria-hidden className="h-6 w-6 text-stone-300" />
   857→          <p className="text-sm text-stone-400">
   858→            Nenhuma pergunta ainda — {isOwner ? 'seus alunos aparecerão aqui.' : 'seja o primeiro a perguntar!'}
   859→          </p>
   860→        </div>
   861→      ) : (
   862→        <ul className="space-y-3">
   863→          {questions.map((q) => (
   864→            <li key={q.id} className="rounded-2xl border border-stone-200 bg-white p-4 sm:p-5">
   865→              <div className="flex items-start gap-3">
   866→                <Avatar name={q.author.name} src={q.author.avatarUrl} size="sm" />
   867→                <div className="min-w-0 flex-1">
   868→                  <p className="text-sm font-bold text-stone-800">
   869→                    {q.author.name}
   870→                    {q.isMine ? <span className="ml-1.5 text-xs font-semibold text-stone-400">(você)</span> : null}
   871→                  </p>
   872→                  <p className="mt-1 whitespace-pre-line text-sm leading-relaxed text-stone-700">
   873→                    {q.body}
   874→                  </p>
   875→                </div>
   876→                {q.isMine && !q.answer ? (
   877→                  <Button
   878→                    variant="ghost"
   879→                    size="icon"
   880→                    className="h-8 w-8 shrink-0 rounded-full text-stone-400 hover:bg-rose-50 hover:text-rose-600"
   881→                    aria-label="Remover minha pergunta"
   882→                    onClick={() => void remove(q.id)}
   883→                  >
   884→                    <Trash2 aria-hidden className="h-3.5 w-3.5" />
   885→                  </Button>
   886→                ) : null}
   887→              </div>
   888→
   889→              {q.answer ? (
   890→                <div className="mt-3 rounded-xl border border-emerald-100 bg-emerald-50/70 p-3.5">
   891→                  <p className="flex items-center gap-1.5 text-xs font-extrabold uppercase tracking-wide text-emerald-700">
   892→                    <Users aria-hidden className="h-3 w-3" /> Resposta de {mentorName}
   893→                  </p>
   894→                  <p className="mt-1.5 whitespace-pre-line text-sm leading-relaxed text-emerald-900">
   895→                    {q.answer}
   896→                  </p>
   897→                </div>
   898→              ) : isOwner ? (
   899→                <div className="mt-3 rounded-xl border border-amber-100 bg-amber-50/60 p-3.5">
   900→                  <p className="text-xs font-extrabold uppercase tracking-wide text-amber-700">
   901→                    Aguardando sua resposta
   902→                  </p>
   903→                  <Textarea
   904→                    rows={2}
   905→                    value={answerDraft[q.id] ?? ''}
   906→                    onChange={(e) => setAnswerDraft((prev) => ({ ...prev, [q.id]: e.target.value }))}
   907→                    placeholder="Escreva a resposta para o aluno..."
   908→                    className="mt-2 resize-none border-white bg-white"
   909→                    maxLength={2000}
   910→                    aria-label={`Responder pergunta de ${q.author.name}`}
   911→                  />
   912→                  <div className="mt-2 flex justify-end">
   913→                    <Button
   914→                      size="sm"
   915→                      className="h-9 rounded-full font-semibold"
   916→                      onClick={() => void answer(q.id)}
   917→                      disabled={answeringId === q.id || (answerDraft[q.id] ?? '').trim().length < 2}
   918→                    >
   919→                      {answeringId === q.id ? 'Enviando…' : 'Responder'}
   920→                    </Button>
   921→                  </div>
   922→                </div>
   923→              ) : (
   924→                <p className="mt-3 inline-flex items-center gap-1.5 rounded-full bg-amber-50 px-3 py-1 text-xs font-semibold text-amber-700">
   925→                  <MessageCircle aria-hidden className="h-3 w-3" /> Aguardando resposta do mentor
   926→                </p>
   927→              )}
   928→            </li>
   929→          ))}
   930→        </ul>
   931→      )}
   932→    </div>
   933→  )

   936→/* ==================== ANOTAÇÕES DA AULA ==================== */
   937→
   938→function LessonNotes({
   939→  lessonId,
   940→  user,
   941→  lessonTitle,
   942→}: {
   943→  lessonId: string
   944→  user: { id: string; name: string } | null
   945→  lessonTitle: string
   946→}) {
   947→  const [body, setBody] = useState('')
   948→  const [loaded, setLoaded] = useState(false)
   949→  const [saveState, setSaveState] = useState<'idle' | 'saving' | 'saved'>('idle')
   950→  const [savedAt, setSavedAt] = useState<string | null>(null)
   951→  const timerRef = useRef<ReturnType<typeof setTimeout> | null>(null)
   952→  const latestRef = useRef('')
   953→
   954→  // Carrega as anotações da aula (o componente é remontado por aula via key)
   955→  useEffect(() => {
   956→    if (!user) return
   957→    let active = true
   958→    api
   959→      .getLessonNote(lessonId, user.id)
   960→      .then((note) => {
   961→        if (!active) return
   962→        setBody(note.body)
   963→        latestRef.current = note.body
   964→        setSavedAt(note.updatedAt)
   965→      })
   966→      .catch(() => {
   967→        /* sem anotações ainda */
   968→      })
   969→      .finally(() => {
   970→        if (active) setLoaded(true)
   971→      })
   972→    return () => {
   973→      active = false
   974→    }
   975→  }, [lessonId, user])
   976→
   977→  // Autosave com debounce (o estado "salvando" aparece no envio)
   978→  useEffect(() => {
   979→    if (!loaded || !user) return
   980→    if (body === latestRef.current) return
   981→    if (timerRef.current) clearTimeout(timerRef.current)
   982→    timerRef.current = setTimeout(async () => {
   983→      setSaveState('saving')
   984→      try {
   985→        const res = await api.saveLessonNote(lessonId, { userId: user.id, body })
   986→        latestRef.current = body
   987→        setSavedAt(res.updatedAt)
   988→        setSaveState('saved')
   989→      } catch {
   990→        setSaveState('idle')
   991→        toast.error('Não foi possível salvar as anotações.')
   992→      }
   993→    }, 900)
   994→    return () => {
   995→      if (timerRef.current) clearTimeout(timerRef.current)
   996→    }
   997→  }, [body, loaded, user, lessonId])
   998→
   999→  const downloadNotes = () => {
  1000→    const text = body.trim()
  1001→    if (!text) {
  1002→      toast.error('Escreva algo antes de baixar suas anotações.')
  1003→      return
  1004→    }
  1005→    const blob = new Blob(
  1006→      [`Anotações — ${lessonTitle}\nMentorHub · ${new Date().toLocaleString('pt-BR')}\n\n${text}\n`],
  1007→      { type: 'text/plain;charset=utf-8' }
  1008→    )
  1009→    const url = URL.createObjectURL(blob)
