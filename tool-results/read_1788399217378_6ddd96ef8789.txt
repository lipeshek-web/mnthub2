    57→    57→/** Máscara leve: 000.000.000-00 ou 00.000.000/0000-00 conforme digita. */
    58→    58→function maskCpfCnpj(raw: string): string {
    59→    59→  const d = raw.replace(/\D/g, "").slice(0, 14);
    60→    60→  if (d.length <= 11) {
    61→    61→    return d
    62→    62→      .replace(/(\d{3})(\d)/, "$1.$2")
    63→    63→      .replace(/(\d{3})(\d)/, "$1.$2")
    64→    64→      .replace(/(\d{3})(\d{1,2})$/, "$1-$2");
    65→    65→  }
    66→    66→  return d
    67→    67→    .replace(/(\d{2})(\d)/, "$1.$2")
    68→    68→    .replace(/(\d{3})(\d)/, "$1.$2")
    69→    69→    .replace(/(\d{3})(\d)/, "$1/$2")
    70→    70→    .replace(/(\d{4})(\d{1,2})$/, "$1-$2");
    71→    71→}
