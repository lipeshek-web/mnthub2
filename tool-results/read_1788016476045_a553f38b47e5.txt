   104→   104→            value: d.price,
   105→   105→            contentName: d.title,
   106→   106→          })

   117→   117→            mentorGaId: d.mentor.tracking?.gaMeasurementId,

   119→   119→          })
   120→   120→          trackEvent('begin_checkout', {
   121→   121→            mentorId: d.mentor.id,
   122→   122→            courseId: d.id,
   123→   123→            value: d.price,
   124→   124→            contentName: d.title,
   125→   125→          })
   126→   126→        }
   127→   127→      } else {
   128→   128→        setError('Item não encontrado.')
   129→   129→      }
   130→   130→    } catch (err) {
   131→   131→      setError(err instanceof Error ? err.message : 'Não foi possível carregar o checkout.')
   132→   132→    } finally {
   133→   133→      setLoading(false)
   134→   134→    }
   135→   135→  }, [courseId, trackId, user?.id])
   136→   136→
   137→   137→  useEffect(() => {
   138→   138→    void fetchSummary()
   139→   139→  }, [fetchSummary])
   140→   140→
   141→   141→  /* ---------- Ações ---------- */
   142→   142→
   143→   143→  const doEnrollFree = async () => {
   144→   144→    if (!user || !item) return
   145→   145→    setFreeEnrolling(true)

   148→   148→        await api.enrollTrack(item.id, user.id)

   154→   154→          res.alreadyEnrolled

   159→   159→      }
   160→   160→    } catch (err) {

   198→   198→    }
   199→   199→  }
   200→   200→
   201→   201→  /* ---------- Skeleton ---------- */

   205→   205→        <Skeleton className="h-9 w-28 rounded-full" />
   206→   206→        <Skeleton className="mt-4 h-64 w-full rounded-2xl" />
   207→   207→        <Skeleton className="mt-6 h-24 w-full rounded-2xl" />

   212→   212→  }
   213→   213→

   217→   217→      <div className="mx-auto w-full max-w-3xl px-4 py-16">

   219→   219→          <CardContent className="flex flex-col items-center gap-3 p-10 text-center">
   220→   220→            <span className="flex h-14 w-14 items-center justify-center rounded-full bg-rose-50">
   221→   221→              <AlertCircle aria-hidden className="h-7 w-7 text-rose-500" />

   225→   225→              {error ?? 'Item não encontrado.'}
   226→   226→            </p>
   227→   227→            <Button variant="outline" className="rounded-full" onClick={() => void fetchSummary()}>
   228→   228→              Tentar novamente
   229→   229→            </Button>
   230→   230→          </CardContent>
   231→   231→        </Card>
   232→   232→      </div>
   233→   233→    )
   234→   234→  }
   235→   235→
   236→   236→  /* ---------- Resumo normalizado (curso ou trilha) ---------- */
   237→   237→  const item: ItemSummary | null = track
   238→   238→    ? {
   239→   239→        kind: 'track',
   240→   240→        id: track.id,
   241→   241→        title: track.title,
   242→   242→        coverUrl: track.coverUrl ?? null,
   243→   243→        category: track.category,
   244→   244→        level: track.level,
   245→   245→        price: track.price,
   246→   246→        mentor: {
   247→   247→          id: track.mentor.id,
   248→   248→          name: track.mentor.name,
   249→   249→          rating: track.mentor.rating,
   250→   250→          reviewCount: track.mentor.reviewCount,
   251→   251→          avatarUrl: track.mentor.avatarUrl,
   252→   252→        },
   253→   253→        isEnrolled: track.myEnrollment !== null,
   254→   254→        trackStats: {
   255→   255→          courseCount: track.courseCount,
   256→   256→          mentorshipSessions: track.mentorshipSessions,
   257→   257→        },
   258→   258→      }
   259→   259→    : course
   260→   260→      ? {
   261→   261→          kind: 'course',
   262→   262→          id: course.id,
   263→   263→          title: course.title,
   264→   264→          coverUrl: course.coverUrl ?? null,
   265→   265→          category: course.category,
   266→   266→          level: course.level,
   267→   267→          price: course.price,
   268→   268→          mentor: {
   269→   269→            id: course.mentor.id,
   270→   270→            name: course.mentor.name,
   271→   271→            rating: course.mentor.rating,
   272→   272→            reviewCount: course.mentor.reviewCount,
   273→   273→            avatarUrl: course.mentor.avatarUrl,
   274→   274→          },
   275→   275→          isEnrolled: course.enrollment !== null,
   276→   276→        }
   277→   277→      : null
   278→   278→
   279→   279→  if (!item) return null

   287→   287→        <Card className="mx-auto max-w-md rounded-2xl">
   288→   288→          <CardContent className="flex flex-col items-center gap-1.5 p-8 text-center">

   339→   339→                  onClick={() => navigate({ name: 'course', courseId: item.id })}

   346→   346→                  onClick={() => navigate({ name: 'marketplace' })}
   347→   347→                  className="w-full rounded-full"
   348→   348→                >
   349→   349→                  Explorar mais cursos
   350→   350→                </Button>

   353→   353→          </CardContent>

   380→   380→      {!user && (
