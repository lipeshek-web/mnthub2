    97→export function ClassroomView({ courseId }: { courseId: string }) {
    98→  const navigate = useAppStore((s) => s.navigate)
    99→  const user = useAppStore((s) => s.user)
   100→
   101→  const [data, setData] = useState<CourseDetailDTO | null>(null)
   102→  const [loading, setLoading] = useState(true)
   103→  const [error, setError] = useState<string | null>(null)
   104→
   105→  const [currentLessonId, setCurrentLessonId] = useState<string | null>(null)
   106→  const [completedIds, setCompletedIds] = useState<string[]>([])
   107→  const [toggling, setToggling] = useState(false)
   108→
   109→  // Auto-retry para falhas transitórias de rede/servidor (ex.: dev server reiniciando)
   110→  const retriesRef = useRef(0)
   111→  const retryTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null)
   112→
   113→  const fetchCourse = useCallback(async () => {
   114→    setLoading(true)
   115→    setError(null)
   116→    try {
   117→      const d = await api.getCourse(courseId, user?.id)
   118→      retriesRef.current = 0
   119→      setData(d)
   120→    } catch (err) {
   121→      const message = err instanceof Error ? err.message : 'Não foi possível carregar o curso.'
   122→      const transient =
   123→        /inesperado|fetch|network|load failed|conex/i.test(message) && retriesRef.current < 2
   124→      if (transient) {
   125→        retriesRef.current += 1
   126→        if (retryTimerRef.current) clearTimeout(retryTimerRef.current)
   127→        retryTimerRef.current = setTimeout(() => void fetchCourse(), 1200)
   128→      } else {
   129→        setError(message)
   130→      }
   131→    } finally {
   132→      setLoading(false)
   133→    }
   134→  }, [courseId, user?.id])
   135→
   136→  useEffect(() => {
   137→    void fetchCourse()
   138→    return () => {
   139→      if (retryTimerRef.current) clearTimeout(retryTimerRef.current)
   140→    }
   141→  }, [fetchCourse])
   142→
   143→  const course = data
   144→  const isOwner = Boolean(user && course && course.mentor.userId === user.id)
   145→  const isEnrolled = course?.enrollment != null
   146→  const hasAccess = isEnrolled || isOwner
   147→
   148→  const lessons = useMemo<CourseLessonDTO[]>(
   149→    () => [...(course?.lessons ?? [])].sort((a, b) => a.order - b.order),
   150→    [course]
   151→  )
   152→
   153→  // Temas (módulos) em ordem; aulas sem tema (ou com tema removido) vão para o
   154→  // grupo virtual "Outros conteúdos", sempre no fim da lista.
   155→  const themeList = useMemo<CourseThemeDTO[]>(
   156→    () => [...(course?.themes ?? [])].sort((a, b) => a.order - b.order),
   157→    [course]
   158→  )
   159→
   160→  const lessonGroups = useMemo<LessonGroup[]>(() => {
   161→    const byTheme = new Map<string, CourseLessonDTO[]>()
   162→    const loose: CourseLessonDTO[] = []
   163→    for (const lesson of lessons) {
   164→      if (lesson.themeId && themeList.some((t) => t.id === lesson.themeId)) {
   165→        const bucket = byTheme.get(lesson.themeId) ?? []
   166→        bucket.push(lesson)
   167→        byTheme.set(lesson.themeId, bucket)
   168→      } else {
   169→        loose.push(lesson)
   170→      }
   171→    }
   172→    const groups: LessonGroup[] = themeList.map((theme) => ({
   173→      key: theme.id,
   174→      title: theme.title,
   175→      theme,
   176→      lessons: byTheme.get(theme.id) ?? [],
   177→    }))
   178→    if (loose.length > 0) {
   179→      groups.push({ key: 'none', title: 'Outros conteúdos', theme: null, lessons: loose })
   180→    }
   181→    return groups
   182→  }, [lessons, themeList])
   183→
   184→  // Lista achatada (temas em ordem → aulas em ordem): base do índice global,
   185→  // do avançar/voltar e do progresso geral.
   186→  const orderedLessons = useMemo(() => lessonGroups.flatMap((g) => g.lessons), [lessonGroups])
   187→
   188→  // Seções recolhidas da sidebar (default: todas expandidas)
   189→  const [collapsedThemes, setCollapsedThemes] = useState<Record<string, boolean>>({})
   190→  // Emissão do certificado (botão da celebração de 100%)
   191→  const [issuing, setIssuing] = useState(false)
   192→
   193→  // Modo foco (teatro): esconde sidebar e abas, centraliza o palco da aula.
   194→  // Persistido na sessão para o aluno retomar a imersão ao voltar para a sala.
   195→  const [focusMode, setFocusMode] = useState(() => {
   196→    if (typeof window === 'undefined') return false
   197→    try {
   198→      return window.sessionStorage.getItem('mentorhub-classroom-focus') === '1'
   199→    } catch {
   200→      return false
   201→    }
   202→  })
   203→  const [contentsOpen, setContentsOpen] = useState(false)
   204→
   205→  useEffect(() => {
   206→    try {
   207→      if (focusMode) window.sessionStorage.setItem('mentorhub-classroom-focus', '1')
   208→      else window.sessionStorage.removeItem('mentorhub-classroom-focus')
   209→    } catch {
   210→      /* storage indisponível */
   211→    }
   212→  }, [focusMode])
   213→
   214→  // Aula inicial: a indicada na view (retorno do leitor) ou a primeira ainda
   215→  // não concluída (só define uma vez por curso)
   216→  const initializedRef = useRef<string | null>(null)
   217→  useEffect(() => {
   218→    if (!course || !hasAccess) return
   219→    if (initializedRef.current === course.id) return
   220→    initializedRef.current = course.id
   221→    const enrollment = course.enrollment
   222→    const completed = enrollment?.completedLessonIds ?? []
   223→    setCompletedIds(completed)
   224→    const view = useAppStore.getState().view
   225→    const returnLessonId = view.name === 'classroom' ? view.lessonId : undefined
   226→    const target =
   227→      (returnLessonId && orderedLessons.some((l) => l.id === returnLessonId) ? returnLessonId : undefined) ??
   228→      orderedLessons.find((l) => !completed.includes(l.id))?.id ??
   229→      orderedLessons[0]?.id ??
   230→      null
   231→    setCurrentLessonId(target)
   232→  }, [course, hasAccess, orderedLessons])
   233→
   234→  const currentLesson = useMemo(
   235→    () => orderedLessons.find((l) => l.id === currentLessonId) ?? orderedLessons[0] ?? null,
   236→    [orderedLessons, currentLessonId]
   237→  )
   238→  const currentIndex = currentLesson ? orderedLessons.findIndex((l) => l.id === currentLesson.id) : -1
   239→  const percent =
   240→    orderedLessons.length > 0 ? Math.round((completedIds.length / orderedLessons.length) * 100) : 0
   241→  const courseCompleted =
   242→    orderedLessons.length > 0 && completedIds.length >= orderedLessons.length
   243→
   244→  const handleToggleTheme = (key: string) =>
   245→    setCollapsedThemes((prev) => ({ ...prev, [key]: !prev[key] }))
   246→
   247→  const selectLesson = (lessonId: string) => {
   248→    setCurrentLessonId(lessonId)
   249→    // Ao selecionar uma aula, garante o tema dela expandido
   250→    const lessonThemeId = orderedLessons.find((l) => l.id === lessonId)?.themeId
   251→    if (lessonThemeId) setCollapsedThemes((prev) => ({ ...prev, [lessonThemeId]: false }))
   252→  }
   253→
   254→  const toggleComplete = async (autoAdvance = false) => {
   255→    if (!user || !course || !currentLesson) return
   256→    setToggling(true)
   257→    try {
   258→      const res = await api.toggleLessonComplete(course.id, {
   259→        userId: user.id,
   260→        lessonId: currentLesson.id,
   261→      })
   262→      const nowCompleted = res.completedLessonIds.includes(currentLesson.id)
   263→      setCompletedIds(res.completedLessonIds)
   264→      if (res.xpAwarded > 0) {
   265→        toast.success(`🎉 Aula concluída! +${res.xpAwarded} XP de estudo`)
   266→      }
   267→      if (nowCompleted && res.courseCompleted) {
   268→        toast.success('🏆 Curso concluído! Parabéns 🎉')
   269→      } else if (nowCompleted && res.xpAwarded <= 0) {
   270→        toast.success('Aula marcada como concluída ✅')
   271→      } else if (!nowCompleted) {
   272→        toast('Marcação removida desta aula.')
   273→      }
   274→      if (autoAdvance && nowCompleted && currentIndex < orderedLessons.length - 1) {
   275→        setCurrentLessonId(orderedLessons[currentIndex + 1].id)
   276→      }
   277→    } catch (err) {
   278→      toast.error(err instanceof Error ? err.message : 'Não foi possível atualizar o progresso.')
   279→    } finally {
   280→      setToggling(false)
   281→    }
   282→  }
   283→
   284→  const goPrev = () => {
   285→    if (currentLesson && currentIndex > 0) setCurrentLessonId(orderedLessons[currentIndex - 1].id)
   286→  }
   287→  const goNext = () => {
   288→    if (currentLesson && currentIndex < orderedLessons.length - 1) {
   289→      setCurrentLessonId(orderedLessons[currentIndex + 1].id)
   290→    }
   291→  }
   292→
   293→  // Emite (ou retorna) o certificado e abre a página pública verificável.
   294→  // O overlay da sala fecha naturalmente: a view global muda para 'certificate'.
   295→  const handleIssueCertificate = async () => {
   296→    if (!user || !course) return
   297→    setIssuing(true)
   298→    try {
   299→      const cert = await api.issueCertificate(course.id, user.id)
   300→      toast.success('Certificado emitido! 🎓')
   301→      navigate({ name: 'certificate', code: cert.code })
   302→    } catch (err) {
   303→      toast.error(err instanceof Error ? err.message : 'Não foi possível emitir o certificado.')
   304→    } finally {
   305→      setIssuing(false)
   306→    }
   307→  }
   308→
   309→  /* ---------- Estados de tela ---------- */
   310→
   311→  if (loading) {
   312→    return (
   313→      <div className="flex h-full min-h-0 flex-col" aria-busy="true">
   314→        <div className="flex items-center gap-3 border-b border-emerald-400/15 bg-emerald-950 px-4 py-3 sm:px-6">
   315→          <Skeleton className="h-9 w-9 rounded-full bg-white/10" />
   316→          <Skeleton className="h-5 w-48 bg-white/10" />
   317→          <Skeleton className="ml-auto h-6 w-28 rounded-full bg-white/10" />
   318→        </div>
   319→        <div className="grid flex-1 grid-cols-1 lg:grid-cols-[1fr_380px]">
   320→          <div className="space-y-4 p-4 sm:p-6">
   321→            <Skeleton className="aspect-video w-full rounded-2xl" />
   322→            <Skeleton className="h-7 w-2/3" />
   323→            <Skeleton className="h-24 w-full rounded-2xl" />
   324→          </div>
   325→          <div className="hidden space-y-2 border-l border-stone-200 bg-white p-4 lg:block dark:border-stone-800 dark:bg-stone-900">
   326→            {Array.from({ length: 6 }).map((_, i) => (
   327→              <Skeleton key={i} className="h-14 rounded-xl" />
   328→            ))}
   329→          </div>
   330→        </div>
   331→        <p className="sr-only">Abrindo sala de aula…</p>
   332→      </div>
   333→    )
   334→  }
   335→
   336→  if (!course) {
   337→    return (
   338→      <div className="flex h-full items-center justify-center p-6">
   339→        <div className="flex w-full max-w-md flex-col items-center gap-3 rounded-2xl border border-dashed border-stone-300 px-6 py-14 text-center dark:border-stone-700">
   340→          <span className="flex h-14 w-14 items-center justify-center rounded-full bg-rose-50 dark:bg-rose-950/50">
   341→            <AlertCircle aria-hidden className="h-7 w-7 text-rose-500" />
   342→          </span>
   343→          <p className="font-bold text-stone-900 dark:text-stone-50">Não foi possível abrir a sala de aula</p>
   344→          <p className="max-w-sm text-sm leading-relaxed text-stone-500 dark:text-stone-400">
   345→            {error ?? 'Curso não encontrado.'}
   346→          </p>
   347→          <Button variant="outline" className="rounded-full" onClick={() => void fetchCourse()}>
   348→            Tentar novamente
   349→          </Button>
   350→        </div>
   351→      </div>
   352→    )
   353→  }
   354→
   355→  if (!hasAccess) {
   356→    return (
   357→      <div className="flex h-full items-center justify-center p-6">
   358→        <Card className="w-full max-w-md rounded-2xl border-stone-200 p-8 text-center shadow-none dark:border-stone-800">
   359→          <span className="mx-auto flex h-14 w-14 items-center justify-center rounded-full bg-amber-50 dark:bg-amber-950/50">
   360→            <Lock aria-hidden className="h-6 w-6 text-amber-600 dark:text-amber-400" />
   361→          </span>
   362→          <h1 className="mt-4 text-lg font-extrabold tracking-tight text-stone-900 dark:text-stone-50">
   363→            {course.title}
   364→          </h1>
   365→          <p className="mt-2 text-sm leading-relaxed text-stone-500 dark:text-stone-400">
   366→            Você ainda não tem acesso a este curso. Inscreva-se para desbloquear todas as aulas,
   367→            perguntas e anotações.
   368→          </p>
   369→          <Button
   370→            className="mt-5 h-11 w-full rounded-full font-bold"
   371→            onClick={() => navigate({ name: 'course', courseId })}
   372→          >
   373→            Ver o curso e se inscrever
   374→          </Button>
   375→        </Card>
   376→      </div>
   377→    )
   378→  }
   379→
   380→  const firstName = course.mentor.name.split(' ')[0]
   381→  // Código do certificado do usuário neste curso (null = ainda não emitido)
   382→  const certificateCode = course.certificateCode
   383→
   384→  return (
   385→    <div className="flex h-full min-h-0 flex-col bg-stone-50 dark:bg-stone-950" aria-label={`Sala de aula: ${course.title}`}>
   386→      {/* ---------- BARRA SUPERIOR IMERSIVA ---------- */}
   387→      <div className="flex shrink-0 items-center gap-1 border-b border-emerald-400/15 bg-emerald-950 px-3 py-2.5 text-white sm:gap-2.5 sm:px-6">
   388→        <Button
   389→          variant="ghost"
   390→          size="icon"
   391→          className="h-9 w-9 shrink-0 rounded-full text-white hover:bg-white/10 hover:text-white"
   392→          aria-label="Sair da sala de aula"
   393→          onClick={() => navigate({ name: 'course', courseId })}
   394→        >
   395→          <ArrowLeft aria-hidden className="h-4 w-4" />
   396→        </Button>
   397→        <div className="min-w-0 flex-1">
   398→          <p className="truncate text-sm font-extrabold tracking-tight text-white sm:text-base">
   399→            {course.title}
   400→          </p>
   401→          <p className="hidden text-xs text-emerald-100/70 sm:block">
   402→            {orderedLessons.length} {orderedLessons.length === 1 ? 'aula' : 'aulas'} ·{' '}
   403→            {formatTotalDuration(course.totalDurationMin)}
   404→          </p>
   405→        </div>
   406→        <Button
   407→          variant="ghost"
   408→          size="icon"
   409→          className="h-9 w-9 shrink-0 rounded-full text-emerald-100 hover:bg-white/10 hover:text-white lg:hidden"
   410→          aria-label="Ver conteúdos do curso"
   411→          onClick={() => setContentsOpen(true)}
   412→        >
   413→          <PanelRight aria-hidden className="h-4 w-4" />
   414→        </Button>
   415→        <Button
   416→          variant="ghost"
   417→          size="icon"
   418→          className="h-9 w-9 shrink-0 rounded-full text-emerald-100 hover:bg-white/10 hover:text-white"
   419→          aria-label={focusMode ? 'Sair do modo foco' : 'Ativar modo foco'}
   420→          title={focusMode ? 'Sair do modo foco' : 'Ativar modo foco'}
   421→          onClick={() => setFocusMode((v) => !v)}
   422→        >
   423→          {focusMode ? (
   424→            <Minimize2 aria-hidden className="h-4 w-4" />
   425→          ) : (
   426→            <Maximize2 aria-hidden className="h-4 w-4" />
   427→          )}
   428→        </Button>
   429→        <div className="hidden items-center gap-2 sm:flex" aria-hidden>
   430→          <Avatar
   431→            name={course.mentor.name}
   432→            src={course.mentor.avatarUrl}
   433→            size="sm"
   434→            className="h-6 w-6 text-[10px]"
   435→          />
   436→          <span className="max-w-36 truncate text-xs font-semibold text-emerald-100/70">
   437→            {course.mentor.name}
   438→          </span>
   439→        </div>
   440→        <div className="flex w-32 shrink-0 items-center gap-2 sm:w-44">
   441→          <div
   442→            role="progressbar"
   443→            aria-label="Progresso do curso"
   444→            aria-valuemin={0}
   445→            aria-valuemax={100}
   446→            aria-valuenow={percent}
   447→            className="h-2 min-w-0 flex-1 overflow-hidden rounded-full bg-white/20"
   448→          >
   449→            <div
   450→              className="h-full rounded-full bg-gradient-to-r from-emerald-600 to-emerald-400 transition-all duration-500"
   451→              style={{ width: `${percent}%` }}
   452→            />
   453→          </div>
   454→          <span className="w-9 shrink-0 text-right text-xs font-bold tabular-nums text-emerald-100">
   455→            {percent}%
   456→          </span>
   457→        </div>
   458→      </div>
   459→
   460→      {/* ---------- CORPO: PLAYER + CONTEÚDOS (sidebar oculta no modo foco) ---------- */}
   461→      <div
   462→        className={cn(
   463→          'grid min-h-0 flex-1 grid-cols-1',
   464→          focusMode ? 'lg:grid-cols-1' : 'lg:grid-cols-[1fr_380px]'
   465→        )}
   466→      >
   467→        {/* Coluna esquerda: aula — conteúdo interno largo p/ vídeos grandes */}
   468→        <div className="min-h-0 overflow-y-auto p-4 sm:p-6 lg:p-8">
   469→          <div className={cn('mx-auto w-full', focusMode ? 'max-w-4xl' : 'max-w-5xl')}>
   470→            <AnimatePresence mode="wait">
   471→              {currentLesson ? (
   472→                <motion.div
   473→                  key={currentLesson.id}
   474→                  initial={{ opacity: 0, y: 12 }}
   475→                  animate={{ opacity: 1, y: 0 }}
   476→                  exit={{ opacity: 0, y: -8 }}
   477→                  transition={{ duration: 0.22 }}
   478→                >
   479→                {currentLesson.kind === 'LIVE' ? (
   480→                  <LivePanel lesson={currentLesson} isOwner={isOwner} />
   481→                ) : currentLesson.kind === 'READING' ? (
   482→                  <ReadingMaterial lesson={currentLesson} courseId={courseId} />
   483→                ) : currentLesson.kind === 'RECORDED' && currentLesson.videoUrl ? (
   484→                  <div className="rounded-3xl bg-gradient-to-b from-emerald-50/70 to-transparent p-2 sm:p-3 dark:from-emerald-950/25 dark:to-transparent">
   485→                    <div className="overflow-hidden rounded-2xl bg-black shadow-2xl shadow-stone-950/25 ring-1 ring-stone-200 dark:ring-stone-800">
   486→                      <LessonPlayer lesson={currentLesson} />
   487→                    </div>
   488→                  </div>
   489→                ) : (
   490→                  <LessonPlayer lesson={currentLesson} />
   491→                )}
   492→
   493→                {/* Celebração + certificado: 100% das aulas concluídas */}

   878→/* ==================== NAVEGAÇÃO DE CONTEÚDOS (sidebar + dialog mobile) ==================== */
   879→
   880→function ContentsNav({
   881→  groups,
   882→  themeList,
   883→  collapsedThemes,
   884→  onToggleTheme,
   885→  currentLessonId,
   886→  completedIds,
   887→  onSelectLesson,
   888→  className,
   889→}: {
   890→  groups: LessonGroup[]
   891→  themeList: CourseThemeDTO[]
   892→  collapsedThemes: Record<string, boolean>
   893→  onToggleTheme: (key: string) => void
   894→  currentLessonId: string | null
   895→  completedIds: string[]
   896→  onSelectLesson: (lessonId: string) => void
   897→  className?: string
   898→}) {
   899→  const totalLessons = groups.reduce((n, g) => n + g.lessons.length, 0)
   900→
   901→  return (
   902→    <nav
   903→      aria-label="Conteúdos do curso"
   904→      className={cn(
   905→        'min-h-0 flex-1 overflow-y-auto px-3 pb-3 [scrollbar-width:thin] [&::-webkit-scrollbar-thumb]:rounded-full [&::-webkit-scrollbar-thumb]:bg-stone-300 dark:[&::-webkit-scrollbar-thumb]:bg-stone-700 [&::-webkit-scrollbar-track]:bg-stone-100 dark:[&::-webkit-scrollbar-track]:bg-stone-900 [&::-webkit-scrollbar]:w-1.5',
   906→        className
   907→      )}
   908→    >
   909→      {groups.map((group) => {
   910→        const collapsible = themeList.length > 0
   911→        const isCollapsed = collapsible && Boolean(collapsedThemes[group.key])
   912→        const groupTotal = group.lessons.length
   913→        const groupDone = group.lessons.filter((l) => completedIds.includes(l.id)).length
   914→        return (
   915→          <div key={group.key} className="space-y-1 py-1">
   916→            {collapsible ? (
   917→              <button
   918→                type="button"
   919→                onClick={() => onToggleTheme(group.key)}
   920→                aria-expanded={!isCollapsed}
   921→                aria-label={`${isCollapsed ? 'Expandir' : 'Recolher'} a seção ${group.title}`}
   922→                className="flex min-h-11 w-full items-center gap-1.5 rounded-xl px-1.5 py-1.5 text-left transition-colors hover:bg-stone-50 dark:hover:bg-stone-800"
   923→              >
   924→                {isCollapsed ? (
   925→                  <ChevronRight aria-hidden className="h-4 w-4 shrink-0 text-stone-400 dark:text-stone-500" />
   926→                ) : (
   927→                  <ChevronDown aria-hidden className="h-4 w-4 shrink-0 text-stone-400 dark:text-stone-500" />
   928→                )}
   929→                {group.theme ? (
   930→                  <span className="flex h-6 w-6 shrink-0 items-center justify-center rounded-full bg-emerald-100 text-[11px] font-bold text-emerald-800 dark:bg-emerald-950/50 dark:text-emerald-300">
   931→                    {themeList.findIndex((t) => t.id === group.key) + 1}

  1280→/* ==================== Q&A DA AULA ==================== */
  1281→
  1282→function LessonQuestions({
  1283→  lesson,
  1284→  isOwner,
  1285→  mentorName,
  1286→  user,
  1287→}: {
  1288→  lesson: CourseLessonDTO
  1289→  isOwner: boolean
  1290→  mentorName: string
  1291→  user: { id: string; name: string } | null
  1292→}) {
  1293→  const [questions, setQuestions] = useState<LessonQuestionDTO[] | null>(null)
  1294→  const [body, setBody] = useState('')
  1295→  const [sending, setSending] = useState(false)
  1296→  const [answerDraft, setAnswerDraft] = useState<Record<string, string>>({})
  1297→  const [answeringId, setAnsweringId] = useState<string | null>(null)
  1298→
  1299→  const load = useCallback(async () => {
  1300→    if (!user) {
  1301→      setQuestions([])
  1302→      return
  1303→    }
  1304→    try {
  1305→      const list = await api.listLessonQuestions(lesson.id, user.id)
  1306→      setQuestions(list)
  1307→    } catch {
  1308→      setQuestions([])
  1309→    }
  1310→  }, [lesson.id, user])
  1311→
  1312→  useEffect(() => {
  1313→    setQuestions(null)
  1314→    void load()
  1315→  }, [load])
  1316→
  1317→  const ask = async () => {
  1318→    if (!user) {
  1319→      toast.error('Entre com uma conta para perguntar.')
  1320→      return
  1321→    }
  1322→    if (body.trim().length < 5) {
  1323→      toast.error('Escreva sua pergunta (mín. 5 caracteres).')
  1324→      return
  1325→    }
  1326→    setSending(true)
  1327→    try {
  1328→      const created = await api.askLessonQuestion(lesson.id, { userId: user.id, body: body.trim() })
  1329→      setQuestions((prev) => [created, ...(prev ?? [])])
  1330→      setBody('')
  1331→      toast.success('Pergunta enviada! O mentor será notificado.')
  1332→    } catch (err) {
  1333→      toast.error(err instanceof Error ? err.message : 'Não foi possível enviar a pergunta.')
  1334→    } finally {
  1335→      setSending(false)
  1336→    }
  1337→  }
  1338→
  1339→  const answer = async (questionId: string) => {
  1340→    const text = (answerDraft[questionId] ?? '').trim()
  1341→    if (!user || text.length < 2) {
  1342→      toast.error('Escreva a resposta antes de enviar.')
  1343→      return
  1344→    }
  1345→    setAnsweringId(questionId)
  1346→    try {
  1347→      const res = await api.answerQuestion(questionId, { userId: user.id, answer: text })
  1348→      setQuestions((prev) =>
  1349→        (prev ?? []).map((q) =>
  1350→          q.id === questionId ? { ...q, answer: res.answer, answeredAt: res.answeredAt } : q
  1351→        )
  1352→      )
  1353→      setAnswerDraft((prev) => ({ ...prev, [questionId]: '' }))
  1354→      toast.success('Resposta publicada!')
  1355→    } catch (err) {
  1356→      toast.error(err instanceof Error ? err.message : 'Não foi possível responder.')
  1357→    } finally {
  1358→      setAnsweringId(null)
  1359→    }
  1360→  }
  1361→
  1362→  const remove = async (questionId: string) => {
  1363→    if (!user) return
  1364→    try {
  1365→      await api.deleteQuestion(questionId, user.id)
  1366→      setQuestions((prev) => (prev ?? []).filter((q) => q.id !== questionId))
  1367→      toast('Pergunta removida.')
  1368→    } catch (err) {
  1369→      toast.error(err instanceof Error ? err.message : 'Não foi possível remover.')
  1370→    }
  1371→  }
  1372→
  1373→  if (!user) {
  1374→    return (
  1375→      <div className="rounded-2xl border border-stone-200 bg-white p-6 text-center text-sm text-stone-500 dark:border-stone-800 dark:bg-stone-900 dark:text-stone-400">
  1376→        Entre com uma conta no topo da página para participar das perguntas.
  1377→      </div>
  1378→    )
  1379→  }
  1380→
  1381→  return (
  1382→    <div className="space-y-4">
  1383→      {/* Formulário de pergunta (aluno) */}
  1384→      {!isOwner ? (
  1385→        <div className="rounded-2xl border border-stone-200 bg-white p-4 sm:p-5 dark:border-stone-800 dark:bg-stone-900">
  1386→          <label htmlFor="question-body" className="text-sm font-bold text-stone-800 dark:text-stone-200">
  1387→            Tirou uma dúvida? Pergunte ao mentor
  1388→          </label>
  1389→          <Textarea
  1390→            id="question-body"
  1391→            rows={3}
  1392→            value={body}
  1393→            onChange={(e) => setBody(e.target.value)}
  1394→            placeholder={`Ex.: "${mentorName.split(' ')[0]}", no minuto 4:30 você menciona X — como aplico isso na prática?`}
  1395→            className="mt-2 resize-none"
  1396→            maxLength={1200}
  1397→          />
  1398→          <div className="mt-3 flex items-center justify-between gap-3">
  1399→            <span className="text-xs text-stone-400 dark:text-stone-500">{body.length}/1200</span>

  1607→      <div className="flex items-center justify-between gap-3">

  1614→            saveState === 'saving' || saveState === 'saved'

  1616→              : 'text-stone-400 dark:text-stone-500'
  1617→          )}
  1618→          aria-live="polite"
  1619→        >
  1620→          {saveState === 'saving' ? (

  1622→              <span aria-hidden className="h-2 w-2 animate-pulse rounded-full bg-emerald-500" />
