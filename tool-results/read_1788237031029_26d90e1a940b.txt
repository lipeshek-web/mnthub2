   209→export function LandingMenteeView() {
   210→  const navigate = useAppStore((s) => s.navigate)
   211→  const setExploreTab = useAppStore((s) => s.setExploreTab)
   212→  const user = useAppStore((s) => s.user)
   213→  const [mentors, setMentors] = useState<MentorListItemDTO[]>([])
   214→  const [loading, setLoading] = useState(true)
   215→  const [courses, setCourses] = useState<CourseListItemDTO[]>([])
   216→  const [coursesLoading, setCoursesLoading] = useState(true)
   217→  const [tracks, setTracks] = useState<TrackListItemDTO[]>([])
   218→  const [tracksLoading, setTracksLoading] = useState(true)
   219→
   220→  // Fetch preguiçoso: cursos e trilhas só saem quando as seções se aproximam da viewport
   221→  const coursesSectionRef = useRef<HTMLElement | null>(null)
   222→  const coursesInView = useInView(coursesSectionRef, { once: true, margin: '600px' })
   223→  const tracksSectionRef = useRef<HTMLElement | null>(null)
   224→  const tracksInView = useInView(tracksSectionRef, { once: true, margin: '600px' })
   225→
   226→  useEffect(() => {
   227→    let active = true
   228→    api
   229→      .listMentors({})
   230→      .then((data) => {
   231→        if (active) setMentors(data)
   232→      })
   233→      .catch(() => {
   234→        if (active) setMentors([])
   235→      })
   236→      .finally(() => {
   237→        if (active) setLoading(false)
   238→      })
   239→    return () => {
   240→      active = false
   241→    }
   242→  }, [])
   243→
   244→  // Cursos para a seção "Cursos em destaque" (disparado quando a seção ou a faixa de números se aproxima)
   245→  const statsSectionRef = useRef<HTMLElement | null>(null)
   246→  const statsInView = useInView(statsSectionRef, { once: true, margin: '600px' })
   247→
   248→  useEffect(() => {
   249→    if (!coursesInView && !statsInView) return
   250→    let active = true
   251→    api
   252→      .listCourses({})
   253→      .then((data) => {
   254→        if (active) setCourses(data)
   255→      })
   256→      .catch(() => {
   257→        if (active) setCourses([])
   258→      })
   259→      .finally(() => {
   260→        if (active) setCoursesLoading(false)
   261→      })
   262→    return () => {
   263→      active = false
   264→    }
   265→  }, [coursesInView, statsInView])
   266→
   267→  // Trilhas populares para a seção "Trilhas em destaque" (seção ou faixa de números se aproximando)
   268→  useEffect(() => {
   269→    if (!tracksInView && !statsInView) return
   270→    let active = true
   271→    api
   272→      .listTracks({ sort: 'popular' })
   273→      .then((data) => {
   274→        if (active) setTracks(data)
   275→      })
   276→      .catch(() => {
   277→        if (active) setTracks([])
   278→      })
   279→      .finally(() => {
   280→        if (active) setTracksLoading(false)
   281→      })
   282→    return () => {
   283→      active = false
   284→    }
   285→  }, [tracksInView, statsInView])
   286→
   287→  // Biblioteca em destaque (disparado quando a seção se aproxima)
   288→  const librarySectionRef = useRef<HTMLElement | null>(null)
   289→  const libraryInView = useInView(librarySectionRef, { once: true, margin: '600px' })
   290→  const [libraryItems, setLibraryItems] = useState<LibraryItemDTO[]>([])
   291→  const [libraryLoading, setLibraryLoading] = useState(true)
   292→
   293→  useEffect(() => {
   294→    if (!libraryInView) return
   295→    let active = true
   296→    api
   297→      .listLibrary({ sort: 'recent' })
   298→      .then((data) => {
   299→        if (active) setLibraryItems(data)
   300→      })
   301→      .catch(() => {
   302→        if (active) setLibraryItems([])
   303→      })
   304→      .finally(() => {
   305→        if (active) setLibraryLoading(false)
   306→      })
   307→    return () => {
   308→      active = false
   309→    }
   310→  }, [libraryInView])
   311→
   312→  // "Continue aprendendo": inscrições do usuário logado (seção oculta para convidados)
   313→  const userId = user?.id
   314→  const [enrollments, setEnrollments] = useState<EnrolledCourseDTO[]>([])
   315→  const [enrollmentsLoading, setEnrollmentsLoading] = useState(true)
   316→
   317→  useEffect(() => {
   318→    // Convidado: a seção fica oculta (user == null) e o estado inicial não precisa mudar.
   319→    if (!userId) return
   320→    let active = true
   321→    api
   322→      .listMyEnrollments(userId)
   323→      .then((data) => {
   324→        if (active) setEnrollments(data)
   325→      })
   326→      .catch(() => {
   327→        if (active) setEnrollments([])
   328→      })
   329→      .finally(() => {
   330→        if (active) setEnrollmentsLoading(false)
   331→      })
   332→    return () => {
   333→      active = false
   334→    }
   335→  }, [userId])
   336→
   337→  const totalReviews = useMemo(
   338→    () => mentors.reduce((acc, m) => acc + m.reviewCount, 0),
   339→    [mentors]
   340→  )
   341→
   342→  const avgRating = useMemo(() => {
   343→    const rated = mentors.filter((m) => m.rating > 0)
   344→    if (rated.length === 0) return null
   345→    return rated.reduce((acc, m) => acc + m.rating, 0) / rated.length
   346→  }, [mentors])
   347→
   348→  // Números da faixa "A plataforma em números" (chegam junto com os fetches preguiçosos)
   349→  const totalLessons = useMemo(
   350→    () => courses.reduce((acc, c) => acc + c.lessonCount, 0),
   351→    [courses]
   352→  )
   353→
   354→  // Top 6 itens da biblioteca (a API já devolve ordenada por data — novidades primeiro)
   355→  const topLibrary = useMemo(() => libraryItems.slice(0, 6), [libraryItems])
   356→
   357→  const featured = useMemo(
   358→    () =>
   359→      mentors
   360→        .filter((m) => m.rating > 0)
   361→        .sort((a, b) => b.rating - a.rating || b.reviewCount - a.reviewCount)
   362→        .slice(0, 3),
   363→    [mentors]
   364→  )
   365→
   366→  // Top 3 cursos por alunos inscritos, desempate pela nota do mentor
   367→  const topCourses = useMemo(
   368→    () =>
   369→      [...courses]
   370→        .sort(
   371→          (a, b) =>
   372→            b.studentCount - a.studentCount || b.mentor.rating - a.mentor.rating
   373→        )
   374→        .slice(0, 3),
   375→    [courses]
   376→  )
   377→
   378→  // Top 3 trilhas (a API já devolve ordenada por popularidade)
   379→  const topTracks = useMemo(() => tracks.slice(0, 3), [tracks])
   380→
   381→  // Até 3 inscrições para "Continue aprendendo" — prioridade: em andamento (0 < x < 100%),
   382→  // depois não iniciadas (0%), concluídas por último; recência (enrolledAt desc) dentro do grupo
   383→  const continueItems = useMemo(() => {
   384→    const byRecency = (a: EnrolledCourseDTO, b: EnrolledCourseDTO) =>
   385→      b.enrolledAt.localeCompare(a.enrolledAt) // ISO-8601: comparação lexicográfica segura
   386→    const isDone = (e: EnrolledCourseDTO) =>
   387→      e.course.lessonCount > 0 && e.completedLessonIds.length >= e.course.lessonCount
   388→    const started = enrollments
   389→      .filter((e) => !isDone(e) && e.completedLessonIds.length > 0)
   390→      .sort(byRecency)
   391→    const untouched = enrollments
   392→      .filter((e) => e.completedLessonIds.length === 0)
   393→      .sort(byRecency)
   394→    const done = enrollments.filter(isDone).sort(byRecency)
   395→    return [...started, ...untouched, ...done].slice(0, 3)
   396→  }, [enrollments])
   397→
   398→  // "Feito para você": recomendações da IA (cruza histórico do aluno com o catálogo).
   399→  // Silencioso em erro/sem resultados — a landing nunca quebra por causa da IA.
   400→  const [recommendations, setRecommendations] = useState<RecommendationDTO[]>([])
   401→  const [recsGenerated, setRecsGenerated] = useState(false)
   402→
   403→  useEffect(() => {
   404→    // Convidado: seção oculta (render gated por user) — nada a buscar.
   405→    if (!userId) return
   406→    let active = true
   407→    api
   408→      .recommendations(userId)
   409→      .then((data) => {
   410→        if (!active) return
   411→        setRecommendations(data.items ?? [])
   412→        setRecsGenerated(Boolean(data.generated))
   413→      })
   414→      .catch(() => {
   415→        if (active) setRecommendations([])
   416→      })
   417→    return () => {
   418→      active = false
   419→    }
   420→  }, [userId])
   421→
   422→  const handleExploreMentors = () => {
   423→    setExploreTab('mentors')
   424→    navigate({ name: 'marketplace' })
   425→  }
   426→
   427→  const handleVerCursos = () => {
   428→    setExploreTab('courses')
   429→    navigate({ name: 'marketplace' })
   430→  }
   431→
   432→  const handleVerTrilhas = () => {
   433→    setExploreTab('tracks')
   434→    navigate({ name: 'marketplace' })
   435→  }
   436→
   437→  const handleVerBiblioteca = () => {
   438→    setExploreTab('library')
   439→    navigate({ name: 'marketplace' })
   440→  }
   441→
   442→  const handleExploreArea = () => {
   443→    setExploreTab('all')
   444→    navigate({ name: 'marketplace' })
   445→  }
   446→
   447→  const handleFormatTab = (tab: (typeof FORMATS)[number]['tab']) => {
   448→    setExploreTab(tab)
   449→    navigate({ name: 'marketplace' })
   450→  }
   451→
   452→  return (
   453→    <div className="flex flex-col bg-white dark:bg-stone-950">
   454→      {/* ---------- HERO (split com carrossel leve dos formatos) ---------- */}
   455→      <section aria-labelledby="hero-title" className="relative overflow-hidden">
   456→        <div className="relative mx-auto max-w-7xl px-4 sm:px-6 pb-20 pt-14 sm:pb-28 sm:pt-20">
   457→          <div className="grid grid-cols-1 items-center gap-14 lg:grid-cols-2 lg:gap-12">
   458→            {/* Coluna esquerda: mensagem + busca + prova social */}
   459→            <motion.div
   460→              initial={{ opacity: 0, y: 16 }}
   461→              animate={{ opacity: 1, y: 0 }}
   462→              transition={{ duration: 0.5 }}
   463→              className="min-w-0"
   464→            >
   465→              <p className="text-sm font-semibold text-emerald-700 dark:text-emerald-300">
   466→                Mentorias, cursos, trilhas e biblioteca — tudo em um só lugar
   467→              </p>
   468→
   469→              <h1
   470→                id="hero-title"
   471→                className="mt-4 bg-gradient-to-b from-stone-900 via-stone-900 to-stone-600 bg-clip-text text-5xl font-semibold leading-[1.05] tracking-[-0.03em] text-transparent sm:text-6xl xl:text-7xl dark:from-stone-50 dark:via-stone-100 dark:to-stone-400"
   472→              >
   473→                {user ? (
   474→                  <>
   475→                    Olá, <span className="text-emerald-700 dark:text-emerald-300">{firstName(user.name)}</span>! Pronto
   476→                    para{' '}
   477→                    <span className="text-emerald-700 dark:text-emerald-300">
   478→                      continuar aprendendo
   479→                    </span>
   480→                    ?
   481→                  </>
   482→                ) : (
   483→                  <>
   484→                    Aprenda com quem{' '}
   485→                    <span className="text-emerald-700 dark:text-emerald-300">
   486→                      vive o que ensina
   487→                    </span>
   488→                  </>
   489→                )}
   490→              </h1>
   491→
   492→              <p className="mt-5 max-w-xl text-lg leading-relaxed text-stone-500 sm:text-xl dark:text-stone-400">
   493→                Especialistas que viveram o caminho que você quer trilhar — em mentorias 1:1, cursos
   494→                gravados, trilhas guiadas e uma biblioteca completa. Tudo dentro da plataforma, do
   495→                primeiro clique ao certificado.
   496→              </p>
   497→
   498→              {/* CTA estilo Apple: pill sólida + link secundário com chevron */}
   499→              <div className="mt-9 flex flex-col gap-3 sm:flex-row sm:items-center sm:gap-6">
   500→                <Button
   501→                  onClick={handleExploreArea}
   502→                  className="h-12 w-full rounded-full bg-stone-950 px-7 text-base font-medium text-white shadow-none hover:bg-stone-800 sm:w-auto dark:bg-white dark:text-stone-950 dark:hover:bg-stone-200"
   503→                >
   504→                  Explorar a plataforma
   505→                </Button>
   506→                <button
   507→                  type="button"
   508→                  onClick={handleVerCursos}
   509→                  className="inline-flex min-h-11 items-center gap-1 rounded-full text-base font-medium text-emerald-700 transition-colors hover:text-emerald-800 focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-emerald-500/40 dark:text-emerald-300 dark:hover:text-emerald-200"
   510→                >
   511→                  Ver cursos
   512→                  <ChevronRight aria-hidden className="h-4 w-4" />
   513→                </button>
   514→              </div>
   515→
   516→              {/* Prova social: avatares + nota média + avaliações reais */}
   517→              {loading ? (
   518→                <div className="mt-9 flex items-center gap-3">
   519→                  <div className="flex -space-x-2.5">
   520→                    {Array.from({ length: 5 }).map((_, i) => (
   521→                      <Skeleton key={i} className="h-8 w-8 rounded-full" />
   522→                    ))}
   523→                  </div>
   524→                  <Skeleton className="h-4 w-44" />
   525→                </div>
   526→              ) : mentors.length > 0 ? (
   527→                <div className="mt-9 flex flex-wrap items-center gap-x-4 gap-y-3">
   528→                  <div className="flex -space-x-2.5">
   529→                    {mentors.slice(0, 5).map((m) => (
   530→                      <Avatar key={m.id} name={m.name} src={m.avatarUrl} size="sm" />
   531→                    ))}
   532→                  </div>
