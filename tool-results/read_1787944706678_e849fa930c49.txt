   184→   184→  onReject,
   185→   185→  onComplete,
   186→   186→  onCancel,
   187→   187→  onJoin,
   188→   188→  onReview,
   189→   189→}: {
   190→   190→  booking: BookingDTO
   191→   191→  userId: string
   192→   192→  busy: boolean
   193→   193→  onConfirm: (booking: BookingDTO) => void
   194→   194→  onReject: (booking: BookingDTO) => void
   195→   195→  onComplete: (booking: BookingDTO) => void
   196→   196→  onCancel: (booking: BookingDTO) => void
   197→   197→  onJoin: (booking: BookingDTO) => void
   198→   198→  onReview: (booking: BookingDTO, rating: number, comment: string) => Promise<void>
   199→   199→}) {
   200→   200→  const isMentorSide = b.mentor.userId === userId
   201→   201→  const otherName = isMentorSide ? b.mentee.name : b.mentor.name
   202→   202→  const relative = relativeDayLabel(b.startsAt)
   203→   203→  const statusMeta = STATUS_META[b.status] ?? {
   204→   204→    label: b.status,
   205→   205→    className: 'bg-stone-100 text-stone-600 border-stone-200',
   206→   206→  }
   207→   207→
   208→   208→  const [dialogAction, setDialogAction] = useState<ConfirmKind | null>(null)
   209→   209→  const [reviewOpen, setReviewOpen] = useState(false)
   210→   210→  const [rating, setRating] = useState(0)
   211→   211→  const [hoverRating, setHoverRating] = useState(0)
   212→   212→  const [comment, setComment] = useState('')
   213→   213→  const [submitting, setSubmitting] = useState(false)
   214→   214→
   215→   215→  const canJoin = b.status === 'CONFIRMED'
   216→   216→  const canReview = !isMentorSide && b.status === 'COMPLETED' && !b.reviewed
   217→   217→
   218→   218→  const dialogContent: Record<ConfirmKind, { title: string; description: string; actionLabel: string }> = {
   219→   219→    reject: {
   220→   220→      title: 'Recusar solicitação?',
   221→   221→      description: `A solicitação de ${b.mentee.name} sobre "${b.topic}" será cancelada e o horário será liberado na sua agenda.`,
   222→   222→      actionLabel: 'Recusar',
   223→   223→    },
   224→   224→    complete: {
   225→   225→      title: 'Marcar como concluída?',
   226→   226→      description: `Confirme que a sessão com ${b.mentee.name} aconteceu. Ela irá para o histórico e o aluno poderá avaliá-la.`,
   227→   227→      actionLabel: 'Concluir sessão',
   228→   228→    },
   229→   229→    cancel: {
   230→   230→      title: 'Cancelar sessão?',
   231→   231→      description: `Sua sessão com ${b.mentor.name} será cancelada e o horário será liberado para outras pessoas.`,
   232→   232→      actionLabel: 'Cancelar sessão',
   233→   233→    },
   234→   234→  }
   235→   235→  const currentDialog = dialogAction ? dialogContent[dialogAction] : null
   236→   236→
   237→   237→  const submitReview = async () => {
   238→   238→    if (rating < 1) {
   239→   239→      toast.error('Escolha uma nota de 1 a 5 estrelas antes de enviar.')
   240→   240→      return
   241→   241→    }
   242→   242→    setSubmitting(true)
   243→   243→    try {
   244→   244→      await onReview(b, rating, comment.trim())
   245→   245→      setReviewOpen(false)
   246→   246→      setRating(0)
   247→   247→      setHoverRating(0)
   248→   248→      setComment('')
   249→   249→    } catch (err) {
   250→   250→      toast.error(err instanceof Error ? err.message : 'Não foi possível enviar sua avaliação.')
   251→   251→    } finally {
   252→   252→      setSubmitting(false)
   253→   253→    }
   254→   254→  }
   255→   255→
   256→   256→  return (
   257→   257→    <Card className="gap-4 py-6">
   258→   258→      <CardContent className="flex flex-col gap-4 px-6">
   259→   259→        <div className="flex flex-wrap items-start justify-between gap-3">
   260→   260→          <div className="flex min-w-0 items-start gap-3">
   261→   261→            <Avatar name={otherName} size="lg" />
   262→   262→            <div className="min-w-0">
   263→   263→              <div className="flex flex-wrap items-center gap-2">
   264→   264→                <p className="font-semibold leading-tight">{otherName}</p>
   265→   265→                <Badge
   266→   266→                  className={
   267→   267→                    isMentorSide
   268→   268→                      ? 'border-emerald-200 bg-emerald-50 text-emerald-700'
   269→   269→                      : 'border-stone-200 bg-stone-100 text-stone-600'
   270→   270→                  }
   271→   271→                >
   272→   272→                  {isMentorSide ? 'Como mentor' : 'Como aluno'}
   273→   273→                </Badge>
   274→   274→              </div>
   275→   275→              {!isMentorSide && b.mentor.headline ? (
   276→   276→                <p className="mt-0.5 text-sm text-muted-foreground">{b.mentor.headline}</p>
   277→   277→              ) : null}
   278→   278→              <p className="mt-1.5 font-medium">{b.topic}</p>
   279→   279→              {b.notes ? <p className="mt-1 text-sm text-muted-foreground">{b.notes}</p> : null}
   280→   280→            </div>
   281→   281→          </div>
   282→   282→          <Badge variant="outline" className={statusMeta.className}>
   283→   283→            {statusMeta.label}
   284→   284→          </Badge>
   285→   285→        </div>
   286→   286→
   287→   287→        <div className="flex flex-wrap items-center gap-x-4 gap-y-1.5 text-sm text-muted-foreground">
   288→   288→          <span className="inline-flex items-center gap-1.5">
   289→   289→            <CalendarDays className="size-4 text-emerald-600" aria-hidden />
   290→   290→            {relative ? `${relative} · ` : ''}
   291→   291→            {formatDayLabelLong(b.startsAt)} · {formatTimeLabel(b.startsAt)} →{' '}
   292→   292→            {addMinutesToTime(b.startsAt, b.durationMin)}
   293→   293→          </span>
   294→   294→          <span className="inline-flex items-center gap-1.5">
   295→   295→            <Clock className="size-4" aria-hidden /> {b.durationMin} min
   296→   296→          </span>
   297→   297→          <span className="inline-flex items-center gap-1.5">
   298→   298→            <Wallet className="size-4" aria-hidden /> {currencyBRL(b.price)}
   299→   299→          </span>
   300→   300→        </div>
   301→   301→
   302→   302→        <div className="flex flex-wrap gap-2">
   303→   303→          {isMentorSide && b.status === 'PENDING' ? (
   304→   304→            <>
   305→   305→              <Button size="sm" disabled={busy} onClick={() => onConfirm(b)}>
   306→   306→                <Check className="size-4" aria-hidden /> Confirmar
   307→   307→              </Button>
   308→   308→              <Button size="sm" variant="outline" disabled={busy} onClick={() => setDialogAction('reject')}>
   309→   309→                <X className="size-4" aria-hidden /> Recusar
   310→   310→              </Button>
   311→   311→            </>
   312→   312→          ) : null}
   313→   313→
   314→   314→          {isMentorSide && b.status === 'CONFIRMED' ? (
   315→   315→            <Button size="sm" variant="outline" disabled={busy} onClick={() => setDialogAction('complete')}>
   316→   316→              <CheckCheck className="size-4" aria-hidden /> Marcar como concluída
   317→   317→            </Button>
   318→   318→          ) : null}
   319→   319→
   320→   320→          {canJoin ? (
   321→   321→            <Button size="sm" onClick={() => onJoin(b)}>
   322→   322→              <Video className="size-4" aria-hidden /> Entrar na sala
   323→   323→            </Button>
   324→   324→          ) : null}
   325→   325→
   326→   326→          {!isMentorSide && (b.status === 'PENDING' || b.status === 'CONFIRMED') ? (
   327→   327→            <Button
   328→   328→              size="sm"
   329→   329→              variant="ghost"
   330→   330→              className="text-rose-600 hover:bg-rose-50 hover:text-rose-700"
   331→   331→              disabled={busy}
   332→   332→              onClick={() => setDialogAction('cancel')}
   333→   333→            >
   334→   334→              Cancelar sessão
   335→   335→            </Button>
   336→   336→          ) : null}
   337→   337→
   338→   338→          {canReview ? (
   339→   339→            <Button size="sm" variant="outline" onClick={() => setReviewOpen(true)}>
   340→   340→              <Star className="size-4 fill-amber-400 text-amber-400" aria-hidden /> Avaliar sessão
   341→   341→            </Button>
   342→   342→          ) : null}
   343→   343→        </div>
   344→   344→      </CardContent>
   345→   345→
   346→   346→      {/* Confirmações: recusar / concluir / cancelar */}
   347→   347→      <AlertDialog
   348→   348→        open={dialogAction !== null}
   349→   349→        onOpenChange={(open) => {
   350→   350→          if (!open) setDialogAction(null)
   351→   351→        }}
   352→   352→      >
   353→   353→        <AlertDialogContent>
