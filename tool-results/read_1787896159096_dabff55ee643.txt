     1→'use client'
     2→
     3→import { useCallback, useEffect, useMemo, useRef, useState } from 'react'
     4→import { AnimatePresence, motion } from 'framer-motion'
     5→import {
     6→  ArrowRight,
     7→  BadgeCheck,
     8→  BookOpen,
     9→  Clock,
    10→  Globe2,
    11→  GraduationCap,
    12→  Library,
    13→  Search,
    14→  SearchX,
    15→  Star,
    16→  Users,
    17→  Video,
    18→  X,
    19→} from 'lucide-react'
    20→import { Button } from '@/components/ui/button'
    21→import { Input } from '@/components/ui/input'
    22→import {
    23→  Select,
    24→  SelectContent,
    25→  SelectItem,
    26→  SelectTrigger,
    27→  SelectValue,
    28→} from '@/components/ui/select'
    29→import { Skeleton } from '@/components/ui/skeleton'
    30→import { Avatar, Stars } from '@/components/platform/avatar'
    31→import { api } from '@/lib/api'
    32→import {
    33→  CATEGORIES,
    34→  LEVEL_LABELS,
    35→  avatarGradient,
    36→  currencyBRL,
    37→  firstName,
    38→  formatTotalDuration,
    39→} from '@/lib/helpers'
    40→import { useAppStore, type ExploreTab } from '@/lib/store'
    41→import type { CourseListItemDTO, MentorListItemDTO } from '@/lib/types'
    42→import { cn } from '@/lib/utils'
    43→
    44→const SPOTLIGHT_INTERVAL_MS = 6000
    45→
    46→export function MarketplaceView() {
    47→  const navigate = useAppStore((s) => s.navigate)
    48→
    49→  // Base completa (sem filtros): alimenta destaque rotativo, contagens e stats
    50→  const [baseMentors, setBaseMentors] = useState<MentorListItemDTO[]>([])
    51→  // Lista filtrada exibida no grid
    52→  const [mentors, setMentors] = useState<MentorListItemDTO[]>([])
    53→  const [loading, setLoading] = useState(true)
    54→  const [search, setSearch] = useState('')
    55→  const [inputValue, setInputValue] = useState('')
    56→  const [category, setCategory] = useState('')
    57→  const [sort, setSort] = useState('relevance')
    58→  const debounceRef = useRef<ReturnType<typeof setTimeout> | null>(null)
    59→  const searchRef = useRef<HTMLInputElement>(null)
    60→
    61→  // Aba do Explorar (mentores/cursos): valor inicial lido da store uma única vez
    62→  const [tab, setTab] = useState<ExploreTab>(() => useAppStore.getState().exploreTab)
    63→  // Cursos: base completa (contagens/destaque/stats) e lista filtrada
    64→  const [baseCourses, setBaseCourses] = useState<CourseListItemDTO[]>([])
    65→  const [courses, setCourses] = useState<CourseListItemDTO[]>([])
    66→  const [coursesLoading, setCoursesLoading] = useState(false)
    67→  const [courseSort, setCourseSort] = useState('relevance')
    68→
    69→  // Consome termo vindo de outra tela (hero da home) uma única vez
    70→  useEffect(() => {
    71→    const q = useAppStore.getState().exploreQuery
    72→    if (q) {
    73→      setInputValue(q)
    74→      setSearch(q)
    75→      useAppStore.setState({ exploreQuery: '' })
    76→    }
    77→  }, [])
    78→
    79→  // Consome a aba pedida por outra tela ("Ver todos os cursos" da home) uma única vez.
    80→  // O valor inicial já foi lido no useState acima; aqui apenas resetamos a store.
    81→  useEffect(() => {
    82→    useAppStore.setState({ exploreTab: 'mentors' })
    83→  }, [])
    84→
    85→  const load = useCallback(async () => {
    86→    setLoading(true)
    87→    try {
    88→      const data = await api.listMentors({ search, category, sort })
    89→      setMentors(data)
    90→    } catch {
    91→      setMentors([])
    92→    } finally {
    93→      setLoading(false)
    94→    }
    95→  }, [search, category, sort])
    96→
    97→  useEffect(() => {
    98→    load()
    99→  }, [load])
   100→
   101→  useEffect(() => {
   102→    api
   103→      .listMentors({})
   104→      .then(setBaseMentors)
   105→      .catch(() => {})
   106→  }, [])
   107→
   108→  // Base de cursos: alimenta destaque, contagens e stats (busca única)
   109→  useEffect(() => {
   110→    api
   111→      .listCourses({})
   112→      .then(setBaseCourses)
   113→      .catch(() => {})
   114→  }, [])
   115→
   116→  // Lista filtrada de cursos — só carrega enquanto a aba Cursos está ativa
   117→  const loadCourses = useCallback(async () => {
   118→    setCoursesLoading(true)
   119→    try {
   120→      const data = await api.listCourses({ search, category, sort: courseSort })
   121→      setCourses(data)
   122→    } catch {
   123→      setCourses([])
   124→    } finally {
   125→      setCoursesLoading(false)
   126→    }
   127→  }, [search, category, courseSort])
   128→
   129→  useEffect(() => {
   130→    if (tab === 'courses') void loadCourses()
   131→  }, [tab, loadCourses])
   132→
   133→  // Atalho "/" foca a busca (sensação de app nativo)
   134→  useEffect(() => {
   135→    const onKey = (e: KeyboardEvent) => {
   136→      const target = e.target as HTMLElement | null
   137→      const typing =
   138→        target instanceof HTMLInputElement ||
   139→        target instanceof HTMLTextAreaElement ||
   140→        target?.isContentEditable
   141→      if (e.key === '/' && !typing) {
   142→        e.preventDefault()
   143→        searchRef.current?.focus()
   144→      }
   145→    }
   146→    window.addEventListener('keydown', onKey)
   147→    return () => window.removeEventListener('keydown', onKey)
   148→  }, [])
   149→
   150→  const onSearchChange = (value: string) => {
   151→    setInputValue(value)
   152→    if (debounceRef.current) clearTimeout(debounceRef.current)
   153→    debounceRef.current = setTimeout(() => setSearch(value), 300)
   154→  }
   155→
   156→  const clearSearch = () => {
   157→    setInputValue('')
   158→    setSearch('')
   159→    if (debounceRef.current) clearTimeout(debounceRef.current)
   160→    searchRef.current?.focus()
   161→  }
   162→
   163→  // ---------- Derivados da base ----------
   164→
   165→  const spotlightPool = useMemo(() => {
   166→    const rated = baseMentors
   167→      .filter((m) => m.rating > 0)
   168→      .sort((a, b) => b.rating - a.rating || b.reviewCount - a.reviewCount)
   169→    if (rated.length > 0) return rated.slice(0, 4)
   170→    return [...baseMentors].sort((a, b) => b.totalSessions - a.totalSessions).slice(0, 3)
   171→  }, [baseMentors])
   172→
   173→  const [spotIdx, setSpotIdx] = useState(0)
   174→  useEffect(() => {
   175→    if (spotlightPool.length < 2) return
   176→    const t = setInterval(
   177→      () => setSpotIdx((i) => (i + 1) % spotlightPool.length),
   178→      SPOTLIGHT_INTERVAL_MS
   179→    )
   180→    return () => clearInterval(t)
   181→  }, [spotlightPool.length])
   182→
   183→  const spot = spotlightPool.length > 0 ? spotlightPool[spotIdx % spotlightPool.length] : null
   184→
   185→  const stats = useMemo(() => {
   186→    const rated = baseMentors.filter((m) => m.rating > 0)
   187→    return {
   188→      mentors: baseMentors.length,
   189→      sessions: baseMentors.reduce((acc, m) => acc + m.totalSessions, 0),
   190→      avg: rated.length > 0 ? rated.reduce((acc, m) => acc + m.rating, 0) / rated.length : null,
   191→    }
   192→  }, [baseMentors])
   193→
   194→  const categoryCounts = useMemo(() => {
   195→    const counts = new Map<string, number>()
   196→    for (const m of baseMentors) {
   197→      for (const c of m.categories) counts.set(c, (counts.get(c) ?? 0) + 1)
   198→    }
   199→    return counts
   200→  }, [baseMentors])
   201→
   202→  const baseReady = baseMentors.length > 0
   203→
   204→  // ---------- Derivados dos cursos ----------
   205→
   206→  const courseCounts = useMemo(() => {
   207→    const counts = new Map<string, number>()
   208→    for (const c of baseCourses) counts.set(c.category, (counts.get(c.category) ?? 0) + 1)
   209→    return counts
   210→  }, [baseCourses])
   211→
   212→  const baseCoursesReady = baseCourses.length > 0
   213→
   214→  const courseStats = useMemo(
   215→    () => ({
   216→      published: baseCourses.length,
   217→      students: baseCourses.reduce((acc, c) => acc + c.studentCount, 0),
   218→      lessons: baseCourses.reduce((acc, c) => acc + c.lessonCount, 0),
   219→    }),
   220→    [baseCourses]
   221→  )
   222→
   223→  const topCourse = useMemo(() => {
   224→    if (baseCourses.length === 0) return null
   225→    return [...baseCourses].sort(
   226→      (a, b) => b.studentCount - a.studentCount || b.mentor.rating - a.mentor.rating
   227→    )[0]
   228→  }, [baseCourses])
   229→
   230→  return (
   231→    <div>
   232→      {/* ---------- BARRA SUPERIOR: título, ordenação, busca e categorias ---------- */}
   233→      <section className="border-b border-stone-200/70 bg-white">
   234→        <div className="mx-auto max-w-6xl px-4 py-7 sm:py-9">
   235→          {/* Controle segmentado: Mentores | Cursos */}
   236→          <div
   237→            role="tablist"
   238→            aria-label="Explorar mentores ou cursos"
   239→            className="inline-flex rounded-full bg-stone-100 p-1"
   240→          >
   241→            <button
   242→              role="tab"
   243→              aria-selected={tab === 'mentors'}
   244→              onClick={() => setTab('mentors')}
   245→              className={cn(
   246→                'inline-flex h-9 items-center gap-1.5 rounded-full px-4 text-sm font-semibold transition-all',
   247→                tab === 'mentors'
   248→                  ? 'bg-white text-stone-900 shadow-sm'
   249→                  : 'text-stone-500 hover:text-stone-700'
   250→              )}
   251→            >
   252→              <Users aria-hidden className="h-4 w-4" /> Mentores
   253→            </button>
   254→            <button
   255→              role="tab"
   256→              aria-selected={tab === 'courses'}
   257→              onClick={() => setTab('courses')}
   258→              className={cn(
   259→                'inline-flex h-9 items-center gap-1.5 rounded-full px-4 text-sm font-semibold transition-all',
   260→                tab === 'courses'
   261→                  ? 'bg-white text-stone-900 shadow-sm'
   262→                  : 'text-stone-500 hover:text-stone-700'
   263→              )}
   264→            >
   265→              <BookOpen aria-hidden className="h-4 w-4" /> Cursos
   266→            </button>
   267→          </div>
   268→
   269→          <div className="mt-5 flex flex-wrap items-end justify-between gap-3">
   270→            <div>
   271→              {tab === 'mentors' ? (
   272→                <>
   273→                  <h1 className="text-2xl font-extrabold tracking-tight text-stone-900 sm:text-3xl">
   274→                    Explorar mentores
   275→                  </h1>
   276→                  <p className="mt-1 text-sm text-stone-500">
   277→                    {loading
   278→                      ? 'Carregando especialistas...'
   279→                      : `${mentors.length} ${mentors.length === 1 ? 'especialista' : 'especialistas'} pronto${
   280→                          mentors.length === 1 ? '' : 's'
   281→                        } para mentoria 1:1`}
   282→                  </p>
   283→                </>
   284→              ) : (
   285→                <>
   286→                  <h1 className="text-2xl font-extrabold tracking-tight text-stone-900 sm:text-3xl">
   287→                    Explorar cursos
   288→                  </h1>
   289→                  <p className="mt-1 text-sm text-stone-500" aria-live="polite">
   290→                    {coursesLoading
   291→                      ? 'Carregando cursos...'
   292→                      : `${courses.length} ${courses.length === 1 ? 'curso publicado' : 'cursos publicados'}`}
   293→                  </p>
   294→                </>
   295→              )}
   296→            </div>
   297→            <div className="w-44">
   298→              <Select
   299→                value={tab === 'mentors' ? sort : courseSort}
   300→                onValueChange={tab === 'mentors' ? setSort : setCourseSort}
   301→              >
   302→                <SelectTrigger
   303→                  aria-label={tab === 'mentors' ? 'Ordenar mentores' : 'Ordenar cursos'}
   304→                  className="bg-white"
   305→                >
   306→                  <SelectValue placeholder="Ordenar" />
   307→                </SelectTrigger>
   308→                <SelectContent>
   309→                  {tab === 'mentors' ? (
   310→                    <>
   311→                      <SelectItem value="relevance">Relevância</SelectItem>
   312→                      <SelectItem value="rating">Melhor avaliados</SelectItem>
   313→                      <SelectItem value="price_asc">Menor preço</SelectItem>
   314→                      <SelectItem value="price_desc">Maior preço</SelectItem>
   315→                      <SelectItem value="experience">Mais experiência</SelectItem>
   316→                    </>
   317→                  ) : (
   318→                    <>
   319→                      <SelectItem value="relevance">Relevância</SelectItem>
   320→                      <SelectItem value="popular">Populares</SelectItem>
   321→                      <SelectItem value="new">Novidades</SelectItem>
   322→                      <SelectItem value="price_asc">Menor preço</SelectItem>
   323→                      <SelectItem value="price_desc">Maior preço</SelectItem>
   324→                    </>
   325→                  )}
   326→                </SelectContent>
   327→              </Select>
   328→            </div>
   329→          </div>
   330→
   331→          <div className="relative mt-5 max-w-2xl">
   332→            <Search
   333→              aria-hidden
   334→              className="absolute left-4 top-1/2 h-4.5 w-4.5 -translate-y-1/2 text-stone-400"
   335→            />
   336→            <Input
   337→              ref={searchRef}
   338→              value={inputValue}
   339→              onChange={(e) => onSearchChange(e.target.value)}
   340→              placeholder={
   341→                tab === 'mentors'
   342→                  ? 'Busque por nome, especialidade ou área...'
   343→                  : 'Busque por curso, tema ou mentor...'
   344→              }
   345→              aria-label={tab === 'mentors' ? 'Buscar mentores' : 'Buscar cursos'}
   346→              className="h-12 rounded-2xl border-stone-200 bg-white pl-11 pr-16 text-stone-900 shadow-none placeholder:text-stone-400 focus-visible:border-emerald-400 focus-visible:ring-emerald-200"
   347→            />
   348→            {inputValue ? (
   349→              <button
   350→                onClick={clearSearch}
   351→                aria-label="Limpar busca"
   352→                className="absolute right-3 top-1/2 flex h-7 w-7 -translate-y-1/2 items-center justify-center rounded-full text-stone-400 transition-colors hover:bg-stone-100 hover:text-stone-700"
   353→              >
   354→                <X className="h-4 w-4" />
   355→              </button>
   356→            ) : (
   357→              <kbd className="pointer-events-none absolute right-3.5 top-1/2 hidden h-6 -translate-y-1/2 items-center rounded-md border border-stone-200 bg-stone-50 px-1.5 font-mono text-[11px] font-medium text-stone-400 sm:inline-flex">
   358→                /
   359→              </kbd>
   360→            )}
   361→          </div>
   362→
   363→          <div
   364→            aria-label="Filtrar por categoria"
   365→            className="-mx-4 mt-4 flex gap-2 overflow-x-auto px-4 pb-0.5 [scrollbar-width:none] [&::-webkit-scrollbar]:hidden"
   366→          >
   367→            <button
   368→              onClick={() => setCategory('')}
   369→              className={cn(
   370→                'shrink-0 rounded-full border px-4 py-1.5 text-sm font-medium transition-colors',
   371→                category === ''
   372→                  ? 'border-emerald-700 bg-emerald-700 text-white'
   373→                  : 'border-stone-200 bg-white text-stone-600 hover:border-emerald-300 hover:text-emerald-700'
   374→              )}
   375→            >
   376→              Todas as áreas
   377→            </button>
   378→            {CATEGORIES.map((c) => {
   379→              const count =
   380→                tab === 'mentors'
   381→                  ? (categoryCounts.get(c) ?? 0)
   382→                  : (courseCounts.get(c) ?? 0)
   383→              return (
   384→                <button
   385→                  key={c}
   386→                  onClick={() => setCategory(category === c ? '' : c)}
   387→                  className={cn(
   388→                    'shrink-0 rounded-full border px-4 py-1.5 text-sm font-medium transition-colors',
   389→                    category === c
   390→                      ? 'border-emerald-700 bg-emerald-700 text-white'
   391→                      : 'border-stone-200 bg-white text-stone-600 hover:border-emerald-300 hover:text-emerald-700'
   392→                  )}
   393→                >
   394→                  {c}
   395→                  {(tab === 'mentors' ? baseReady : baseCoursesReady) && (
   396→                    <span className={cn('ml-1.5 text-[10px]', category === c ? 'text-emerald-100' : 'text-stone-400')}>
   397→                      {count}
   398→                    </span>
   399→                  )}
   400→                </button>
   401→              )
   402→            })}
   403→          </div>
   404→        </div>
   405→      </section>
   406→
   407→      {/* ---------- BENTO: destaque + estatísticas (por aba) ---------- */}
   408→      <section aria-label="Destaques e estatísticas" className="mx-auto w-full max-w-6xl px-4 pt-6">
   409→        <div className="grid grid-cols-1 gap-4 lg:grid-cols-3">
   410→          {/* Spotlight */}
   411→          <div className="lg:col-span-2">
   412→            {tab === 'mentors' ? (
   413→              spot ? (
   414→                <SpotlightCard mentor={spot} index={spotIdx} total={spotlightPool.length} onSelect={(i) => setSpotIdx(i)} />
   415→              ) : (
   416→                <div className="h-full min-h-56 animate-pulse rounded-2xl bg-emerald-950/90" aria-hidden />
   417→              )
   418→            ) : topCourse ? (
   419→              <CourseSpotlightCard course={topCourse} />
   420→            ) : (
   421→              <div className="h-full min-h-56 animate-pulse rounded-2xl bg-emerald-950/90" aria-hidden />
   422→            )}
   423→          </div>
   424→
   425→          {/* Stats */}
   426→          <div className="grid grid-cols-3 gap-4 lg:grid-cols-1">
   427→            {tab === 'mentors' ? (
   428→              <>
   429→                <StatTile icon={<Users className="h-4.5 w-4.5" />} value={`+${stats.mentors}`} label="mentores especialistas" />
   430→                <StatTile icon={<Video className="h-4.5 w-4.5" />} value={`+${stats.sessions}`} label="sessões realizadas" />
   431→                <StatTile
   432→                  icon={<Star className="h-4.5 w-4.5" />}
   433→                  value={stats.avg !== null ? stats.avg.toFixed(1).replace('.', ',') : '—'}
   434→                  label="nota média da comunidade"
   435→                />
   436→              </>
   437→            ) : (
   438→              <>
   439→                <StatTile icon={<Library className="h-4.5 w-4.5" />} value={`+${courseStats.published}`} label="cursos publicados" />
   440→                <StatTile icon={<Users className="h-4.5 w-4.5" />} value={`+${courseStats.students}`} label="alunos inscritos" />
   441→                <StatTile icon={<BookOpen className="h-4.5 w-4.5" />} value={`+${courseStats.lessons}`} label="aulas publicadas" />
   442→              </>
   443→            )}
   444→          </div>
   445→        </div>
   446→      </section>
   447→
   448→      {/* ---------- RESULTADOS ---------- */}
   449→      <section aria-labelledby="resultado-title" className="mx-auto w-full max-w-6xl px-4 pb-12 pt-8">
   450→        <div className="flex flex-wrap items-end justify-between gap-2">
   451→          <h2 id="resultado-title" className="text-lg font-extrabold tracking-tight text-stone-900">
   452→            {tab === 'mentors' ? 'Todos os mentores' : 'Todos os cursos'}
   453→          </h2>
   454→          {!(tab === 'mentors' ? loading : coursesLoading) && (
   455→            <p className="text-xs font-medium text-stone-400">
   456→              {search || category ? 'Resultado da busca' : 'Ordenado por relevância'}
   457→            </p>
   458→          )}
   459→        </div>
   460→
   461→        {tab === 'mentors' ? (
   462→          <>
   463→            <div className="mt-5 grid gap-5 sm:grid-cols-2 lg:grid-cols-3">
   464→              {loading
   465→                ? Array.from({ length: 6 }).map((_, i) => (
   466→                    <div key={i} className="rounded-2xl border border-stone-200 p-5">
   467→                      <div className="flex items-center gap-3">
   468→                        <Skeleton className="h-14 w-14 rounded-full" />
   469→                        <div className="flex-1 space-y-2">
   470→                          <Skeleton className="h-4 w-3/4" />
   471→                          <Skeleton className="h-3 w-1/2" />
   472→                        </div>
   473→                      </div>
   474→                      <Skeleton className="mt-4 h-3 w-full" />
   475→                      <Skeleton className="mt-2 h-3 w-5/6" />
   476→                      <div className="mt-4 flex gap-2">
   477→                        <Skeleton className="h-6 w-20 rounded-full" />
   478→                        <Skeleton className="h-6 w-16 rounded-full" />
   479→                      </div>
   480→                      <Skeleton className="mt-4 h-10 w-full rounded-full" />
   481→                    </div>
   482→                  ))
   483→                : mentors.map((m) => <MentorCard key={m.id} mentor={m} />)}
   484→            </div>
   485→
   486→            {!loading && mentors.length === 0 && (
   487→              <div className="mt-2 flex flex-col items-center gap-3 rounded-2xl border border-dashed border-stone-300 px-6 py-14 text-center">
   488→                <span className="flex h-14 w-14 items-center justify-center rounded-full bg-stone-100">
   489→                  <SearchX className="h-7 w-7 text-stone-400" />
   490→                </span>
   491→                <p className="font-bold text-stone-900">Nenhum mentor encontrado</p>
   492→                <p className="max-w-sm text-sm leading-relaxed text-stone-500">
   493→                  Tente remover os filtros ou buscar por outro termo — temos especialistas em várias
   494→                  áreas.
   495→                </p>
   496→                <Button
   497→                  variant="outline"
   498→                  className="rounded-full"
   499→                  onClick={() => {
   500→                    setSearch('')
   501→                    setInputValue('')
   502→                    setCategory('')
   503→                  }}
   504→                >
   505→                  Limpar filtros
   506→                </Button>
   507→              </div>
   508→            )}
   509→          </>
   510→        ) : (
   511→          <>
   512→            <div className="mt-5 grid gap-5 sm:grid-cols-2 lg:grid-cols-3">
   513→              {coursesLoading
   514→                ? Array.from({ length: 6 }).map((_, i) => (
   515→                    <div key={i} className="overflow-hidden rounded-2xl border border-stone-200">
   516→                      <Skeleton className="h-28 w-full rounded-none" />
   517→                      <div className="space-y-3 p-5">
   518→                        <Skeleton className="h-4 w-3/4" />
   519→                        <Skeleton className="h-3 w-1/3" />
   520→                        <Skeleton className="h-3 w-full" />
   521→                        <Skeleton className="h-3 w-5/6" />
