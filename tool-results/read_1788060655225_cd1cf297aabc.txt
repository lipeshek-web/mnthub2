    71→/** Tempo relativo compacto: "agora", "há 5 min", "há 2 h", "há 3 d" (data p/ antigos) */
    72→function relativeTime(iso: string) {
    73→  const diffMin = Math.max(0, Math.floor((Date.now() - new Date(iso).getTime()) / 60_000))
    74→  if (diffMin < 1) return 'agora'
    75→  if (diffMin < 60) return `há ${diffMin} min`
    76→  const hours = Math.floor(diffMin / 60)
    77→  if (hours < 24) return `há ${hours} h`
    78→  const days = Math.floor(hours / 24)
    79→  if (days < 30) return `há ${days} d`
    80→  return new Date(iso).toLocaleDateString('pt-BR', { day: '2-digit', month: 'short' })
    81→}

    85→/** Ícone de mensagens com badge de não lidas (polling leve a cada 45s) */
    86→function MessagesButton() {
    87→  const user = useAppStore((s) => s.user)
    88→  const navigate = useAppStore((s) => s.navigate)
    89→  const [unread, setUnread] = useState(0)
    90→  const aliveRef = useRef(true)
    91→
    92→  useEffect(() => {
    93→    const userId = user?.id
    94→    if (!userId) return
    95→    aliveRef.current = true
    96→    let alive = true
    97→    const load = () =>
    98→      api
    99→        .unreadMessages(userId)
   100→        .then((r) => {
   101→          if (alive) setUnread(r.count)
   102→        })
   103→        .catch(() => {
   104→          /* silencioso */
   105→        })
   106→    load()
   107→    const timer = setInterval(load, 45_000)
   108→    return () => {
   109→      alive = false
   110→      aliveRef.current = false
   111→      clearInterval(timer)
   112→    }
   113→  }, [user?.id])
   114→
   115→  if (!user) return null
   116→
   117→  return (
   118→    <button
   119→      type="button"
   120→      onClick={() => navigate({ name: 'messages' })}
   121→      aria-label={
   122→        unread > 0 ? `Mensagens, ${unread} não lida${unread === 1 ? '' : 's'}` : 'Mensagens'
   123→      }
   124→      title="Mensagens"
   125→      className="relative flex h-9 w-9 shrink-0 items-center justify-center rounded-full text-stone-500 transition-colors hover:bg-stone-100 hover:text-stone-900 dark:text-stone-400 dark:hover:bg-stone-800 dark:hover:text-stone-100"
   126→    >
   127→      <MessageCircle className="h-4.5 w-4.5" />
   128→      {unread > 0 && (
   129→        <span
   130→          aria-live="polite"
   131→          className="absolute -right-0.5 -top-0.5 flex h-4 min-w-4 items-center justify-center rounded-full bg-emerald-600 px-1 text-[10px] font-bold leading-none text-white"
   132→        >
   133→          {unread > 99 ? '99+' : unread}

   142→function ThemeToggle() {
   143→  const { resolvedTheme, setTheme } = useTheme()
   144→  // next-themes só sabe o tema no cliente — evita mismatch de hidratação
   145→  // (useSyncExternalStore em vez de setState em effect: sem render cascata)
   146→  const emptySubscribe = useCallback(() => () => {}, [])
   147→  const mounted = useSyncExternalStore(
   148→    emptySubscribe,
   149→    () => true,
   150→    () => false
   151→  )
   152→
   153→  const isDark = mounted && resolvedTheme === 'dark'
   154→  return (
   155→    <button
   156→      type="button"
   157→      onClick={() => setTheme(isDark ? 'light' : 'dark')}
   158→      aria-label={isDark ? 'Mudar para o tema claro' : 'Mudar para o tema escuro'}
   159→      title={isDark ? 'Tema claro' : 'Tema escuro'}
   160→      className="flex h-9 w-9 shrink-0 items-center justify-center rounded-full text-stone-500 transition-colors hover:bg-stone-100 hover:text-stone-900 dark:text-stone-400 dark:hover:bg-stone-800 dark:hover:text-stone-100"
   161→    >
   162→      {mounted ? (
   163→        isDark ? (
   164→          <Sun className="h-4.5 w-4.5" />
   165→        ) : (
   166→          <Moon className="h-4.5 w-4.5" />
   167→        )
   168→      ) : (
   169→        <span className="h-4.5 w-4.5" aria-hidden />
   170→      )}
   171→    </button>
   172→  )
   173→}

   175→function NotificationsBell() {
   176→  const user = useAppStore((s) => s.user)
   177→  const navigate = useAppStore((s) => s.navigate)
   178→
   179→  const [open, setOpen] = useState(false)
   180→  const [items, setItems] = useState<NotificationDTO[]>([])
   181→  const [unreadCount, setUnreadCount] = useState(0)
   182→
   183→  // Guard `alive` + contador de requisição: evita setState de respostas
   184→  // obsoletas (unmount, troca de usuário ou fetch sobreposto)
   185→  const aliveRef = useRef(true)
   186→  const reqIdRef = useRef(0)
   187→
   188→  const loadNotifications = useCallback(async (userId: string) => {
   189→    const reqId = ++reqIdRef.current
   190→    try {
   191→      const res = await api.listNotifications(userId)
   192→      if (!aliveRef.current || reqId !== reqIdRef.current) return
   193→      setItems(res.items)
   194→      setUnreadCount(res.unreadCount)
   195→    } catch {
   196→      // Erro silencioso: badge apenas não aparece (sem toast)
   197→    }
   198→  }, [])
   199→
   200→  const userId = user?.id
   201→  useEffect(() => {
   202→    if (!userId) return
   203→    aliveRef.current = true
   204→    let alive = true
   205→    // Fetch inicial (setState em callback assíncrono) com guard anti-race:
   206→    // descartado se o componente desmontar ou se um fetch mais novo existir
   207→    const reqId = ++reqIdRef.current
   208→    api
   209→      .listNotifications(userId)
   210→      .then((res) => {
   211→        if (!alive || reqId !== reqIdRef.current) return
   212→        setItems(res.items)
   213→        setUnreadCount(res.unreadCount)
   214→      })
   215→      .catch(() => {
   216→        // Erro silencioso: badge apenas não aparece
   217→      })
   218→    // Polling a cada 60s enquanto logado
   219→    const timer = setInterval(() => {
   220→      loadNotifications(userId)
   221→    }, 60_000)
   222→    return () => {
   223→      alive = false
   224→      aliveRef.current = false
   225→      clearInterval(timer)
   226→    }
   227→  }, [userId, loadNotifications])
   228→
   229→  const handleOpenChange = (next: boolean) => {
   230→    setOpen(next)
   231→    if (next && userId) loadNotifications(userId) // refetch ao abrir o dropdown
   232→  }
   233→
   234→  const handleItemClick = (item: NotificationDTO) => {
   235→    // Marca como lida de forma otimista (badge/zumbido atualizam na hora)
   236→    if (!item.read) {
   237→      setItems((prev) => prev.map((n) => (n.id === item.id ? { ...n, read: true } : n)))
   238→      setUnreadCount((count) => Math.max(0, count - 1))
   239→      if (userId) api.markNotificationsRead(userId, [item.id]).catch(() => {})
   240→    }
   241→    // Navega conforme o destino sugerido pela notificação
   242→    if (item.linkView === 'dashboard') navigate({ name: 'dashboard' })
   243→    else if (item.linkView === 'course' && item.refId) navigate({ name: 'course', courseId: item.refId })
   244→    else if (item.linkView === 'onboarding') navigate({ name: 'onboarding' })
   245→    else if (item.linkView === 'messages') navigate({ name: 'messages', peerId: item.refId ?? undefined })
   246→    else if (item.linkView === 'referrals') navigate({ name: 'referrals' })
   247→    setOpen(false)
   248→  }
   249→
   250→  const handleMarkAll = () => {
   251→    if (!userId || unreadCount === 0) return
   252→    const prevItems = items
   253→    setItems((prev) => prev.map((n) => ({ ...n, read: true })))
   254→    setUnreadCount(0)
   255→    api.markNotificationsRead(userId).catch(() => {
   256→      // Falhou: restaura o estado anterior
   257→      setItems(prevItems)
   258→      setUnreadCount(prevItems.filter((n) => !n.read).length)
   259→    })
   260→  }
   261→
   262→  if (!user) return null
   263→
   264→  return (
   265→    <DropdownMenu open={open} onOpenChange={handleOpenChange}>
   266→      <DropdownMenuTrigger asChild>
   267→        <button
   268→          type="button"
   269→          aria-label={
   270→            unreadCount > 0
   271→              ? `Notificações, ${unreadCount} não lida${unreadCount === 1 ? '' : 's'}`
   272→              : 'Notificações'
   273→          }
   274→          title="Notificações"
   275→          className="relative flex h-9 w-9 shrink-0 items-center justify-center rounded-full text-stone-500 transition-colors hover:bg-stone-100 hover:text-stone-900"
   276→        >
   277→          <Bell className="h-4.5 w-4.5" />
   278→          {unreadCount > 0 && (
   279→            <span
   280→              aria-live="polite"
   281→              className="absolute -right-0.5 -top-0.5 flex h-4 min-w-4 items-center justify-center rounded-full bg-rose-500 px-1 text-[10px] font-bold leading-none text-white"
   282→            >
   283→              {unreadCount > 99 ? '99+' : unreadCount}
   284→            </span>
   285→          )}
   286→        </button>
   287→      </DropdownMenuTrigger>
   288→
   289→      <DropdownMenuContent align="end" sideOffset={8} className="w-88 overflow-hidden p-0 sm:w-96">
   290→        {/* Cabeçalho do painel */}
   291→        <div className="flex items-center justify-between gap-2 border-b border-stone-100 px-4 py-2.5 dark:border-stone-800">
   292→          <DropdownMenuLabel className="p-0 font-semibold text-stone-900 dark:text-stone-100">
   293→            Notificações
   294→          </DropdownMenuLabel>
   295→          <button
   296→            type="button"
   297→            onClick={handleMarkAll}
   298→            disabled={unreadCount === 0}
   299→            className="rounded text-xs font-semibold text-emerald-700 transition-colors hover:text-emerald-900 disabled:pointer-events-none disabled:text-stone-300 dark:text-emerald-400 dark:hover:text-emerald-300 dark:disabled:text-stone-600"
   300→          >
   301→            Marcar todas como lidas
   302→          </button>
   303→        </div>
   304→
   305→        {items.length === 0 ? (
   306→          /* Empty state */
   307→          <div className="flex flex-col items-center gap-2 px-6 py-8 text-center">
   308→            <BellOff className="h-8 w-8 text-stone-300 dark:text-stone-600" aria-hidden />
   309→            <p className="text-sm text-stone-500 dark:text-stone-400">Você está em dia! Nenhuma notificação.</p>
   310→          </div>
   311→        ) : (
   312→          <div
   313→            role="list"
   314→            aria-label="Lista de notificações"
   315→            className="max-h-[380px] overflow-y-auto [scrollbar-width:thin] [&::-webkit-scrollbar]:h-1.5 [&::-webkit-scrollbar-thumb]:rounded-full [&::-webkit-scrollbar-thumb]:bg-stone-200 dark:[&::-webkit-scrollbar-thumb]:bg-stone-700 [&::-webkit-scrollbar-track]:bg-transparent"
   316→          >
   317→            {items.map((item) => {
   318→              const chip = NOTIFICATION_KINDS[item.kind] ?? DEFAULT_KIND
   319→              const Icon = chip.icon
   320→              return (
   321→                <div key={item.id} role="listitem">
   322→                  <button
   323→                    type="button"
   324→                    onClick={() => handleItemClick(item)}
   325→                    className={cn(
   326→                      'relative flex w-full items-start gap-3 px-4 py-3 text-left transition-colors focus-visible:outline-none',
   327→                      item.read
   328→                        ? 'hover:bg-stone-50 focus-visible:bg-stone-50 dark:hover:bg-stone-800/60 dark:focus-visible:bg-stone-800/60'
   329→                        : 'bg-emerald-50/50 hover:bg-emerald-50 focus-visible:bg-emerald-50 dark:bg-emerald-950/40 dark:hover:bg-emerald-900/30 dark:focus-visible:bg-emerald-900/30'
   330→                    )}
   331→                  >
   332→                    {!item.read && (
   333→                      <span
   334→                        aria-hidden
   335→                        className="absolute left-1.5 top-1/2 h-1.5 w-1.5 -translate-y-1/2 rounded-full bg-emerald-500"
   336→                      />
   337→                    )}
   338→                    <span
   339→                      className={cn(
   340→                        'flex h-8 w-8 shrink-0 items-center justify-center rounded-full',
   341→                        chip.chipClass
   342→                      )}
   343→                    >
   344→                      <Icon className="h-4 w-4" aria-hidden />
   345→                    </span>
   346→                    <span className="min-w-0 flex-1">
   347→                      <span className="flex items-baseline justify-between gap-2">
   348→                        <span
   349→                          className={cn(
   350→                            'truncate text-sm font-semibold',
   351→                            item.read ? 'text-stone-500' : 'text-stone-900'
   352→                          )}
   353→                        >
   354→                          {item.title}
   355→                        </span>
   356→                        <span className="shrink-0 text-[11px] text-stone-400 dark:text-stone-500">
   357→                          {relativeTime(item.createdAt)}
   358→                        </span>
   359→                      </span>
   360→                      {item.body && (
   361→                        <span className="mt-0.5 block text-xs leading-snug text-stone-500 line-clamp-2 dark:text-stone-400">
   362→                          {item.body}
   363→                        </span>
   364→                      )}
   365→                    </span>
   366→                  </button>
   367→                </div>
   368→              )
   369→            })}
   370→          </div>
   371→        )}
   372→      </DropdownMenuContent>
   373→    </DropdownMenu>
   374→  )
   375→}

   384→  const mobileSearchRef = useRef<HTMLInputElement>(null)

   386→  const submitSearch = (value: string) => {
   387→    const q = value.trim()

   395→  // Atalho "/" foca a busca do header (sensação de app nativo)

   398→      const target = e.target as HTMLElement | null
   399→      const typing =
   400→        target instanceof HTMLInputElement ||
   401→        target instanceof HTMLTextAreaElement ||
   402→        target?.isContentEditable
   403→      if (e.key === '/' && !typing) {
   404→        e.preventDefault()

   406→          desktopSearchRef.current?.focus()
   407→        } else {
   408→          setMobileSearchOpen(true)

   410→      }
   411→    }
