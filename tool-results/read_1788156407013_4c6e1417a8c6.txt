    62→export function CourseView({ courseId }: { courseId: string }) {
    63→  const navigate = useAppStore((s) => s.navigate)
    64→  const user = useAppStore((s) => s.user)
    65→
    66→  const [data, setData] = useState<CourseDetailDTO | null>(null)
    67→  const [loading, setLoading] = useState(true)
    68→  const [error, setError] = useState<string | null>(null)
    69→
    70→  // Estado da inscrição (modo visitante)
    71→  const [enrolling, setEnrolling] = useState(false)
    72→
    73→  // Estado de envio da avaliação do aluno (modo inscrito)
    74→  const [reviewSaving, setReviewSaving] = useState(false)
    75→
    76→  // Tracking (GA4/Meta Pixel do mentor + view_item) — uma única vez por curso/montagem
    77→  const trackedCourseRef = useRef<string | null>(null)
    78→
    79→  const fetchCourse = useCallback(async () => {
    80→    setLoading(true)
    81→    setError(null)
    82→    try {
    83→      const d = await api.getCourse(courseId, user?.id)
    84→      setData(d)
    85→    } catch (err) {
    86→      setError(err instanceof Error ? err.message : 'Não foi possível carregar o curso.')
    87→    } finally {
    88→      setLoading(false)
    89→    }
    90→  }, [courseId, user?.id])
    91→
    92→  useEffect(() => {
    93→    void fetchCourse()
    94→  }, [fetchCourse])
    95→
    96→  const lessons = useMemo<CourseLessonDTO[]>(
    97→    () => [...(data?.lessons ?? [])].sort((a, b) => a.order - b.order),
    98→    [data]
    99→  )
   100→
   101→  const course = data
   102→  const isEnrolled = course?.enrollment != null
   103→  const isOwner = Boolean(user && course && course.mentor.userId === user.id)
   104→
   105→  // Dispara view_item + injeta os pixels do mentor quando os dados do curso chegam
   106→  // (visitante ou inscrito), no máximo uma vez por curso por montagem.
   107→  useEffect(() => {
   108→    if (!course || trackedCourseRef.current === course.id) return
   109→    trackedCourseRef.current = course.id
   110→    loadTrackingScripts({
   111→      mentorGaId: course.mentor.tracking?.gaMeasurementId,
   112→      mentorPixelId: course.mentor.tracking?.metaPixelId,
   113→    })
   114→    trackEvent('view_item', {
   115→      mentorId: course.mentor.id,
   116→      courseId: course.id,
   117→      value: course.price || undefined,
   118→      contentName: course.title,
   119→    })
   120→  }, [course])
   121→
   122→  /* ---------- Avaliação do curso (aluno inscrito) ---------- */
   123→
   124→  const handleSaveReview = async (rating: number, comment: string) => {
   125→    if (!user || !course) return
   126→    setReviewSaving(true)
   127→    try {
   128→      const res: { id: string; updated: boolean; createdAt?: string } =
   129→        await api.saveCourseReview(course.id, { userId: user.id, rating, comment })
   130→
   131→      const saved: CourseReviewDTO = {
   132→        id: res.id,
   133→        rating,
   134→        comment,
   135→        createdAt: res.createdAt ?? new Date().toISOString(),
   136→        student: { id: user.id, name: user.name, avatarUrl: user.avatarUrl ?? null },
   137→      }
   138→
   139→      // Reflete localmente: atualiza/insere a review na lista, recalcula o resumo
   140→      // (distribuição → média) e preenche myReview para pré-preencher o formulário.
   141→      setData((prev) => {
   142→        if (!prev) return prev
   143→        const list = [...(prev.reviews ?? [])]
   144→        const idx = list.findIndex((r) => r.student.id === user.id)
   145→        const oldRating = idx >= 0 ? list[idx].rating : null
   146→        if (idx >= 0) list[idx] = saved
   147→        else list.unshift(saved)
   148→
   149→        const dist = [...(prev.reviewSummary?.distribution ?? [0, 0, 0, 0, 0])]
   150→        if (oldRating != null) {
   151→          dist[5 - oldRating] = Math.max(0, (dist[5 - oldRating] ?? 0) - 1)
   152→        }
   153→        dist[5 - rating] = (dist[5 - rating] ?? 0) + 1
   154→        const total = dist.reduce((acc, n, i) => acc + n * (5 - i), 0)
   155→        const newCount = dist.reduce((acc, n) => acc + n, 0)
   156→        const avg = newCount > 0 ? Math.round((total / newCount) * 10) / 10 : 0
   157→
   158→        return {
   159→          ...prev,
   160→          reviews: list.slice(0, 20),
   161→          reviewSummary: { rating: avg, count: newCount, distribution: dist },
   162→          myReview: { rating, comment },
   163→          rating: avg,
   164→          reviewCount: newCount,
   165→        }
   166→      })
   167→
   168→      toast.success(
   169→        res.updated ? 'Avaliação atualizada!' : 'Avaliação enviada! Obrigada pela contribuição 💚'
   170→      )
   171→    } catch (err) {
   172→      toast.error(err instanceof Error ? err.message : 'Erro ao salvar sua avaliação.')
   173→    } finally {
   174→      setReviewSaving(false)
   175→    }
   176→  }
   177→
   178→  /* ---------- Inscrição (overview) ---------- */
   179→
   180→  const doEnroll = async () => {
   181→    if (!user || !course) return
   182→    setEnrolling(true)
   183→    try {
   184→      const res = await api.enrollCourse(course.id, user.id)
   185→      if (res.alreadyEnrolled) {
   186→        toast.info('Você já estava inscrito neste curso.')
   187→        // Busca novamente com o usuário: enrollment preenchido mostra o modo inscrito
   188→        const fresh = await api.getCourse(course.id, user.id)
   189→        setData(fresh)
   190→      } else {
   191→        toast.success('Inscrição realizada! Boa jornada 🎉')
   192→        await api.getCourse(course.id, user.id)
   193→        // Nova inscrição: entra direto na sala de aula
   194→        navigate({ name: 'classroom', courseId: course.id })
   195→      }
   196→    } catch (err) {
   197→      toast.error(err instanceof Error ? err.message : 'Erro ao realizar inscrição.')
   198→    } finally {
   199→      setEnrolling(false)
   200→    }
   201→  }
   202→
   203→  const handleEnrollClick = () => {
   204→    if (!user) {
   205→      toast.info('Entre com uma conta para se inscrever.')
   206→      navigate({ name: 'auth', mode: 'login' })
   207→      return
   208→    }
   209→    if (!course) return
   210→    if (course.price === 0) {
   211→      void doEnroll()
   212→    } else {
   213→      // Curso pago: fluxo completo de checkout (com atribuição e pixels carregados)
   214→      trackEvent('begin_checkout', {
   215→        mentorId: course.mentor.id,
   216→        courseId: course.id,
   217→        value: course.price || undefined,
   218→        contentName: course.title,
   219→      })
   220→      navigate({ name: 'checkout', courseId: course.id })
   221→    }
   222→  }
   223→
   224→  /* ---------- Renders auxiliares ---------- */
   225→
   226→  if (loading) {
   227→    return (
   228→      <div className="mx-auto w-full max-w-7xl px-4 sm:px-6 py-6 sm:py-8" aria-busy="true">
   229→        <Skeleton className="h-9 w-24 rounded-full" />
   230→        <Skeleton className="mt-4 h-52 w-full rounded-2xl" />
   231→        <div className="mt-6 grid grid-cols-1 gap-6 lg:grid-cols-3">
   232→          <div className="space-y-6 lg:col-span-2">
   233→            <Skeleton className="h-44 rounded-2xl" />
   234→            <Skeleton className="h-64 rounded-2xl" />
   235→          </div>
   236→          <Skeleton className="h-72 rounded-2xl" />
   237→        </div>
   238→        <p className="sr-only">Carregando curso…</p>
   239→      </div>
   240→    )
   241→  }
   242→
   243→  if (!course) {
   244→    return (
   245→      <div className="mx-auto w-full max-w-7xl px-4 sm:px-6 py-6 sm:py-8">
   246→        <div className="flex flex-col items-center gap-3 rounded-2xl border border-dashed border-stone-300 dark:border-stone-700 px-6 py-14 text-center">
   247→          <span className="flex h-14 w-14 items-center justify-center rounded-full bg-rose-50 dark:bg-rose-950/50">
   248→            <AlertCircle aria-hidden className="h-7 w-7 text-rose-500" />
   249→          </span>
   250→          <p className="font-bold text-stone-900 dark:text-stone-50">Não foi possível carregar o curso</p>
   251→          <p className="max-w-sm text-sm leading-relaxed text-stone-500 dark:text-stone-400">
   252→            {error ?? 'Curso não encontrado.'}
   253→          </p>
   254→          <Button variant="outline" className="rounded-full" onClick={() => void fetchCourse()}>
   255→            Tentar novamente
   256→          </Button>
   257→        </div>
   258→      </div>
   259→    )
   260→  }
   261→
   262→  if (isEnrolled || isOwner) {
   263→    const done = course.enrollment?.completedLessonIds.length ?? 0
   264→    const pct = course.lessonCount > 0 ? Math.round((done / course.lessonCount) * 100) : 0
   265→    const enr = course.enrollment
   266→    // Curso 100% concluído → destaque suave no card de avaliação
   267→    const allDone =
   268→      enr != null && lessons.length > 0 && enr.completedLessonIds.length >= lessons.length
   269→    // Remonta o formulário quando a avaliação salva do servidor muda (pré-preencher)
   270→    const reviewFormKey = course.myReview
   271→      ? `${course.id}-saved-${course.myReview.rating}-${course.myReview.comment.length}`
   272→      : `${course.id}-new`
   273→    return (
   274→      <div className="mx-auto w-full max-w-7xl px-4 sm:px-6 py-6 sm:py-8">
   275→        <Button
   276→          variant="ghost"
   277→          onClick={() => navigate({ name: 'marketplace' })}
   278→          className="-ml-2 h-10 gap-1.5 rounded-full px-3 font-semibold text-stone-600 dark:text-stone-300 hover:text-stone-900 dark:hover:text-stone-50"
   279→        >
   280→          <ArrowLeft aria-hidden className="h-4 w-4" /> Voltar
   281→        </Button>
   282→
   283→        {/* Hero do aluno inscrito */}
   284→        <section
   285→          aria-label="Seu curso"
   286→          className="relative mt-2 overflow-hidden rounded-3xl bg-emerald-950 p-6 text-white sm:p-8"
   287→        >
   288→          <div aria-hidden className="absolute -right-16 -top-20 h-56 w-56 rounded-full bg-emerald-500/20 blur-3xl" />
   289→          <div aria-hidden className="absolute -bottom-24 left-10 h-48 w-48 rounded-full bg-teal-400/10 blur-3xl" />
   290→          <p className="relative inline-flex items-center gap-1.5 text-[10px] font-bold uppercase tracking-[0.22em] text-emerald-300">
   291→            {isOwner && !isEnrolled ? 'Seu curso' : 'Inscrição ativa'}
   292→          </p>
   293→          <h1 className="relative mt-2 max-w-2xl text-2xl font-extrabold tracking-tight sm:text-3xl">
   294→            {course.title}
   295→          </h1>
   296→          <p className="relative mt-2 flex flex-wrap items-center gap-x-3 gap-y-1 text-sm text-emerald-100/80">
   297→            <span>por {course.mentor.name}</span>
   298→            <span aria-hidden>·</span>
   299→            <span>{course.lessonCount} {course.lessonCount === 1 ? 'aula' : 'aulas'}</span>
   300→            <span aria-hidden>·</span>
   301→            <span>{formatTotalDuration(course.totalDurationMin)}</span>
   302→          </p>
   303→          <div className="relative mt-5 flex flex-wrap items-center gap-4">
   304→            <Button
   305→              size="lg"
   306→              onClick={() => navigate({ name: 'classroom', courseId: course.id })}
   307→              aria-label="Abrir a sala de aula"
   308→              className="h-12 rounded-full bg-white px-7 font-bold text-emerald-950 hover:bg-emerald-100"
   309→            >
   310→              <PlayCircle aria-hidden className="h-5 w-5" />
   311→              {done > 0 ? 'Continuar curso' : 'Começar agora'}
   312→            </Button>
   313→            <div className="min-w-40 flex-1 max-w-xs" aria-hidden>
   314→              <div className="flex items-center justify-between text-xs font-semibold text-emerald-100/80">
   315→                <span>Seu progresso</span>
   316→                <span className="tabular-nums">{pct}%</span>
   317→              </div>
   318→              <Progress
   319→                value={pct}
   320→                aria-label="Progresso do curso"
   321→                className="mt-1.5 h-2 bg-white/15 [&_[data-slot=progress-indicator]]:bg-emerald-400"
   322→              />
   323→              <p className="mt-1 text-[11px] text-emerald-100/60">
   324→                {done} de {course.lessonCount} {course.lessonCount === 1 ? 'aula concluída' : 'aulas concluídas'}
   325→              </p>
   326→            </div>
   327→          </div>
   328→        </section>
   329→
   330→        <OverviewContent
   331→          course={course}
   332→          lessons={lessons}
   333→          enrolling={false}
   334→          isLoggedIn={Boolean(user)}
   335→          enrolled
   336→          currentUserId={user?.id ?? null}
   337→          reviewFormNode={
   338→            enr ? (
   339→              <ReviewFormCard
   340→                key={reviewFormKey}
   341→                myReview={course.myReview}
   342→                completed={allDone}
   343→                saving={reviewSaving}
   344→                onSubmit={handleSaveReview}
   345→              />
   346→            ) : null
   347→          }
   348→          onContinue={() => navigate({ name: 'classroom', courseId: course.id })}
   349→          onEnrollClick={handleEnrollClick}
   350→          onLogin={() => navigate({ name: 'auth', mode: 'login' })}
   351→          onViewMentor={(mentorId) => navigate({ name: 'mentor', mentorId })}
   352→        />
   353→      </div>
   354→    )
   355→  }
   356→
   357→  return (
   358→    <div className="mx-auto w-full max-w-7xl px-4 sm:px-6 py-6 sm:py-8">
   359→      <Button
   360→        variant="ghost"
   361→        onClick={() => navigate({ name: 'marketplace' })}
   362→        className="-ml-2 h-10 gap-1.5 rounded-full px-3 font-semibold text-stone-600 dark:text-stone-300 hover:text-stone-900 dark:hover:text-stone-50"
   363→      >
   364→        <ArrowLeft aria-hidden className="h-4 w-4" /> Voltar
   365→      </Button>
   366→
   367→      <OverviewContent
   368→        course={course}
   369→        lessons={lessons}
   370→        enrolling={enrolling}
   371→        isLoggedIn={Boolean(user)}
   372→        currentUserId={user?.id ?? null}
   373→        onContinue={
   374→          course.enrollment || isOwner
   375→            ? () => navigate({ name: 'classroom', courseId: course.id })
   376→            : undefined
   377→        }
   378→        onEnrollClick={handleEnrollClick}
   379→        onLogin={() => navigate({ name: 'auth', mode: 'login' })}
   380→        onViewMentor={(mentorId) => navigate({ name: 'mentor', mentorId })}
   381→      />
   382→    </div>
   383→  )
   384→}

   386→/* ==================== MODO VISITANTE (overview) ==================== */
   387→
   388→function OverviewContent({
   389→  course,
   390→  lessons,
   391→  enrolling,
   392→  isLoggedIn,
   393→  enrolled = false,
   394→  currentUserId = null,
   395→  reviewFormNode = null,
   396→  onContinue,
   397→  onEnrollClick,
   398→  onLogin,
   399→  onViewMentor,
   400→}: {
   401→  course: CourseDetailDTO
   402→  lessons: CourseLessonDTO[]
   403→  enrolling: boolean
   404→  isLoggedIn: boolean
   405→  enrolled?: boolean
   406→  currentUserId?: string | null
   407→  reviewFormNode?: ReactNode
   408→  onContinue?: () => void
   409→  onEnrollClick: () => void
   410→  onLogin: () => void
   411→  onViewMentor: (mentorId: string) => void
   412→}) {
   413→  // Currículo agrupado por temas (em ordem); aulas sem tema → "Outros conteúdos" no fim
   414→  const themes = useMemo(
   415→    () => [...(course.themes ?? [])].sort((a, b) => a.order - b.order),
   416→    [course]
   417→  )
   418→  const curriculumGroups = useMemo(() => {
   419→    const byTheme = new Map<string, CourseLessonDTO[]>()
   420→    const loose: CourseLessonDTO[] = []
   421→    for (const lesson of lessons) {
   422→      if (lesson.themeId && themes.some((t) => t.id === lesson.themeId)) {
   423→        const bucket = byTheme.get(lesson.themeId) ?? []
   424→        bucket.push(lesson)
   425→        byTheme.set(lesson.themeId, bucket)
   426→      } else {
   427→        loose.push(lesson)
   428→      }
   429→    }
   430→    const groups: { key: string; title: string; lessons: CourseLessonDTO[] }[] = themes.map((t) => ({
   431→      key: t.id,
   432→      title: t.title,
   433→      lessons: byTheme.get(t.id) ?? [],
   434→    }))
   435→    if (loose.length > 0) groups.push({ key: 'none', title: 'Outros conteúdos', lessons: loose })
   436→    return groups
   437→  }, [lessons, themes])
   438→  return (
   439→    <>
   440→      {/* ---------- HERO (oculto no modo inscrito: já existe o hero de progresso) ---------- */}
   441→      {!enrolled && (
   442→      <section
   443→        aria-label={`Curso: ${course.title}`}
   444→        className="relative overflow-hidden rounded-2xl p-6 text-white sm:p-8"
   445→        style={avatarGradient(course.title)}
   446→      >
   447→        {course.coverUrl && (
   448→          <>
   449→            <img
   450→              src={course.coverUrl}
   451→              alt=""
   452→              aria-hidden
   453→              className="absolute inset-0 h-full w-full object-cover"
   454→            />
   455→            <div aria-hidden className="absolute inset-0 bg-stone-950/55" />
   456→          </>
   457→        )}
   458→        {!course.coverUrl && (
   459→          <Library aria-hidden className="pointer-events-none absolute -right-6 -top-10 h-48 w-48 text-white/15" />
   460→        )}
   461→        <div className="relative">
   462→          <div className="flex flex-wrap items-center gap-2">
   463→            <Badge className="rounded-full border-white/25 bg-white/15 text-white hover:bg-white/15">
   464→              {course.category}
   465→            </Badge>
   466→            <Badge className="rounded-full border-white/25 bg-stone-900/30 text-white hover:bg-stone-900/30">
   467→              {LEVEL_LABELS[course.level] ?? course.level}
   468→            </Badge>
   469→          </div>
   470→          <h1 className="mt-4 text-2xl font-extrabold tracking-tight sm:text-3xl">{course.title}</h1>
   471→          <p className="mt-2 max-w-2xl text-sm leading-relaxed text-white/85 sm:text-[15px]">
   472→            {course.description}
   473→          </p>
   474→          <div className="mt-5 flex flex-wrap items-center gap-x-5 gap-y-2 text-sm text-white/90">
   475→            <span className="inline-flex items-center gap-1.5">
   476→              <BookOpen aria-hidden className="h-4 w-4" />
   477→              {course.lessonCount} {course.lessonCount === 1 ? 'aula' : 'aulas'}
   478→            </span>
   479→            {course.liveCount > 0 ? (
   480→              <span className="inline-flex items-center gap-1.5 rounded-full bg-rose-500/25 px-2.5 py-0.5 font-semibold">
   481→                <Radio aria-hidden className="h-4 w-4" />
   482→                {course.liveCount} ao vivo
   483→              </span>
   484→            ) : null}
   485→            <span className="inline-flex items-center gap-1.5">
   486→              <Clock aria-hidden className="h-4 w-4" />
   487→              {formatTotalDuration(course.totalDurationMin)}
   488→            </span>
   489→            {course.mentorshipCount > 0 ? (
   490→              <span className="inline-flex items-center gap-1.5">
   491→                <UserRound aria-hidden className="h-4 w-4" />
   492→                {course.mentorshipCount} mentoria{course.mentorshipCount > 1 ? 's' : ''} 1:1
   493→              </span>
   494→            ) : null}
   495→            <span className="inline-flex items-center gap-1.5">
   496→              <Users aria-hidden className="h-4 w-4" />
   497→              {course.studentCount} {course.studentCount === 1 ? 'aluno' : 'alunos'}
   498→            </span>
   499→            <span className="font-extrabold">
   500→              {course.price === 0 ? 'Grátis' : currencyBRL(course.price)}
   501→            </span>
   502→          </div>
   503→        </div>
   504→      </section>
   505→      )}
   506→
   507→      {/* ---------- CONTEÚDO ---------- */}
   508→      <div className="mt-6 grid grid-cols-1 gap-6 lg:grid-cols-3">
   509→        <div className="space-y-6 lg:col-span-2">
   510→          {/* O que você vai aprender */}
   511→          <section
   512→            aria-labelledby="aprender-title"
   513→            className="rounded-2xl border border-stone-200 dark:border-stone-800 bg-white dark:bg-stone-900 p-5 sm:p-6"
   514→          >
   515→            <h2 id="aprender-title" className="text-lg font-extrabold tracking-tight text-stone-900 dark:text-stone-50">

   832→/** Busca pacotes publicados que contêm este curso; esconde silenciosamente se não houver */
   833→function BundleCallout({
   834→  courseId,
   835→  currentUserId,
   836→}: {
   837→  courseId: string
   838→  currentUserId?: string | null
   839→}) {
   840→  const navigate = useAppStore((s) => s.navigate)
   841→  const [bundle, setBundle] = useState<BundleDTO | null>(null)
   842→
   843→  useEffect(() => {
   844→    let active = true
   845→    api
   846→      .listBundles({ courseId, userId: currentUserId ?? undefined })
   847→      .then((res) => {
   848→        if (active && res.bundles.length > 0) setBundle(res.bundles[0])
   849→      })
   850→      .catch(() => {})
   851→    return () => {
   852→      active = false
   853→    }
   854→  }, [courseId, currentUserId])
   855→
   856→  if (!bundle) return null
   857→
   858→  const brl = (v: number) =>
   859→    v.toLocaleString('pt-BR', { style: 'currency', currency: 'BRL', minimumFractionDigits: 2 })
   860→
   861→  return (
   862→    <div className="rounded-2xl border border-emerald-200 dark:border-emerald-900 bg-emerald-50/60 dark:bg-emerald-950/50 p-4">
   863→      <p className="flex items-center gap-2 text-sm font-bold text-emerald-900 dark:text-emerald-200">
   864→        <Layers aria-hidden className="h-4 w-4 text-emerald-700 dark:text-emerald-300" />
   865→        Este curso faz parte do pacote “{bundle.title}”
   866→      </p>
   867→      <p className="mt-1 text-xs leading-relaxed text-emerald-800/80 dark:text-emerald-300/80">
   868→        Leve {bundle.courseCount} cursos de {bundle.mentor.name} por {bundle.price === 0 ? 'grátis' : brl(bundle.price)}
   869→        {bundle.coursesTotal > bundle.price && bundle.price > 0
   870→          ? ` — economize ${bundle.discountPercent}% em relação à compra separada`
   871→          : ''}
   872→        .
   873→      </p>
   874→      <Button
   875→        size="sm"
   876→        onClick={() => navigate({ name: 'checkout', bundleId: bundle.id })}
   877→        className="mt-3 h-10 w-full rounded-full bg-emerald-700 font-bold hover:bg-emerald-800"
   878→      >
   879→        <BadgePercent aria-hidden className="h-4 w-4" />
   880→        Ver pacote
   881→      </Button>
   882→    </div>
   883→  )
   884→}

   888→/** Plano de assinatura publicado do mentor; esconde silenciosamente se não houver */
   889→function MembershipCallout({
   890→  mentorId,
   891→  currentUserId,
   892→}: {
   893→  mentorId: string
   894→  currentUserId?: string | null
   895→}) {

   897→  const membershipTuple = useState<MembershipDTO | null>(null)
   898→  const membership = membershipTuple[0]
   899→  const setMembership = membershipTuple[1]
   900→
   901→  useEffect(() => {
   902→    let active = true
   903→    api
   904→      .listMemberships({ mentorId, userId: currentUserId ?? undefined })
   905→      .then((res) => {
   906→        const m = res.memberships.find((x) => x.isPublished && x.myStatus !== 'ACTIVE')
   907→        if (active && m) setMembership(m)
   908→      })
   909→      .catch(() => {})
   910→    return () => {
   911→      active = false
   912→    }
   913→  }, [mentorId, currentUserId])
   914→
   915→  if (!membership) return null
   916→
   917→  const brl = (v: number) =>
   918→    v.toLocaleString('pt-BR', { style: 'currency', currency: 'BRL', minimumFractionDigits: 2 })
   919→
   920→  return (
   921→    <div className="rounded-2xl border border-emerald-200 dark:border-emerald-900 bg-emerald-50/60 dark:bg-emerald-950/50 p-4">
   922→      <p className="flex items-center gap-2 text-sm font-bold text-emerald-900 dark:text-emerald-200">
   923→        <CreditCard aria-hidden className="h-4 w-4 text-emerald-700 dark:text-emerald-300" />

   926→      <p className="mt-1 text-xs leading-relaxed text-emerald-800/80 dark:text-emerald-300/80">
   927→        Todos os {membership.coursesCount} cursos de {membership.mentor.name} + sessão em grupo
   928→        mensal por {brl(membership.price)}/mês — cancele quando quiser.
   929→      </p>
   930→      <Button
   931→        size="sm"
   932→        onClick={() => navigate({ name: 'checkout', membershipId: membership.id })}
   933→        className="mt-3 h-10 w-full rounded-full bg-emerald-700 font-bold hover:bg-emerald-800"
   934→      >
   935→        <Sparkles aria-hidden className="h-4 w-4" />

   938→    </div>

   963→  const shown = hover || rating

  1007→              className="flex h-9 w-9 items-center justify-center rounded-full transition-transform hover:scale-110 focus-visible:outline-2 focus-visible:outline-offset-2 focus-visible:outline-emerald-700 disabled:pointer-events-none"
  1008→            >
  1009→              <Star
  1010→                aria-hidden
  1011→                className={cn(

  1024→      </div>

  1027→        value={comment}

  1029→        maxLength={800}
  1030→        rows={4}

  1032→        aria-label="Comentário da avaliação (opcional)"

  1034→      />

  1037→      <Button
  1038→        onClick={submit}
  1039→        disabled={rating < 1 || saving}
  1040→        className="mt-2 h-11 w-full rounded-full bg-emerald-700 font-bold text-white hover:bg-emerald-800"
  1041→      >

  1050→/** Data relativa/curta em pt-BR: "agora", "há 5 min", "ontem", "12 de mai. de 2025" */
  1051→function reviewDateLabel(iso: string): string {
  1052→  const d = new Date(iso)
  1053→  const diffMin = Math.floor((Date.now() - d.getTime()) / 60000)
  1054→  if (diffMin < 1) return 'agora'
  1055→  if (diffMin < 60) return `há ${diffMin} min`
  1056→  const hours = Math.floor(diffMin / 60)
  1057→  if (hours < 24) return `há ${hours} h`
  1058→  const days = Math.floor(hours / 24)
  1059→  if (days === 1) return 'ontem'
  1060→  if (days < 30) return `há ${days} dias`
  1061→  return d.toLocaleDateString('pt-BR', { day: '2-digit', month: 'short', year: 'numeric' })
  1062→}
