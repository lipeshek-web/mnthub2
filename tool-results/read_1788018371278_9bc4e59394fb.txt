   159→/** Valida o tamanho (máx. 5MB), envia o arquivo e devolve a URL — falhas são exibidas via toast */
   160→async function uploadImageFile(file: File | null | undefined): Promise<string | null> {
   161→  if (!file) return null
   162→  if (file.size > MAX_IMAGE_BYTES) {
   163→    toast.error('A imagem deve ter no máximo 5MB.')
   164→    return null
   165→  }
   166→  try {
   167→    return await api.uploadImage(file)
   168→  } catch (err) {
   169→    toast.error(err instanceof Error ? err.message : 'Não foi possível enviar a imagem.')
   170→    return null
   171→  }
   172→}

   174→function copyToClipboard(text: string, successMessage = 'Link copiado!') {
   175→  if (typeof navigator === 'undefined' || !navigator.clipboard?.writeText) {
   176→    toast.error('Não foi possível copiar o link.')
   177→    return
   178→  }
   179→  navigator.clipboard
   180→    .writeText(text)
   181→    .then(() => toast.success(successMessage))
   182→    .catch(() => toast.error('Não foi possível copiar o link.'))
   183→}

   442→function MentorProfileForm({
   443→  initial,
   444→  submitLabel,
   445→  onSubmit,
   446→  mentorName,
   447→}: {
   448→  initial?: ProfileFormValues
   449→  submitLabel: string
   450→  onSubmit: (values: ProfileFormValues) => Promise<void>
   451→  /** Nome usado como fallback do avatar (iniciais) */
   452→  mentorName: string
   453→}) {
   454→  const [headline, setHeadline] = useState(initial?.headline ?? '')
   455→  const [description, setDescription] = useState(initial?.description ?? '')
   456→  const [categories, setCategories] = useState<string[]>(initial?.categories ?? [])
   457→  const [hourlyRate, setHourlyRate] = useState(initial ? String(initial.hourlyRate) : '')
   458→  const [experienceYears, setExperienceYears] = useState(initial ? String(initial.experienceYears) : '')
   459→  const [languages, setLanguages] = useState(initial?.languages || 'Português')
   460→  const [instagram, setInstagram] = useState(initial?.socials?.instagram ?? '')
   461→  const [linkedin, setLinkedin] = useState(initial?.socials?.linkedin ?? '')
   462→  const [github, setGithub] = useState(initial?.socials?.github ?? '')
   463→  const [website, setWebsite] = useState(initial?.socials?.website ?? '')
   464→  const [errors, setErrors] = useState<ProfileFormErrors>({})
   465→  const [saving, setSaving] = useState(false)
   466→  const [avatarUrl, setAvatarUrl] = useState<string | null>(initial?.avatarUrl ?? null)
   467→  const [coverUrl, setCoverUrl] = useState<string | null>(initial?.coverUrl ?? null)
   468→  const [fontHeading, setFontHeading] = useState<string | null>(initial?.fontHeading ?? null)
   469→  const [fontBody, setFontBody] = useState<string | null>(initial?.fontBody ?? null)
   470→  const [uploadingPhoto, setUploadingPhoto] = useState<'avatar' | 'cover' | null>(null)
   471→  // Só envia no payload a foto que o usuário realmente mexeu: undefined = manter atual, null = remover
   472→  const [photoTouched, setPhotoTouched] = useState<{ avatar: boolean; cover: boolean }>({
   473→    avatar: false,
   474→    cover: false,
   475→  })
   476→
   477→  const toggleCategory = (category: string) => {
   478→    setCategories((prev) =>
   479→      prev.includes(category) ? prev.filter((c) => c !== category) : [...prev, category]
   480→    )
   481→    setErrors((prev) => ({ ...prev, categories: undefined }))
   482→  }
   483→
   484→  const handlePhotoFile = async (kind: 'avatar' | 'cover', file: File | null | undefined) => {
   485→    if (!file || uploadingPhoto) return
   486→    setUploadingPhoto(kind)
   487→    const url = await uploadImageFile(file)
   488→    if (url) {
   489→      if (kind === 'avatar') {
   490→        setAvatarUrl(url)
   491→        setPhotoTouched((prev) => ({ ...prev, avatar: true }))
   492→      } else {
   493→        setCoverUrl(url)
   494→        setPhotoTouched((prev) => ({ ...prev, cover: true }))
   495→      }
   496→    }
   497→    setUploadingPhoto(null)
   498→  }
   499→
   500→  const validate = (): ProfileFormErrors => {
   501→    const errs: ProfileFormErrors = {}
   502→    if (headline.trim().length < 8) {
   503→      errs.headline = 'O título deve ter pelo menos 8 caracteres.'
   504→    }
   505→    if (description.trim().length < 30) {
   506→      errs.description = 'Descreva sua mentoria com pelo menos 30 caracteres.'
   507→    }
   508→    if (categories.length === 0) {
   509→      errs.categories = 'Selecione pelo menos uma categoria.'
   510→    }
   511→    const rate = Number(hourlyRate)
   512→    if (hourlyRate.trim() === '' || Number.isNaN(rate) || rate < 0) {
   513→      errs.hourlyRate = 'Informe um valor válido por hora (maior ou igual a zero).'
   514→    }
   515→    const years = Number(experienceYears)
   516→    if (experienceYears.trim() === '' || Number.isNaN(years) || years < 0) {
   517→      errs.experienceYears = 'Informe seus anos de experiência.'
   518→    }
   519→    return errs
   520→  }
   521→
   522→  const handleSubmit = async (event: FormEvent<HTMLFormElement>) => {
   523→    event.preventDefault()
   524→    const errs = validate()
   525→    setErrors(errs)
   526→    if (Object.values(errs).some(Boolean)) return
   527→    setSaving(true)
   528→    try {
   529→      await onSubmit({
   530→        headline: headline.trim(),
   531→        description: description.trim(),
   532→        categories,
   533→        hourlyRate: Number(hourlyRate),
   534→        experienceYears: Number(experienceYears),
   535→        languages: languages.trim() || 'Português',
   536→        socials: {
   537→          instagram: instagram.trim() || undefined,
   538→          linkedin: linkedin.trim() || undefined,
   539→          github: github.trim() || undefined,
   540→          website: website.trim() || undefined,
   541→        },
   542→        // undefined = não alterar no servidor; null = remover; string = definir
   543→        avatarUrl: photoTouched.avatar ? avatarUrl : undefined,
   544→        coverUrl: photoTouched.cover ? coverUrl : undefined,
   545→        // Tipografia sempre enviada (null volta ao padrão da plataforma)
   546→        fontHeading,
   547→        fontBody,
   548→      })
   549→    } catch (err) {
   550→      toast.error(err instanceof Error ? err.message : 'Não foi possível salvar o perfil.')
   551→    } finally {
   552→      setSaving(false)
   553→    }

   871→function AvailabilityEditor({
   872→  initialSlots,
   873→  onSave,
   874→}: {
   875→  initialSlots: AvailabilitySlotInput[]
   876→  onSave: (slots: AvailabilitySlotInput[]) => Promise<void>
   877→}) {
   878→  const [slots, setSlots] = useState<AvailabilitySlotInput[]>(() =>
   879→    [...initialSlots].sort((a, b) => a.weekday - b.weekday || a.startHour - b.startHour)
   880→  )
   881→  const [addingDay, setAddingDay] = useState<number | null>(null)
   882→  const [newStart, setNewStart] = useState('09:00')
   883→  const [newEnd, setNewEnd] = useState('12:00')
   884→  const [saving, setSaving] = useState(false)
   885→
   886→  const openAdd = (weekday: number) => {
   887→    setAddingDay(weekday)
   888→    setNewStart('09:00')
   889→    setNewEnd('12:00')
   890→  }
   891→
   892→  const removeSlot = (weekday: number, startHour: number, endHour: number) => {
   893→    setSlots((prev) =>
   894→      prev.filter((s) => !(s.weekday === weekday && s.startHour === startHour && s.endHour === endHour))
   895→    )
   896→  }
   897→
   898→  const confirmAdd = () => {
   899→    if (addingDay === null) return
   900→    const startHour = labelToHour(newStart)
   901→    const endHour = labelToHour(newEnd)
   902→    if (endHour - startHour < 1) {
   903→      toast.error('A faixa de horário deve ter pelo menos 1 hora de duração.')
   904→      return
   905→    }
   906→    const conflicts = slots.some(
   907→      (s) => s.weekday === addingDay && startHour < s.endHour && endHour > s.startHour
   908→    )
   909→    if (conflicts) {
   910→      toast.error('Esta faixa conflita com um horário já cadastrado neste dia.')
   911→      return
   912→    }
   913→    setSlots((prev) => [...prev, { weekday: addingDay, startHour, endHour }])
   914→    setAddingDay(null)
   915→  }
   916→
   917→  const handleSave = async () => {
   918→    setSaving(true)
   919→    try {
   920→      const ordered = [...slots].sort((a, b) => a.weekday - b.weekday || a.startHour - b.startHour)
   921→      await onSave(ordered)
   922→    } finally {
   923→      setSaving(false)
   924→    }
   925→  }
   926→
   927→  return (
   928→    <div className="flex flex-col gap-4">
   929→      <div className="flex flex-col divide-y divide-stone-200 rounded-lg border border-stone-200">
   930→        {WEEKDAYS_FULL_PT.map((day, weekday) => {
   931→          const daySlots = slots
   932→            .filter((s) => s.weekday === weekday)
   933→            .sort((a, b) => a.startHour - b.startHour)
   934→          return (
   935→            <div key={day} className="flex flex-col gap-3 p-4 sm:flex-row sm:items-center sm:justify-between">
   936→              <div className="flex min-w-0 flex-1 flex-wrap items-center gap-2">
   937→                <span className="w-28 shrink-0 text-sm font-medium sm:w-32">{day}</span>
   938→                {daySlots.length === 0 ? (
   939→                  <span className="text-xs text-stone-400">Sem disponibilidade</span>
   940→                ) : (
   941→                  <span className="flex flex-wrap items-center gap-1.5">
   942→                    {daySlots.map((s) => (
   943→                      <Badge
   944→                        key={`${s.weekday}-${s.startHour}-${s.endHour}`}
   945→                        variant="outline"
   946→                        className="gap-1 border-emerald-200 bg-emerald-50 py-1 text-emerald-800"
   947→                      >
   948→                        {hourToLabel(s.startHour)} – {hourToLabel(s.endHour)}
   949→                        <button
   950→                          type="button"
   951→                          aria-label={`Remover faixa de ${hourToLabel(s.startHour)} às ${hourToLabel(
   952→                            s.endHour
   953→                          )} na ${day.toLowerCase()}`}
   954→                          onClick={() => removeSlot(s.weekday, s.startHour, s.endHour)}
   955→                          className="rounded-full p-0.5 transition-colors hover:bg-emerald-200/70 focus-visible:outline-2 focus-visible:outline-emerald-600"
   956→                        >
   957→                          <X className="size-3" aria-hidden />
   958→                        </button>
   959→                      </Badge>
   960→                    ))}
   961→                  </span>
   962→                )}
   963→              </div>
   964→
   965→              <div className="shrink-0">
   966→                {addingDay === weekday ? (
   967→                  <div className="flex flex-wrap items-center gap-2">
   968→                    <Select value={newStart} onValueChange={setNewStart}>
   969→                      <SelectTrigger size="sm" className="w-24" aria-label="Início da faixa">
   970→                        <SelectValue />
   971→                      </SelectTrigger>
   972→                      <SelectContent>
   973→                        {TIME_OPTIONS.map((time) => (
   974→                          <SelectItem key={time} value={time}>
   975→                            {time}
   976→                          </SelectItem>
   977→                        ))}
   978→                      </SelectContent>
   979→                    </Select>
   980→                    <span className="text-sm text-muted-foreground" aria-hidden>
   981→                      →
   982→                    </span>
   983→                    <Select value={newEnd} onValueChange={setNewEnd}>
   984→                      <SelectTrigger size="sm" className="w-24" aria-label="Fim da faixa">
   985→                        <SelectValue />
   986→                      </SelectTrigger>
   987→                      <SelectContent>
   988→                        {TIME_OPTIONS.map((time) => (
   989→                          <SelectItem key={time} value={time}>
   990→                            {time}
   991→                          </SelectItem>
   992→                        ))}
   993→                      </SelectContent>
   994→                    </Select>
   995→                    <Button size="sm" onClick={confirmAdd}>
   996→                      <Check className="size-4" aria-hidden /> Confirmar
   997→                    </Button>
   998→                    <Button size="sm" variant="ghost" onClick={() => setAddingDay(null)}>
   999→                      Cancelar
  1000→                    </Button>
  1001→                  </div>
  1002→                ) : (
  1003→                  <Button
  1004→                    size="sm"
  1005→                    variant="ghost"
  1006→                    onClick={() => openAdd(weekday)}
  1007→                    aria-label={`Adicionar faixa de horário na ${day.toLowerCase()}`}
  1008→                  >
  1009→                    <Plus className="size-4" aria-hidden /> Adicionar
  1010→                  </Button>
  1011→                )}
  1012→              </div>
  1013→            </div>
  1014→          )
  1015→        })}
  1016→      </div>
  1017→
  1018→      <div className="flex flex-col gap-3 sm:flex-row sm:items-center sm:justify-between">
  1019→        <p className="text-xs text-muted-foreground">
  1020→          Alunos só podem agendar dentro destas faixas. Use intervalos de pelo menos 1 hora.
  1021→        </p>
  1022→        <Button onClick={() => void handleSave()} disabled={saving} className="shrink-0">
  1023→          <Save className="size-4" aria-hidden /> {saving ? 'Salvando...' : 'Salvar disponibilidade'}
  1024→        </Button>
  1025→      </div>
  1026→    </div>
  1027→  )
  1028→}
