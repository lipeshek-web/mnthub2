    96→  const options: string[] = []
    97→  for (let minutes = 6 * 60; minutes <= 21 * 60 + 30; minutes += 30) {
    98→    const h = Math.floor(minutes / 60)
    99→    const m = minutes % 60
   100→    options.push(`${String(h).padStart(2, '0')}:${String(m).padStart(2, '0')}`)
   101→  }
   102→  return options
   103→})()

   403→// ---------- Editor de disponibilidade semanal ----------
   404→
   405→function AvailabilityEditor({
   406→  initialSlots,
   407→  onSave,
   408→}: {
   409→  initialSlots: AvailabilitySlotInput[]
   410→  onSave: (slots: AvailabilitySlotInput[]) => Promise<void>
   411→}) {
   412→  const [slots, setSlots] = useState<AvailabilitySlotInput[]>(() =>
   413→    [...initialSlots].sort((a, b) => a.weekday - b.weekday || a.startHour - b.startHour)
   414→  )
   415→  const [addingDay, setAddingDay] = useState<number | null>(null)
   416→  const [newStart, setNewStart] = useState('09:00')
   417→  const [newEnd, setNewEnd] = useState('12:00')
   418→  const [saving, setSaving] = useState(false)
   419→
   420→  const openAdd = (weekday: number) => {
   421→    setAddingDay(weekday)
   422→    setNewStart('09:00')
   423→    setNewEnd('12:00')
   424→  }
   425→
   426→  const removeSlot = (weekday: number, startHour: number, endHour: number) => {
   427→    setSlots((prev) =>
   428→      prev.filter((s) => !(s.weekday === weekday && s.startHour === startHour && s.endHour === endHour))
   429→    )
   430→  }
   431→
   432→  const confirmAdd = () => {
   433→    if (addingDay === null) return
   434→    const startHour = labelToHour(newStart)
   435→    const endHour = labelToHour(newEnd)
   436→    if (endHour - startHour < 1) {
   437→      toast.error('A faixa de horário deve ter pelo menos 1 hora de duração.')
   438→      return
   439→    }
   440→    const conflicts = slots.some(
   441→      (s) => s.weekday === addingDay && startHour < s.endHour && endHour > s.startHour
   442→    )
   443→    if (conflicts) {
   444→      toast.error('Esta faixa conflita com um horário já cadastrado neste dia.')
   445→      return
   446→    }
   447→    setSlots((prev) => [...prev, { weekday: addingDay, startHour, endHour }])
   448→    setAddingDay(null)
   449→  }
   450→
   451→  const handleSave = async () => {
   452→    setSaving(true)
   453→    try {
   454→      const ordered = [...slots].sort((a, b) => a.weekday - b.weekday || a.startHour - b.startHour)
   455→      await onSave(ordered)
   456→    } finally {
   457→      setSaving(false)
   458→    }
   459→  }
   460→
   461→  return (
   462→    <div className="flex flex-col gap-4">
   463→      <div className="flex flex-col divide-y divide-stone-200 rounded-lg border border-stone-200">
   464→        {WEEKDAYS_FULL_PT.map((day, weekday) => {
   465→          const daySlots = slots
   466→            .filter((s) => s.weekday === weekday)
   467→            .sort((a, b) => a.startHour - b.startHour)
   468→          return (
   469→            <div key={day} className="flex flex-col gap-3 p-4 sm:flex-row sm:items-center sm:justify-between">
   470→              <div className="flex min-w-0 flex-1 flex-wrap items-center gap-2">
   471→                <span className="w-28 shrink-0 text-sm font-medium sm:w-32">{day}</span>
   472→                {daySlots.length === 0 ? (
   473→                  <span className="text-xs text-stone-400">Sem disponibilidade</span>
   474→                ) : (
   475→                  <span className="flex flex-wrap items-center gap-1.5">
   476→                    {daySlots.map((s) => (
   477→                      <Badge
   478→                        key={`${s.weekday}-${s.startHour}-${s.endHour}`}
   479→                        variant="outline"
   480→                        className="gap-1 border-emerald-200 bg-emerald-50 py-1 text-emerald-800"
   481→                      >
   482→                        {hourToLabel(s.startHour)} – {hourToLabel(s.endHour)}
   483→                        <button
   484→                          type="button"
   485→                          aria-label={`Remover faixa de ${hourToLabel(s.startHour)} às ${hourToLabel(
   486→                            s.endHour
   487→                          )} na ${day.toLowerCase()}`}
   488→                          onClick={() => removeSlot(s.weekday, s.startHour, s.endHour)}
   489→                          className="rounded-full p-0.5 transition-colors hover:bg-emerald-200/70 focus-visible:outline-2 focus-visible:outline-emerald-600"
   490→                        >
   491→                          <X className="size-3" aria-hidden />
   492→                        </button>
   493→                      </Badge>
   494→                    ))}
   495→                  </span>
   496→                )}
   497→              </div>
   498→
   499→              <div className="shrink-0">
   500→                {addingDay === weekday ? (
   501→                  <div className="flex flex-wrap items-center gap-2">
   502→                    <Select value={newStart} onValueChange={setNewStart}>
   503→                      <SelectTrigger size="sm" className="w-24" aria-label="Início da faixa">
   504→                        <SelectValue />
   505→                      </SelectTrigger>
   506→                      <SelectContent>
   507→                        {TIME_OPTIONS.map((time) => (
   508→                          <SelectItem key={time} value={time}>
   509→                            {time}
   510→                          </SelectItem>
   511→                        ))}
   512→                      </SelectContent>
   513→                    </Select>
   514→                    <span className="text-sm text-muted-foreground" aria-hidden>
   515→                      →
   516→                    </span>
   517→                    <Select value={newEnd} onValueChange={setNewEnd}>
   518→                      <SelectTrigger size="sm" className="w-24" aria-label="Fim da faixa">
   519→                        <SelectValue />
   520→                      </SelectTrigger>
   521→                      <SelectContent>
   522→                        {TIME_OPTIONS.map((time) => (
   523→                          <SelectItem key={time} value={time}>
   524→                            {time}
   525→                          </SelectItem>
   526→                        ))}
   527→                      </SelectContent>
   528→                    </Select>
   529→                    <Button size="sm" onClick={confirmAdd}>
   530→                      <Check className="size-4" aria-hidden /> Confirmar
   531→                    </Button>
   532→                    <Button size="sm" variant="ghost" onClick={() => setAddingDay(null)}>
   533→                      Cancelar
   534→                    </Button>
   535→                  </div>
   536→                ) : (
   537→                  <Button
   538→                    size="sm"
   539→                    variant="ghost"
   540→                    onClick={() => openAdd(weekday)}
   541→                    aria-label={`Adicionar faixa de horário na ${day.toLowerCase()}`}
   542→                  >
   543→                    <Plus className="size-4" aria-hidden /> Adicionar
   544→                  </Button>
   545→                )}
   546→              </div>
   547→            </div>
   548→          )
   549→        })}
   550→      </div>
   551→
   552→      <div className="flex flex-col gap-3 sm:flex-row sm:items-center sm:justify-between">
   553→        <p className="text-xs text-muted-foreground">
   554→          Alunos só podem agendar dentro destas faixas. Use intervalos de pelo menos 1 hora.
   555→        </p>
   556→        <Button onClick={() => void handleSave()} disabled={saving} className="shrink-0">
   557→          <Save className="size-4" aria-hidden /> {saving ? 'Salvando...' : 'Salvar disponibilidade'}
   558→        </Button>
   559→      </div>
   560→    </div>
   561→  )
   562→}
