     1→/**
     2→ * Detalhe do curso (tela mais rica do app):
     3→ * - Não inscrito: hero com capa em gradiente, mentor, rating, preço e botão
     4→ *   "Inscrever-se" (402 = curso pago → Alert com preço e opção de comprar no site).
     5→ * - Inscrito: progresso destacado, card "Aula atual" (descrição, assistir/materiais,
     6→ *   concluir +XP, navegação anterior/próxima) e lista compacta de aulas com
     7→ *   círculo numerado/check — toque na linha seleciona a aula, círculo alterna
     8→ *   a conclusão.
     9→ */
    10→import React, { useCallback, useEffect, useRef, useState } from "react";
    11→import { savePendingCheckout } from "../lib/pendingCheckout";
    12→import {
    13→  ActivityIndicator,
    14→  Alert,
    15→  Linking,
    16→  RefreshControl,
    17→  ScrollView,
    18→  StyleSheet,
    19→  Text,
    20→  TouchableOpacity,
    21→  View,
    22→} from "react-native";
    23→import { useNavigation, useRoute } from "@react-navigation/native";
    24→import * as WebBrowser from "expo-web-browser";
    25→import { Ionicons } from "@expo/vector-icons";
    26→import { LinearGradient } from "expo-linear-gradient";
    27→import {
    28→  ApiError,
    29→  siteUrl,
    30→  enrollCourse,
    31→  errMessage,
    32→  getCourse,
    33→  toggleLessonCompletion,
    34→  type CourseDetailResponse,
    35→  type Lesson,
    36→  type LessonKind,
    37→} from "../lib/api";
    38→import {
    39→  formatDuration,
    40→  formatNaiveLong,
    41→  formatNumber,
    42→  formatPrice,
    43→  levelLabel,
    44→} from "../lib/format";
    45→import { theme } from "../theme";
    46→import { Avatar } from "../components/Avatar";
    47→import { Chip } from "../components/Chip";
    48→import { EmptyState } from "../components/EmptyState";
    49→import { ErrorBox } from "../components/ErrorBox";
    50→import { LoadingList } from "../components/LoadingList";
    51→import { ProgressBar } from "../components/ProgressBar";
    52→import { RemoteImage } from "../components/RemoteImage";
    53→import { RichText } from "../components/RichText";
    54→import { Screen } from "../components/Screen";
    55→import { ScreenHeader } from "../components/ScreenHeader";
    56→import { Stars } from "../components/Stars";
    57→
    58→const KIND_META: Record<LessonKind, { label: string; icon: keyof typeof Ionicons.glyphMap }> = {
    59→  RECORDED: { label: "Vídeo", icon: "play-circle-outline" },
    60→  TEXT: { label: "Leitura", icon: "document-text-outline" },
    61→  LIVE: { label: "Ao vivo", icon: "videocam-outline" },
    62→};
    63→
    64→export default function CourseDetailScreen() {
    65→  const styles = makeStyles();
    66→  const navigation = useNavigation<any>();
    67→  const params = (useRoute<any>().params ?? {}) as { id: string };
    68→  const courseId = Array.isArray(params.id) ? params.id[0] : params.id;
    69→
    70→  const [detail, setDetail] = useState<CourseDetailResponse | null>(null);
    71→  const [loading, setLoading] = useState(true);
    72→  const [refreshing, setRefreshing] = useState(false);
    73→  const [error, setError] = useState<string | null>(null);
    74→  const [enrolling, setEnrolling] = useState(false);
    75→  const [togglingIds, setTogglingIds] = useState<string[]>([]);
    76→  const [expandedLessonId, setExpandedLessonId] = useState<string | null>(null);
    77→  // Aula em destaque no card "Aula atual" (selecionada pelo usuário ou a próxima pendente).
    78→  const [selectedLessonId, setSelectedLessonId] = useState<string | null>(null);
    79→  const [xpToast, setXpToast] = useState<string | null>(null);
    80→  const xpTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
    81→
    82→  const load = useCallback(
    83→    async (mode: "initial" | "refresh") => {
    84→      if (!courseId) return;
    85→      if (mode === "initial") setLoading(true);
    86→      else setRefreshing(true);
    87→      setError(null);
    88→      try {
    89→        const res = await getCourse(courseId);
    90→        setDetail(res);
    91→      } catch (err) {
    92→        setError(errMessage(err));
    93→      } finally {
    94→        setLoading(false);
    95→        setRefreshing(false);
    96→      }
    97→    },
    98→    [courseId]
    99→  );
   100→
   101→  useEffect(() => {
   102→    void load("initial");
   103→  }, [load]);
   104→
   105→  // Limpa o timer do toast de XP ao desmontar.
   106→  useEffect(() => {
   107→    return () => {
   108→      if (xpTimerRef.current) clearTimeout(xpTimerRef.current);
   109→    };
   110→  }, []);
   111→
   112→  function showXpToast(text: string) {
   113→    setXpToast(text);
   114→    if (xpTimerRef.current) clearTimeout(xpTimerRef.current);
   115→    xpTimerRef.current = setTimeout(() => setXpToast(null), 2400);
   116→  }
   117→
   118→  async function openInBrowser(url: string) {
   119→    try {
   120→      await WebBrowser.openBrowserAsync(url);
   121→    } catch {
   122→      Alert.alert("Não foi possível abrir", "Verifique sua conexão e tente novamente.");
   123→    }
   124→  }
   125→
   126→  
   127→async function openExternal(url: string) {
   128→    try {
   129→      await Linking.openURL(url);
   130→    } catch {
   131→      Alert.alert("Não foi possível abrir o link", url);
   132→    }
   133→  }
   134→
   135→  async function handleEnroll() {
   136→    if (!detail || enrolling) return;
   137→    setEnrolling(true);
   138→    try {
   139→      await enrollCourse(detail.course.id);
   140→      Alert.alert("Inscrição confirmada!", "Agora você tem acesso a todas as aulas deste curso.");
   141→      await load("refresh");
   142→    } catch (err) {
   143→      if (err instanceof ApiError && err.status === 402) {
   144→        // Curso pago: checkout no site seguro. Guarda a intenção de compra —
   145→        // se a sessão cair antes de concluir, o app retoma direto neste curso
   146→        // após o novo login.
   147→        const price =
   148→          typeof err.payload?.price === "number" ? err.payload.price : detail.course.price;
   149→        await savePendingCheckout(detail.course.id);
   150→        Alert.alert(
   151→          "Finalizar compra do curso",
   152→          `${err.message}${price > 0 ? `\n\nValor do curso: ${formatPrice(price)}` : ""}\n\nO pagamento é processado no site seguro do MentorHub — você volta pro app depois de concluir.`,
   153→          [
   154→            { text: "Depois", style: "cancel" },
   155→            {
   156→              text: "Continuar para o pagamento",
   157→              onPress: () =>
   158→                void openExternal(`${siteUrl()}/cursos/${detail.course.id}?checkout=1&from=app`),
   159→            },
   160→          ]
   161→        );
   162→      } else {
   163→        Alert.alert("Não foi possível se inscrever", errMessage(err));
   164→      }
   165→    } finally {
   166→      setEnrolling(false);
   167→    }
   168→  }
   169→
   170→  async function handleToggleLesson(lesson: Lesson) {
   171→    if (!detail || togglingIds.includes(lesson.id)) return;
   172→    setTogglingIds((prev) => [...prev, lesson.id]);
   173→    try {
   174→      const res = await toggleLessonCompletion(detail.course.id, lesson.id);
   175→      setDetail((prev) =>
   176→        prev
   177→          ? {
   178→              ...prev,
   179→              enrollment: {
   180→                completedLessonIds: res.completedLessonIds,
   181→                completedAt: prev.enrollment?.completedAt ?? null,
   182→              },
   183→            }
   184→          : prev
   185→      );
   186→      if (res.courseCompleted) {
   187→        Alert.alert(
   188→          "Curso concluído!",
   189→          `Parabéns! Você finalizou "${detail.course.title}".` +
   190→            (res.xpAwarded > 0 ? ` +${res.xpAwarded} XP de recompensa.` : "")
   191→        );
   192→      } else if (res.xpAwarded > 0) {
   193→        showXpToast(`Aula concluída! +${res.xpAwarded} XP`);
   194→      }
   195→    } catch (err) {
   196→      Alert.alert("Não foi possível atualizar", errMessage(err));
   197→    } finally {
   198→      setTogglingIds((prev) => prev.filter((id) => id !== lesson.id));
   199→    }
   200→  }
   201→
   202→  /* ----------------------------- Derivados ----------------------------- */
   203→
   204→  const course = detail?.course ?? null;
   205→  const enrollment = detail?.enrollment ?? null;
   206→  const enrolled = enrollment !== null;
   207→  const completedIds = enrollment?.completedLessonIds ?? [];
   208→  const allLessons = detail
   209→    ? [...detail.lessons, ...detail.themes.flatMap((courseTheme) => courseTheme.lessons)]
   210→    : [];
   211→  const totalLessons = detail
   212→    ? detail.course.lessonCount > 0
   213→      ? detail.course.lessonCount
   214→      : allLessons.length
   215→    : 0;
   216→  const pct = totalLessons > 0 ? Math.round((completedIds.length / totalLessons) * 100) : 0;
   217→
   218→  /* Aula atual: selecionada > primeira não concluída > primeira do curso. */
   219→  const currentLesson =
   220→    allLessons.find((lesson) => lesson.id === selectedLessonId) ??
   221→    allLessons.find((lesson) => !lesson.locked && !completedIds.includes(lesson.id)) ??
   222→    allLessons[0] ??
   223→    null;
   224→  const currentIndex = currentLesson
   225→    ? allLessons.findIndex((lesson) => lesson.id === currentLesson.id)
   226→    : -1;
   227→
   228→  function goToLesson(offset: number) {
   229→    const next = allLessons[currentIndex + offset];
   230→    if (next) setSelectedLessonId(next.id);
   231→  }
   232→
   233→  const sections: Array<{ key: string; title: string; description: string | null; lessons: Lesson[] }> = [];
   234→  if (detail) {
   235→    if (detail.lessons.length > 0) {
   236→      sections.push({
   237→        key: "no-theme",
   238→        title: detail.themes.length > 0 ? "Aulas iniciais" : "Aulas",
   239→        description: null,
   240→        lessons: detail.lessons,
   241→      });
   242→    }
   243→    [...detail.themes]
   244→      .sort((a, b) => a.order - b.order)
   245→      .forEach((courseTheme) => {
   246→        if (courseTheme.lessons.length > 0) {
   247→          sections.push({
   248→            key: courseTheme.id,
   249→            title: courseTheme.title,
   250→            description: courseTheme.description,
   251→            lessons: courseTheme.lessons,
   252→          });
   253→        }
   254→      });
   255→  }
   256→
   257→  return (
   258→    <Screen edges={["top", "left", "right", "bottom"]}>
   259→      {loading ? (
   260→        <LoadingList label="Carregando curso..." />
   261→      ) : error && !detail ? (
   262→        <ErrorBox message={error} onRetry={() => void load("initial")} />
   263→      ) : detail && course ? (
   264→        <View style={styles.flex}>
   265→          <ScreenHeader title="Curso" onBack={() => navigation.goBack()} />
   266→          <ScrollView
   267→            style={styles.flex}
   268→            contentContainerStyle={styles.content}
   269→            showsVerticalScrollIndicator={false}
   270→            refreshControl={
   271→              <RefreshControl
   272→                refreshing={refreshing}
   273→                onRefresh={() => void load("refresh")}
   274→                tintColor={theme.colors.accent}
   275→                colors={[theme.colors.accent]}
   276→                progressBackgroundColor={theme.colors.surface}
   277→              />
   278→            }
   279→          >
   280→            {/* Hero: capa com gradiente e título sobreposto */}
   281→            <View style={styles.hero}>
   282→              <RemoteImage
   283→                uri={course.coverUrl}
   284→                style={styles.heroCover}
   285→                recyclingKey={course.id}
   286→                fallbackIcon="play-circle-outline"
   287→                errorIcon="image-outline"
   288→                iconSize={38}
   289→              />
   290→              <LinearGradient
   291→                colors={["transparent", theme.colors.overlay]}
   292→                start={{ x: 0, y: 0 }}
   293→                end={{ x: 0, y: 1 }}
   294→                style={styles.heroGradient}
   295→              >
   296→                <Text style={styles.heroTitle} numberOfLines={3}>
   297→                  {course.title}
   298→                </Text>
   299→              </LinearGradient>
   300→            </View>
   301→
   302→            {/* Mentor + rating */}
   303→            <TouchableOpacity
   304→              style={styles.mentorRow}
   305→              onPress={() => navigation.navigate("Mentor", { id: course.mentor.id })}
   306→              activeOpacity={0.85}
   307→            >
   308→              <Avatar uri={course.mentor.avatarUrl} name={course.mentor.name} size={38} />
   309→              <View style={styles.mentorInfo}>
   310→                <Text style={styles.mentorName} numberOfLines={1}>
   311→                  Por {course.mentor.name}
   312→                </Text>
   313→                <View style={styles.ratingRow}>
   314→                  <Stars value={course.rating} size={12} />
   315→                  <Text style={styles.ratingText}>
   316→                    {(course.rating ?? 0).toFixed(1)} ({course.reviewCount})
   317→                  </Text>
   318→                </View>
   319→              </View>
   320→              <Ionicons name="chevron-forward" size={16} color={theme.colors.textFaint} />
   321→            </TouchableOpacity>
   322→
   323→            {/* Metadados */}
   324→            <View style={styles.chipsRow}>
   325→              <Chip label={course.category} tone="accent" />
   326→              {levelLabel(course.level) ? <Chip label={levelLabel(course.level)} tone="outline" /> : null}
   327→              {course.liveCount > 0 ? <Chip label={`${course.liveCount} ao vivo`} /> : null}
   328→            </View>
   329→            <Text style={styles.metaLine}>
   330→              {course.lessonCount} aulas
   331→              {course.totalDurationMin > 0 ? ` · ${formatDuration(course.totalDurationMin)}` : ""}
   332→              {course.studentCount > 0 ? ` · ${formatNumber(course.studentCount)} alunos` : ""}
   333→            </Text>
   334→
   335→            {/* Preço */}
   336→            <View style={styles.priceRow}>
   337→              <Text style={[styles.price, course.price > 0 ? null : styles.priceFree]}>
   338→                {formatPrice(course.price)}
   339→              </Text>
   340→              {enrolled ? <Chip label="Inscrito" tone="accent" /> : null}
   341→            </View>
   342→
   343→            {course.description ? <Text style={styles.description}>{course.description}</Text> : null}
   344→
   345→            {/* Progresso (inscrito) ou aviso de bloqueio */}
   346→            {enrolled ? (
   347→              <View style={styles.progressCard}>
   348→                <View style={styles.progressHeader}>
   349→                  <Text style={styles.progressTitle}>Seu progresso</Text>
   350→                  <View style={styles.progressPill}>
   351→                    <Text style={styles.progressPct}>{pct}%</Text>
   352→                  </View>
   353→                </View>
   354→                <ProgressBar pct={pct} height={10} />
   355→                <Text style={styles.progressMeta}>
   356→                  {completedIds.length} de {totalLessons} aulas concluídas
   357→                </Text>
   358→                {pct >= 100 ? (
   359→                  <Text style={styles.progressDone}>Curso concluído — parabéns!</Text>
   360→                ) : null}
   361→              </View>
   362→            ) : (
   363→              <View style={styles.lockedBanner}>
   364→                <Ionicons name="lock-closed" size={14} color={theme.colors.warning} />
   365→                <Text style={styles.lockedBannerText}>
   366→                  Inscreva-se para desbloquear todas as aulas deste curso.
   367→                </Text>
   368→              </View>
   369→            )}
   370→
   371→            {/* Card "Aula atual" em destaque (inscrito) */}
   372→            {enrolled && currentLesson ? (
   373→              <CurrentLessonCard
   374→                lesson={currentLesson}
   375→                position={currentIndex + 1}
   376→                total={allLessons.length}
   377→                completed={completedIds.includes(currentLesson.id)}
   378→                toggling={togglingIds.includes(currentLesson.id)}
   379→                expanded={expandedLessonId === currentLesson.id}
   380→                canPrev={currentIndex > 0}
   381→                canNext={currentIndex < allLessons.length - 1}
   382→                onPrev={() => goToLesson(-1)}
   383→                onNext={() => goToLesson(1)}
   384→                onToggleComplete={() => void handleToggleLesson(currentLesson)}
   385→                onToggleExpand={() =>
   386→                  setExpandedLessonId((prev) => (prev === currentLesson.id ? null : currentLesson.id))
   387→                }
   388→                onOpen={(url) => void openInBrowser(url)}
   389→              />
   390→            ) : null}
   391→
   392→            {/* Lista compacta de aulas por tema (aulas sem tema ficam no topo) */}
   393→            {sections.length === 0 && enrolled ? (
   394→              <EmptyState
   395→                icon="library-outline"
   396→                title="Nenhuma aula publicada"
   397→                message="As aulas deste curso aparecerão aqui."
   398→              />
   399→            ) : (
   400→              sections.map((section) => (
   401→                <View key={section.key} style={styles.section}>
   402→                  <Text style={styles.sectionTitle}>{section.title}</Text>
   403→                  {section.description ? (
   404→                    <Text style={styles.sectionDescription}>{section.description}</Text>
   405→                  ) : null}
   406→                  <View style={styles.lessonsList}>
   407→                    {section.lessons.map((lesson, index) => (
   408→                      <LessonRow
   409→                        key={lesson.id}
   410→                        lesson={lesson}
   411→                        number={index + 1}
   412→                        completed={completedIds.includes(lesson.id)}
   413→                        locked={lesson.locked}
   414→                        toggling={togglingIds.includes(lesson.id)}
   415→                        active={currentLesson?.id === lesson.id}
   416→                        onSelect={() => setSelectedLessonId(lesson.id)}
   417→                        onToggleComplete={() => void handleToggleLesson(lesson)}
   418→                      />
   419→                    ))}
   420→                  </View>
   421→                </View>
   422→              ))
   423→            )}
   424→
   425→            <View style={styles.bottomSpacer} />
   426→          </ScrollView>
   427→
   428→          {/* CTA de inscrição */}
   429→          {!enrolled ? (
   430→            <View style={styles.footer}>
   431→              <TouchableOpacity
   432→                style={[styles.cta, enrolling && styles.ctaDisabled]}
   433→                onPress={() => void handleEnroll()}
   434→                disabled={enrolling}
   435→                activeOpacity={0.85}
   436→                accessibilityRole="button"
   437→                accessibilityLabel="Inscrever-se no curso"
   438→              >
   439→                {enrolling ? (
   440→                  <ActivityIndicator size="small" color={theme.colors.onAccent} />
   441→                ) : (
   442→                  <Text style={styles.ctaText}>
   443→                    {course.price > 0
   444→                      ? `Inscrever-se · ${formatPrice(course.price)}`
   445→                      : "Inscrever-se gratuitamente"}
   446→                  </Text>
   447→                )}
   448→              </TouchableOpacity>
   449→            </View>
   450→          ) : null}
   451→
   452→          {/* Toast de XP */}
   453→          {xpToast ? (
   454→            <View pointerEvents="none" style={styles.xpToast}>
   455→              <Ionicons name="flash" size={13} color={theme.colors.onAccent} />
   456→              <Text style={styles.xpToastText}>{xpToast}</Text>
   457→            </View>
   458→          ) : null}
   459→        </View>
   460→      ) : null}
   461→    </Screen>
   462→  );
   463→}

   466→
   467→/** Card "Aula atual": descrição, ações, concluir (+XP) e navegação anterior/próxima. */
   468→function CurrentLessonCard({
   469→  lesson,
   470→  position,
   471→  total,
   472→  completed,
   473→  toggling,
   474→  expanded,
   475→  canPrev,
   476→  canNext,
   477→  onPrev,
   478→  onNext,
   479→  onToggleComplete,
   480→  onToggleExpand,
   481→  onOpen,
   482→}: {
   483→  lesson: Lesson;
   484→  position: number;
   485→  total: number;
   486→  completed: boolean;
   487→  toggling: boolean;
   488→  expanded: boolean;
   489→  canPrev: boolean;
   490→  canNext: boolean;
   491→  onPrev: () => void;
   492→  onNext: () => void;
   493→  onToggleComplete: () => void;
   494→  onToggleExpand: () => void;
   495→  onOpen: (url: string) => void;
   496→}) {
   497→  const styles = makeStyles();
   498→  const kindMeta = KIND_META[lesson.kind];
   499→  const videoUrl = !lesson.locked && lesson.kind === "RECORDED" ? lesson.videoUrl : null;
   500→  const meetingUrl = !lesson.locked && lesson.kind === "LIVE" ? lesson.meetingUrl : null;
   501→  const textContent = !lesson.locked && lesson.kind === "TEXT" ? lesson.content : null;
   502→  const liveWhen =
   503→    !lesson.locked && lesson.kind === "LIVE" && lesson.startsAt ? formatNaiveLong(lesson.startsAt) : null;
   504→  const attachments = lesson.locked ? [] : lesson.attachments ?? [];
   505→
   506→  return (
   507→    <View style={styles.currentCard}>
   508→      <View style={styles.currentHeader}>
   509→        <Text style={styles.currentOverline}>
   510→          Aula atual · {position} de {total}
   511→        </Text>
   512→        <Chip label={kindMeta.label} tone="accent" />
   513→      </View>
   514→      <Text style={styles.currentTitle}>{lesson.title}</Text>
   515→      {lesson.description ? (
   516→        <Text style={styles.currentDescription} numberOfLines={4}>
   517→          {lesson.description}
   518→        </Text>
   519→      ) : null}
   520→      <View style={styles.currentMetaRow}>
   521→        {lesson.durationMin > 0 ? (
   522→          <>
   523→            <Ionicons name="time-outline" size={12} color={theme.colors.textFaint} />
   524→            <Text style={styles.currentMeta}>{formatDuration(lesson.durationMin)}</Text>
   525→          </>
   526→        ) : null}
   527→        {liveWhen ? <Text style={styles.currentLiveWhen}>Encontro: {liveWhen}</Text> : null}
   528→      </View>
   529→
   530→      {/* Ações da aula: assistir / sala / conteúdo / materiais */}
   531→      {!lesson.locked ? (
   532→        <View style={styles.lessonActions}>
   533→          {videoUrl ? (
   534→            <ActionButton icon="play" label="Assistir aula" onPress={() => onOpen(videoUrl)} />
   535→          ) : null}
   536→          {meetingUrl ? (
   537→            <ActionButton
   538→              icon="videocam"
   539→              label="Abrir sala de transmissão"
   540→              onPress={() => onOpen(meetingUrl)}
   541→            />
   542→          ) : null}
   543→          {textContent ? (
   544→            <>
   545→              <ActionButton
   546→                icon={expanded ? "eye-off-outline" : "eye-outline"}
   547→                label={expanded ? "Ocultar conteúdo" : "Ver conteúdo da aula"}
   548→                onPress={onToggleExpand}
   549→              />
   550→              {expanded ? (
   551→                <View style={styles.lessonContent}>
   552→                  <RichText text={textContent} />
   553→                </View>
   554→              ) : null}
   555→            </>

   647→/** Linha compacta de aula: número/check, título e duração. Toque seleciona a aula atual. */
   648→function LessonRow({
   649→  lesson,
   650→  number,

   652→  locked,
   653→  toggling,
   654→  active,

   687→            styles.stepCircle,
   688→            completed ? styles.stepCircleDone : null,
   689→            active && !completed ? styles.stepCircleActive : null,
   690→          ]}
   691→          onPress={onToggleComplete}
   692→          disabled={toggling}
   693→          activeOpacity={0.8}
   694→          accessibilityRole="checkbox"
   695→          accessibilityState={{ checked: completed }}
   696→          accessibilityLabel={
   697→            completed
   698→              ? `Desmarcar aula ${lesson.title} como concluída`
   699→              : `Marcar aula ${lesson.title} como concluída`
   700→          }
   701→        >
   702→          {toggling ? (
   703→            <ActivityIndicator size="small" color={completed ? theme.colors.onAccent : theme.colors.accent} />
   704→          ) : completed ? (
   705→            <Ionicons name="checkmark" size={15} color={theme.colors.onAccent} />
   706→          ) : (
   707→            <Text style={styles.stepNumber}>{number}</Text>
   708→          )}
   709→        </TouchableOpacity>
   710→      )}
   711→      <Text
   712→        style={[
   713→          styles.lessonRowTitle,
   714→          completed ? styles.lessonRowTitleDone : null,
   715→          locked ? styles.lessonRowTitleLocked : null,
   716→          active ? styles.lessonRowTitleActive : null,
   717→        ]}
   718→        numberOfLines={2}
   719→      >
   720→        {lesson.title}
   721→      </Text>
   722→      <Text style={styles.lessonRowEnd}>{endLabel}</Text>
   723→    </TouchableOpacity>
   724→  );
   725→}
