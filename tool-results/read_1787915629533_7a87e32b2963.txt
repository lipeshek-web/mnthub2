     1→'use client'
     2→
     3→import { useCallback, useEffect, useMemo, useRef, useState } from 'react'
     4→import { AnimatePresence, motion } from 'framer-motion'
     5→import {
     6→  ArrowRight,
     7→  BadgeCheck,
     8→  BookOpen,
     9→  Clock,
    10→  Globe2,

    16→  Star,

    20→} from 'lucide-react'
    21→import { Badge } from '@/components/ui/badge'
    22→import { Button } from '@/components/ui/button'
    23→import { Input } from '@/components/ui/input'
    24→import {
    25→  Select,
    26→  SelectContent,
    27→  SelectItem,
    28→  SelectTrigger,
    29→  SelectValue,
    30→} from '@/components/ui/select'
    31→import { Skeleton } from '@/components/ui/skeleton'
    32→import { Avatar, Stars } from '@/components/platform/avatar'
    33→import { api } from '@/lib/api'
    34→import {
    35→  CATEGORIES,
    36→  LEVEL_LABELS,
    37→  avatarGradient,
    38→  currencyBRL,
    39→  firstName,
    40→  formatTotalDuration,
    41→} from '@/lib/helpers'
    42→import { useAppStore, type ExploreTab } from '@/lib/store'
    43→import type { CourseListItemDTO, MentorListItemDTO, TrackListItemDTO } from '@/lib/types'
    44→import { cn } from '@/lib/utils'
    45→
    46→const SPOTLIGHT_INTERVAL_MS = 6000
    47→
    48→export function MarketplaceView() {
    49→  const navigate = useAppStore((s) => s.navigate)
    50→
    51→  // Base completa (sem filtros): alimenta destaque rotativo, contagens e stats
    52→  const [baseMentors, setBaseMentors] = useState<MentorListItemDTO[]>([])
    53→  // Lista filtrada exibida no grid
    54→  const [mentors, setMentors] = useState<MentorListItemDTO[]>([])
    55→  const [loading, setLoading] = useState(true)
    56→  const [search, setSearch] = useState('')
    57→  const [inputValue, setInputValue] = useState('')
    58→  const [category, setCategory] = useState('')
    59→  const [sort, setSort] = useState('relevance')
    60→  const debounceRef = useRef<ReturnType<typeof setTimeout> | null>(null)
    61→  const searchRef = useRef<HTMLInputElement>(null)
    62→
    63→  // Aba do Explorar (mentores/cursos): valor inicial lido da store uma única vez
    64→  const [tab, setTab] = useState<ExploreTab>(() => useAppStore.getState().exploreTab)
    65→  // Cursos: base completa (contagens/destaque/stats) e lista filtrada
    66→  const [baseCourses, setBaseCourses] = useState<CourseListItemDTO[]>([])
    67→  const [courses, setCourses] = useState<CourseListItemDTO[]>([])
    68→  const [coursesLoading, setCoursesLoading] = useState(false)
    69→  const [courseSort, setCourseSort] = useState('relevance')
    70→  // Trilhas: base completa (contagens/destaque/stats) e lista filtrada
    71→  const [baseTracks, setBaseTracks] = useState<TrackListItemDTO[]>([])
    72→  const [tracks, setTracks] = useState<TrackListItemDTO[]>([])
    73→  const [tracksLoading, setTracksLoading] = useState(false)
    74→  const [trackSort, setTrackSort] = useState('relevance')
    75→
    76→  // Consome termo vindo de outra tela (hero da home) uma única vez
    77→  useEffect(() => {
    78→    const q = useAppStore.getState().exploreQuery
    79→    if (q) {
    80→      setInputValue(q)
    81→      setSearch(q)
    82→      useAppStore.setState({ exploreQuery: '' })
    83→    }
    84→  }, [])
    85→
    86→  // Consome a aba pedida por outra tela ("Ver todos os cursos" da home) uma única vez.
    87→  // O valor inicial já foi lido no useState acima; aqui apenas resetamos a store.
    88→  useEffect(() => {
    89→    useAppStore.setState({ exploreTab: 'mentors' })
    90→  }, [])
    91→
    92→  const load = useCallback(async () => {
    93→    setLoading(true)
    94→    try {
    95→      const data = await api.listMentors({ search, category, sort })
    96→      setMentors(data)
    97→    } catch {
    98→      setMentors([])
    99→    } finally {
   100→      setLoading(false)
   101→    }
   102→  }, [search, category, sort])
   103→
   104→  useEffect(() => {
   105→    load()
   106→  }, [load])
   107→
   108→  useEffect(() => {
   109→    api
   110→      .listMentors({})
   111→      .then(setBaseMentors)
   112→      .catch(() => {})
   113→  }, [])
   114→
   115→  // Base de cursos: alimenta destaque, contagens e stats (busca única)
   116→  useEffect(() => {
   117→    api
   118→      .listCourses({})
   119→      .then(setBaseCourses)
   120→      .catch(() => {})
   121→  }, [])
   122→
   123→  // Base de trilhas: alimenta destaque, contagens e stats (busca única)
   124→  useEffect(() => {
   125→    api
   126→      .listTracks({})
   127→      .then(setBaseTracks)
   128→      .catch(() => {})
   129→  }, [])
   130→
   131→  // Lista filtrada de cursos — só carrega enquanto a aba Cursos está ativa
   132→  const loadCourses = useCallback(async () => {
   133→    setCoursesLoading(true)
   134→    try {
   135→      const data = await api.listCourses({ search, category, sort: courseSort })
   136→      setCourses(data)
   137→    } catch {
   138→      setCourses([])
   139→    } finally {
   140→      setCoursesLoading(false)
   141→    }
   142→  }, [search, category, courseSort])
   143→
   144→  useEffect(() => {
   145→    if (tab === 'courses') void loadCourses()
   146→  }, [tab, loadCourses])
   147→
   148→  // Lista filtrada de trilhas — só carrega enquanto a aba Trilhas está ativa
   149→  const loadTracks = useCallback(async () => {
   150→    setTracksLoading(true)
   151→    try {
   152→      const data = await api.listTracks({ search, category, sort: trackSort })
   153→      setTracks(data)
   154→    } catch {
   155→      setTracks([])
   156→    } finally {
   157→      setTracksLoading(false)
   158→    }
   159→  }, [search, category, trackSort])
   160→
   161→  useEffect(() => {
   162→    if (tab === 'tracks') void loadTracks()
   163→  }, [tab, loadTracks])
   164→
   165→  // Atalho "/" foca a busca (sensação de app nativo)
   166→  useEffect(() => {
   167→    const onKey = (e: KeyboardEvent) => {
   168→      const target = e.target as HTMLElement | null
   169→      const typing =
   170→        target instanceof HTMLInputElement ||
   171→        target instanceof HTMLTextAreaElement ||
   172→        target?.isContentEditable
   173→      if (e.key === '/' && !typing) {
   174→        e.preventDefault()
   175→        searchRef.current?.focus()
   176→      }
   177→    }
   178→    window.addEventListener('keydown', onKey)
   179→    return () => window.removeEventListener('keydown', onKey)
   180→  }, [])
   181→
   182→  const onSearchChange = (value: string) => {
   183→    setInputValue(value)
   184→    if (debounceRef.current) clearTimeout(debounceRef.current)
   185→    debounceRef.current = setTimeout(() => setSearch(value), 300)
   186→  }
   187→
   188→  const clearSearch = () => {
   189→    setInputValue('')
   190→    setSearch('')
   191→    if (debounceRef.current) clearTimeout(debounceRef.current)
   192→    searchRef.current?.focus()
   193→  }
   194→
   195→  // ---------- Derivados da base ----------
   196→
   197→  const spotlightPool = useMemo(() => {
   198→    const rated = baseMentors
   199→      .filter((m) => m.rating > 0)
   200→      .sort((a, b) => b.rating - a.rating || b.reviewCount - a.reviewCount)
   201→    if (rated.length > 0) return rated.slice(0, 4)
   202→    return [...baseMentors].sort((a, b) => b.totalSessions - a.totalSessions).slice(0, 3)
   203→  }, [baseMentors])
   204→
   205→  const [spotIdx, setSpotIdx] = useState(0)
   206→  useEffect(() => {
   207→    if (spotlightPool.length < 2) return
   208→    const t = setInterval(
   209→      () => setSpotIdx((i) => (i + 1) % spotlightPool.length),
   210→      SPOTLIGHT_INTERVAL_MS
   211→    )
   212→    return () => clearInterval(t)
   213→  }, [spotlightPool.length])
   214→
   215→  const spot = spotlightPool.length > 0 ? spotlightPool[spotIdx % spotlightPool.length] : null
   216→
   217→  const stats = useMemo(() => {
   218→    const rated = baseMentors.filter((m) => m.rating > 0)
   219→    return {
   220→      mentors: baseMentors.length,
   221→      sessions: baseMentors.reduce((acc, m) => acc + m.totalSessions, 0),
   222→      avg: rated.length > 0 ? rated.reduce((acc, m) => acc + m.rating, 0) / rated.length : null,
   223→    }
   224→  }, [baseMentors])
   225→
   226→  const categoryCounts = useMemo(() => {
   227→    const counts = new Map<string, number>()
   228→    for (const m of baseMentors) {
   229→      for (const c of m.categories) counts.set(c, (counts.get(c) ?? 0) + 1)
   230→    }
   231→    return counts
   232→  }, [baseMentors])
   233→
   234→  const baseReady = baseMentors.length > 0
   235→
   236→  // ---------- Derivados dos cursos ----------
   237→
   238→  const courseCounts = useMemo(() => {
   239→    const counts = new Map<string, number>()
   240→    for (const c of baseCourses) counts.set(c.category, (counts.get(c.category) ?? 0) + 1)
   241→    return counts
   242→  }, [baseCourses])
   243→
   244→  const baseCoursesReady = baseCourses.length > 0
   245→
   246→  const courseStats = useMemo(
   247→    () => ({
   248→      published: baseCourses.length,
   249→      students: baseCourses.reduce((acc, c) => acc + c.studentCount, 0),
   250→      lessons: baseCourses.reduce((acc, c) => acc + c.lessonCount, 0),
   251→    }),
   252→    [baseCourses]
   253→  )
   254→
   255→  const topCourse = useMemo(() => {
   256→    if (baseCourses.length === 0) return null
   257→    return [...baseCourses].sort(
   258→      (a, b) => b.studentCount - a.studentCount || b.mentor.rating - a.mentor.rating
   259→    )[0]
   260→  }, [baseCourses])
   261→
   262→  // ---------- Derivados das trilhas ----------
   263→
   264→  const trackCounts = useMemo(() => {
   265→    const counts = new Map<string, number>()
   266→    for (const t of baseTracks) counts.set(t.category, (counts.get(t.category) ?? 0) + 1)
   267→    return counts
   268→  }, [baseTracks])
   269→
   270→  const baseTracksReady = baseTracks.length > 0
   271→
   272→  const trackStats = useMemo(
   273→    () => ({
   274→      published: baseTracks.length,
   275→      students: baseTracks.reduce((acc, t) => acc + t.studentCount, 0),
   276→      mentorships: baseTracks.reduce((acc, t) => acc + t.mentorshipSessions, 0),
   277→    }),
   278→    [baseTracks]
   279→  )
   280→
   281→  const topTrack = useMemo(() => {
   282→    if (baseTracks.length === 0) return null
   283→    return [...baseTracks].sort(
   284→      (a, b) => b.studentCount - a.studentCount || b.mentor.rating - a.mentor.rating
   285→    )[0]
   286→  }, [baseTracks])
   287→
   288→  return (
   289→    <div>
   290→      {/* ---------- BARRA SUPERIOR: título, ordenação, busca e categorias ---------- */}
   291→      <section className="border-b border-stone-200/70 bg-white">
   292→        <div className="mx-auto max-w-6xl px-4 py-7 sm:py-9">
   293→          {/* Controle segmentado: Mentores | Cursos | Trilhas */}
   294→          <div
   295→            role="tablist"
   296→            aria-label="Explorar mentores, cursos ou trilhas"
   297→            className="inline-flex max-w-full flex-wrap rounded-full bg-stone-100 p-1"
   298→          >
   299→            <button
   300→              role="tab"
   301→              aria-selected={tab === 'mentors'}
   302→              onClick={() => setTab('mentors')}
   303→              className={cn(
   304→                'inline-flex h-9 items-center gap-1.5 rounded-full px-4 text-sm font-semibold transition-all',
   305→                tab === 'mentors'
   306→                  ? 'bg-white text-stone-900 shadow-sm'
   307→                  : 'text-stone-500 hover:text-stone-700'
   308→              )}
   309→            >
   310→              <Users aria-hidden className="h-4 w-4" /> Mentores
   311→            </button>
   312→            <button
   313→              role="tab"
   314→              aria-selected={tab === 'courses'}
   315→              onClick={() => setTab('courses')}
   316→              className={cn(
   317→                'inline-flex h-9 items-center gap-1.5 rounded-full px-4 text-sm font-semibold transition-all',
   318→                tab === 'courses'
   319→                  ? 'bg-white text-stone-900 shadow-sm'
   320→                  : 'text-stone-500 hover:text-stone-700'
   321→              )}
   322→            >
   323→              <BookOpen aria-hidden className="h-4 w-4" /> Cursos
   324→            </button>
   325→            <button
   326→              role="tab"
   327→              aria-selected={tab === 'tracks'}
   328→              onClick={() => setTab('tracks')}
   329→              className={cn(
   330→                'inline-flex h-9 items-center gap-1.5 rounded-full px-4 text-sm font-semibold transition-all',
   331→                tab === 'tracks'
   332→                  ? 'bg-white text-stone-900 shadow-sm'
   333→                  : 'text-stone-500 hover:text-stone-700'
   334→              )}
   335→            >
   336→              <Route aria-hidden className="h-4 w-4" /> Trilhas
   337→            </button>
   338→          </div>
   339→
   340→          <div className="mt-5 flex flex-wrap items-end justify-between gap-3">
   341→            <div>
   342→              {tab === 'mentors' ? (
   343→                <>
   344→                  <h1 className="text-2xl font-extrabold tracking-tight text-stone-900 sm:text-3xl">
   345→                    Explorar mentores
   346→                  </h1>
   347→                  <p className="mt-1 text-sm text-stone-500">
   348→                    {loading
   349→                      ? 'Carregando especialistas...'
   350→                      : `${mentors.length} ${mentors.length === 1 ? 'especialista' : 'especialistas'} pronto${
   351→                          mentors.length === 1 ? '' : 's'
   352→                        } para mentoria 1:1`}
   353→                  </p>
   354→                </>
   355→              ) : tab === 'courses' ? (
   356→                <>
   357→                  <h1 className="text-2xl font-extrabold tracking-tight text-stone-900 sm:text-3xl">
   358→                    Explorar cursos
   359→                  </h1>
   360→                  <p className="mt-1 text-sm text-stone-500" aria-live="polite">
   361→                    {coursesLoading
   362→                      ? 'Carregando cursos...'
   363→                      : `${courses.length} ${courses.length === 1 ? 'curso publicado' : 'cursos publicados'}`}
   364→                  </p>
   365→                </>
   366→              ) : (
   367→                <>
   368→                  <h1 className="text-2xl font-extrabold tracking-tight text-stone-900 sm:text-3xl">
   369→                    Explorar trilhas
   370→                  </h1>
   371→                  <p className="mt-1 text-sm text-stone-500" aria-live="polite">
   372→                    {tracksLoading
   373→                      ? 'Carregando trilhas...'
   374→                      : `${tracks.length} ${tracks.length === 1 ? 'trilha publicada' : 'trilhas publicadas'}`}
   375→                  </p>
   376→                </>
   377→              )}
   378→            </div>
   379→            <div className="w-44">
   380→              <Select
   381→                value={tab === 'mentors' ? sort : tab === 'courses' ? courseSort : trackSort}
   382→                onValueChange={
   383→                  tab === 'mentors' ? setSort : tab === 'courses' ? setCourseSort : setTrackSort
   384→                }
   385→              >
   386→                <SelectTrigger
   387→                  aria-label={
   388→                    tab === 'mentors'
   389→                      ? 'Ordenar mentores'
   390→                      : tab === 'courses'
   391→                        ? 'Ordenar cursos'
   392→                        : 'Ordenar trilhas'
   393→                  }
   394→                  className="bg-white"
   395→                >
   396→                  <SelectValue placeholder="Ordenar" />
   397→                </SelectTrigger>
   398→                <SelectContent>
   399→                  {tab === 'mentors' ? (
   400→                    <>
   401→                      <SelectItem value="relevance">Relevância</SelectItem>
   402→                      <SelectItem value="rating">Melhor avaliados</SelectItem>
   403→                      <SelectItem value="price_asc">Menor preço</SelectItem>
   404→                      <SelectItem value="price_desc">Maior preço</SelectItem>
   405→                      <SelectItem value="experience">Mais experiência</SelectItem>
   406→                    </>
   407→                  ) : (
   408→                    <>
   409→                      <SelectItem value="relevance">Relevância</SelectItem>
   410→                      <SelectItem value="popular">Populares</SelectItem>
   411→                      <SelectItem value="new">Novidades</SelectItem>
   412→                      <SelectItem value="price_asc">Menor preço</SelectItem>
   413→                      <SelectItem value="price_desc">Maior preço</SelectItem>
   414→                    </>
   415→                  )}
   416→                </SelectContent>
   417→              </Select>
   418→            </div>
   419→          </div>
   420→
   421→          <div className="relative mt-5 max-w-2xl">
   422→            <Search
   423→              aria-hidden
   424→              className="absolute left-4 top-1/2 h-4.5 w-4.5 -translate-y-1/2 text-stone-400"
   425→            />
   426→            <Input
   427→              ref={searchRef}
   428→              value={inputValue}
   429→              onChange={(e) => onSearchChange(e.target.value)}
   430→              placeholder={
   431→                tab === 'mentors'
   432→                  ? 'Busque por nome, especialidade ou área...'
   433→                  : tab === 'courses'
   434→                    ? 'Busque por curso, tema ou mentor...'
   435→                    : 'Busque por trilha, tema ou mentor...'
   436→              }
   437→              aria-label={
   438→                tab === 'mentors'
   439→                  ? 'Buscar mentores'
   440→                  : tab === 'courses'
   441→                    ? 'Buscar cursos'
   442→                    : 'Buscar trilhas'
   443→              }
   444→              className="h-12 rounded-2xl border-stone-200 bg-white pl-11 pr-16 text-stone-900 shadow-none placeholder:text-stone-400 focus-visible:border-emerald-400 focus-visible:ring-emerald-200"
   445→            />
   446→            {inputValue ? (
   447→              <button
   448→                onClick={clearSearch}
   449→                aria-label="Limpar busca"
   450→                className="absolute right-3 top-1/2 flex h-7 w-7 -translate-y-1/2 items-center justify-center rounded-full text-stone-400 transition-colors hover:bg-stone-100 hover:text-stone-700"
   451→              >
   452→                <X className="h-4 w-4" />
   453→              </button>
   454→            ) : (
   455→              <kbd className="pointer-events-none absolute right-3.5 top-1/2 hidden h-6 -translate-y-1/2 items-center rounded-md border border-stone-200 bg-stone-50 px-1.5 font-mono text-[11px] font-medium text-stone-400 sm:inline-flex">
   456→                /
   457→              </kbd>
   458→            )}
   459→          </div>
   460→
   461→          <div
   462→            aria-label="Filtrar por categoria"
   463→            className="-mx-4 mt-4 flex gap-2 overflow-x-auto px-4 pb-0.5 [scrollbar-width:none] [&::-webkit-scrollbar]:hidden"
   464→          >
   465→            <button
   466→              onClick={() => setCategory('')}
   467→              className={cn(
   468→                'shrink-0 rounded-full border px-4 py-1.5 text-sm font-medium transition-colors',
   469→                category === ''
   470→                  ? 'border-emerald-700 bg-emerald-700 text-white'
   471→                  : 'border-stone-200 bg-white text-stone-600 hover:border-emerald-300 hover:text-emerald-700'
   472→              )}
   473→            >
   474→              Todas as áreas
   475→            </button>
   476→            {CATEGORIES.map((c) => {
   477→              const count =
   478→                tab === 'mentors'
   479→                  ? (categoryCounts.get(c) ?? 0)
   480→                  : tab === 'courses'
   481→                    ? (courseCounts.get(c) ?? 0)
   482→                    : (trackCounts.get(c) ?? 0)
   483→              const countsReady =
   484→                tab === 'mentors'
   485→                  ? baseReady
   486→                  : tab === 'courses'
   487→                    ? baseCoursesReady
   488→                    : baseTracksReady
   489→              return (
   490→                <button
   491→                  key={c}
   492→                  onClick={() => setCategory(category === c ? '' : c)}
   493→                  className={cn(
   494→                    'shrink-0 rounded-full border px-4 py-1.5 text-sm font-medium transition-colors',
   495→                    category === c
   496→                      ? 'border-emerald-700 bg-emerald-700 text-white'
   497→                      : 'border-stone-200 bg-white text-stone-600 hover:border-emerald-300 hover:text-emerald-700'
   498→                  )}
   499→                >
   500→                  {c}
   501→                  {countsReady && (
   502→                    <span className={cn('ml-1.5 text-[10px]', category === c ? 'text-emerald-100' : 'text-stone-400')}>
   503→                      {count}
   504→                    </span>
   505→                  )}
   506→                </button>
   507→              )
   508→            })}
   509→          </div>
   510→        </div>
   511→      </section>
   512→
   513→      {/* ---------- BENTO: destaque + estatísticas (por aba) ---------- */}
   514→      <section aria-label="Destaques e estatísticas" className="mx-auto w-full max-w-6xl px-4 pt-6">
   515→        <div className="grid grid-cols-1 gap-4 lg:grid-cols-3">
   516→          {/* Spotlight */}
   517→          <div className="lg:col-span-2">
   518→            {tab === 'mentors' ? (
   519→              spot ? (
   520→                <SpotlightCard mentor={spot} index={spotIdx} total={spotlightPool.length} onSelect={(i) => setSpotIdx(i)} />
   521→              ) : (
   522→                <div className="h-full min-h-56 animate-pulse rounded-2xl bg-emerald-950/90" aria-hidden />
   523→              )
   524→            ) : tab === 'courses' ? (
   525→              topCourse ? (
   526→                <CourseSpotlightCard course={topCourse} />
   527→              ) : (
   528→                <div className="h-full min-h-56 animate-pulse rounded-2xl bg-emerald-950/90" aria-hidden />
   529→              )
   530→            ) : topTrack ? (
   531→              <TrackSpotlightCard track={topTrack} />
   532→            ) : (
   533→              <div className="h-full min-h-56 animate-pulse rounded-2xl bg-emerald-950/90" aria-hidden />
   534→            )}
   535→          </div>
   536→
   537→          {/* Stats */}
   538→          <div className="grid grid-cols-3 gap-4 lg:grid-cols-1">
   539→            {tab === 'mentors' ? (
