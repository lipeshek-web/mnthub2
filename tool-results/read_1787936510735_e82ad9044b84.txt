   172→              className="-mt-12 h-24 w-24 shadow-lg"

   500→          <BookingWidget mentor={mentor} />
   501→        </aside>
   502→      </div>
   503→    </div>
   504→  )
   505→}

   507→// ==================== WIDGET DE AGENDAMENTO ====================
   508→
   509→function BookingWidget({ mentor }: { mentor: MentorDetailDTO }) {
   510→  const navigate = useAppStore((s) => s.navigate)
   511→  const user = useAppStore((s) => s.user)
   512→  const bookingTopic = useAppStore((s) => s.bookingTopic)
   513→  const setBookingTopic = useAppStore((s) => s.setBookingTopic)
   514→
   515→  const days = Array.from({ length: 14 }, (_, i) => addDays(new Date(), i))
   516→  const [selectedDate, setSelectedDate] = useState(() => dateKey(new Date()))
   517→  const [slots, setSlots] = useState<string[]>([])
   518→  const [slotsLoading, setSlotsLoading] = useState(true)
   519→  const [selectedSlot, setSelectedSlot] = useState<string | null>(null)
   520→  const [topic, setTopic] = useState(() => bookingTopic)
   521→  const [notes, setNotes] = useState('')
   522→  const [confirmOpen, setConfirmOpen] = useState(false)
   523→  const [submitting, setSubmitting] = useState(false)
   524→
   525→  // Consome o pré-preenchimento (mentoria inclusa em curso/trilha) uma única vez
   526→  useEffect(() => {
   527→    if (bookingTopic) {
   528→      setTopic((prev) => prev || bookingTopic)
   529→      setBookingTopic('')
   530→    }
   531→  }, [bookingTopic, setBookingTopic])
   532→
   533→  const loadSlots = useCallback(async (date: string) => {
   534→    setSlotsLoading(true)
   535→    setSelectedSlot(null)
   536→    try {
   537→      const { slots } = await api.getSlots(mentor.id, date)
   538→      setSlots(slots)
   539→    } catch {
   540→      setSlots([])
   541→    } finally {
   542→      setSlotsLoading(false)
   543→    }
   544→  }, [mentor.id])
   545→
   546→  useEffect(() => {
   547→    loadSlots(selectedDate)
   548→  }, [loadSlots, selectedDate])
   549→
   550→  const isOwner = user?.id === mentor.userId
   551→
   552→  const submit = async () => {
   553→    if (!user) {
   554→      toast.error('Entre com um usuário no topo da página para agendar.')
   555→      return
   556→    }
   557→    if (!selectedSlot) {
   558→      toast.error('Escolha um horário disponível.')
   559→      return
   560→    }
   561→    if (topic.trim().length < 5) {
   562→      toast.error('Descreva o tema da sessão (mín. 5 caracteres).')
   563→      return
   564→    }
   565→    setSubmitting(true)
   566→    try {
   567→      await api.createBooking({
   568→        menteeId: user.id,
   569→        mentorId: mentor.id,
   570→        startsAt: selectedSlot,
   571→        durationMin: 60,
   572→        topic: topic.trim(),
   573→        notes: notes.trim() || undefined,
   574→      })
   575→      toast.success('Solicitação enviada! O mentor confirmará em breve. 🎉')
   576→      navigate({ name: 'dashboard' })
   577→    } catch (e) {
   578→      toast.error(e instanceof Error ? e.message : 'Erro ao agendar')
   579→      loadSlots(selectedDate)
   580→    } finally {
   581→      setSubmitting(false)
   582→      setConfirmOpen(false)
   583→    }
   584→  }
   585→
   586→  if (isOwner) {
   587→    return (
   588→      <Card className="border-emerald-200">
   589→        <CardContent className="flex flex-col items-center gap-3 p-6 text-center">
   590→          <Avatar name={mentor.name} size="lg" />
   591→          <p className="font-bold">Este é o seu perfil público de mentor</p>
   592→          <p className="text-sm text-muted-foreground">
   593→            Gerencie suas informações, agenda e mural de conteúdos.
   594→          </p>
   595→          <Button className="w-full" onClick={() => navigate({ name: 'onboarding' })}>
   596→            Gerenciar meu perfil
   597→          </Button>
   598→        </CardContent>
   599→      </Card>
   600→    )
   601→  }
   602→
   603→  return (
   604→    <Card className="border-stone-200 shadow-md">
   605→      <CardContent className="space-y-5 p-6">
   606→        <div className="flex items-end justify-between">
   607→          <p className="text-3xl font-black tracking-tight text-stone-900">
   608→            {currencyBRL(mentor.hourlyRate)}
   609→            <span className="text-sm font-medium text-muted-foreground"> / sessão</span>
   610→          </p>
   611→          <Badge variant="secondary" className="bg-stone-100 text-stone-600">60 min · vídeo</Badge>
   612→        </div>
   613→
   614→        {/* Dias */}
   615→        <div>
   616→          <Label className="text-xs uppercase tracking-wide text-muted-foreground">Escolha o dia</Label>
   617→          <div className="mt-2 flex gap-2 overflow-x-auto pb-1.5 [scrollbar-width:thin]">
   618→            {days.map((d) => {
   619→              const key = dateKey(d)
   620→              const active = key === selectedDate
   621→              return (
   622→                <button
   623→                  key={key}
   624→                  onClick={() => setSelectedDate(key)}
   625→                  aria-pressed={active}
   626→                  aria-label={`${WEEKDAYS_PT[d.getDay()]}, ${d.getDate()} de ${MONTHS_PT[d.getMonth()]}`}
   627→                  className={cn(
   628→                    'flex h-[62px] w-[52px] shrink-0 flex-col items-center justify-center rounded-xl border text-center transition-all',
   629→                    active
   630→                      ? 'border-emerald-700 bg-emerald-700 text-white shadow-md'
   631→                      : 'border-stone-200 bg-white text-stone-600 hover:border-emerald-300 hover:bg-emerald-50'
   632→                  )}
   633→                >
   634→                  <span className={cn('text-[10px] font-semibold uppercase', active ? 'text-emerald-200' : 'text-stone-400')}>
   635→                    {WEEKDAYS_PT[d.getDay()]}
   636→                  </span>
   637→                  <span className="text-lg font-extrabold leading-tight">{d.getDate()}</span>
   638→                  <span className={cn('text-[10px]', active ? 'text-emerald-200' : 'text-stone-400')}>
   639→                    {MONTHS_PT[d.getMonth()]}
   640→                  </span>
   641→                </button>
   642→              )
   643→            })}
   644→          </div>
   645→        </div>
   646→
   647→        {/* Horários */}
   648→        <div>
   649→          <Label className="text-xs uppercase tracking-wide text-muted-foreground">
   650→            Horários livres · {formatDayLabel(`${selectedDate}T12:00`)}
   651→          </Label>
   652→          {slotsLoading ? (
   653→            <div className="mt-2 grid grid-cols-4 gap-2">
   654→              {Array.from({ length: 4 }).map((_, i) => (
   655→                <Skeleton key={i} className="h-10 rounded-lg" />
   656→              ))}
   657→            </div>
   658→          ) : slots.length === 0 ? (
   659→            <div className="mt-2 flex items-center gap-2 rounded-xl border border-dashed border-stone-200 bg-stone-50 p-4 text-sm text-stone-500">
   660→              <CalendarOff className="h-4 w-4 shrink-0" />
   661→              Sem horários livres neste dia. Tente outra data.
   662→            </div>
   663→          ) : (
   664→            <div className="mt-2 grid max-h-44 grid-cols-4 gap-2 overflow-y-auto pr-1">
   665→              {slots.map((s) => {
   666→                const active = s === selectedSlot
   667→                return (
   668→                  <button
   669→                    key={s}
   670→                    onClick={() => setSelectedSlot(s)}
   671→                    aria-pressed={active}
   672→                    className={cn(
   673→                      'h-10 rounded-lg border text-sm font-semibold transition-all',
   674→                      active
   675→                        ? 'border-emerald-700 bg-emerald-700 text-white shadow'
   676→                        : 'border-stone-200 bg-white text-stone-700 hover:border-emerald-400 hover:text-emerald-700'
   677→                    )}
   678→                  >
   679→                    {formatTimeLabel(s)}
   680→                  </button>
   681→                )
   682→              })}
   683→            </div>
   684→          )}
   685→        </div>
   686→
   687→        {/* Resumo */}
   688→        {selectedSlot && (
   689→          <div className="rounded-xl border border-emerald-200 bg-emerald-50 p-3.5 text-sm">
   690→            <p className="font-bold text-emerald-900">{formatDayLabelLong(selectedSlot)}</p>
   691→            <p className="mt-0.5 text-emerald-800">
   692→              {formatTimeLabel(selectedSlot)} → {addMinutesToTime(selectedSlot, 60)} · {currencyBRL(mentor.hourlyRate)}
   693→            </p>
   694→          </div>
   695→        )}
   696→
   697→        {/* Tema */}
   698→        <div className="space-y-3">
   699→          <div className="space-y-1.5">
   700→            <Label htmlFor="booking-topic">Tema da sessão *</Label>
   701→            <Input
   702→              id="booking-topic"
   703→              value={topic}
   704→              onChange={(e) => setTopic(e.target.value)}
   705→              placeholder="Ex.: Revisão de portfólio e próximos passos"
   706→            />
   707→          </div>
   708→          <div className="space-y-1.5">
   709→            <Label htmlFor="booking-notes">Observações (opcional)</Label>
   710→            <Textarea
   711→              id="booking-notes"
   712→              value={notes}
   713→              onChange={(e) => setNotes(e.target.value)}
   714→              placeholder="Contexto, links ou materiais para o mentor se preparar..."
   715→              rows={2}
   716→            />
   717→          </div>
   718→        </div>
   719→
   720→        <Button
   721→          className="h-11 w-full text-base font-bold"
   722→          disabled={submitting}
   723→          onClick={() => {
   724→            if (!user) {
   725→              toast.error('Entre com um usuário no topo da página para agendar.')
   726→              return
   727→            }
   728→            setConfirmOpen(true)
   729→          }}
   730→        >
   731→          {submitting ? 'Enviando...' : 'Solicitar agendamento'}
   732→        </Button>
   733→        <p className="-mt-2 text-center text-[11px] leading-relaxed text-muted-foreground">
   734→          A sessão nasce como solicitação e é confirmada pelo mentor. Acompanhe tudo em{' '}
   735→          <strong>Minhas sessões</strong>.
   736→        </p>
   737→      </CardContent>
   738→
   739→      {/* Confirmação */}
   740→      <Dialog open={confirmOpen} onOpenChange={setConfirmOpen}>
   741→        <DialogContent className="sm:max-w-md">
   742→          <DialogHeader>
   743→            <DialogTitle>Confirmar solicitação</DialogTitle>
   744→            <DialogDescription>
   745→              Revise os dados antes de enviar para {mentor.name.split(' ')[0]}.
   746→            </DialogDescription>
   747→          </DialogHeader>
   748→          <div className="space-y-2.5 rounded-xl border border-stone-200 bg-stone-50 p-4 text-sm">
   749→            <div className="flex justify-between gap-4">
   750→              <span className="text-muted-foreground">Mentor</span>
   751→              <span className="font-semibold">{mentor.name}</span>
   752→            </div>
   753→            <div className="flex justify-between gap-4">
   754→              <span className="text-muted-foreground">Quando</span>
   755→              <span className="text-right font-semibold">
   756→                {selectedSlot && formatDayLabelLong(selectedSlot)}
   757→                <br />
   758→                {selectedSlot &&
   759→                  `${formatTimeLabel(selectedSlot)} → ${addMinutesToTime(selectedSlot, 60)}`}
   760→              </span>
   761→            </div>
   762→            <div className="flex justify-between gap-4">
   763→              <span className="text-muted-foreground">Tema</span>
   764→              <span className="max-w-56 text-right font-semibold">{topic || '—'}</span>
   765→            </div>
   766→            <div className="flex justify-between gap-4 border-t border-stone-200 pt-2.5">
   767→              <span className="text-muted-foreground">Valor</span>
   768→              <span className="font-extrabold text-emerald-700">{currencyBRL(mentor.hourlyRate)}</span>
   769→            </div>
   770→          </div>
   771→          <DialogFooter>
   772→            <Button variant="outline" onClick={() => setConfirmOpen(false)}>
   773→              Revisar
   774→            </Button>
   775→            <Button onClick={submit} disabled={submitting}>
   776→              {submitting ? 'Enviando...' : 'Enviar solicitação'}
   777→            </Button>
   778→          </DialogFooter>
   779→        </DialogContent>
   780→      </Dialog>
   781→    </Card>
   782→  )
   783→}
