   189→function BookingCard({
   190→  booking: b,
   191→  userId,
   192→  busy,
   193→  onConfirm,
   194→  onReject,
   195→  onComplete,
   196→  onCancel,
   197→  onJoin,
   198→  onReview,
   199→}: {
   200→  booking: BookingDTO
   201→  userId: string
   202→  busy: boolean
   203→  onConfirm: (booking: BookingDTO) => void
   204→  onReject: (booking: BookingDTO) => void
   205→  onComplete: (booking: BookingDTO) => void
   206→  onCancel: (booking: BookingDTO) => void
   207→  onJoin: (booking: BookingDTO) => void
   208→  onReview: (booking: BookingDTO, rating: number, comment: string) => Promise<void>
   209→}) {
   210→  const isMentorSide = b.mentor.userId === userId
   211→  const otherName = isMentorSide ? b.mentee.name : b.mentor.name
   212→  const relative = relativeDayLabel(b.startsAt)
   213→  const statusMeta = STATUS_META[b.status] ?? {
   214→    label: b.status,
   215→    className: 'bg-stone-100 text-stone-600 border-stone-200',
   216→  }
   217→
   218→  const [dialogAction, setDialogAction] = useState<ConfirmKind | null>(null)
   219→  const [reviewOpen, setReviewOpen] = useState(false)
   220→  const [rating, setRating] = useState(0)
   221→  const [hoverRating, setHoverRating] = useState(0)
   222→  const [comment, setComment] = useState('')
   223→  const [submitting, setSubmitting] = useState(false)
   224→
   225→  const canJoin = b.status === 'CONFIRMED'
   226→  const canReview = !isMentorSide && b.status === 'COMPLETED' && !b.reviewed
   227→
   228→  const dialogContent: Record<ConfirmKind, { title: string; description: string; actionLabel: string }> = {
   229→    reject: {
   230→      title: 'Recusar solicitação?',
   231→      description: `A solicitação de ${b.mentee.name} sobre "${b.topic}" será cancelada e o horário será liberado na sua agenda.`,
   232→      actionLabel: 'Recusar',
   233→    },
   234→    complete: {
   235→      title: 'Marcar como concluída?',
   236→      description: `Confirme que a sessão com ${b.mentee.name} aconteceu. Ela irá para o histórico e o aluno poderá avaliá-la.`,
   237→      actionLabel: 'Concluir sessão',
   238→    },
   239→    cancel: {
   240→      title: 'Cancelar sessão?',
   241→      description: `Sua sessão com ${b.mentor.name} será cancelada e o horário será liberado para outras pessoas.`,
   242→      actionLabel: 'Cancelar sessão',
   243→    },
   244→  }
   245→  const currentDialog = dialogAction ? dialogContent[dialogAction] : null
   246→
   247→  const submitReview = async () => {
   248→    if (rating < 1) {
   249→      toast.error('Escolha uma nota de 1 a 5 estrelas antes de enviar.')
   250→      return
   251→    }
   252→    setSubmitting(true)
   253→    try {
   254→      await onReview(b, rating, comment.trim())
   255→      setReviewOpen(false)
   256→      setRating(0)
   257→      setHoverRating(0)
   258→      setComment('')
   259→    } catch (err) {
   260→      toast.error(err instanceof Error ? err.message : 'Não foi possível enviar sua avaliação.')
   261→    } finally {
   262→      setSubmitting(false)
   263→    }
   264→  }
   265→
   266→  return (
   267→    <Card className="gap-4 py-6">
   268→      <CardContent className="flex flex-col gap-4 px-6">
   269→        <div className="flex flex-wrap items-start justify-between gap-3">
   270→          <div className="flex min-w-0 items-start gap-3">
   271→            <Avatar name={otherName} size="lg" />
   272→            <div className="min-w-0">
   273→              <div className="flex flex-wrap items-center gap-2">
   274→                <p className="font-semibold leading-tight">{otherName}</p>
   275→                <Badge
   276→                  className={
   277→                    isMentorSide
   278→                      ? 'border-emerald-200 bg-emerald-50 text-emerald-700'
   279→                      : 'border-stone-200 bg-stone-100 text-stone-600'
   280→                  }
   281→                >
   282→                  {isMentorSide ? 'Como mentor' : 'Como aluno'}
   283→                </Badge>
   284→              </div>
   285→              {!isMentorSide && b.mentor.headline ? (
   286→                <p className="mt-0.5 text-sm text-muted-foreground">{b.mentor.headline}</p>
   287→              ) : null}
   288→              <p className="mt-1.5 font-medium">{b.topic}</p>
   289→              {b.notes ? <p className="mt-1 text-sm text-muted-foreground">{b.notes}</p> : null}
   290→            </div>
   291→          </div>
   292→          <Badge variant="outline" className={statusMeta.className}>
   293→            {statusMeta.label}
   294→          </Badge>
   295→        </div>
   296→
   297→        <div className="flex flex-wrap items-center gap-x-4 gap-y-1.5 text-sm text-muted-foreground">
   298→          <span className="inline-flex items-center gap-1.5">
   299→            <CalendarDays className="size-4 text-emerald-600" aria-hidden />
   300→            {relative ? `${relative} · ` : ''}
   301→            {formatDayLabelLong(b.startsAt)} · {formatTimeLabel(b.startsAt)} →{' '}
   302→            {addMinutesToTime(b.startsAt, b.durationMin)}
   303→          </span>
   304→          <span className="inline-flex items-center gap-1.5">
   305→            <Clock className="size-4" aria-hidden /> {b.durationMin} min
   306→          </span>
   307→          <span className="inline-flex items-center gap-1.5">
   308→            <Wallet className="size-4" aria-hidden /> {currencyBRL(b.price)}
   309→          </span>
   310→        </div>
   311→
   312→        <div className="flex flex-wrap gap-2">
   313→          {isMentorSide && b.status === 'PENDING' ? (
   314→            <>
   315→              <Button size="sm" disabled={busy} onClick={() => onConfirm(b)}>
   316→                <Check className="size-4" aria-hidden /> Confirmar
   317→              </Button>
   318→              <Button size="sm" variant="outline" disabled={busy} onClick={() => setDialogAction('reject')}>
   319→                <X className="size-4" aria-hidden /> Recusar
   320→              </Button>
   321→            </>
   322→          ) : null}
   323→
   324→          {isMentorSide && b.status === 'CONFIRMED' ? (
   325→            <Button size="sm" variant="outline" disabled={busy} onClick={() => setDialogAction('complete')}>
   326→              <CheckCheck className="size-4" aria-hidden /> Marcar como concluída
   327→            </Button>
   328→          ) : null}
   329→
   330→          {canJoin ? (
   331→            <Button size="sm" onClick={() => onJoin(b)}>
   332→              <Video className="size-4" aria-hidden /> Entrar na sala
   333→            </Button>
   334→          ) : null}
   335→
   336→          {!isMentorSide && (b.status === 'PENDING' || b.status === 'CONFIRMED') ? (
   337→            <Button
   338→              size="sm"
   339→              variant="ghost"
   340→              className="text-rose-600 hover:bg-rose-50 hover:text-rose-700"
   341→              disabled={busy}
   342→              onClick={() => setDialogAction('cancel')}
   343→            >
   344→              Cancelar sessão
   345→            </Button>
   346→          ) : null}
   347→
   348→          {canReview ? (
   349→            <Button size="sm" variant="outline" onClick={() => setReviewOpen(true)}>
   350→              <Star className="size-4 fill-amber-400 text-amber-400" aria-hidden /> Avaliar sessão
   351→            </Button>
   352→          ) : null}
   353→        </div>
   354→      </CardContent>
   355→
   356→      {/* Confirmações: recusar / concluir / cancelar */}
   357→      <AlertDialog
   358→        open={dialogAction !== null}
   359→        onOpenChange={(open) => {
   360→          if (!open) setDialogAction(null)
   361→        }}
   362→      >
   363→        <AlertDialogContent>
   364→          <AlertDialogHeader>
   365→            <AlertDialogTitle>{currentDialog?.title}</AlertDialogTitle>
   366→            <AlertDialogDescription>{currentDialog?.description}</AlertDialogDescription>
   367→          </AlertDialogHeader>
   368→          <AlertDialogFooter>
   369→            <AlertDialogCancel>Voltar</AlertDialogCancel>
   370→            <AlertDialogAction
   371→              className="bg-rose-600 text-white hover:bg-rose-700"
   372→              disabled={busy}
   373→              onClick={() => {
   374→                const action = dialogAction
   375→                setDialogAction(null)
   376→                if (action === 'reject') onReject(b)
   377→                else if (action === 'complete') onComplete(b)
   378→                else if (action === 'cancel') onCancel(b)
   379→              }}
   380→            >
   381→              {currentDialog?.actionLabel}
   382→            </AlertDialogAction>
   383→          </AlertDialogFooter>
   384→        </AlertDialogContent>
   385→      </AlertDialog>
   386→
   387→      {/* Avaliar sessão */}
   388→      <Dialog open={reviewOpen} onOpenChange={setReviewOpen}>
   389→        <DialogContent className="sm:max-w-md">
   390→          <DialogHeader>
   391→            <DialogTitle>Avaliar sessão</DialogTitle>
   392→            <DialogDescription>
   393→              Como foi sua mentoria com {b.mentor.name} sobre &quot;{b.topic}&quot;? Seu feedback ajuda outros
   394→              alunos.
   395→            </DialogDescription>
   396→          </DialogHeader>
   397→          <div className="flex flex-col gap-4">
   398→            <div className="flex items-center justify-center gap-1" role="radiogroup" aria-label="Nota da avaliação">
   399→              {[1, 2, 3, 4, 5].map((value) => {
   400→                const active = (hoverRating || rating) >= value
   401→                return (
   402→                  <button
   403→                    key={value}
   404→                    type="button"
   405→                    role="radio"
   406→                    aria-checked={rating === value}
   407→                    aria-label={`${value} ${value === 1 ? 'estrela' : 'estrelas'}`}
   408→                    onClick={() => setRating(value)}
   409→                    onMouseEnter={() => setHoverRating(value)}
   410→                    onMouseLeave={() => setHoverRating(0)}
   411→                    className="rounded-full p-1 transition-transform hover:scale-110 focus-visible:outline-2 focus-visible:outline-amber-500"
   412→                  >
   413→                    <Star
   414→                      size={28}
   415→                      className={cn(
   416→                        'transition-colors',
   417→                        active ? 'fill-amber-400 text-amber-400' : 'fill-stone-200 text-stone-300'
   418→                      )}
   419→                      aria-hidden
   420→                    />
   421→                  </button>
   422→                )
   423→              })}
   424→            </div>
   425→            <div className="flex flex-col gap-2">
   426→              <Label htmlFor={`review-comment-${b.id}`}>Como foi sua experiência?</Label>
   427→              <Textarea
   428→                id={`review-comment-${b.id}`}
   429→                rows={4}
   430→                value={comment}
   431→                onChange={(event) => setComment(event.target.value)}
   432→                placeholder="Conte o que funcionou bem e o que pode melhorar..."
   433→              />
   434→            </div>
   435→          </div>
   436→          <DialogFooter>
   437→            <Button variant="ghost" onClick={() => setReviewOpen(false)} disabled={submitting}>
   438→              Cancelar
   439→            </Button>
   440→            <Button onClick={() => void submitReview()} disabled={submitting}>
   441→              {submitting ? 'Enviando...' : 'Enviar avaliação'}
   442→            </Button>
   443→          </DialogFooter>
   444→        </DialogContent>
   445→      </Dialog>
   446→    </Card>
   447→  )
   448→}

   922→function EnrolledCourseCard({
   923→  enrollment,
   924→  onOpen,
   925→}: {
   926→  enrollment: EnrolledCourseDTO
   927→  onOpen: () => void
   928→}) {
   929→  const { course, completedLessonIds } = enrollment

   931→  const completed = completedLessonIds.length
   932→  const pct = Math.round((completed / total) * 100)
   933→  const isDone = completed >= course.lessonCount && course.lessonCount > 0
   934→
   935→  return (
   936→    <Card className="overflow-hidden p-0">
   937→      <div className="flex flex-col gap-4 p-5 sm:flex-row sm:items-center">

   939→          <img
   940→            src={course.coverUrl}
   941→            alt=""
   942→            aria-hidden
   943→            className="h-14 w-14 shrink-0 rounded-2xl object-cover"
   944→          />
   945→        ) : (
   946→          <span
   947→            aria-hidden
   948→            style={avatarGradient(course.title)}
   949→            className="flex h-14 w-14 shrink-0 items-center justify-center rounded-2xl text-white"
   950→          >
   951→            <Library className="size-6" />
   952→          </span>
   953→        )}
   954→
   955→        <div className="min-w-0 flex-1">
   956→          <div className="flex flex-wrap items-center gap-2">
   957→            <p className="truncate font-bold text-stone-900">{course.title}</p>
   958→            {isDone && (
   959→              <Badge className="border-emerald-200 bg-emerald-100 text-emerald-800">
   960→                <CheckCircle2 className="size-3" aria-hidden /> Concluído
   961→              </Badge>
   962→            )}
   963→          </div>
   964→          <p className="mt-0.5 text-xs text-muted-foreground">
   965→            por {course.mentor.name} · {course.lessonCount} aulas ·{' '}
   966→            {formatTotalDuration(course.totalDurationMin)}
   967→          </p>
   968→          <div className="mt-2.5 flex items-center gap-3">
   969→            <Progress value={pct} aria-label={`${pct}% do curso concluído`} className="h-2 max-w-64 flex-1" />
   970→            <span className="shrink-0 text-xs font-semibold text-stone-500">
   971→              {completed}/{course.lessonCount} aulas
   972→            </span>
   973→          </div>
   974→        </div>
   975→
   976→        <Button onClick={onOpen} className="shrink-0 self-start rounded-full sm:self-auto">
   977→          <PlayCircle className="size-4" aria-hidden /> {isDone ? 'Revisar curso' : 'Continuar'}
   978→        </Button>
   979→      </div>
   980→    </Card>
   981→  )
   982→}
