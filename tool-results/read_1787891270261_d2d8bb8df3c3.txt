   374→   374→  const navigate = useAppStore((s) => s.navigate)
   375→   375→  const user = useAppStore((s) => s.user)
   376→   376→
   377→   377→  const days = Array.from({ length: 14 }, (_, i) => addDays(new Date(), i))
   378→   378→  const [selectedDate, setSelectedDate] = useState(() => dateKey(new Date()))
   379→   379→  const [slots, setSlots] = useState<string[]>([])
   380→   380→  const [slotsLoading, setSlotsLoading] = useState(true)

   382→   382→  const [topic, setTopic] = useState('')

   386→   386→
   387→   387→  const loadSlots = useCallback(async (date: string) => {
   388→   388→    setSlotsLoading(true)
   389→   389→    setSelectedSlot(null)
   390→   390→    try {
   391→   391→      const { slots } = await api.getSlots(mentor.id, date)
   392→   392→      setSlots(slots)
   393→   393→    } catch {
   394→   394→      setSlots([])
   395→   395→    } finally {
   396→   396→      setSlotsLoading(false)
   397→   397→    }
   398→   398→  }, [mentor.id])
   399→   399→
   400→   400→  useEffect(() => {
   401→   401→    loadSlots(selectedDate)
   402→   402→  }, [loadSlots, selectedDate])
   403→   403→
   404→   404→  const isOwner = user?.id === mentor.userId
   405→   405→
   406→   406→  const submit = async () => {
   407→   407→    if (!user) {
   408→   408→      toast.error('Entre com um usuário no topo da página para agendar.')
   409→   409→      return

   411→   411→    if (!selectedSlot) {
   412→   412→      toast.error('Escolha um horário disponível.')
   413→   413→      return
   414→   414→    }
   415→   415→    if (topic.trim().length < 5) {
   416→   416→      toast.error('Descreva o tema da sessão (mín. 5 caracteres).')
   417→   417→      return
   418→   418→    }
   419→   419→    setSubmitting(true)
   420→   420→    try {
   421→   421→      await api.createBooking({
   422→   422→        menteeId: user.id,
