     1→     1→'use client'

     4→     4→import {
     5→     5→  AlertCircle,
     6→     6→  ArrowLeft,
     7→     7→  BookOpen,
     8→     8→  Check,

    19→    19→import { Badge } from '@/components/ui/badge'
    20→    20→import { Button } from '@/components/ui/button'
    21→    21→import { Card } from '@/components/ui/card'
    22→    22→import { Progress } from '@/components/ui/progress'

    29→    29→  currencyBRL,

    32→    32→} from '@/lib/helpers'
    33→    33→import { loadTrackingScripts, trackEvent } from '@/lib/tracking'
    34→    34→import { useAppStore } from '@/lib/store'

    37→    37→
    38→    38→export function CourseView({ courseId }: { courseId: string }) {
    39→    39→  const navigate = useAppStore((s) => s.navigate)
    40→    40→  const user = useAppStore((s) => s.user)
    41→    41→
    42→    42→  const [data, setData] = useState<CourseDetailDTO | null>(null)
    43→    43→  const [loading, setLoading] = useState(true)
    44→    44→  const [error, setError] = useState<string | null>(null)
    45→    45→
    46→    46→  // Estado da sala de aula (modo inscrito)
    47→    47→  const [completedIds, setCompletedIds] = useState<string[]>([])
    48→    48→  const [currentLessonId, setCurrentLessonId] = useState<string | null>(null)
    49→    49→  const [toggling, setToggling] = useState(false)
    50→    50→
    51→    51→  // Estado da inscrição (modo visitante)
    52→    52→  const [enrolling, setEnrolling] = useState(false)
    53→    53→
    54→    54→  // Tracking (GA4/Meta Pixel do mentor + view_item) — uma única vez por curso/montagem
    55→    55→  const trackedCourseRef = useRef<string | null>(null)
    56→    56→
    57→    57→  const fetchCourse = useCallback(async () => {
    58→    58→    setLoading(true)
    59→    59→    setError(null)
    60→    60→    try {
    61→    61→      const d = await api.getCourse(courseId, user?.id)
    62→    62→      setData(d)
    63→    63→    } catch (err) {
    64→    64→      setError(err instanceof Error ? err.message : 'Não foi possível carregar o curso.')
    65→    65→    } finally {
    66→    66→      setLoading(false)
    67→    67→    }
    68→    68→  }, [courseId, user?.id])
    69→    69→
    70→    70→  useEffect(() => {
    71→    71→    void fetchCourse()
    72→    72→  }, [fetchCourse])
    73→    73→
    74→    74→  const lessons = useMemo<CourseLessonDTO[]>(
    75→    75→    () => [...(data?.lessons ?? [])].sort((a, b) => a.order - b.order),
    76→    76→    [data]
    77→    77→  )
    78→    78→
    79→    79→  // Inicializa a sala de aula quando os dados (com matrícula) chegam:
    80→    80→  // a aula aberta por padrão é a primeira ainda não concluída.
    81→    81→  useEffect(() => {
    82→    82→    const enrollment = data?.enrollment
    83→    83→    if (!enrollment) return
    84→    84→    const sorted = [...data.lessons].sort((a, b) => a.order - b.order)
    85→    85→    setCompletedIds(enrollment.completedLessonIds)
    86→    86→    const firstIncomplete = sorted.find((l) => !enrollment.completedLessonIds.includes(l.id))
    87→    87→    setCurrentLessonId(firstIncomplete?.id ?? sorted[0]?.id ?? null)
    88→    88→  }, [data])
    89→    89→
    90→    90→  const course = data
    91→    91→
    92→    92→  // Dispara view_item + injeta os pixels do mentor quando os dados do curso chegam
    93→    93→  // (visitante ou inscrito), no máximo uma vez por curso por montagem.
    94→    94→  useEffect(() => {
    95→    95→    if (!course || trackedCourseRef.current === course.id) return
    96→    96→    trackedCourseRef.current = course.id
    97→    97→    loadTrackingScripts({
    98→    98→      mentorGaId: course.mentor.tracking?.gaMeasurementId,
    99→    99→      mentorPixelId: course.mentor.tracking?.metaPixelId,
   100→   100→    })
   101→   101→    trackEvent('view_item', {
   102→   102→      mentorId: course.mentor.id,
   103→   103→      courseId: course.id,
   104→   104→      value: course.price || undefined,
   105→   105→      contentName: course.title,
   106→   106→    })
   107→   107→  }, [course])
   108→   108→
   109→   109→  const currentLesson = useMemo(
   110→   110→    () => lessons.find((l) => l.id === currentLessonId) ?? lessons[0] ?? null,
   111→   111→    [lessons, currentLessonId]
   112→   112→  )
   113→   113→  const currentIndex = currentLesson ? lessons.findIndex((l) => l.id === currentLesson.id) : -1
   114→   114→  const percent =
   115→   115→    lessons.length > 0 ? Math.round((completedIds.length / lessons.length) * 100) : 0
   116→   116→
   117→   117→  /* ---------- Inscrição (overview) ---------- */
   118→   118→
   119→   119→  const doEnroll = async () => {
   120→   120→    if (!user || !course) return
   121→   121→    setEnrolling(true)
   122→   122→    try {
   123→   123→      const res = await api.enrollCourse(course.id, user.id)
   124→   124→      if (res.alreadyEnrolled) {
   125→   125→        toast.info('Você já estava inscrito neste curso.')
   126→   126→      } else {
   127→   127→        toast.success('Inscrição realizada! Boa jornada 🎉')
   128→   128→      }
   129→   129→      // Busca novamente com o usuário: enrollment preenchido ativa o modo sala de aula
   130→   130→      const fresh = await api.getCourse(course.id, user.id)
   131→   131→      setData(fresh)
   132→   132→    } catch (err) {
   133→   133→      toast.error(err instanceof Error ? err.message : 'Erro ao realizar inscrição.')
   134→   134→    } finally {
   135→   135→      setEnrolling(false)
   136→   136→    }
   137→   137→  }
   138→   138→
   139→   139→  const handleEnrollClick = () => {
   140→   140→    if (!user) {
   141→   141→      toast.error('Entre com uma conta no topo da página para se inscrever.')
   142→   142→      return
   143→   143→    }
   144→   144→    if (!course) return
   145→   145→    if (course.price === 0) {
   146→   146→      void doEnroll()
   147→   147→    } else {
   148→   148→      // Curso pago: fluxo completo de checkout (com atribuição e pixels carregados)
   149→   149→      trackEvent('begin_checkout', {
   150→   150→        mentorId: course.mentor.id,
   151→   151→        courseId: course.id,
   152→   152→        value: course.price || undefined,
   153→   153→        contentName: course.title,
   154→   154→      })
   155→   155→      navigate({ name: 'checkout', courseId: course.id })
   156→   156→    }
   157→   157→  }
   158→   158→
   159→   159→  /* ---------- Progresso (classroom) ---------- */
   160→   160→
   161→   161→  const toggleComplete = async () => {
   162→   162→    if (!user || !course || !currentLesson) return
   163→   163→    setToggling(true)
   164→   164→    try {
   165→   165→      const res = await api.toggleLessonComplete(course.id, {
   166→   166→        userId: user.id,
   167→   167→        lessonId: currentLesson.id,
   168→   168→      })
   169→   169→      const nowCompleted = res.completedLessonIds.includes(currentLesson.id)
   170→   170→      setCompletedIds(res.completedLessonIds)
   171→   171→      if (nowCompleted && res.completedLessonIds.length === lessons.length) {
   172→   172→        toast.success('Curso concluído! Parabéns 🎉')
   173→   173→      } else if (nowCompleted) {
   174→   174→        toast.success('Aula marcada como concluída.')
   175→   175→      } else {
   176→   176→        toast('Marcação removida desta aula.')
   177→   177→      }
   178→   178→    } catch (err) {
   179→   179→      toast.error(err instanceof Error ? err.message : 'Não foi possível atualizar o progresso.')
   180→   180→    } finally {
   181→   181→      setToggling(false)
   182→   182→    }
   183→   183→  }
   184→   184→
   185→   185→  const goPrev = () => {
   186→   186→    if (currentLesson && currentIndex > 0) setCurrentLessonId(lessons[currentIndex - 1].id)
   187→   187→  }
   188→   188→  const goNext = () => {
   189→   189→    if (currentLesson && currentIndex < lessons.length - 1) {
   190→   190→      setCurrentLessonId(lessons[currentIndex + 1].id)
   191→   191→    }
   192→   192→  }
   193→   193→
   194→   194→  /* ---------- Renders auxiliares ---------- */
   195→   195→
   196→   196→  if (loading) {
   197→   197→    return (
   198→   198→      <div className="mx-auto w-full max-w-6xl px-4 py-6 sm:py-8" aria-busy="true">
   199→   199→        <Skeleton className="h-9 w-24 rounded-full" />
   200→   200→        <Skeleton className="mt-4 h-52 w-full rounded-2xl" />
   201→   201→        <div className="mt-6 grid grid-cols-1 gap-6 lg:grid-cols-3">
   202→   202→          <div className="space-y-6 lg:col-span-2">
   203→   203→            <Skeleton className="h-44 rounded-2xl" />
   204→   204→            <Skeleton className="h-64 rounded-2xl" />
   205→   205→          </div>
   206→   206→          <Skeleton className="h-72 rounded-2xl" />
   207→   207→        </div>
   208→   208→        <p className="sr-only">Carregando curso…</p>
   209→   209→      </div>
   210→   210→    )
   211→   211→  }
   212→   212→
   213→   213→  if (!course) {
   214→   214→    return (
   215→   215→      <div className="mx-auto w-full max-w-6xl px-4 py-6 sm:py-8">
   216→   216→        <div className="flex flex-col items-center gap-3 rounded-2xl border border-dashed border-stone-300 px-6 py-14 text-center">
   217→   217→          <span className="flex h-14 w-14 items-center justify-center rounded-full bg-rose-50">
   218→   218→            <AlertCircle aria-hidden className="h-7 w-7 text-rose-500" />
   219→   219→          </span>
   220→   220→          <p className="font-bold text-stone-900">Não foi possível carregar o curso</p>
   221→   221→          <p className="max-w-sm text-sm leading-relaxed text-stone-500">
   222→   222→            {error ?? 'Curso não encontrado.'}
   223→   223→          </p>
   224→   224→          <Button variant="outline" className="rounded-full" onClick={() => void fetchCourse()}>
   225→   225→            Tentar novamente
   226→   226→          </Button>
   227→   227→        </div>
   228→   228→      </div>
   229→   229→    )
   230→   230→  }
   231→   231→
   232→   232→  const isEnrolled = course.enrollment !== null
   233→   233→
   234→   234→  return (
   235→   235→    <div className="mx-auto w-full max-w-6xl px-4 py-6 sm:py-8">
   236→   236→      <Button
   237→   237→        variant="ghost"
   238→   238→        onClick={() => navigate({ name: 'marketplace' })}
   239→   239→        className="-ml-2 h-10 gap-1.5 rounded-full px-3 font-semibold text-stone-600 hover:text-stone-900"
   240→   240→      >
   241→   241→        <ArrowLeft aria-hidden className="h-4 w-4" /> Voltar
   242→   242→      </Button>
   243→   243→
   244→   244→      {isEnrolled ? (
   245→   245→        <ClassroomContent
   246→   246→          course={course}
   247→   247→          lessons={lessons}
   248→   248→          currentLesson={currentLesson}
   249→   249→          currentIndex={currentIndex}
   250→   250→          percent={percent}
   251→   251→          completedIds={completedIds}
   252→   252→          toggling={toggling}
   253→   253→          onSelectLesson={setCurrentLessonId}
   254→   254→          onToggleComplete={() => void toggleComplete()}
   255→   255→          onPrev={goPrev}
   256→   256→          onNext={goNext}
   257→   257→        />
   258→   258→      ) : (
   259→   259→        <OverviewContent
   260→   260→          course={course}
   261→   261→          lessons={lessons}
   262→   262→          enrolling={enrolling}
   263→   263→          isLoggedIn={Boolean(user)}
   264→   264→          onEnrollClick={handleEnrollClick}
   265→   265→          onViewMentor={(mentorId) => navigate({ name: 'mentor', mentorId })}
   266→   266→        />
   267→   267→      )}
   268→   268→    </div>
   269→   269→  )
   270→   270→}
   271→   271→
   272→   272→/* ==================== MODO VISITANTE (overview) ==================== */
   273→   273→
   274→   274→function OverviewContent({
   275→   275→  course,
   276→   276→  lessons,
   277→   277→  enrolling,
   278→   278→  isLoggedIn,
   279→   279→  onEnrollClick,
   280→   280→  onViewMentor,
   281→   281→}: {
   282→   282→  course: CourseDetailDTO
   283→   283→  lessons: CourseLessonDTO[]
   284→   284→  enrolling: boolean
   285→   285→  isLoggedIn: boolean
   286→   286→  onEnrollClick: () => void
   287→   287→  onViewMentor: (mentorId: string) => void
   288→   288→}) {
   289→   289→  return (
   290→   290→    <>
   291→   291→      {/* ---------- HERO ---------- */}
   292→   292→      <section
   293→   293→        aria-label={`Curso: ${course.title}`}
   294→   294→        className="relative overflow-hidden rounded-2xl p-6 text-white sm:p-8"
   295→   295→        style={avatarGradient(course.title)}
   296→   296→      >
   297→   297→        {course.coverUrl && (
   298→   298→          <>
   299→   299→            <img
   300→   300→              src={course.coverUrl}
   301→   301→              alt=""
   302→   302→              aria-hidden
   303→   303→              className="absolute inset-0 h-full w-full object-cover"
   304→   304→            />
   305→   305→            <div aria-hidden className="absolute inset-0 bg-stone-950/55" />
   306→   306→          </>
   307→   307→        )}
   308→   308→        {!course.coverUrl && (
   309→   309→          <Library aria-hidden className="pointer-events-none absolute -right-6 -top-10 h-48 w-48 text-white/15" />
   310→   310→        )}
   311→   311→        <div className="relative">
   312→   312→          <div className="flex flex-wrap items-center gap-2">
   313→   313→            <Badge className="rounded-full border-white/25 bg-white/15 text-white hover:bg-white/15">
   314→   314→              {course.category}
   315→   315→            </Badge>
   316→   316→            <Badge className="rounded-full border-white/25 bg-stone-900/30 text-white hover:bg-stone-900/30">
   317→   317→              {LEVEL_LABELS[course.level] ?? course.level}
   318→   318→            </Badge>
   319→   319→          </div>
   320→   320→          <h1 className="mt-4 text-2xl font-extrabold tracking-tight sm:text-3xl">{course.title}</h1>
   321→   321→          <p className="mt-2 max-w-2xl text-sm leading-relaxed text-white/85 sm:text-[15px]">
   322→   322→            {course.description}
   323→   323→          </p>
   324→   324→          <div className="mt-5 flex flex-wrap items-center gap-x-5 gap-y-2 text-sm text-white/90">
   325→   325→            <span className="inline-flex items-center gap-1.5">
   326→   326→              <BookOpen aria-hidden className="h-4 w-4" />
   327→   327→              {course.lessonCount} {course.lessonCount === 1 ? 'aula' : 'aulas'}
   328→   328→            </span>
   329→   329→            <span className="inline-flex items-center gap-1.5">
   330→   330→              <Clock aria-hidden className="h-4 w-4" />
   331→   331→              {formatTotalDuration(course.totalDurationMin)}
   332→   332→            </span>
   333→   333→            <span className="inline-flex items-center gap-1.5">
   334→   334→              <Users aria-hidden className="h-4 w-4" />
   335→   335→              {course.studentCount} {course.studentCount === 1 ? 'aluno' : 'alunos'}
   336→   336→            </span>
   337→   337→            <span className="font-extrabold">
   338→   338→              {course.price === 0 ? 'Grátis' : currencyBRL(course.price)}
   339→   339→            </span>
   340→   340→          </div>
   341→   341→        </div>
   342→   342→      </section>
   343→   343→
   344→   344→      {/* ---------- CONTEÚDO ---------- */}
   345→   345→      <div className="mt-6 grid grid-cols-1 gap-6 lg:grid-cols-3">
   346→   346→        <div className="space-y-6 lg:col-span-2">
   347→   347→          {/* O que você vai aprender */}
   348→   348→          <section
   349→   349→            aria-labelledby="aprender-title"
   350→   350→            className="rounded-2xl border border-stone-200 bg-white p-5 sm:p-6"
   351→   351→          >
   352→   352→            <h2 id="aprender-title" className="text-lg font-extrabold tracking-tight text-stone-900">
   353→   353→              O que você vai aprender
   354→   354→            </h2>
   355→   355→            {lessons.length > 0 ? (
   356→   356→              <ul className="mt-4 grid gap-x-6 gap-y-3 sm:grid-cols-2">
   357→   357→                {lessons.slice(0, 6).map((lesson) => (
   358→   358→                  <li key={lesson.id} className="flex items-start gap-2.5 text-sm text-stone-700">
   359→   359→                    <Check aria-hidden className="mt-0.5 h-4 w-4 shrink-0 text-emerald-600" />
   360→   360→                    <span className="leading-snug">{lesson.title}</span>
   361→   361→                  </li>
   362→   362→                ))}
   363→   363→              </ul>
   364→   364→            ) : (
   365→   365→              <p className="mt-3 text-sm text-stone-400">
   366→   366→                As aulas deste curso estão sendo preparadas.
   367→   367→              </p>
   368→   368→            )}
   369→   369→          </section>
   370→   370→
   371→   371→          {/* Currículo */}
   372→   372→          <section
   373→   373→            aria-labelledby="curriculo-title"
   374→   374→            className="rounded-2xl border border-stone-200 bg-white p-5 sm:p-6"
   375→   375→          >
   376→   376→            <div className="flex flex-wrap items-center justify-between gap-2">
   377→   377→              <h2 id="curriculo-title" className="text-lg font-extrabold tracking-tight text-stone-900">
   378→   378→                Currículo do curso
   379→   379→              </h2>
   380→   380→              {lessons.length > 0 && (
   381→   381→                <span className="text-xs font-medium text-stone-400">
   382→   382→                  {lessons.length} {lessons.length === 1 ? 'aula' : 'aulas'} ·{' '}
   383→   383→                  {formatTotalDuration(course.totalDurationMin)}
   384→   384→                </span>
   385→   385→              )}
   386→   386→            </div>
   387→   387→            <ol className="mt-3 divide-y divide-stone-100">
   388→   388→              {lessons.map((lesson, i) => (
   389→   389→                <li key={lesson.id} className="flex items-center gap-3 py-3.5 first:pt-4 last:pb-0">
   390→   390→                  <span className="flex h-7 w-7 shrink-0 items-center justify-center rounded-full bg-stone-100 text-xs font-bold text-stone-500">
   391→   391→                    {i + 1}
   392→   392→                  </span>
   393→   393→                  <div className="min-w-0 flex-1">
   394→   394→                    <p className="truncate text-sm font-semibold text-stone-800">{lesson.title}</p>
   395→   395→                    <p className="mt-0.5 line-clamp-1 text-xs text-stone-400">{lesson.description}</p>
   396→   396→                  </div>
   397→   397→                  <span className="hidden shrink-0 items-center gap-1 text-xs text-stone-400 sm:inline-flex">
   398→   398→                    {lesson.videoUrl ? (
   399→   399→                      <PlayCircle aria-hidden className="h-4 w-4" />
   400→   400→                    ) : (
   401→   401→                      <FileText aria-hidden className="h-4 w-4" />
   402→   402→                    )}
   403→   403→                    {lesson.durationMin} min
   404→   404→                  </span>
   405→   405→                  <Lock
   406→   406→                    aria-label="Aula bloqueada — inscreva-se para acessar"
   407→   407→                    className="h-3.5 w-3.5 shrink-0 text-stone-300"
   408→   408→                  />
   409→   409→                </li>
   410→   410→              ))}
   411→   411→            </ol>
   412→   412→            {lessons.length === 0 && (
   413→   413→              <p className="py-4 text-sm text-stone-400">Nenhuma aula publicada ainda.</p>
   414→   414→            )}
   415→   415→          </section>
   416→   416→
   417→   417→          {/* Mentor */}
   418→   418→          <section
   419→   419→            aria-label="Mentor do curso"
   420→   420→            className="rounded-2xl border border-stone-200 bg-white p-5 sm:p-6"
   421→   421→          >
   422→   422→            <div className="flex flex-col gap-4 sm:flex-row sm:items-center">
   423→   423→              <Avatar name={course.mentor.name} src={course.mentor.avatarUrl} size="lg" />
   424→   424→              <div className="min-w-0 flex-1">
   425→   425→                <p className="font-bold text-stone-900">{course.mentor.name}</p>
   426→   426→                <p className="line-clamp-1 text-sm text-stone-500">{course.mentor.headline}</p>
   427→   427→                <div className="mt-1.5 flex items-center gap-1.5">
   428→   428→                  <Stars rating={course.mentor.rating} size={13} />
   429→   429→                  <span className="text-xs font-semibold text-stone-700">
   430→   430→                    {course.mentor.rating > 0 ? course.mentor.rating.toFixed(1) : 'Novo'}
   431→   431→                  </span>
   432→   432→                  <span className="text-xs text-stone-400">
   433→   433→                    ({course.mentor.reviewCount} {course.mentor.reviewCount === 1 ? 'avaliação' : 'avaliações'})
   434→   434→                  </span>
   435→   435→                </div>
   436→   436→              </div>
