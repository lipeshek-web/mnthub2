   864→/* ==================== NAVEGAÇÃO DE CONTEÚDOS (sidebar + dialog mobile) ==================== */
   865→
   866→function ContentsNav({
   867→  groups,
   868→  themeList,
   869→  collapsedThemes,
   870→  onToggleTheme,
   871→  currentLessonId,
   872→  completedIds,
   873→  onSelectLesson,
   874→  className,
   875→}: {
   876→  groups: LessonGroup[]
   877→  themeList: CourseThemeDTO[]
   878→  collapsedThemes: Record<string, boolean>
   879→  onToggleTheme: (key: string) => void
   880→  currentLessonId: string | null
   881→  completedIds: string[]
   882→  onSelectLesson: (lessonId: string) => void
   883→  className?: string
   884→}) {
   885→  const totalLessons = groups.reduce((n, g) => n + g.lessons.length, 0)
   886→
   887→  return (
   888→    <nav
   889→      aria-label="Conteúdos do curso"
   890→      className={cn(
   891→        'min-h-0 flex-1 overflow-y-auto px-3 pb-3 [scrollbar-width:thin] [&::-webkit-scrollbar-thumb]:rounded-full [&::-webkit-scrollbar-thumb]:bg-stone-300 dark:[&::-webkit-scrollbar-thumb]:bg-stone-700 [&::-webkit-scrollbar-track]:bg-stone-100 dark:[&::-webkit-scrollbar-track]:bg-stone-900 [&::-webkit-scrollbar]:w-1.5',
   892→        className
   893→      )}
   894→    >
   895→      {groups.map((group) => {
   896→        const collapsible = themeList.length > 0
   897→        const isCollapsed = collapsible && Boolean(collapsedThemes[group.key])
   898→        const groupTotal = group.lessons.length
   899→        const groupDone = group.lessons.filter((l) => completedIds.includes(l.id)).length
   900→        return (
   901→          <div key={group.key} className="space-y-1 py-1">
   902→            {collapsible ? (
   903→              <button
   904→                type="button"
   905→                onClick={() => onToggleTheme(group.key)}
   906→                aria-expanded={!isCollapsed}
   907→                aria-label={`${isCollapsed ? 'Expandir' : 'Recolher'} a seção ${group.title}`}
   908→                className="flex min-h-11 w-full items-center gap-1.5 rounded-xl px-1.5 py-1.5 text-left transition-colors hover:bg-stone-50 dark:hover:bg-stone-800"
   909→              >
   910→                {isCollapsed ? (
   911→                  <ChevronRight aria-hidden className="h-4 w-4 shrink-0 text-stone-400 dark:text-stone-500" />
   912→                ) : (
   913→                  <ChevronDown aria-hidden className="h-4 w-4 shrink-0 text-stone-400 dark:text-stone-500" />
   914→                )}
   915→                {group.theme ? (
   916→                  <span className="flex h-6 w-6 shrink-0 items-center justify-center rounded-full bg-emerald-100 text-[11px] font-bold text-emerald-800 dark:bg-emerald-950/50 dark:text-emerald-300">
   917→                    {themeList.findIndex((t) => t.id === group.key) + 1}
   918→                  </span>
   919→                ) : (
   920→                  <span className="flex h-6 w-6 shrink-0 items-center justify-center rounded-full bg-stone-100 dark:bg-stone-800">
   921→                    <Folder aria-hidden className="h-3.5 w-3.5 text-stone-500 dark:text-stone-400" />
   922→                  </span>
   923→                )}
   924→                <span className="min-w-0 flex-1 truncate text-sm font-bold text-stone-700 dark:text-stone-200">
   925→                  {group.title}
   926→                </span>
   927→                <span className="shrink-0 text-[11px] font-semibold text-stone-400 dark:text-stone-500">
   928→                  {groupTotal} {groupTotal === 1 ? 'aula' : 'aulas'}
   929→                </span>
