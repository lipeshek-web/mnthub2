    81→  for (let minutes = 6 * 60; minutes <= 21 * 60 + 30; minutes += 30) {
    82→    const h = Math.floor(minutes / 60)
    83→    const m = minutes % 60
    84→    options.push(`${String(h).padStart(2, '0')}:${String(m).padStart(2, '0')}`)
    85→  }
    86→  return options

   111→function MentorProfileForm({

   116→  initial?: ProfileFormValues

   120→  const [headline, setHeadline] = useState(initial?.headline ?? '')
   121→  const [description, setDescription] = useState(initial?.description ?? '')
   122→  const [categories, setCategories] = useState<string[]>(initial?.categories ?? [])

   124→  const [experienceYears, setExperienceYears] = useState(initial ? String(initial.experienceYears) : '')

   131→      prev.includes(category) ? prev.filter((c) => c !== category) : [...prev, category]
   132→    )

   134→  }
   135→

   137→    const errs: ProfileFormErrors = {}
   138→    if (headline.trim().length < 8) {
   139→      errs.headline = 'O título deve ter pelo menos 8 caracteres.'
   140→    }
   141→    if (description.trim().length < 30) {
   142→      errs.description = 'Descreva sua mentoria com pelo menos 30 caracteres.'
   143→    }
   144→    if (categories.length === 0) {
   145→      errs.categories = 'Selecione pelo menos uma categoria.'
   146→    }

   148→    if (hourlyRate.trim() === '' || Number.isNaN(rate) || rate < 0) {
   149→      errs.hourlyRate = 'Informe um valor válido por hora (maior ou igual a zero).'

   152→    if (experienceYears.trim() === '' || Number.isNaN(years) || years < 0) {

   190→        />
   191→        {errors.headline ? (
   192→          <p className="text-xs text-rose-600">{errors.headline}</p>
   193→        ) : (
   194→          <p className="text-xs text-muted-foreground">
   195→            Uma frase curta que resume quem você é e o que faz.

   207→          placeholder="Conte como você trabalha: o que o aluno recebe, como são as sessões e quais resultados você pode ajudar a alcançar..."
   208→          aria-invalid={Boolean(errors.description)}
   209→        />

   211→          <p className="text-xs text-rose-600">{errors.description}</p>

   214→            Detalhe sua metodologia, público ideal e o que o aluno leva de cada sessão (mínimo 30 caracteres).

   313→// ---------- Editor de disponibilidade semanal ----------
   314→
   315→function AvailabilityEditor({
   316→  initialSlots,
   317→  onSave,
   318→}: {
   319→  initialSlots: AvailabilitySlotInput[]
   320→  onSave: (slots: AvailabilitySlotInput[]) => Promise<void>
   321→}) {
   322→  const [slots, setSlots] = useState<AvailabilitySlotInput[]>(() =>
   323→    [...initialSlots].sort((a, b) => a.weekday - b.weekday || a.startHour - b.startHour)
   324→  )
   325→  const [addingDay, setAddingDay] = useState<number | null>(null)
   326→  const [newStart, setNewStart] = useState('09:00')
   327→  const [newEnd, setNewEnd] = useState('12:00')
   328→  const [saving, setSaving] = useState(false)
   329→
   330→  const openAdd = (weekday: number) => {
   331→    setAddingDay(weekday)
   332→    setNewStart('09:00')
   333→    setNewEnd('12:00')
   334→  }
   335→
   336→  const removeSlot = (weekday: number, startHour: number, endHour: number) => {
   337→    setSlots((prev) =>
   338→      prev.filter((s) => !(s.weekday === weekday && s.startHour === startHour && s.endHour === endHour))
   339→    )
   340→  }
   341→
   342→  const confirmAdd = () => {
   343→    if (addingDay === null) return
   344→    const startHour = labelToHour(newStart)
   345→    const endHour = labelToHour(newEnd)
   346→    if (endHour - startHour < 1) {
   347→      toast.error('A faixa de horário deve ter pelo menos 1 hora de duração.')
   348→      return
   349→    }
   350→    const conflicts = slots.some(
   351→      (s) => s.weekday === addingDay && startHour < s.endHour && endHour > s.startHour
   352→    )
   353→    if (conflicts) {
   354→      toast.error('Esta faixa conflita com um horário já cadastrado neste dia.')
   355→      return
   356→    }
   357→    setSlots((prev) => [...prev, { weekday: addingDay, startHour, endHour }])
   358→    setAddingDay(null)
   359→  }
   360→
   361→  const handleSave = async () => {
   362→    setSaving(true)
   363→    try {
   364→      const ordered = [...slots].sort((a, b) => a.weekday - b.weekday || a.startHour - b.startHour)
   365→      await onSave(ordered)
   366→    } finally {
   367→      setSaving(false)
   368→    }
   369→  }
   370→
   371→  return (
   372→    <div className="flex flex-col gap-4">
   373→      <div className="flex flex-col divide-y divide-stone-200 rounded-lg border border-stone-200">
   374→        {WEEKDAYS_FULL_PT.map((day, weekday) => {
   375→          const daySlots = slots
   376→            .filter((s) => s.weekday === weekday)
   377→            .sort((a, b) => a.startHour - b.startHour)
   378→          return (
   379→            <div key={day} className="flex flex-col gap-3 p-4 sm:flex-row sm:items-center sm:justify-between">
   380→              <div className="flex min-w-0 flex-1 flex-wrap items-center gap-2">
   381→                <span className="w-28 shrink-0 text-sm font-medium sm:w-32">{day}</span>
   382→                {daySlots.length === 0 ? (
   383→                  <span className="text-xs text-stone-400">Sem disponibilidade</span>
   384→                ) : (
   385→                  <span className="flex flex-wrap items-center gap-1.5">
   386→                    {daySlots.map((s) => (
   387→                      <Badge
   388→                        key={`${s.weekday}-${s.startHour}-${s.endHour}`}
   389→                        variant="outline"
   390→                        className="gap-1 border-emerald-200 bg-emerald-50 py-1 text-emerald-800"
   391→                      >
   392→                        {hourToLabel(s.startHour)} – {hourToLabel(s.endHour)}
   393→                        <button
   394→                          type="button"
   395→                          aria-label={`Remover faixa de ${hourToLabel(s.startHour)} às ${hourToLabel(
   396→                            s.endHour
   397→                          )} na ${day.toLowerCase()}`}
   398→                          onClick={() => removeSlot(s.weekday, s.startHour, s.endHour)}
   399→                          className="rounded-full p-0.5 transition-colors hover:bg-emerald-200/70 focus-visible:outline-2 focus-visible:outline-emerald-600"
   400→                        >
   401→                          <X className="size-3" aria-hidden />
   402→                        </button>
   403→                      </Badge>
   404→                    ))}
   405→                  </span>
   406→                )}
   407→              </div>
   408→
   409→              <div className="shrink-0">
   410→                {addingDay === weekday ? (
   411→                  <div className="flex flex-wrap items-center gap-2">
   412→                    <Select value={newStart} onValueChange={setNewStart}>
   413→                      <SelectTrigger size="sm" className="w-24" aria-label="Início da faixa">
   414→                        <SelectValue />
   415→                      </SelectTrigger>
   416→                      <SelectContent>
   417→                        {TIME_OPTIONS.map((time) => (
   418→                          <SelectItem key={time} value={time}>
   419→                            {time}
   420→                          </SelectItem>
   421→                        ))}
   422→                      </SelectContent>
   423→                    </Select>
   424→                    <span className="text-sm text-muted-foreground" aria-hidden>
   425→                      →
   426→                    </span>
   427→                    <Select value={newEnd} onValueChange={setNewEnd}>
   428→                      <SelectTrigger size="sm" className="w-24" aria-label="Fim da faixa">
   429→                        <SelectValue />
   430→                      </SelectTrigger>
   431→                      <SelectContent>
   432→                        {TIME_OPTIONS.map((time) => (
   433→                          <SelectItem key={time} value={time}>
   434→                            {time}
   435→                          </SelectItem>
   436→                        ))}
   437→                      </SelectContent>
   438→                    </Select>
   439→                    <Button size="sm" onClick={confirmAdd}>
   440→                      <Check className="size-4" aria-hidden /> Confirmar
   441→                    </Button>
   442→                    <Button size="sm" variant="ghost" onClick={() => setAddingDay(null)}>
   443→                      Cancelar
   444→                    </Button>
   445→                  </div>
   446→                ) : (
   447→                  <Button
   448→                    size="sm"
   449→                    variant="ghost"
   450→                    onClick={() => openAdd(weekday)}
   451→                    aria-label={`Adicionar faixa de horário na ${day.toLowerCase()}`}
   452→                  >
   453→                    <Plus className="size-4" aria-hidden /> Adicionar
   454→                  </Button>
   455→                )}
   456→              </div>
   457→            </div>
   458→          )
   459→        })}
   460→      </div>
   461→
   462→      <div className="flex flex-col gap-3 sm:flex-row sm:items-center sm:justify-between">
   463→        <p className="text-xs text-muted-foreground">
   464→          Alunos só podem agendar dentro destas faixas. Use intervalos de pelo menos 1 hora.
   465→        </p>
   466→        <Button onClick={() => void handleSave()} disabled={saving} className="shrink-0">
   467→          <Save className="size-4" aria-hidden /> {saving ? 'Salvando...' : 'Salvar disponibilidade'}
   468→        </Button>
   469→      </div>
   470→    </div>
   471→  )
   472→}

   476→function ContentsManager({
   477→  contents,
   478→  userId,
   479→  onChanged,
   480→}: {
   481→  contents: ContentPostDTO[]
   482→  userId: string
   483→  onChanged: () => Promise<void>
   484→}) {
   485→  const [createOpen, setCreateOpen] = useState(false)
   486→  const [toDelete, setToDelete] = useState<ContentPostDTO | null>(null)
   487→  const [deleting, setDeleting] = useState(false)
   488→  const [title, setTitle] = useState('')
   489→  const [description, setDescription] = useState('')
   490→  const [type, setType] = useState<string>('ARTICLE')
   491→  const [level, setLevel] = useState<string>('INICIANTE')
   492→  const [durationMin, setDurationMin] = useState('45')
   493→  const [tags, setTags] = useState('')
   494→  const [saving, setSaving] = useState(false)
   495→  const [formErrors, setFormErrors] = useState<{
   496→    title?: string
   497→    description?: string
   498→    durationMin?: string
   499→  }>({})
   500→
   501→  const handleCreate = async () => {
   502→    const errs: typeof formErrors = {}
   503→    if (title.trim().length < 3) {
   504→      errs.title = 'Dê um título ao conteúdo (mínimo 3 caracteres).'
   505→    }
   506→    if (description.trim().length < 10) {
   507→      errs.description = 'Escreva uma breve descrição (mínimo 10 caracteres).'
   508→    }
   509→    const duration = Number(durationMin)
   510→    if (durationMin.trim() === '' || Number.isNaN(duration) || duration <= 0) {
   511→      errs.durationMin = 'Informe a duração em minutos.'
   512→    }
   513→    setFormErrors(errs)
   514→    if (Object.values(errs).some(Boolean)) return
   515→
   516→    setSaving(true)
   517→    try {
   518→      await api.createContent({
   519→        userId,
   520→        title: title.trim(),
   521→        description: description.trim(),
   522→        type,
   523→        level,
   524→        durationMin: duration,
   525→        tags: tags
   526→          .split(',')
   527→          .map((t) => t.trim())
   528→          .filter((t) => t.length > 0),
   529→      })
   530→      toast.success('Conteúdo publicado no mural!')
   531→      setCreateOpen(false)
   532→      setTitle('')
   533→      setDescription('')
   534→      setType('ARTICLE')
   535→      setLevel('INICIANTE')
   536→      setDurationMin('45')
   537→      setTags('')
   538→      await onChanged()
   539→    } catch (err) {
   540→      toast.error(err instanceof Error ? err.message : 'Não foi possível publicar o conteúdo.')
   541→    } finally {
   542→      setSaving(false)
   543→    }
   544→  }
   545→
   546→  const handleDelete = async () => {
   547→    if (!toDelete) return
   548→    setDeleting(true)
   549→    try {
   550→      await api.deleteContent(toDelete.id, userId)
   551→      toast.success('Conteúdo removido do mural.')
   552→      setToDelete(null)
   553→      await onChanged()
   554→    } catch (err) {
   555→      toast.error(err instanceof Error ? err.message : 'Não foi possível remover o conteúdo.')
   556→    } finally {
   557→      setDeleting(false)
   558→    }
   559→  }
   560→
   561→  return (
   562→    <Card>
   563→      <CardHeader>
   564→        <CardTitle className="flex items-center gap-2 text-lg">
   565→          <span className="flex size-8 shrink-0 items-center justify-center rounded-lg bg-emerald-50 text-emerald-600 ring-1 ring-emerald-100">
   566→            <Newspaper className="size-4" aria-hidden />
   567→          </span>
