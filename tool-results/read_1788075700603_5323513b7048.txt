   208→  const navigate = useAppStore((s) => s.navigate)
   209→  const setExploreTab = useAppStore((s) => s.setExploreTab)

   211→  const [mentors, setMentors] = useState<MentorListItemDTO[]>([])
   212→  const [loading, setLoading] = useState(true)
   213→  const [courses, setCourses] = useState<CourseListItemDTO[]>([])
   214→  const [coursesLoading, setCoursesLoading] = useState(true)
   215→  const [tracks, setTracks] = useState<TrackListItemDTO[]>([])
   216→  const [tracksLoading, setTracksLoading] = useState(true)
   217→
   218→  // Fetch preguiçoso: cursos e trilhas só saem quando as seções se aproximam da viewport
   219→  const coursesSectionRef = useRef<HTMLElement | null>(null)
   220→  const coursesInView = useInView(coursesSectionRef, { once: true, margin: '600px' })
   221→  const tracksSectionRef = useRef<HTMLElement | null>(null)
   222→  const tracksInView = useInView(tracksSectionRef, { once: true, margin: '600px' })
   223→
   224→  useEffect(() => {
   225→    let active = true
   226→    api
   227→      .listMentors({})
   228→      .then((data) => {
   229→        if (active) setMentors(data)
   230→      })
   231→      .catch(() => {
   232→        if (active) setMentors([])
   233→      })
   234→      .finally(() => {
   235→        if (active) setLoading(false)
   236→      })
   237→    return () => {
   238→      active = false
   239→    }
   240→  }, [])
   241→
   242→  // Cursos para a seção "Cursos em destaque" (disparado quando a seção ou a faixa de números se aproxima)
   243→  const statsSectionRef = useRef<HTMLElement | null>(null)
   244→  const statsInView = useInView(statsSectionRef, { once: true, margin: '600px' })
   245→
   246→  useEffect(() => {
   247→    if (!coursesInView && !statsInView) return
   248→    let active = true
   249→    api
   250→      .listCourses({})
   251→      .then((data) => {
   252→        if (active) setCourses(data)
   253→      })
   254→      .catch(() => {
   255→        if (active) setCourses([])
   256→      })
   257→      .finally(() => {
   258→        if (active) setCoursesLoading(false)
   259→      })
   260→    return () => {
   261→      active = false
   262→    }
   263→  }, [coursesInView, statsInView])
   264→
   265→  // Trilhas populares para a seção "Trilhas em destaque" (seção ou faixa de números se aproximando)
   266→  useEffect(() => {
   267→    if (!tracksInView && !statsInView) return
   268→    let active = true
   269→    api
   270→      .listTracks({ sort: 'popular' })
   271→      .then((data) => {
   272→        if (active) setTracks(data)
   273→      })
   274→      .catch(() => {
   275→        if (active) setTracks([])
   276→      })
   277→      .finally(() => {
   278→        if (active) setTracksLoading(false)
   279→      })
   280→    return () => {
   281→      active = false
   282→    }
   283→  }, [tracksInView, statsInView])
   284→
   285→  // Biblioteca em destaque (disparado quando a seção se aproxima)
   286→  const librarySectionRef = useRef<HTMLElement | null>(null)
   287→  const libraryInView = useInView(librarySectionRef, { once: true, margin: '600px' })
   288→  const [libraryItems, setLibraryItems] = useState<LibraryItemDTO[]>([])
   289→  const [libraryLoading, setLibraryLoading] = useState(true)
   290→
   291→  useEffect(() => {
   292→    if (!libraryInView) return
   293→    let active = true
   294→    api
   295→      .listLibrary({ sort: 'popular' })
   296→      .then((data) => {
   297→        if (active) setLibraryItems(data)
   298→      })
   299→      .catch(() => {
   300→        if (active) setLibraryItems([])
   301→      })
   302→      .finally(() => {
   303→        if (active) setLibraryLoading(false)
   304→      })
   305→    return () => {
   306→      active = false
   307→    }
   308→  }, [libraryInView])
   309→
   310→  // "Continue aprendendo": inscrições do usuário logado (seção oculta para convidados)
   311→  const userId = user?.id
   312→  const [enrollments, setEnrollments] = useState<EnrolledCourseDTO[]>([])
   313→  const [enrollmentsLoading, setEnrollmentsLoading] = useState(true)
   314→
   315→  useEffect(() => {
   316→    // Convidado: a seção fica oculta (user == null) e o estado inicial não precisa mudar.
   317→    if (!userId) return
   318→    let active = true
   319→    api
   320→      .listMyEnrollments(userId)
   321→      .then((data) => {
   322→        if (active) setEnrollments(data)
   323→      })
   324→      .catch(() => {
   325→        if (active) setEnrollments([])
   326→      })
   327→      .finally(() => {
   328→        if (active) setEnrollmentsLoading(false)
   329→      })
   330→    return () => {
   331→      active = false
   332→    }
   333→  }, [userId])
   334→
   335→  const totalReviews = useMemo(
   336→    () => mentors.reduce((acc, m) => acc + m.reviewCount, 0),
   337→    [mentors]
   338→  )
   339→
   340→  const avgRating = useMemo(() => {
   341→    const rated = mentors.filter((m) => m.rating > 0)
   342→    if (rated.length === 0) return null
   343→    return rated.reduce((acc, m) => acc + m.rating, 0) / rated.length
   344→  }, [mentors])
   345→
   346→  // Números da faixa "A plataforma em números" (chegam junto com os fetches preguiçosos)
   347→  const totalLessons = useMemo(
   348→    () => courses.reduce((acc, c) => acc + c.lessonCount, 0),
   349→    [courses]
   350→  )
   351→
   352→  // Top 3 itens da biblioteca (a API já devolve ordenada por popularidade)
   353→  const topLibrary = useMemo(() => libraryItems.slice(0, 3), [libraryItems])
   354→
   355→  const featured = useMemo(
   356→    () =>
   357→      mentors
   358→        .filter((m) => m.rating > 0)
   359→        .sort((a, b) => b.rating - a.rating || b.reviewCount - a.reviewCount)
   360→        .slice(0, 3),
   361→    [mentors]
   362→  )
   363→
   364→  // Top 3 cursos por alunos inscritos, desempate pela nota do mentor
   365→  const topCourses = useMemo(
   366→    () =>
   367→      [...courses]
   368→        .sort(
   369→          (a, b) =>
   370→            b.studentCount - a.studentCount || b.mentor.rating - a.mentor.rating
   371→        )
   372→        .slice(0, 3),
   373→    [courses]
   374→  )
   375→
   376→  // Top 3 trilhas (a API já devolve ordenada por popularidade)
   377→  const topTracks = useMemo(() => tracks.slice(0, 3), [tracks])
   378→
   379→  // Até 3 inscrições para "Continue aprendendo" — prioridade: em andamento (0 < x < 100%),
   380→  // depois não iniciadas (0%), concluídas por último; recência (enrolledAt desc) dentro do grupo
   381→  const continueItems = useMemo(() => {
   382→    const byRecency = (a: EnrolledCourseDTO, b: EnrolledCourseDTO) =>
   383→      b.enrolledAt.localeCompare(a.enrolledAt) // ISO-8601: comparação lexicográfica segura
   384→    const isDone = (e: EnrolledCourseDTO) =>
   385→      e.course.lessonCount > 0 && e.completedLessonIds.length >= e.course.lessonCount
   386→    const started = enrollments
   387→      .filter((e) => !isDone(e) && e.completedLessonIds.length > 0)
   388→      .sort(byRecency)
   389→    const untouched = enrollments
   390→      .filter((e) => e.completedLessonIds.length === 0)
   391→      .sort(byRecency)
   392→    const done = enrollments.filter(isDone).sort(byRecency)
   393→    return [...started, ...untouched, ...done].slice(0, 3)
   394→  }, [enrollments])
   395→
   396→  // "Feito para você": recomendações da IA (cruza histórico do aluno com o catálogo).
   397→  // Silencioso em erro/sem resultados — a landing nunca quebra por causa da IA.
   398→  const [recommendations, setRecommendations] = useState<RecommendationDTO[]>([])
   399→  const [recsGenerated, setRecsGenerated] = useState(false)
   400→
   401→  useEffect(() => {
   402→    // Convidado: seção oculta (render gated por user) — nada a buscar.
   403→    if (!userId) return
   404→    let active = true
   405→    api
   406→      .recommendations(userId)
   407→      .then((data) => {
   408→        if (!active) return
   409→        setRecommendations(data.items ?? [])
   410→        setRecsGenerated(Boolean(data.generated))
   411→      })
   412→      .catch(() => {
   413→        if (active) setRecommendations([])
   414→      })
   415→    return () => {
   416→      active = false
   417→    }
   418→  }, [userId])
   419→
   420→  const handleExploreMentors = () => {

   422→    navigate({ name: 'marketplace' })
   423→  }
   424→
   425→  const handleVerCursos = () => {
   426→    setExploreTab('courses')
   427→    navigate({ name: 'marketplace' })
   428→  }
   429→
   430→  const handleVerTrilhas = () => {
   431→    setExploreTab('tracks')
   432→    navigate({ name: 'marketplace' })
   433→  }
   434→
   435→  const handleVerBiblioteca = () => {
   436→    setExploreTab('library')
   437→    navigate({ name: 'marketplace' })
   438→  }
   439→
   440→  const handleExploreArea = () => {
   441→    setExploreTab('all')
   442→    navigate({ name: 'marketplace' })
   443→  }
   444→
   445→  const handleFormatTab = (tab: (typeof FORMATS)[number]['tab']) => {
   446→    setExploreTab(tab)
   447→    navigate({ name: 'marketplace' })
   448→  }
   449→
   450→  return (
   451→    <div className="flex flex-col bg-white dark:bg-stone-950">
   452→      {/* ---------- HERO (split com carrossel leve dos formatos) ---------- */}
   453→      <section aria-labelledby="hero-title" className="relative overflow-hidden">
   454→        {/* Decoração de fundo: um único brilho suave — leve, sem ruído */}
   455→        <div
   456→          aria-hidden
   457→          className="absolute -left-32 -top-32 h-96 w-96 rounded-full bg-emerald-100/60 blur-3xl dark:bg-emerald-950/40"
   458→        />
   459→
   460→        <div className="relative mx-auto max-w-6xl px-4 pb-16 pt-12 sm:pt-16 lg:pb-24">
   461→          <div className="grid grid-cols-1 items-center gap-14 lg:grid-cols-2 lg:gap-12">
   462→            {/* Coluna esquerda: mensagem + busca + prova social */}
   463→            <motion.div
   464→              initial={{ opacity: 0, y: 16 }}
   465→              animate={{ opacity: 1, y: 0 }}
   466→              transition={{ duration: 0.5 }}
   467→              className="min-w-0"
   468→            >
   469→              <span className="inline-flex items-center gap-1.5 rounded-full border border-emerald-200 bg-emerald-50 px-3.5 py-1.5 text-xs font-semibold text-emerald-800 dark:border-emerald-900 dark:bg-emerald-950/50 dark:text-emerald-300">
   470→                <Sparkles aria-hidden className="h-3.5 w-3.5 text-emerald-700 dark:text-emerald-300" />
   471→                Mentorias, cursos, trilhas e biblioteca — tudo em um só lugar
   472→              </span>
   473→
   474→              <h1
   475→                id="hero-title"
   476→                className="mt-5 text-4xl font-extrabold leading-[1.08] tracking-tight text-stone-900 sm:text-5xl xl:text-6xl dark:text-stone-50"
   477→              >
   478→                {user ? (
   479→                  <>
   480→                    Olá, <span className="text-emerald-700 dark:text-emerald-300">{firstName(user.name)}</span>! Pronto
   481→                    para{' '}
   482→                    <span className="underline decoration-emerald-400/60 decoration-[0.12em] underline-offset-[0.16em]">
   483→                      continuar aprendendo
   484→                    </span>
   485→                    ?
   486→                  </>
   487→                ) : (
   488→                  <>
   489→                    Aprenda com quem{' '}
   490→                    <span className="text-emerald-700 underline decoration-emerald-400/60 decoration-[0.12em] underline-offset-[0.16em] dark:text-emerald-300">
   491→                      vive o que ensina
   492→                    </span>
   493→                  </>
   494→                )}
   495→              </h1>
   496→
   497→              <p className="mt-5 max-w-xl text-base leading-relaxed text-stone-600 sm:text-lg dark:text-stone-300">
   498→                Especialistas que viveram o caminho que você quer trilhar — em mentorias 1:1, cursos
   499→                gravados, trilhas guiadas e uma biblioteca completa. Tudo dentro da plataforma, do
   500→                primeiro clique ao certificado.
   501→              </p>
   502→
   503→              {/* CTA: a busca vive na barra central do header — aqui, o convite direto */}
   504→              <div className="mt-8 flex max-w-xl flex-col gap-2.5 sm:flex-row sm:items-center">
   505→                <Button
   506→                  onClick={handleExploreArea}
   507→                  className="h-13 w-full rounded-full px-8 font-bold shadow-sm shadow-emerald-700/20 sm:w-auto"
   508→                >
   509→                  <Search aria-hidden className="h-4.5 w-4.5" /> Explorar tudo
   510→                </Button>
   511→                <p className="text-sm text-stone-500 dark:text-stone-400">
   512→                  ou use a busca acima ✨
   513→                </p>
   514→              </div>
   515→
   516→              {/* Prova social: avatares + nota média + avaliações reais */}
   517→              {loading ? (
   518→                <div className="mt-7 flex items-center gap-3">
   519→                  <div className="flex -space-x-2.5">
   520→                    {Array.from({ length: 5 }).map((_, i) => (
   521→                      <Skeleton key={i} className="h-8 w-8 rounded-full" />
   522→                    ))}
   523→                  </div>
   524→                  <Skeleton className="h-4 w-44" />
   525→                </div>
   526→              ) : mentors.length > 0 ? (
   527→                <div className="mt-7 flex flex-wrap items-center gap-x-4 gap-y-3">
   528→                  <div className="flex -space-x-2.5">
   529→                    {mentors.slice(0, 5).map((m) => (
   530→                      <Avatar key={m.id} name={m.name} src={m.avatarUrl} size="sm" />
